
// TextEncoder / TextDecoder — engine globals (LLRT tier: real for the
// supported label set). Pure JS: the codecs are the handwritten state
// machines the buffer/string_decoder shims use (the QuickJS engine has no
// native TextDecoder/Intl), lifted here into standalone classes.
//
// Supported labels: utf-8 (with fatal + streaming + BOM handling), the
// windows-1252 encoding (the WHATWG label table maps latin1/iso-8859-1/ascii/
// us-ascii/cp1252/... to windows-1252, so bytes 0x80-0x9F decode to the
// CP1252 punctuation, exactly like node) and utf-16le (utf-16/ucs-2/...,
// little-endian only). Anything else throws Node's
// RangeError ERR_ENCODING_NOT_SUPPORTED.

(function () {
  // Host fast path (§6.5 host-first policy): when the pir extension runtime
  // registered its native codec (`__pirBufferCodec`, see
  // src/extensions/runtime/host_functions/buffer_ops.rs), the decode cores
  // below dispatch to it; `undefined` results (unmarshalable inputs like
  // lone surrogates) or thrown errors fall back to the JS state machines —
  // behavior never changes either way.
  const pirNativeCodec = typeof globalThis.__pirBufferCodec === "function" ? globalThis.__pirBufferCodec : null;
  const PIR_EMPTY_BYTES = new Uint8Array(0);

  // Node's WHATWG error for invalid data under `fatal`.
  function pirFatalDecodeError(encoding) {
    const error = new TypeError("The encoded data was not valid for encoding " + encoding);
    error.code = "ERR_ENCODING_INVALID_ENCODED_DATA";
    return error;
  }

  // Node's ERR_ENCODING_NOT_SUPPORTED for an unknown label.
  function pirEncodingNotSupported(label) {
    const error = new RangeError('The "' + label + '" encoding is not supported');
    error.code = "ERR_ENCODING_NOT_SUPPORTED";
    return error;
  }

  // Node validates options with util.validateObject (objects and null only).
  function pirValidateOptions(options, name) {
    if (options === undefined || options === null) return null;
    if (typeof options !== "object" && typeof options !== "function") {
      const error = new TypeError(
        'The "' + name + '" argument must be of type object. Received type ' + typeof options
      );
      error.code = "ERR_INVALID_ARG_TYPE";
      throw error;
    }
    return options;
  }

  // WHATWG windows-1252 index for 0x80-0x9F (encoding.spec.whatwg.org). The
  // five bytes the standard leaves unmapped (0x81/0x8D/0x8F/0x90/0x9D) map to
  // themselves in Node's ICU-backed decoder, so they are included here.
  const PIR_WIN1252_HIGH = String.fromCharCode(
    0x20ac, 0x81, 0x201a, 0x192, 0x201e, 0x2026, 0x2020, 0x2021,
    0x2c6, 0x2030, 0x160, 0x2039, 0x152, 0x8d, 0x17d, 0x8f,
    0x90, 0x2018, 0x2019, 0x201c, 0x201d, 0x2022, 0x2013, 0x2014,
    0x2dc, 0x2122, 0x161, 0x203a, 0x153, 0x9d, 0x17e, 0x178
  );
  const PIR_WIN1252_TEST = /[\u0080-\u009f]/;
  const PIR_WIN1252_REPLACE = /[\u0080-\u009f]/g;

  // Map a decoded ISO-8859-1 string's 0x80-0x9F characters to CP1252.
  function pirWin1252FromLatin1(text) {
    if (!PIR_WIN1252_TEST.test(text)) return text;
    return text.replace(PIR_WIN1252_REPLACE, (ch) => PIR_WIN1252_HIGH[ch.charCodeAt(0) - 0x80]);
  }

  // Node's `Received ...` suffix for ERR_INVALID_ARG_TYPE. Only the shapes
  // that reach the streamed decode path matter here (parseInput's validator).
  function pirReceivedDetail(value) {
    if (value === null) return " Received null";
    if (value === undefined) return " Received undefined";
    const type = typeof value;
    if (type === "string") return " Received type string ('" + value + "')";
    if (type === "number" || type === "boolean") return " Received type " + type + " (" + String(value) + ")";
    if (type === "symbol") return " Received type symbol (" + String(value) + ")";
    if (type === "bigint") return " Received type bigint (" + String(value) + "n)";
    if (type === "function") return " Received function " + (value.name || "<anonymous>");
    if (Array.isArray(value)) return " Received an instance of Array";
    let name = "Object";
    try {
      const prototype = Object.getPrototypeOf(value);
      const constructor = prototype === null ? undefined : prototype.constructor;
      if (constructor !== undefined && typeof constructor.name === "string" && constructor.name !== "") {
        name = constructor.name;
      }
    } catch (error) { /* keep Object */ }
    return " Received an instance of " + name;
  }

  // BufferSource → Uint8Array *view* (no copy): the native codecs borrow the
  // backing store and honour the view's byteOffset/byteLength, so a Buffer,
  // a subarray or any typed-array view can cross the bridge as-is.
  function pirInputBytes(input, stream) {
    if (input instanceof Uint8Array) return input;
    if (typeof SharedArrayBuffer !== "undefined" && input instanceof SharedArrayBuffer) {
      return new Uint8Array(input);
    }
    if (typeof ArrayBuffer !== "undefined" && input instanceof ArrayBuffer) {
      return new Uint8Array(input);
    }
    if (typeof ArrayBuffer !== "undefined" && ArrayBuffer.isView(input)) {
      return new Uint8Array(input.buffer, input.byteOffset, input.byteLength);
    }
    // Node's `parseInput` (the streamed path) names the argument "input";
    // the buffered single-shot binding names it "list".
    const error = new TypeError(
      stream
        ? 'The "input" argument must be an instance of ArrayBuffer or ArrayBufferView.' + pirReceivedDetail(input)
        : 'The "list" argument must be an instance of SharedArrayBuffer, ArrayBuffer or ArrayBufferView.'
    );
    error.code = "ERR_INVALID_ARG_TYPE";
    throw error;
  }

  // Unpaired UTF-16 code units in a decoded string (valid astral pairs are
  // not reported). A native utf-16le result containing one cannot be trusted
  // for Node's streaming hold rules, so the JS decoder takes the chunk.
  const PIR_LONE_HIGH_SURROGATE = /[\uD800-\uDBFF](?![\uDC00-\uDFFF])/;
  const PIR_LONE_LOW_SURROGATE = /(?:^|[^\uD800-\uDBFF])[\uDC00-\uDFFF]/;
  function pirUtf16leHasLoneSurrogate(text) {
    return PIR_LONE_HIGH_SURROGATE.test(text) || PIR_LONE_LOW_SURROGATE.test(text);
  }

  // Node's `DecodeData` walk-back: how many trailing bytes form an incomplete
  // UTF-8 character. WHATWG text decoding buffers exactly the same prefix.
  function pirUtf8HoldCount(bytes) {
    const length = bytes.length;
    if (length === 0) return 0;
    const last = bytes[length - 1];
    if (last < 0x80) return 0;
    let start;
    if ((last & 0xc0) === 0x80) {
      start = length - 1;
      while (start > 0 && (bytes[start - 1] & 0xc0) === 0x80) start -= 1;
      if (start === 0) return 0;
      start -= 1;
    } else {
      start = length - 1;
    }
    const lead = bytes[start];
    // Exact WHATWG lead classification. The broad bit-mask classes this
    // replaced accepted invalid leads (C0/C1, F5..FF) and every continuation
    // as a possible second byte, so a definitively invalid pair such as
    // `E0 80` was held for the next chunk instead of being decoded now (Node:
    // two U+FFFD in this chunk).
    let expected;
    let secondLow;
    let secondHigh;
    if (lead >= 0xc2 && lead <= 0xdf) {
      expected = 2; secondLow = 0x80; secondHigh = 0xbf;
    } else if (lead === 0xe0) {
      expected = 3; secondLow = 0xa0; secondHigh = 0xbf;
    } else if (lead >= 0xe1 && lead <= 0xec) {
      expected = 3; secondLow = 0x80; secondHigh = 0xbf;
    } else if (lead === 0xed) {
      expected = 3; secondLow = 0x80; secondHigh = 0x9f;
    } else if (lead >= 0xee && lead <= 0xef) {
      expected = 3; secondLow = 0x80; secondHigh = 0xbf;
    } else if (lead === 0xf0) {
      expected = 4; secondLow = 0x90; secondHigh = 0xbf;
    } else if (lead >= 0xf1 && lead <= 0xf3) {
      expected = 4; secondLow = 0x80; secondHigh = 0xbf;
    } else if (lead === 0xf4) {
      expected = 4; secondLow = 0x80; secondHigh = 0x8f;
    } else {
      return 0; // not a valid lead: the tail can never complete
    }
    const nbuf = length - start;
    // The walk-back proved every byte after the lead is a continuation; only
    // the lead-specific second-byte window can still be violated (the tail is
    // then invalid now, not incomplete).
    if (nbuf >= 2) {
      const second = bytes[start + 1];
      if (second < secondLow || second > secondHigh) return 0;
    }
    return nbuf < expected ? nbuf : 0;
  }

  // Node's brand checks: `decode.call({})` is ERR_INVALID_THIS, not a random
  // property error. A symbol keeps the marker out of the enumerable surface.
  const PIR_ENCODER_BRAND = Symbol("TextEncoder");
  const PIR_DECODER_BRAND = Symbol("TextDecoder");
  function pirInvalidThis(type) {
    const error = new TypeError('Value of "this" must be of type ' + type);
    error.code = "ERR_INVALID_THIS";
    return error;
  }

  // WHATWG "ASCII whitespace": tab, LF, FF, CR, space. String#trim also
  // strips U+00A0 and friends, which node does not.
  function pirTrimAsciiWhitespace(text) {
    let start = 0;
    let end = text.length;
    while (start < end) {
      const code = text.charCodeAt(start);
      if (code !== 0x09 && code !== 0x0a && code !== 0x0c && code !== 0x0d && code !== 0x20) break;
      start += 1;
    }
    while (end > start) {
      const code = text.charCodeAt(end - 1);
      if (code !== 0x09 && code !== 0x0a && code !== 0x0c && code !== 0x0d && code !== 0x20) break;
      end -= 1;
    }
    return text.slice(start, end);
  }

  // Node: `encoding = `${encoding}`` (so only `undefined` defaults to utf-8;
  // null becomes "null"), then the WHATWG label table. The label sets below
  // are the exact windows-1252/utf-8/utf-16le rows of node's encodings map.
  function pirNormalizeLabel(label) {
    const raw = label === undefined ? "utf-8" : String(label);
    const text = pirTrimAsciiWhitespace(raw.toLowerCase());
    if (text === "unicode-1-1-utf-8" || text === "unicode11utf8" ||
        text === "unicode20utf8" || text === "utf8" || text === "utf-8" ||
        text === "x-unicode20utf8") {
      return "utf-8";
    }
    if (text === "ansi_x3.4-1968" || text === "ascii" || text === "cp1252" ||
        text === "cp819" || text === "csisolatin1" || text === "ibm819" ||
        text === "iso-8859-1" || text === "iso-ir-100" || text === "iso8859-1" ||
        text === "iso88591" || text === "iso_8859-1" || text === "iso_8859-1:1987" ||
        text === "l1" || text === "latin1" || text === "us-ascii" ||
        text === "windows-1252" || text === "x-cp1252") {
      return "windows-1252";
    }
    if (text === "csunicode" || text === "iso-10646-ucs-2" || text === "ucs-2" ||
        text === "unicode" || text === "unicodefeff" || text === "utf-16le" ||
        text === "utf-16") {
      return "utf-16le";
    }
    throw pirEncodingNotSupported(raw);
  }

  function pirEncodeIntoUtf8(text, target, targetStart) {
    // Returns {read, written}: `read` counts code units consumed, `written`
    // bytes. A character that would not fit is left for the next call; lone
    // surrogates encode as U+FFFD (WHATWG parity).
    let read = 0;
    let written = 0;
    let i = 0;
    const length = text.length;
    while (i < length) {
      let cp = text.codePointAt(i);
      let units = cp > 0xffff ? 2 : 1;
      let bytes;
      if (cp >= 0xd800 && cp <= 0xdfff) {
        cp = 0xfffd;
        bytes = 3;
      } else if (cp < 0x80) {
        bytes = 1;
      } else if (cp < 0x800) {
        bytes = 2;
      } else if (cp < 0x10000) {
        bytes = 3;
      } else {
        bytes = 4;
      }
      if (written + bytes > target.length - targetStart) break;
      if (bytes === 1) {
        target[targetStart + written++] = cp;
      } else if (bytes === 2) {
        target[targetStart + written++] = 0xc0 | (cp >> 6);
        target[targetStart + written++] = 0x80 | (cp & 63);
      } else if (bytes === 3) {
        target[targetStart + written++] = 0xe0 | (cp >> 12);
        target[targetStart + written++] = 0x80 | ((cp >> 6) & 63);
        target[targetStart + written++] = 0x80 | (cp & 63);
      } else {
        target[targetStart + written++] = 0xf0 | (cp >> 18);
        target[targetStart + written++] = 0x80 | ((cp >> 12) & 63);
        target[targetStart + written++] = 0x80 | ((cp >> 6) & 63);
        target[targetStart + written++] = 0x80 | (cp & 63);
      }
      i += units;
      read += units;
    }
    return { read: read, written: written };
  }

  function pirUtf8Bytes(text) {
    // Straight encoding (no target-size limit): one pass over code points.
    const out = [];
    for (let i = 0; i < text.length; i++) {
      let cp = text.codePointAt(i);
      if (cp > 0xffff) i++;
      if (cp >= 0xd800 && cp <= 0xdfff) cp = 0xfffd;
      if (cp < 0x80) out.push(cp);
      else if (cp < 0x800) out.push(0xc0 | (cp >> 6), 0x80 | (cp & 63));
      else if (cp < 0x10000) {
        out.push(0xe0 | (cp >> 12), 0x80 | ((cp >> 6) & 63), 0x80 | (cp & 63));
      } else {
        out.push(0xf0 | (cp >> 18), 0x80 | ((cp >> 12) & 63), 0x80 | ((cp >> 6) & 63), 0x80 | (cp & 63));
      }
    }
    return Uint8Array.from(out);
  }

  function pirDecodeUtf8Chunk(bytes, state) {
    // bytes: Uint8Array of the current chunk; state carries the incomplete
    // trailing sequence across streaming decode() calls.
    let all;
    if (state.pending.length > 0) {
      all = new Uint8Array(state.pending.length + bytes.length);
      all.set(state.pending, 0);
      all.set(bytes, state.pending.length);
      state.pending = new Uint8Array(0);
    } else {
      all = bytes;
    }
    let out = "";
    let i = 0;
    while (i < all.length) {
      const b0 = all[i];
      if (b0 < 0x80) {
        out += String.fromCharCode(b0);
        i += 1;
        continue;
      }
      let need;
      if (b0 >= 0xc2 && b0 <= 0xdf) need = 1;
      else if (b0 >= 0xe0 && b0 <= 0xef) need = 2;
      else if (b0 >= 0xf0 && b0 <= 0xf4) need = 3;
      else {
        if (state.fatal) throw pirFatalDecodeError("utf-8");
        out += "\uFFFD";
        i += 1;
        continue;
      }
      if (all.length - i - 1 < need) {
        // Incomplete trailing sequence: carried to the next chunk when
        // streaming, replaced (or fatal) when the stream ends.
        state.pending = all.slice(i);
        break;
      }
      let valid = true;
      for (let k = 1; k <= need; k++) {
        const cont = all[i + k];
        if (cont < 0x80 || cont > 0xbf) {
          valid = false;
          break;
        }
      }
      if (!valid) {
        if (state.fatal) throw pirFatalDecodeError("utf-8");
        out += "\uFFFD";
        i += 1;
        continue;
      }
      let cp;
      if (need === 1) cp = ((b0 & 0x1f) << 6) | (all[i + 1] & 0x3f);
      else if (need === 2) {
        cp = ((b0 & 0x0f) << 12) | ((all[i + 1] & 0x3f) << 6) | (all[i + 2] & 0x3f);
      } else {
        cp = ((b0 & 0x07) << 18) | ((all[i + 1] & 0x3f) << 12) |
          ((all[i + 2] & 0x3f) << 6) | (all[i + 3] & 0x3f);
      }
      out += String.fromCodePoint(cp);
      i += need + 1;
    }
    return out;
  }

  class TextEncoder {
    constructor() {
      this[PIR_ENCODER_BRAND] = true;
      this.encoding = "utf-8";
    }
    encode(input) {
      if (this === null || typeof this !== "object" || this[PIR_ENCODER_BRAND] !== true) {
        throw pirInvalidThis("TextEncoder");
      }
      // WebIDL: only `undefined` (the optional default) is empty; `null`
      // stringifies to "null" like Node.
      const text = input === undefined ? "" : String(input);
      // Native core (rquickjs C-ABI, installed by __pirInstallTextCodec):
      // returns a Uint8Array, or undefined when the string cannot cross the
      // host bridge (lone surrogates) — the JS state machine is the fallback.
      const native = TextEncoder.prototype.__pirNativeEncode;
      if (native) {
        try {
          const out = native(text);
          if (out !== undefined) return out;
        } catch (error) { /* fall through to the JS encoder */ }
      }
      return pirUtf8Bytes(text);
    }
    encodeInto(source, target) {
      if (this === null || typeof this !== "object" || this[PIR_ENCODER_BRAND] !== true) {
        throw pirInvalidThis("TextEncoder");
      }
      if (!(target instanceof Uint8Array)) {
        const error = new TypeError(
          'The "dest" argument must be an instance of Uint8Array.' + pirReceivedDetail(target)
        );
        error.code = "ERR_INVALID_ARG_TYPE";
        throw error;
      }
      const text = source === undefined ? "" : String(source);
      const native = TextEncoder.prototype.__pirNativeEncodeInto;
      if (native) {
        try {
          const out = native(text, target);
          if (out !== undefined) return out;
        } catch (error) { /* fall through to the JS encoder */ }
      }
      return pirEncodeIntoUtf8(text, target, 0);
    }
  }

  class TextDecoder {
    constructor(label, options) {
      this.encoding = pirNormalizeLabel(label);
      this[PIR_DECODER_BRAND] = true;
      options = pirValidateOptions(options, "options");
      this.fatal = options !== null && Boolean(options.fatal);
      this.ignoreBOM = options !== null && Boolean(options.ignoreBOM);
      this.__pending = PIR_EMPTY_BYTES;
      // BOM sniffing is per *stream* in Node, not per decoder instance: the
      // flag resets when a non-stream decode (or a decode error) ends the
      // stream. `decoder.decode(); decoder.decode(bom)` therefore strips the
      // BOM again.
      this.__bomChecked = false;
    }

    decode(input, options) {
      if (this === null || typeof this !== "object" || this[PIR_DECODER_BRAND] !== true) {
        throw pirInvalidThis("TextDecoder");
      }
      const settings = pirValidateOptions(options, "options");
      const stream = settings !== null && !!settings.stream;
      // Bytes first: an invalid input throws before any stream state changes
      // (Node leaves a pending tail alone when `parseInput` rejects).
      const bytes = input === undefined ? PIR_EMPTY_BYTES : pirInputBytes(input, stream);
      try {
        // WHATWG BOM sniffing runs over the decoder's byte stream, so it sees
        // the carried prefix plus this chunk: a partial BOM prefix defers the
        // decision to the next call, and an empty chunk keeps it open (Node
        // strips a BOM after `d.decode(); d.decode(bom)`).
        let chunk = bytes;
        if (!this.__bomChecked && !this.ignoreBOM &&
            (this.encoding === "utf-8" || this.encoding === "utf-16le")) {
          const bomLength = this.encoding === "utf-8" ? 3 : 2;
          let head = chunk;
          if (this.__pending.length > 0) {
            head = new Uint8Array(this.__pending.length + chunk.length);
            head.set(this.__pending, 0);
            head.set(chunk, this.__pending.length);
            this.__pending = PIR_EMPTY_BYTES;
          }
          const length = head.length;
          if (length >= bomLength) {
            const bom = this.encoding === "utf-8"
              ? head[0] === 0xef && head[1] === 0xbb && head[2] === 0xbf
              : head[0] === 0xff && head[1] === 0xfe;
            chunk = bom ? head.subarray(bomLength) : head;
            this.__bomChecked = true;
          } else if (length > 0) {
            // A chunk shorter than the BOM can only defer when it is a prefix
            // of one; otherwise the stream is known not to start with a BOM.
            const prefix = this.encoding === "utf-8"
              ? (length === 1 ? head[0] === 0xef : head[0] === 0xef && head[1] === 0xbb)
              : head[0] === 0xff;
            chunk = head;
            if (!(stream && prefix)) this.__bomChecked = true;
          }
        }
        if (this.encoding === "utf-8") return this.__decodeUtf8(chunk, stream);
        if (this.encoding === "windows-1252") return this.__decodeWindows1252(chunk);
        return this.__decodeUtf16le(chunk, stream);
      } catch (error) {
        // A decode error ends the stream: Node drops the held tail and the
        // BOM decision before the exception reaches the caller.
        this.__pending = PIR_EMPTY_BYTES;
        this.__bomChecked = false;
        throw error;
      } finally {
        if (!stream) {
          // End of stream: nothing may stay held, and the next decode() is a
          // fresh stream whose BOM is sniffed again.
          this.__pending = PIR_EMPTY_BYTES;
          this.__bomChecked = false;
        }
      }
    }

    __decodeUtf8(bytes, stream) {
      if (!pirNativeCodec) {
        const state = { pending: this.__pending, fatal: this.fatal };
        let out = pirDecodeUtf8Chunk(bytes, state);
        this.__pending = state.pending;
        if (!stream && this.__pending.length > 0) {
          // End of the stream: an incomplete trailing sequence becomes
          // U+FFFD (or a fatal TypeError).
          if (this.fatal) throw pirFatalDecodeError("utf-8");
          out += "\uFFFD";
          this.__pending = PIR_EMPTY_BYTES;
        }
        return out;
      }
      let all = bytes;
      if (this.__pending.length > 0) {
        all = new Uint8Array(this.__pending.length + bytes.length);
        all.set(this.__pending, 0);
        all.set(bytes, this.__pending.length);
        this.__pending = PIR_EMPTY_BYTES;
      }
      if (this.fatal) {
        // The validating kernel owns fatal mode: it throws on the first
        // invalid sequence and otherwise reports the unconsumed tail.
        let result;
        try {
          result = pirNativeCodec("utf8DecodeChunk", all, !stream, true);
        } catch (error) {
          if (error instanceof TypeError) throw pirFatalDecodeError("utf-8");
          throw error;
        }
        this.__pending = result.consumed < all.length ? all.slice(result.consumed) : PIR_EMPTY_BYTES;
        return result.text;
      }
      // Common path: the plain `utf8Decode` kernel (std UTF-8 validate + copy)
      // is ~6x cheaper than the streaming kernel, and its replacement output
      // is never worse (for the maximal-subpart set it is at least as close
      // to WHATWG as the chunk kernel; both share the same gap on overlong
      // E0/F0 that the buffer workstream owns).
      const hold = stream ? pirUtf8HoldCount(all) : 0;
      const bodyEnd = all.length - hold;
      if (bodyEnd <= 0) {
        this.__pending = all.length > 0 ? all.slice(0) : PIR_EMPTY_BYTES;
        return "";
      }
      const body = hold > 0 ? all.subarray(0, bodyEnd) : all;
      let text;
      try {
        text = pirNativeCodec("utf8Decode", body);
      } catch (error) {
        text = undefined;
      }
      if (typeof text === "string") {
        this.__pending = hold > 0 ? all.slice(bodyEnd) : PIR_EMPTY_BYTES;
        return text;
      }
      const result = pirNativeCodec("utf8DecodeChunk", all, !stream, false);
      this.__pending = result.consumed < all.length ? all.slice(result.consumed) : PIR_EMPTY_BYTES;
      return result.text;
    }

    __decodeWindows1252(bytes) {
      // Single-byte decoder: no streaming tail, no BOM sniffing, never fatal.
      // The native `latin1Decode` kernel maps byte->U+00xx (that is what
      // Buffer#toString("latin1") must keep doing); translate only the
      // 0x80-0x9F band that separates ISO-8859-1 from windows-1252.
      if (pirNativeCodec) {
        try {
          const latin = pirNativeCodec("latin1Decode", bytes);
          if (typeof latin === "string") return pirWin1252FromLatin1(latin);
        } catch (error) { /* fall through to the JS loop */ }
      }
      let latin = "";
      for (let i = 0; i < bytes.length; i++) latin += String.fromCharCode(bytes[i] & 0xff);
      return pirWin1252FromLatin1(latin);
    }

    __decodeUtf16le(bytes, stream) {
      let all = bytes;
      if (this.__pending.length > 0) {
        all = new Uint8Array(this.__pending.length + bytes.length);
        all.set(this.__pending, 0);
        all.set(bytes, this.__pending.length);
        this.__pending = PIR_EMPTY_BYTES;
      }
      if (!this.fatal && pirNativeCodec) {
        try {
          const result = pirNativeCodec("utf16leDecodeChunk", all);
          if (typeof result === "object" && result !== null &&
              typeof result.text === "string" &&
              !pirUtf16leHasLoneSurrogate(result.text)) {
            if (result.consumed < all.length) {
              // Odd trailing byte: carried while streaming, one U+FFFD at
              // end of stream (Node parity).
              if (stream) this.__pending = all.slice(result.consumed);
              else return result.text + "\uFFFD";
            }
            return result.text;
          }
          // Lone surrogates need Node's hold rules: a trailing high
          // surrogate is carried for its low partner, a lone low one is
          // replaced immediately. The JS decoder below applies them.
        } catch (error) { /* fall through to the JS loop */ }
      }
      return this.__decodeUtf16leJs(all, stream);
    }

    __decodeUtf16leJs(all, stream) {
      // WHATWG utf-16 decoder (little-endian): a high surrogate is held for
      // its low partner; at end of stream a lone high surrogate, a lone low
      // surrogate and an odd trailing byte each become one U+FFFD (fatal
      // mode throws instead). The held prefix is carried in `__pending`, so
      // resuming re-derives the same state from the start of the high
      // surrogate (or of the partial unit).
      let text = "";
      const length = all.length;
      let i = 0;
      let leadSurrogate = -1;
      let pendingStart = length;
      while (i + 1 < length) {
        let unit = all[i] | (all[i + 1] << 8);
        if (leadSurrogate !== -1) {
          if (unit >= 0xdc00 && unit <= 0xdfff) {
            text += String.fromCharCode(leadSurrogate, unit);
            leadSurrogate = -1;
            i += 2;
            continue;
          }
          // Not a low surrogate: replace the held high one, then reprocess
          // this unit as a fresh code unit.
          if (this.fatal) throw pirFatalDecodeError("utf-16le");
          text += "\uFFFD";
          leadSurrogate = -1;
        }
        if (unit >= 0xd800 && unit <= 0xdbff) {
          leadSurrogate = unit;
          pendingStart = i;
          i += 2;
          continue;
        }
        if (unit >= 0xdc00 && unit <= 0xdfff) {
          if (this.fatal) throw pirFatalDecodeError("utf-16le");
          text += "\uFFFD";
          i += 2;
          continue;
        }
        text += String.fromCharCode(unit);
        i += 2;
      }
      if (i < length) {
        // Odd trailing byte: carried with a pending high surrogate while
        // streaming; at end of stream it is one error (which also covers the
        // pending high surrogate, matching Node).
        if (stream) {
          if (leadSurrogate !== -1 && pendingStart < i) this.__pending = all.slice(pendingStart);
          else this.__pending = all.slice(i);
          return text;
        }
        if (this.fatal) throw pirFatalDecodeError("utf-16le");
        return text + "\uFFFD";
      }
      if (leadSurrogate !== -1) {
        if (stream) {
          this.__pending = all.slice(pendingStart);
          return text;
        }
        if (this.fatal) throw pirFatalDecodeError("utf-16le");
        text += "\uFFFD";
      }
      this.__pending = PIR_EMPTY_BYTES;
      return text;
    }
  }

  globalThis.TextEncoder ?? (globalThis.TextEncoder = TextEncoder);
  globalThis.TextDecoder ?? (globalThis.TextDecoder = TextDecoder);

  // Host fast path: native encode/encodeInto cores onto the prototype (see
  // buffer_ops.rs). The class methods stay the entry points so the decoder's
  // per-instance streaming state remains here; the native hooks are advisory
  // (undefined result ⇒ JS state machine). A failed install changes nothing.
  if (typeof globalThis.__pirInstallTextCodec === "function") {
    try {
      globalThis.__pirInstallTextCodec(TextEncoder.prototype, TextDecoder.prototype);
    } catch (error) {
      /* native install failed: the JS implementations above stay active */
    }
    delete globalThis.__pirInstallTextCodec;
  }
})();
