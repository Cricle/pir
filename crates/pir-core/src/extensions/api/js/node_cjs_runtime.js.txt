// ---------------------------------------------------------------------------
// require(esm) body preparation
//
// `esm_to_cjs` (host transform) emits `require("…")` for the module's
// imports and `exports[…]` for its exports. Evaluated with a CJS wrapper's
// ordinary `require`/`exports` parameters, a module that DECLARES those
// names (`const require = createRequire(import.meta.url)` — Node's own
// `test/common/index.mjs`) is a syntax error ("invalid redefinition of
// parameter name"), and `import.meta` is not valid script code at all. Node
// compiles a required ES module with ES module scope, so:
//   - the generated identifiers are renamed to wrapper-private ones,
//   - `import.meta` is replaced with a populated module object,
//   - `require`/`module`/`exports` are NOT provided (Node exposes none of
//     them in ES module scope; a module's own declarations win).
// The CJS wrapper above is unchanged — `const require` in a CJS file is a
// SyntaxError in Node too.
// ---------------------------------------------------------------------------

globalThis.__pirEsmRequirableBody ?? (function () {
const __pirIdentChar = /[A-Za-z0-9_$]/;

/// Native-separator spelling of a module path for `import.meta.filename` /
/// `import.meta.dirname` (Node hands out platform paths, not URLs).
function __pirEsmNativePath(path) {
  const text = String(path);
  const platform =
    typeof globalThis.__pirPlatform === "string"
      ? globalThis.__pirPlatform
      : typeof process === "object" && process !== null
        ? String(process.platform || "")
        : "";
  return platform === "win32" ? text.replace(/\//g, "\\") : text;
}

/// `file:` URL of a module path, percent-encoded per path segment (UTF-8),
/// with the Windows drive colon left literal (`file:///C:/x`).
function __pirEsmFileUrl(path) {
  let text = String(path).replace(/\\/g, "/");
  if (text.charAt(0) !== "/") text = "/" + text;
  try {
    return (
      "file://" +
      text
        .split("/")
        .map((segment) => encodeURIComponent(segment).replace(/%3A/gi, ":"))
        .join("/")
    );
  } catch (_error) {
    return "file://" + text;
  }
}

/// Copy of `source` with the contents of strings, template literals and
/// comments blanked out (positions and newlines preserved), so a token search
/// never matches inside a literal. `${ … }` template expressions stay code.
/// Regex literals are treated as division (a `/` inside one can be seen as a
/// comment start); generated import/export code never carries one.
function __pirMaskCode(source) {
  const out = source.split("");
  const templateStack = [];
  let mode = "code";
  let quote = "";
  let depth = 0;
  let i = 0;
  const n = source.length;
  const hide = (index) => {
    const c = source[index];
    if (c !== "\n" && c !== "\r") out[index] = " ";
  };
  while (i < n) {
    const c = source[i];
    if (mode === "line") {
      if (c === "\n" || c === "\r") mode = "code";
      else out[i] = " ";
      i += 1;
      continue;
    }
    if (mode === "block") {
      if (c === "*" && source[i + 1] === "/") {
        out[i] = " ";
        out[i + 1] = " ";
        i += 2;
        mode = "code";
      } else {
        hide(i);
        i += 1;
      }
      continue;
    }
    if (mode === "string") {
      if (c === "\\") {
        out[i] = " ";
        if (i + 1 < n) out[i + 1] = " ";
        i += 2;
        continue;
      }
      if (c === quote) {
        mode = "code";
        i += 1;
        continue;
      }
      hide(i);
      i += 1;
      continue;
    }
    if (mode === "template") {
      if (c === "\\") {
        out[i] = " ";
        if (i + 1 < n) out[i + 1] = " ";
        i += 2;
        continue;
      }
      if (c === "`") {
        mode = "code";
        i += 1;
        continue;
      }
      if (c === "$" && source[i + 1] === "{") {
        out[i] = " ";
        out[i + 1] = " ";
        templateStack.push(depth);
        depth = 0;
        mode = "code";
        i += 2;
        continue;
      }
      hide(i);
      i += 1;
      continue;
    }
    // code
    if (c === "/" && source[i + 1] === "/") {
      mode = "line";
      i += 2;
      continue;
    }
    if (c === "/" && source[i + 1] === "*") {
      mode = "block";
      i += 2;
      continue;
    }
    if (c === "'" || c === '"') {
      mode = "string";
      quote = c;
      i += 1;
      continue;
    }
    if (c === "`") {
      mode = "template";
      i += 1;
      continue;
    }
    if (c === "{") {
      depth += 1;
      i += 1;
      continue;
    }
    if (c === "}") {
      if (depth === 0 && templateStack.length > 0) {
        depth = templateStack.pop();
        mode = "template";
      } else if (depth > 0) {
        depth -= 1;
      }
      i += 1;
      continue;
    }
    i += 1;
  }
  return out.join("");
}

/// Replace whole-identifier occurrences of `token` that are outside string /
/// comment literals.
function __pirReplaceCodeToken(source, token, replacement) {
  const masked = __pirMaskCode(source);
  let out = "";
  let last = 0;
  let replaced = false;
  let index = masked.indexOf(token);
  while (index !== -1) {
    const before = index === 0 ? "" : masked.charAt(index - 1);
    const after = masked.charAt(index + token.length);
    if (!__pirIdentChar.test(before) && !__pirIdentChar.test(after)) {
      out += source.slice(last, index) + replacement;
      last = index + token.length;
      replaced = true;
    }
    index = masked.indexOf(token, index + 1);
  }
  return replaced ? out + source.slice(last) : source;
}

/// Make an `esm_to_cjs` body evaluable with ES module scope: populate
/// `import.meta` and rename the transform's generated `require`/`exports`
/// references to the wrapper-private names, leaving user declarations (and
/// user uses of `require`) alone.
function __pirEsmRequirableBody(source, key, dir) {
  const meta =
    "{ url: " + JSON.stringify(__pirEsmFileUrl(key)) +
    ", filename: " + JSON.stringify(__pirEsmNativePath(key)) +
    ", dirname: " + JSON.stringify(__pirEsmNativePath(dir)) +
    " }";
  let body = __pirReplaceCodeToken(source, "import.meta", "__pirEsmImportMeta");
  body = "const __pirEsmImportMeta = " + meta + ";\n" + body;
  // Generated import statements (the transform's exact shapes).
  body = body.replace(
    /__pirInteropNamespace\(require\(/g,
    "__pirInteropNamespace(__pirCjsRequire("
  );
  body = body.replace(/(^|\n)require\(/g, "$1__pirCjsRequire(");
  // Generated export assignments.
  body = body.replace(/(^|\n)exports\./g, "$1__pirCjsExports.");
  body = body.replace(/(^|\n)exports\[/g, "$1__pirCjsExports[");
  body = body.replace(/ exports\[/g, " __pirCjsExports[");
  return body;
}
