// node:_stream_duplex / _stream_readable / _stream_writable — private
// stream internals. `from` aliases the stream shim's factory; the web
// interop pair (`fromWeb`/`toWeb`) cannot be served by the minimal stream
// pad and throws the unified error; the internal state classes are inert
// skeletons; `_fromList`/`wrap` are documented-minimal helpers.
import { Duplex, PassThrough, Readable, Transform, Writable } from "@node/stream";

function streamInternalUnsupported(api) {
  const error = new Error("node:_stream_" + api + " is not supported by the pir extension runtime (minimal stream pad)");
  error.code = "ERR_NOT_SUPPORTED";
  return error;
}

// Internal state holders: inert skeletons with the real field names.
class ReadableState {
  constructor(stream, options) {
    this.stream = stream;
    this.highWaterMark = options && typeof options.highWaterMark === "number" ? options.highWaterMark : 16384;
    this.buffer = [];
    this.bufferIndex = 0;
    this.length = 0;
    this.ended = false;
    this.endedEmitted = false;
    this.reading = false;
    this.flowing = null;
  }
}

class WritableState {
  constructor(stream, options) {
    this.stream = stream;
    this.highWaterMark = options && typeof options.highWaterMark === "number" ? options.highWaterMark : 16384;
    this.buffered = [];
    this.bufferedIndex = 0;
    this.length = 0;
    this.ended = false;
    this.finished = false;
    this.writing = false;
  }
}

// Minimal `_fromList`: returns the first buffered chunk (the real one drains
// n bytes across buffers; the minimal pad buffers whole chunks).
function _fromList(_n, state) {
  if (state && Array.isArray(state.buffer)) {
    const index = typeof state.bufferIndex === "number" ? state.bufferIndex : 0;
    if (index < state.buffer.length) {
      const chunk = state.buffer[index];
      state.buffer[index] = null;
      state.bufferIndex = index + 1;
      return chunk;
    }
  }
  return null;
}

// `Readable.from(iterable)` — minimal real: drains the (sync or async)
// iterator through the pad's push surface on microtasks.
function from(iterable) {
  const readable = new Readable();
  // This pad is push-driven; a direct Readable's default _read reports
  // ERR_METHOD_NOT_IMPLEMENTED, so supply the no-op like node's from().
  readable._read = function () {};
  if (iterable === undefined || iterable === null || typeof iterable[Symbol.asyncIterator] !== "function" && typeof iterable[Symbol.iterator] !== "function") {
    globalThis.__pirMicrotask(() => readable.destroy(new TypeError("The list argument must be an iterable")));
    return readable;
  }
  const iterator = typeof iterable[Symbol.asyncIterator] === "function"
    ? iterable[Symbol.asyncIterator]()
    : iterable[Symbol.iterator]();
  const step = () => {
    let result;
    try {
      result = iterator.next();
    } catch (error) {
      readable.destroy(error);
      return;
    }
    Promise.resolve(result).then(({ done, value }) => {
      if (done) {
        readable.push(null);
        return;
      }
      readable.push(value);
      globalThis.__pirMicrotask(step);
    }, (error) => {
      readable.destroy(error);
    });
  };
  globalThis.__pirMicrotask(step);
  return readable;
}

// `wrap` adapts an old-style (event-only) stream; the minimal pad returns
// the input as-is (it already exposes the event surface).
function wrap(stream) {
  return stream;
}

// The Web-streams interop pair landed on the stream pad itself
// (`Duplex.fromWeb`/`Duplex.toWeb`, node's own statics); the private specifiers
// delegate instead of throwing now that the pad can serve them.
function fromWeb(pair, options) {
  return Duplex.fromWeb(pair, options);
}

function toWeb(stream, options) {
  return Duplex.toWeb(stream, options);
}

// Aggregated namespace: each _stream_* specifier maps to the same source and
// keeps the union of the real private surfaces (documented superset).
const streamInternals = {
  Duplex: Duplex,
  Readable: Readable,
  Writable: Writable,
  from: from,
  fromWeb: fromWeb,
  toWeb: toWeb,
  ReadableState: ReadableState,
  WritableState: WritableState,
  _fromList: _fromList,
  wrap: wrap,
};

// --- node's constructor statics + a `require("_stream_*")` identity ---------
//
// Node's private modules export their constructor directly
// (`module.exports = Readable`), and each constructor carries the real
// internals as own properties. The loader serves these specifiers from this
// one file, so the named module objects cannot be the constructors themselves;
// instead, preseed the builtin require cache with the five constructor
// identities before user code runs (all shims are imported at boot). This makes
// `require("stream").Readable === require("_stream_readable")` hold exactly as
// test-stream-aliases-legacy requires.
function pirInstallStreamStatics() {
  if (Readable.ReadableState === undefined) Readable.ReadableState = ReadableState;
  if (Readable._fromList === undefined) Readable._fromList = _fromList;
  if (Readable.fromWeb === undefined) Readable.fromWeb = fromWeb;
  if (Readable.toWeb === undefined) Readable.toWeb = toWeb;
  if (Readable.wrap === undefined) Readable.wrap = wrap;
  if (Writable.WritableState === undefined) Writable.WritableState = WritableState;
  if (Writable.fromWeb === undefined) Writable.fromWeb = fromWeb;
  if (Writable.toWeb === undefined) Writable.toWeb = toWeb;
  if (Duplex.from === undefined) Duplex.from = Readable.from;
  if (Duplex.fromWeb === undefined) Duplex.fromWeb = fromWeb;
  if (Duplex.toWeb === undefined) Duplex.toWeb = toWeb;

  const cache = globalThis.__pirBuiltinRequireCache;
  if (cache === undefined || cache === null || typeof cache.has !== "function") return;
  const identities = [
    ["_stream_readable", Readable],
    ["_stream_writable", Writable],
    ["_stream_duplex", Duplex],
    ["_stream_transform", Transform],
    ["_stream_passthrough", PassThrough],
  ];
  for (let i = 0; i < identities.length; i++) {
    const name = identities[i][0];
    if (!cache.has(name)) cache.set(name, identities[i][1]);
  }
}

pirInstallStreamStatics();

export default streamInternals;
export { Duplex, Readable, Writable, from, fromWeb, toWeb, ReadableState, WritableState, _fromList, wrap };
