// ---------------------------------------------------------------------------
// Node-shaped argument errors
// ---------------------------------------------------------------------------
// Same rendering the zlib shim builds (each shim is an independent module, so
// the two keep their own copy): the `Received …` tail is Node's
// `determineSpecificType`, the noun before it is `property` for a dotted name
// and `argument` otherwise.
function pirTypeTail(value) {
  if (value === null) return "null";
  const type = typeof value;
  if (type === "undefined") return "undefined";
  if (type === "string") return "type string ('" + value + "')";
  if (type === "number") return "type number (" + String(value) + ")";
  if (type === "boolean") return "type boolean (" + String(value) + ")";
  if (type === "bigint") return "type bigint (" + String(value) + "n)";
  if (type === "symbol") return "type symbol (" + String(value) + ")";
  if (type === "function") return "function " + (value.name === undefined ? "" : value.name);
  if (Buffer.isBuffer(value)) return "an instance of Buffer";
  if (value instanceof ArrayBuffer) return "an instance of ArrayBuffer";
  if (ArrayBuffer.isView(value)) {
    return "an instance of " + (value.constructor && value.constructor.name ? value.constructor.name : "TypedArray");
  }
  const ctor = value.constructor;
  return "an instance of " + (ctor && ctor.name ? ctor.name : "Object");
}

function pirInvalidArgType(name, expected, value) {
  const kind = name.indexOf(".") === -1 ? "argument" : "property";
  const error = new TypeError(
    "The \"" + name + "\" " + kind + " must be " + expected +
    ". Received " + pirTypeTail(value),
  );
  error.code = "ERR_INVALID_ARG_TYPE";
  return error;
}

function pirUnknownEncoding(encoding) {
  const error = new TypeError("Unknown encoding: " + String(encoding));
  error.code = "ERR_UNKNOWN_ENCODING";
  return error;
}

const PIR_INPUT_EXPECTED = "of type string or an instance of Buffer, TypedArray, or DataView";

// Node's `addNumericalSeparator`: groups the integer part in threes.
function pirAddNumericalSeparator(value) {
  let result = "";
  let index = value.length;
  const start = value[0] === "-" ? 1 : 0;
  for (; index >= start + 4; index -= 3) {
    result = "_" + value.slice(index - 3, index) + result;
  }
  return value.slice(0, index) + result;
}

// ERR_OUT_OF_RANGE's Received rendering: integers above 2^32 are grouped,
// everything else uses inspect's number spelling.
function pirOutOfRange(name, range, value) {
  let received;
  if (Number.isInteger(value) && Math.abs(value) > 4294967296) {
    received = pirAddNumericalSeparator(String(value));
  } else if (typeof value === "bigint") {
    received = String(value);
    if (value > 4294967296n || value < -4294967296n) received = pirAddNumericalSeparator(received);
    received += "n";
  } else if (typeof value === "number" && Object.is(value, -0)) {
    received = "-0";
  } else {
    received = String(value);
  }
  const error = new RangeError("The value of \"" + name + "\" is out of range. It must be " + range + ". Received " + received);
  error.code = "ERR_OUT_OF_RANGE";
  return error;
}

function pirInvalidArgValue(name, value, reason) {
  const type = name.indexOf(".") === -1 ? "argument" : "property";
  const inspected = typeof value === "string" ? "'" + value + "'" : String(value);
  const error = new TypeError("The " + type + " '" + name + "' " + (reason === undefined ? "is invalid" : reason) + ". Received " + inspected);
  error.code = "ERR_INVALID_ARG_VALUE";
  return error;
}

// internal/validators validateNumber: only a non-number or an explicit
// min/max violation throws (NaN included only when a bound is given).
function pirValidateNumber(value, name, min, max) {
  if (typeof value !== "number") throw pirInvalidArgType(name, "of type number", value);
  if ((min !== undefined && value < min) || (max !== undefined && value > max) ||
      ((min !== undefined || max !== undefined) && Number.isNaN(value))) {
    const range = (min !== undefined ? ">= " + min : "") +
      (min !== undefined && max !== undefined ? " && " : "") +
      (max !== undefined ? "<= " + max : "");
    throw pirOutOfRange(name, range, value);
  }
}

function pirValidateFunction(value, name) {
  if (typeof value !== "function") throw pirInvalidArgType(name, "of type function", value);
}

function pirValidateObject(value, name) {
  if (value === null || Array.isArray(value) || typeof value !== "object") {
    throw pirInvalidArgType(name, "of type object", value);
  }
}

// Hash.update()/Sign.update() encoding validation (internal/validators
// validateEncoding): only an odd-length hex payload is rejected up front; the
// decoder itself accepts a bad digit run and stops there, exactly like
// Buffer.from.
function pirValidateEncoding(data, encoding) {
  if (typeof encoding === "string" && encoding.length === 3 &&
      encoding.toLowerCase() === "hex" && data.length % 2 !== 0) {
    throw pirInvalidArgValue("encoding", encoding, "is invalid for data of length " + data.length);
  }
}

// Node's Hash.update decodes strings through the C++ StringBytes decoder,
// which falls back to UTF-8 for an unknown/non-string encoding (unlike
// Buffer.from, which throws ERR_UNKNOWN_ENCODING).
function pirDecodeString(data, encoding) {
  if (typeof encoding === "string" && Buffer.isEncoding(encoding)) {
    return Buffer.from(data, encoding);
  }
  return Buffer.from(data, "utf8");
}

// One conversion for every hash/sign input: strings decode per the encoding
// rules above, any ArrayBufferView (TypedArray or DataView, Buffer included)
// passes through as a zero-copy view of its own byte range, and everything
// else (ArrayBuffer, SharedArrayBuffer, numbers, null, ...) is Node's
// ERR_INVALID_ARG_TYPE.
/// A zero-copy Buffer over a typed-array view using the backing store's real
/// extent. Node's C++ reads the internal typed-array length, not the JS
/// `byteLength` getter (test-crypto-hkdf defines a lying `byteLength` getter
/// on a 2048-byte info array); a view whose offset/length falls outside its
/// buffer is Node's ERR_OUT_OF_RANGE.
function pirViewBytes(value) {
  const elementSize = value.BYTES_PER_ELEMENT || 1;
  const length = value.length;
  const offset = value.byteOffset;
  const total = typeof length === "number" ? length * elementSize : value.byteLength;
  const capacity = value.buffer === undefined || value.buffer === null ? 0 : value.buffer.byteLength;
  if (!(offset >= 0) || !(total >= 0) || offset + total > capacity) {
    const error = new RangeError("input buffer is out of bounds");
    error.code = "ERR_OUT_OF_RANGE";
    throw error;
  }
  return Buffer.from(value.buffer, offset, total);
}

function pirHashInputBytes(data, encoding) {
  if (typeof data === "string") {
    pirValidateEncoding(data, encoding);
    return pirDecodeString(data, encoding);
  }
  if (ArrayBuffer.isView(data)) {
    return pirViewBytes(data);
  }
  throw pirInvalidArgType("data", PIR_INPUT_EXPECTED, data);
}

// A view of any ArrayBuffer family member (ArrayBuffer or SharedArrayBuffer),
// or Buffer.from-able bytes, for key material.
function pirArrayBufferBytes(value) {
  if (ArrayBuffer.isView(value)) {
    return pirViewBytes(value);
  }
  if (value instanceof ArrayBuffer ||
      (typeof SharedArrayBuffer === "function" && value instanceof SharedArrayBuffer)) {
    return Buffer.from(new Uint8Array(value));
  }
  return undefined;
}

// Sign/Verify buffer their chunks and concatenate later, so the view must be
// copied at update() time (Node's C++ side ingests it immediately).
function pushData(chunks, data, encoding) {
  const bytes = pirHashInputBytes(data, encoding);
  chunks.push(typeof data === "string" ? bytes : Buffer.from(bytes));
}

// Key material coercion shared by the KDF entry points. Node's message lists
// `ArrayBuffer` (the older helper omitted it); each caller passes its exact
// expected-noun phrase because `ikm` also accepts a SecretKeyObject.
const PIR_KEY_EXPECTED = "of type string or an instance of ArrayBuffer, Buffer, TypedArray, or DataView";
const PIR_IKM_EXPECTED = "of type string or an instance of SecretKeyObject, ArrayBuffer, TypedArray, DataView, or Buffer";
const PIR_HKDF_BYTES_EXPECTED = "of type string or an instance of ArrayBuffer, TypedArray, DataView, or Buffer";

function toKeyBytes(value, name, expected) {
  if (typeof value === "string") return Buffer.from(value, "utf8");
  const bytes = pirArrayBufferBytes(value);
  if (bytes !== undefined) return bytes;
  throw pirInvalidArgType(name, expected === undefined ? PIR_KEY_EXPECTED : expected, value);
}

// The wider key surface HMAC/cipher accept: string / bytes / KeyObject (with
// the secret material) / CryptoKey, with Node's ERR_INVALID_ARG_TYPE otherwise.
function pirKeyBytes(value, name) {
  if (KeyObject !== undefined && value instanceof KeyObject) {
    if (value._secretB64 === null) {
      throw pirInvalidArgType(name, "of type string or an instance of ArrayBuffer, Buffer, TypedArray, DataView, KeyObject, or CryptoKey", value);
    }
    return Buffer.from(value._secretB64, "base64");
  }
  if (CryptoKey !== undefined && value instanceof CryptoKey) {
    return Buffer.from(value._raw);
  }
  if (typeof value === "string") return Buffer.from(value, "utf8");
  const bytes = pirArrayBufferBytes(value);
  if (bytes !== undefined) return bytes;
  throw pirInvalidArgType(
    name,
    "of type string or an instance of ArrayBuffer, Buffer, TypedArray, DataView, KeyObject, or CryptoKey",
    value,
  );
}

// Node digest-name normalization ("SHA-256" -> "sha256") shared by every
// hash/hmac/hkdf/kdf entry point; `dss1` is OpenSSL's historical SHA-1 alias
// and Node resolves it to SHA-1 (test-crypto-hmac asserts the lowercase and
// uppercase spellings give the SHA-1 HMAC).
const PIR_DIGEST_ALIASES = { dss1: "sha1" };

function normalizeDigest(algo) {
  let name = String(algo === undefined || algo === null ? "" : algo).toLowerCase();
  // Skip the regex on the common unhyphenated spelling.
  if (name.indexOf("-") !== -1) name = name.replace(/-/g, "");
  return PIR_DIGEST_ALIASES[name] === undefined ? name : PIR_DIGEST_ALIASES[name];
}

function pirRotr32(value, bits) {
  return (value >>> bits) | (value << (32 - bits));
}

function pirSha2_256Family(bytes, is224) {
  const h = is224
    ? [0xc1059ed8, 0x367cd507, 0x3070dd17, 0xf70e5939, 0xffc00b31, 0x68581511, 0x64f98fa7, 0xbefa4fa4]
    : [0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a, 0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19];
  const length = bytes.length;
  const padded = new Uint8Array(Math.ceil((length + 9) / 64) * 64);
  padded.set(bytes);
  padded[length] = 0x80;
  const bits = length * 8;
  // 64-bit big-endian bit length in the last 8 bytes; JS string/buffer sizes
  // stay below 2^48 bits, so the top two bytes are zero.
  const tail = padded.length - 8;
  padded[tail + 2] = Math.floor(bits / 1099511627776) & 0xff;
  padded[tail + 3] = Math.floor(bits / 4294967296) & 0xff;
  padded[tail + 4] = Math.floor(bits / 16777216) & 0xff;
  padded[tail + 5] = Math.floor(bits / 65536) & 0xff;
  padded[tail + 6] = Math.floor(bits / 256) & 0xff;
  padded[tail + 7] = bits & 0xff;
  const w = new Uint32Array(64);
  for (let offset = 0; offset < padded.length; offset += 64) {
    for (let i = 0; i < 16; i++) {
      const p = offset + i * 4;
      w[i] = (padded[p] << 24) | (padded[p + 1] << 16) | (padded[p + 2] << 8) | padded[p + 3];
    }
    for (let i = 16; i < 64; i++) {
      const x = w[i - 15];
      const y = w[i - 2];
      const s0 = pirRotr32(x, 7) ^ pirRotr32(x, 18) ^ (x >>> 3);
      const s1 = pirRotr32(y, 17) ^ pirRotr32(y, 19) ^ (y >>> 10);
      w[i] = (w[i - 16] + s0 + w[i - 7] + s1) | 0;
    }
    let a = h[0], b = h[1], c = h[2], d = h[3];
    let e = h[4], f = h[5], g = h[6], hh = h[7];
    for (let i = 0; i < 64; i++) {
      const s1 = pirRotr32(e, 6) ^ pirRotr32(e, 11) ^ pirRotr32(e, 25);
      const ch = (e & f) ^ (~e & g);
      const t1 = (hh + s1 + ch + PIR_SHA256_K[i] + w[i]) | 0;
      const s0 = pirRotr32(a, 2) ^ pirRotr32(a, 13) ^ pirRotr32(a, 22);
      const maj = (a & b) ^ (a & c) ^ (b & c);
      const t2 = (s0 + maj) | 0;
      hh = g; g = f; f = e; e = (d + t1) | 0;
      d = c; c = b; b = a; a = (t1 + t2) | 0;
    }
    h[0] = (h[0] + a) | 0; h[1] = (h[1] + b) | 0; h[2] = (h[2] + c) | 0; h[3] = (h[3] + d) | 0;
    h[4] = (h[4] + e) | 0; h[5] = (h[5] + f) | 0; h[6] = (h[6] + g) | 0; h[7] = (h[7] + hh) | 0;
  }
  const words = is224 ? 7 : 8;
  const out = Buffer.alloc(words * 4);
  for (let i = 0; i < words; i++) {
    out[i * 4] = (h[i] >>> 24) & 0xff;
    out[i * 4 + 1] = (h[i] >>> 16) & 0xff;
    out[i * 4 + 2] = (h[i] >>> 8) & 0xff;
    out[i * 4 + 3] = h[i] & 0xff;
  }
  return out;
}

const PIR_MASK64 = (1n << 64n) - 1n;

function pirRotr64(value, bits) {
  return ((value >> bits) | (value << (64n - bits))) & PIR_MASK64;
}

function pirSha2_512Family(bytes, is384) {
  const h = is384
    ? [0xcbbb9d5dc1059ed8n, 0x629a292a367cd507n, 0x9159015a3070dd17n, 0x152fecd8f70e5939n,
      0x67332667ffc00b31n, 0x8eb44a8768581511n, 0xdb0c2e0d64f98fa7n, 0x47b5481dbefa4fa4n]
    : [0x6a09e667f3bcc908n, 0xbb67ae8584caa73bn, 0x3c6ef372fe94f82bn, 0xa54ff53a5f1d36f1n,
      0x510e527fade682d1n, 0x9b05688c2b3e6c1fn, 0x1f83d9abfb41bd6bn, 0x5be0cd19137e2179n];
  const length = bytes.length;
  const padded = new Uint8Array(Math.ceil((length + 17) / 128) * 128);
  padded.set(bytes);
  padded[length] = 0x80;
  // 128-bit big-endian bit length; JS inputs stay far below 2^64 bits, so the
  // high half is zero and only the low 64 bits are written.
  const bits = BigInt(length) * 8n;
  const tail = padded.length - 16;
  for (let i = 0; i < 8; i++) {
    padded[tail + 8 + i] = Number((bits >> BigInt(8 * (7 - i))) & 0xffn);
  }
  const w = new Array(80).fill(0n);
  for (let offset = 0; offset < padded.length; offset += 128) {
    for (let i = 0; i < 16; i++) {
      let word = 0n;
      for (let b = 0; b < 8; b++) word = (word << 8n) | BigInt(padded[offset + i * 8 + b]);
      w[i] = word;
    }
    for (let i = 16; i < 80; i++) {
      const x = w[i - 15];
      const y = w[i - 2];
      const s0 = pirRotr64(x, 1n) ^ pirRotr64(x, 8n) ^ (x >> 7n);
      const s1 = pirRotr64(y, 19n) ^ pirRotr64(y, 61n) ^ (y >> 6n);
      w[i] = (w[i - 16] + s0 + w[i - 7] + s1) & PIR_MASK64;
    }
    let a = h[0], b = h[1], c = h[2], d = h[3];
    let e = h[4], f = h[5], g = h[6], hh = h[7];
    for (let i = 0; i < 80; i++) {
      const s1 = pirRotr64(e, 14n) ^ pirRotr64(e, 18n) ^ pirRotr64(e, 41n);
      const ch = (e & f) ^ (~e & g & PIR_MASK64);
      const t1 = (hh + s1 + ch + PIR_SHA512_K[i] + w[i]) & PIR_MASK64;
      const s0 = pirRotr64(a, 28n) ^ pirRotr64(a, 34n) ^ pirRotr64(a, 39n);
      const maj = (a & b) ^ (a & c) ^ (b & c);
      const t2 = (s0 + maj) & PIR_MASK64;
      hh = g; g = f; f = e; e = (d + t1) & PIR_MASK64;
      d = c; c = b; b = a; a = (t1 + t2) & PIR_MASK64;
    }
    h[0] = (h[0] + a) & PIR_MASK64; h[1] = (h[1] + b) & PIR_MASK64;
    h[2] = (h[2] + c) & PIR_MASK64; h[3] = (h[3] + d) & PIR_MASK64;
    h[4] = (h[4] + e) & PIR_MASK64; h[5] = (h[5] + f) & PIR_MASK64;
    h[6] = (h[6] + g) & PIR_MASK64; h[7] = (h[7] + hh) & PIR_MASK64;
  }
  const words = is384 ? 6 : 8;
  const out = Buffer.alloc(words * 8);
  for (let i = 0; i < words; i++) {
    for (let b = 0; b < 8; b++) {
      out[i * 8 + b] = Number((h[i] >> BigInt(8 * (7 - b))) & 0xffn);
    }
  }
  return out;
}

function pirKeccakF(state) {
  const bc = new Array(5).fill(0n);
  for (let round = 0; round < 24; round++) {
    for (let i = 0; i < 5; i++) {
      bc[i] = state[i] ^ state[i + 5] ^ state[i + 10] ^ state[i + 15] ^ state[i + 20];
    }
    for (let i = 0; i < 5; i++) {
      const t = bc[(i + 4) % 5] ^ (((bc[(i + 1) % 5] << 1n) | (bc[(i + 1) % 5] >> 63n)) & PIR_MASK64);
      for (let j = 0; j < 25; j += 5) state[j + i] ^= t;
    }
    let t = state[1];
    for (let i = 0; i < 24; i++) {
      const j = PIR_KECCAK_PILN[i];
      const saved = state[j];
      state[j] = ((t << BigInt(PIR_KECCAK_ROTC[i])) | (t >> BigInt(64 - PIR_KECCAK_ROTC[i]))) & PIR_MASK64;
      t = saved;
    }
    for (let j = 0; j < 25; j += 5) {
      for (let i = 0; i < 5; i++) bc[i] = state[j + i];
      for (let i = 0; i < 5; i++) {
        state[j + i] = bc[i] ^ (~bc[(i + 1) % 5] & bc[(i + 2) % 5] & PIR_MASK64);
      }
    }
    state[0] ^= PIR_KECCAK_RC[round];
  }
}

function pirSha3(bytes, rate, outLength) {
  const state = new Array(25).fill(0n);
  const padded = new Uint8Array(Math.ceil((bytes.length + 1) / rate) * rate);
  padded.set(bytes);
  padded[bytes.length] ^= 0x06; // SHA-3 domain separation (pad10*1, bits 0b01)
  padded[padded.length - 1] ^= 0x80;
  for (let offset = 0; offset < padded.length; offset += rate) {
    for (let i = 0; i < rate / 8; i++) {
      let word = 0n;
      for (let b = 0; b < 8; b++) word |= BigInt(padded[offset + i * 8 + b]) << BigInt(8 * b);
      state[i] ^= word;
    }
    pirKeccakF(state);
  }
  const out = Buffer.alloc(outLength);
  let written = 0;
  while (written < outLength) {
    const take = Math.min(rate, outLength - written);
    for (let i = 0; i < take; i++) {
      out[written + i] = Number((state[(i / 8) | 0] >> BigInt(8 * (i % 8))) & 0xffn);
    }
    written += take;
    if (written < outLength) pirKeccakF(state);
  }
  return out;
}

// algo -> {hash, blockSize}; `hash` takes/returns byte arrays. `blockSize` is
// the HMAC pad length (RFC 2104 uses the digest's internal block size).
const PIR_JS_DIGESTS = {
  sha224: { hash: (bytes) => pirSha2_256Family(bytes, true), blockSize: 64 },
  sha384: { hash: (bytes) => pirSha2_512Family(bytes, true), blockSize: 128 },
  sha3512: { hash: (bytes) => pirSha3(bytes, 72, 64), blockSize: 72 },
  sha3256: { hash: (bytes) => pirSha3(bytes, 136, 32), blockSize: 136 },
  sha3384: { hash: (bytes) => pirSha3(bytes, 104, 48), blockSize: 104 },
};

// Digests the host op serves directly (RustCrypto: SHA-2, SHA-3, SHA-1,
// MD5). The JS implementations below stay as the fallback tier, but the host
// path is what runs: the SHA-3 family used to take seconds per MiB as BigInt
// Keccak and is now ~1-3 ms/MiB in Rust.
const PIR_HOST_DIGESTS = {
  md5: true,
  sha1: true,
  sha224: true,
  sha256: true,
  sha384: true,
  sha512: true,
  sha3224: true,
  sha3256: true,
  sha3384: true,
  sha3512: true,
  shake128: true,
  shake256: true,
};

// HMAC rejects the SHAKE XOF pair (Node: OpenSSL "xof digests not allowed");
// everything else the digest table serves works as an HMAC PRF.
const PIR_HOST_HMACS = {
  md5: true,
  sha1: true,
  sha224: true,
  sha256: true,
  sha384: true,
  sha512: true,
  sha3224: true,
  sha3256: true,
  sha3384: true,
  sha3512: true,
};

// Fixed digest output sizes (bytes) and the SHAKE default lengths. Node's
// `options.outputLength` is only free-form for the XOF pair; for a fixed
// digest it must equal the digest size (and OpenSSL rejects any other value).
const PIR_DIGEST_SIZES = {
  md5: 16,
  sha1: 20,
  sha224: 28,
  sha256: 32,
  sha384: 48,
  sha512: 64,
  sha3224: 28,
  sha3256: 32,
  sha3384: 48,
  sha3512: 64,
};
const PIR_SHAKE_DEFAULTS = { shake128: 16, shake256: 32 };

function pirDigestUnsupported() {
  // Node's `Hash` constructor reports an unknown/unsupported digest as a
  // plain Error with no `code` (unlike `createHmac`, which raises
  // ERR_CRYPTO_INVALID_DIGEST).
  return new Error("Digest method not supported");
}

function pirNotXof(algo, length) {
  const error = new Error(
    "error:030000B2:digital envelope routines::not XOF or invalid length");
  error.code = "ERR_OSSL_EVP_NOT_XOF_OR_INVALID_LENGTH";
  return error;
}

/// Node's `validateInteger(options.outputLength, …)` window (0..2^32-1). The
/// non-integer spelling covers NaN/Infinity/fractions exactly like
/// internal/validators.
function pirHashOutputRange(value) {
  if (!Number.isInteger(value)) {
    return pirOutOfRange("options.outputLength", "an integer", value);
  }
  return pirOutOfRange("options.outputLength", ">= 0 && <= 4294967295", value);
}

/// Read `options.outputLength` for `createHash`/`Hash.copy`. `options` is
/// deliberately not type-checked (Node accepts a string/number and ignores
/// it); only a present `outputLength` property is validated.
function pirParseHashOutputLength(options) {
  if (options === null || typeof options !== "object") return undefined;
  const value = options.outputLength;
  if (value === undefined) return undefined;
  if (typeof value !== "number") {
    throw pirInvalidArgType("options.outputLength", "of type number", value);
  }
  if (!Number.isInteger(value) || value < 0 || value > 4294967295) {
    throw pirHashOutputRange(value);
  }
  return value;
}

/// Apply the digest-specific rule once the algorithm is known: SHAKE accepts
/// any validated length; a fixed digest only its own size.
function pirHashOutputLength(algo, value) {
  if (value === undefined) return undefined;
  if (PIR_SHAKE_DEFAULTS[algo] !== undefined) return value;
  const size = PIR_DIGEST_SIZES[algo];
  if (size !== undefined && value !== size) throw pirNotXof(algo, value);
  return value;
}

function pirInvalidDigest(algo) {
  const error = new TypeError("Invalid digest: " + algo);
  error.code = "ERR_CRYPTO_INVALID_DIGEST";
  return error;
}

function pirDigestKnown(algo) {
  return PIR_HOST_DIGESTS[algo] === true || PIR_JS_DIGESTS[algo] !== undefined;
}

function pirDigestBytes(algo, bytes, outputLength) {
  if (PIR_HOST_DIGESTS[algo] === true) {
    // Typed C-ABI entry: the bytes cross as a Uint8Array view, so a multi-MB
    // `update()` does not pay a base64 encode/decode round trip on both legs.
    const native = globalThis.__pirCryptoDigest;
    if (typeof native === "function") {
      const digest = outputLength === undefined ? native(algo, bytes) : native(algo, bytes, outputLength);
      if (digest !== undefined) {
        return Buffer.from(digest.buffer, digest.byteOffset, digest.byteLength);
      }
    }
    const payload = { algo: algo, data: Buffer.from(bytes).toString("base64") };
    if (outputLength !== undefined) payload.outputLength = outputLength;
    return Buffer.from(call("crypto", "hash", payload), "base64");
  }
  const impl = PIR_JS_DIGESTS[algo];
  if (impl === undefined) throw pirDigestUnsupported();
  return impl.hash(bytes);
}

/// RFC 2104 HMAC over whichever digest backend owns `algo`. `key`/`data` are
/// raw byte arrays.
function pirHmacBytes(algo, key, data) {
  if (PIR_HOST_DIGESTS[algo] === true) {
    const native = globalThis.__pirCryptoHmac;
    if (typeof native === "function") {
      const digest = native(algo, key, data);
      if (digest !== undefined) {
        return Buffer.from(digest.buffer, digest.byteOffset, digest.byteLength);
      }
    }
    return Buffer.from(call("crypto", "hmac", {
      algo: algo,
      key: Buffer.from(key).toString("base64"),
      data: Buffer.from(data).toString("base64"),
    }), "base64");
  }
  const impl = PIR_JS_DIGESTS[algo];
  if (impl === undefined) throw pirInvalidDigest(algo);
  const blockSize = impl.blockSize;
  let keyBytes = Buffer.from(key);
  if (keyBytes.length > blockSize) keyBytes = impl.hash(keyBytes);
  const padded = Buffer.alloc(blockSize);
  keyBytes.copy(padded);
  const inner = Buffer.alloc(blockSize);
  const outer = Buffer.alloc(blockSize);
  for (let i = 0; i < blockSize; i++) {
    inner[i] = padded[i] ^ 0x36;
    outer[i] = padded[i] ^ 0x5c;
  }
  return impl.hash(Buffer.concat([outer, impl.hash(Buffer.concat([inner, data]))]));
}


// Native digest entries answer a bare Uint8Array for the output; Node's
// digest() is a Buffer. Re-tag the view (no copy) and leave fast-path strings
// alone.
function pirDigestValue(value) {
  if (typeof value === "string") return value;
  return Buffer.from(value.buffer, value.byteOffset, value.byteLength);
}

// Node's `hash.digest(encoding)`: a falsy encoding (and the "buffer" spelling)
// gives the raw bytes; a Buffer encoding decodes the digest bytes exactly the
// way Buffer.toString does; anything else (the C++ decoder treats it as an
// unknown name) still hands back the raw bytes.
function pirDigestResult(bytes, encoding) {
  if (!encoding) return bytes;
  const name = typeof encoding === "string" ? encoding : String(encoding);
  if (name.toLowerCase() === "buffer") return bytes;
  if (Buffer.isEncoding(name)) return bytes.toString(name);
  return bytes;
}

// Hash.update()'s normalization. Unlike `pushData` (used by Sign/Verify, whose
// chunks are concatenated later and therefore need their own copy), a view is
// handed to the native state as-is: Rust copies it into the digest before the
// call returns, matching Node's immediate-update semantics and removing the
// per-chunk `Buffer.from` copy.
function pirUpdateBytes(data, encoding) {
  return pirHashInputBytes(data, encoding);
}

// Snapshot for the buffered tier: Node's update() ingests the bytes
// immediately, so the caller's view must not be retained. `new
// Uint8Array(view)` is the engine's memcpy (0.3 µs for 4 KiB), far cheaper
// than the old `Buffer.from` wrapper path.
function pirSnapshotBytes(data, encoding) {
  return new Uint8Array(pirUpdateBytes(data, encoding));
}

// Digest encodings the native finalizer renders itself (`digest("hex")` was
// dominated by the JS per-byte hex loop). Everything else stays on the raw
// bytes + `Buffer.toString(encoding)` path, which keeps object-valued
// encodings (`{toString}`) and Node's encoding errors intact.
const PIR_HASH_TEXT_ENCODINGS = { hex: true, base64: true, base64url: true };

// GC release for a hash the caller dropped without calling digest(). The
// registry itself is per-runtime (context userdata), so this only covers the
// live-runtime case; runtime teardown drops every session anyway. Handles are
// never reused, so a late callback cannot free a different hash.
const pirHashFinalizer = typeof FinalizationRegistry === "function"
  ? new FinalizationRegistry((handle) => {
      if (PIR_HASH_RELEASE !== null) PIR_HASH_RELEASE(handle);
    })
  : null;

// All three entries must exist before a hash commits to the streaming path;
// a host without them keeps the old buffered copy + one-shot digest. Cached at
// module load (the host installs them before any shim runs): the property
// lookups otherwise sit on `createHash()`/`update()`'s hot path.
const PIR_HASH_NEW = typeof globalThis.__pirCryptoHashNew === "function"
  ? globalThis.__pirCryptoHashNew : null;
const PIR_HMAC_NEW = typeof globalThis.__pirCryptoHmacNew === "function"
  ? globalThis.__pirCryptoHmacNew : null;
const PIR_HASH_UPDATE = typeof globalThis.__pirCryptoHashUpdate === "function"
  ? globalThis.__pirCryptoHashUpdate : null;
const PIR_HASH_COPY = typeof globalThis.__pirCryptoHashCopy === "function"
  ? globalThis.__pirCryptoHashCopy : null;
const PIR_HASH_FINAL = typeof globalThis.__pirCryptoHashFinal === "function"
  ? globalThis.__pirCryptoHashFinal : null;
const PIR_HASH_RELEASE = typeof globalThis.__pirCryptoHashRelease === "function"
  ? globalThis.__pirCryptoHashRelease : null;
const PIR_DIGEST_ENC = typeof globalThis.__pirCryptoDigestEnc === "function"
  ? globalThis.__pirCryptoDigestEnc : null;
const PIR_HMAC_ENC = typeof globalThis.__pirCryptoHmacEnc === "function"
  ? globalThis.__pirCryptoHmacEnc : null;
const PIR_RANDOM_UUID = typeof globalThis.__pirCryptoRandomUUID === "function"
  ? globalThis.__pirCryptoRandomUUID : null;
const PIR_CAN_STREAM_HASH = PIR_HASH_NEW !== null && PIR_HASH_UPDATE !== null && PIR_HASH_FINAL !== null;
const PIR_CAN_STREAM_HMAC = PIR_HMAC_NEW !== null && PIR_HASH_UPDATE !== null && PIR_HASH_FINAL !== null;
const PIR_EMPTY_BYTES = new Uint8Array(0);
// A single chunk at least this large goes straight into the Rust state (no JS
// copy); smaller chunks take the cheap snapshot + one-call digest path.
const PIR_STREAM_UPDATE_BYTES = 64 * 1024;

function pirHashFinalizedError() {
  const error = new Error("Digest already called");
  error.code = "ERR_CRYPTO_HASH_FINALIZED";
  return error;
}

// ---------------------------------------------------------------------------
// Hash / Hmac
// ---------------------------------------------------------------------------
// Both carry Node's HashBase surface: `update()…digest()` plus the
// Writable+Readable transform sides (`write`/`end` on the way in, `data`/`end`
// on the way out, `read()`/`setEncoding()` for consumers).
//
// The digest state itself lives in Rust behind an integer handle when this
// host exposes the streaming entries: `update()` feeds each chunk straight to
// the Rust state (no per-chunk copy, no `Buffer.concat` at the end), and
// `digest()` is still exactly one host call. Without those entries the old
// buffered path stays as the fallback tier.
class PirHashBase {
  constructor() {
    // Lazy: the streaming path never allocates the chunk array, and the
    // stream/readable flags only matter if those methods are used.
    this._chunks = null;
    this._handle = null;
    this._canStream = false;
    this._digestBytes = null;
  }
  // Subclasses open a state over their algorithm (and key, for HMAC).
  _openState() { return undefined; }
  // Drops a live native session without finalizing it (destroy()/abandonment).
  _releaseHandle() {
    const handle = this._handle;
    if (handle === null) return;
    this._handle = null;
    if (pirHashFinalizer !== null) pirHashFinalizer.unregister(this);
    if (PIR_HASH_RELEASE !== null) PIR_HASH_RELEASE(handle);
  }
  // Open the Rust state on the first update (Node's Hash ingests each
  // update() immediately, so a later caller mutation must not be observable).
  _openHandle() {
    if (this._handle !== null) return this._handle;
    if (!this._canStream) return null;
    const handle = this._openState();
    if (handle === undefined || handle === null) {
      // The host does not serve this algorithm; stay on the buffered tier.
      this._canStream = false;
      return null;
    }
    this._handle = handle;
    if (pirHashFinalizer !== null) pirHashFinalizer.register(this, handle, this);
    return handle;
  }
  // Feed every buffered chunk into a freshly opened state.
  _drainChunks(handle) {
    const chunks = this._chunks;
    this._chunks = null;
    for (let index = 0; index < chunks.length; index++) PIR_HASH_UPDATE(handle, chunks[index]);
  }
  _push(data, encoding) {
    if (this._handle !== null) {
      PIR_HASH_UPDATE(this._handle, pirUpdateBytes(data, encoding));
      return;
    }
    if (this._canStream) {
      const view = pirUpdateBytes(data, encoding);
      if (view.byteLength >= PIR_STREAM_UPDATE_BYTES) {
        // Large chunk: let Rust snapshot it directly (after any earlier small
        // chunks that are still buffered).
        const handle = this._openHandle();
        if (handle !== null) {
          if (this._chunks !== null) this._drainChunks(handle);
          PIR_HASH_UPDATE(handle, view);
          return;
        }
      }
      // Small (or host-fallback) chunk: keep a copy, which is what Node's
      // update() effectively does, and switch to streaming on the second one.
      if (this._chunks === null) this._chunks = [];
      this._chunks.push(typeof data === "string" ? view : new Uint8Array(view));
      if (this._chunks.length > 1) {
        const handle = this._openHandle();
        if (handle !== null) this._drainChunks(handle);
      }
      return;
    }
    if (this._chunks === null) this._chunks = [];
    this._chunks.push(pirSnapshotBytes(data, encoding));
  }
  update(data, encoding) {
    // Node checks the finalized state before it looks at the data argument.
    if (this._digestCalled) throw pirHashFinalizedError();
    this._push(data, encoding);
    return this;
  }
  // --- Writable side ------------------------------------------------------
  write(chunk, encoding, callback) {
    if (typeof encoding === "function") { callback = encoding; encoding = undefined; }
    if (this._ended) {
      const error = new Error("write after end");
      error.code = "ERR_STREAM_WRITE_AFTER_END";
      if (typeof callback === "function") { globalThis.__pirMicrotask(() => callback(error)); return false; }
      throw error;
    }
    this._push(chunk, encoding);
    if (typeof callback === "function") globalThis.__pirMicrotask(callback);
    return true;
  }
  end(chunk, encoding, callback) {
    if (typeof chunk === "function") { callback = chunk; chunk = undefined; encoding = undefined; }
    else if (typeof encoding === "function") { callback = encoding; encoding = undefined; }
    if (chunk !== undefined && chunk !== null) this._push(chunk, encoding);
    if (!this._ended) {
      this._ended = true;
      this._digestBytes = this._computeDigest();
      this.emit("data", this._encodedDigest());
      this.emit("end");
      this.emit("finish");
    }
    if (typeof callback === "function") callback();
    return this;
  }
  _encodedDigest() {
    return this._encoding === undefined
      ? this._digestBytes
      : this._digestBytes.toString(this._encoding);
  }
  // --- Readable side ------------------------------------------------------
  read(_size) {
    if (this._digestBytes === null || this._readConsumed) return null;
    this._readConsumed = true;
    return this._encodedDigest();
  }
  setEncoding(encoding) {
    this._encoding = encoding;
    return this;
  }
  pipe(destination, options) {
    this.on("data", (chunk) => destination.write(chunk));
    this.on("end", () => {
      if (!options || options.end !== false) destination.end();
    });
    return destination;
  }
  destroy(error) {
    this._ended = true;
    this._releaseHandle();
    if (error) this.emit("error", error);
    this.emit("close");
    return this;
  }
  // One-shot digest over a single byte view, implemented by Hash/Hmac.
  _computeOneShot(_bytes, _encoding) { return Buffer.alloc(0); }
  _finishHandle(encoding) {
    const handle = this._handle;
    this._handle = null;
    if (pirHashFinalizer !== null) pirHashFinalizer.unregister(this);
    // `hex`/`base64`/`base64url` are rendered host-side. Any other encoding
    // (including an object with toString) takes the raw-bytes contract.
    const fast = typeof encoding === "string" && PIR_HASH_TEXT_ENCODINGS[encoding] === true;
    // Never pass an explicit `undefined` in an optional position: the host
    // entry converts its arguments by value, and an undefined where a number
    // or string is expected fails conversion instead of meaning "absent".
    // `outputLength` only exists for SHAKE, so the fixed-digest path keeps the
    // shorter call, and the raw-bytes spelling ("buffer") is passed when SHAKE
    // needs a length without a text encoding: the host returns the same
    // Uint8Array for it as for an absent encoding.
    let value;
    if (this._outputLength === undefined) {
      value = fast ? PIR_HASH_FINAL(handle, encoding) : PIR_HASH_FINAL(handle);
    } else {
      value = PIR_HASH_FINAL(handle, fast ? encoding : "buffer", this._outputLength);
    }
    if (value !== undefined) return pirDigestValue(value);
    throw new Error("crypto hash state lost");
  }
  _computeDigest(encoding) {
    if (this._handle !== null) return this._finishHandle(encoding);
    const chunks = this._chunks;
    const bytes = chunks === null || chunks.length === 0
      ? PIR_EMPTY_BYTES
      : chunks.length === 1 ? chunks[0] : Buffer.concat(chunks);
    return this._computeOneShot(bytes, encoding);
  }
  // After the stream path finalized, Node's Hash hands back the digest the
  // stream produced; Hmac instead returns an empty value (test-crypto-hmac
  // pins `streamHmac.digest()` === Buffer.alloc(0)).
  _streamDigest(encoding) { return pirDigestResult(this._digestBytes, encoding); }
  // --- digest() -----------------------------------------------------------
  digest(encoding) {
    if (this._ended) return this._streamDigest(encoding);
    if (this._digestCalled) return this._repeatDigest(encoding);
    this._digestCalled = true;
    const value = this._computeDigest(encoding);
    this._chunks = null;
    // The host already rendered hex/base64/base64url; a primitive string's
    // toString ignores the encoding argument, so skip the second dispatch.
    if (typeof value === "string") return value;
    return pirDigestResult(value, encoding);
  }
  _repeatDigest(encoding) {
    // Hmac's second digest: an empty Buffer, rendered as a string only for a
    // truthy non-"buffer" encoding (node's own branch).
    if (encoding && encoding !== "buffer") return Buffer.alloc(0).toString(encoding);
    return Buffer.alloc(0);
  }
}

// Node's Hash/Hmac are Transform streams, hence EventEmitters. Linking the
// prototype directly (instead of `extends`) skips EventEmitter's constructor
// and the derived-constructor prologue on every hash; the events shim already
// lazily creates `_events` for util.inherits-style instances, and
// `instanceof EventEmitter` still holds through the prototype chain.
Object.setPrototypeOf(PirHashBase.prototype, EventEmitter.prototype);

class Hash extends PirHashBase {
  constructor(algo, options) {
    super();
    if (typeof algo !== "string") {
      throw pirInvalidArgType("algorithm", "of type string", algo);
    }
    // WebCrypto spells algorithms "SHA-256"; Node digests use "sha256".
    this._algo = normalizeDigest(algo);
    // Node validates `options.outputLength` before it resolves the digest.
    const requested = pirParseHashOutputLength(options);
    // Node reports an unknown digest at construction, not at digest().
    if (!pirDigestKnown(this._algo)) throw pirDigestUnsupported();
    this._outputLength = pirHashOutputLength(this._algo, requested);
    this._canStream = PIR_HOST_DIGESTS[this._algo] === true && PIR_CAN_STREAM_HASH;
  }
  _openState() { return PIR_HASH_NEW(this._algo); }
  _computeOneShot(bytes, encoding) {
    if (PIR_HOST_DIGESTS[this._algo] === true && PIR_DIGEST_ENC !== null) {
      const fast = typeof encoding === "string" && PIR_HASH_TEXT_ENCODINGS[encoding] === true;
      // Same arity rule as `_finishHandle`: never pass an explicit undefined
      // into an optional host position, and spell the raw-bytes choice as
      // "buffer" when a SHAKE length has to be supplied without an encoding.
      let value;
      if (this._outputLength === undefined) {
        value = fast
          ? PIR_DIGEST_ENC(this._algo, bytes, encoding)
          : PIR_DIGEST_ENC(this._algo, bytes);
      } else {
        value = PIR_DIGEST_ENC(this._algo, bytes, fast ? encoding : "buffer", this._outputLength);
      }
      if (value !== undefined) return pirDigestValue(value);
    }
    return pirDigestBytes(this._algo, bytes, this._outputLength);
  }
  // Node's Hash throws on a second digest(); Hmac (below) returns empty.
  _repeatDigest(encoding) {
    throw pirHashFinalizedError();
  }
  copy(options) {
    // Node's Hash.copy() refuses a finalized state, unlike the Hmac wrapper.
    if (this._digestCalled) throw pirHashFinalizedError();
    // Node resets to the digest's default outputLength when copy() gets no
    // options (`createHash('shake256', { outputLength: 0 }).copy()` hashes 32
    // bytes), and honours an explicit `{ outputLength }` otherwise.
    const clone = new Hash(this._algo, options);
    if (this._handle !== null) {
      const handle = PIR_HASH_COPY(this._handle);
      if (handle !== undefined) {
        clone._handle = handle;
        if (pirHashFinalizer !== null) pirHashFinalizer.register(clone, handle, clone);
        return clone;
      }
    }
    if (this._chunks !== null) clone._chunks = this._chunks.slice();
    return clone;
  }
}

class Hmac extends PirHashBase {
  constructor(algo, key) {
    super();
    this._algo = normalizeDigest(algo);
    this._key = pirKeyBytes(key, "key");
    this._canStream = PIR_HOST_HMACS[this._algo] === true && PIR_CAN_STREAM_HMAC;
  }
  _openState() { return PIR_HMAC_NEW(this._algo, this._key); }
  _streamDigest(encoding) { return this._repeatDigest(encoding); }
  _computeOneShot(bytes, encoding) {
    if (PIR_HOST_HMACS[this._algo] === true && PIR_HMAC_ENC !== null) {
      const fast = typeof encoding === "string" && PIR_HASH_TEXT_ENCODINGS[encoding] === true;
      const value = fast
        ? PIR_HMAC_ENC(this._algo, this._key, bytes, encoding)
        : PIR_HMAC_ENC(this._algo, this._key, bytes);
      if (value !== undefined) return pirDigestValue(value);
    }
    return pirHmacBytes(this._algo, this._key, bytes);
  }
}

function createHash(algo, options) { return new Hash(algo, options); }

// Node's createHmac validates the algorithm *name* as a string first, then the
// key, then the digest name (ERR_CRYPTO_INVALID_DIGEST for an unknown one).
function createHmac(algo, key) {
  if (typeof algo !== "string") {
    throw pirInvalidArgType("hmac", "of type string", algo);
  }
  const hmac = new Hmac(algo, key);
  if (!pirDigestKnown(hmac._algo) || PIR_HOST_HMACS[hmac._algo] !== true) {
    throw pirInvalidDigest(hmac._algo);
  }
  return hmac;
}

}

// node's `crypto.Hash`/`Hmac`/`Sign`/`Verify`/`Cipheriv`/`Decipheriv` are
// callable without `new` (test-crypto-hmac.js line 12 builds an Hmac that way),
// while `KeyObject`/`X509Certificate` are classes that throw when called
// without it. Each exported binding below is the callable form, sharing the
// class prototype so instanceof and `constructor` are unchanged.
function pirCallableCtor(impl, before) {
  const callable = function (...args) {
    if (before !== undefined) before();
    return new.target === undefined ? new impl(...args) : Reflect.construct(impl, args, new.target);
  };
  callable.prototype = impl.prototype;
  impl.prototype.constructor = callable;
  Object.defineProperty(callable, "name", { value: impl.name, configurable: true });
  Object.defineProperty(callable, "length", { value: impl.length, configurable: true });
  return callable;
}

// One-shot `crypto.hash(algorithm, data, outputEncoding)` (Node 21+).
// Semantics from lib/internal/crypto/hash.js `hash()`: the third argument is
// either an encoding string or an options object (`{ outputEncoding,
// outputLength }`), the output defaults to **hex**, and "buffer" returns the
// raw bytes.
function hash(algorithm, input, options) {
  if (typeof algorithm !== "string") throw pirInvalidArgType("algorithm", "of type string", algorithm);
  if (typeof input !== "string" && !ArrayBuffer.isView(input)) {
    throw pirInvalidArgType("input", PIR_INPUT_EXPECTED, input);
  }
  let outputEncoding;
  let outputLength;
  if (options === undefined) {
    outputEncoding = "hex";
  } else if (typeof options === "string") {
    outputEncoding = options;
  } else {
    pirValidateObject(options, "options");
    outputEncoding = options.outputEncoding;
    outputLength = pirOneShotOutputLength(options.outputLength);
  }
  if (outputEncoding === undefined || outputEncoding === null) outputEncoding = "hex";

  let encoding = outputEncoding;
  if (encoding !== "hex") {
    if (typeof encoding !== "string") throw pirInvalidArgType("outputEncoding", "of type string", encoding);
    if (!Buffer.isEncoding(encoding) && encoding.toLowerCase() !== "buffer") {
      throw pirInvalidArgValue("outputEncoding", encoding);
    }
  }

  const algo = normalizeDigest(algorithm);
  if (!pirDigestKnown(algo)) {
    // oneShotDigest's own error, distinct from createHash's.
    throw new Error("Digest method " + algorithm + " is not supported");
  }
  // oneShotDigest's non-XOF message differs from the createHash constructor's
  // ERR_OSSL_EVP_NOT_XOF_OR_INVALID_LENGTH.
  if (outputLength !== undefined && PIR_SHAKE_DEFAULTS[algo] === undefined) {
    const size = PIR_DIGEST_SIZES[algo];
    if (size !== undefined && outputLength !== size) {
      throw new Error("Output length " + outputLength + " is invalid for " +
        algorithm + ", which does not support XOF");
    }
  }
  const hashOptions = outputLength === undefined ? undefined : { outputLength: outputLength };
  const bytes = new Hash(algorithm, hashOptions).update(input).digest();
  if (encoding.toLowerCase() === "buffer") return bytes;
  return bytes.toString(encoding);
}

/// One-shot `crypto.hash` spells its XOF length as a *positional option*, so
/// Node's validator names it `outputLength` (not `options.outputLength`).
function pirOneShotOutputLength(value) {
  if (value === undefined) return undefined;
  if (typeof value !== "number") {
    throw pirInvalidArgType("outputLength", "of type number", value);
  }
  if (!Number.isInteger(value)) throw pirOutOfRange("outputLength", "an integer", value);
  if (value < 0 || value > 4294967295) {
    throw pirOutOfRange("outputLength", ">= 0 && <= 4294967295", value);
  }
  return value;
}

function randomBytes(size, callback) {
  // Node's assertSize(size, 1, 0, Infinity): a non-number is
  // ERR_INVALID_ARG_TYPE, then the range check covers NaN/negative/Infinity
  // and the 2^31-1 allocation cap.
  const count = pirAssertSize(size, 1, 0, Infinity);
  if (callback !== undefined) pirValidateFunction(callback, "callback");
  if (callback === undefined) return pirRandomBytes(count);
  microtaskRandom(count, callback);
}

/// OS entropy as a Buffer. The typed entry hands the bytes over directly; the
/// base64 JSON op stays as the fallback for boot paths without it.
function pirRandomBytes(count) {
  if (typeof globalThis.__pirCryptoRandom === "function") {
    const bytes = globalThis.__pirCryptoRandom(count);
    if (bytes !== undefined) return Buffer.from(bytes.buffer, bytes.byteOffset, bytes.byteLength);
  }
  return Buffer.from(call("crypto", "randomBytes", { n: count }), "base64");
}

function microtaskRandom(count, callback) {
  globalThis.__pirMicrotask(() => {
    let value;
    try {
      value = pirRandomBytes(count);
    } catch (error) {
      callback(error);
      return;
    }
    callback(null, value);
  });
}

// Node's random.js `assertOffset`/`assertSize`: element-size aware bounds with
// ERR_OUT_OF_RANGE. `>>> 0` is Node's uint32 truncation (a fractional offset
// or size is accepted and truncated).
function pirAssertOffset(offset, elementSize, length) {
  pirValidateNumber(offset, "offset");
  offset *= elementSize;
  const maxLength = Math.min(length, 2147483647);
  if (Number.isNaN(offset) || offset > maxLength || offset < 0) {
    throw pirOutOfRange("offset", ">= 0 && <= " + maxLength, offset);
  }
  return offset >>> 0;
}

function pirAssertSize(size, elementSize, offset, length) {
  pirValidateNumber(size, "size");
  size *= elementSize;
  if (Number.isNaN(size) || size > 2147483647 || size < 0) {
    throw pirOutOfRange("size", ">= 0 && <= 2147483647", size);
  }
  if (size + offset > length) {
    throw pirOutOfRange("size + offset", "<= " + length, size + offset);
  }
  return size >>> 0;
}

function pirIsAnyArrayBuffer(value) {
  return value instanceof ArrayBuffer ||
    (typeof SharedArrayBuffer === "function" && value instanceof SharedArrayBuffer);
}

// The destination window for a fill. ArrayBuffer/SharedArrayBuffer address
// bytes directly; a view offsets from its own byteOffset.
function pirFillWindow(buffer, offset, size) {
  if (pirIsAnyArrayBuffer(buffer)) return new Uint8Array(buffer, offset, size);
  return new Uint8Array(buffer.buffer, buffer.byteOffset + offset, size);
}

function randomFillSync(buffer, offset, size) {
  if (!pirIsAnyArrayBuffer(buffer) && !ArrayBuffer.isView(buffer)) {
    throw pirInvalidArgType("buf", "an instance of ArrayBuffer or ArrayBufferView", buffer);
  }
  const elementSize = buffer.BYTES_PER_ELEMENT || 1;
  const length = buffer.byteLength;
  offset = pirAssertOffset(offset === undefined ? 0 : offset, elementSize, length);
  size = size === undefined
    ? length - offset
    : pirAssertSize(size, elementSize, offset, length);
  if (size === 0) return buffer;
  pirFillWindow(buffer, offset, size).set(pirRandomBytes(size));
  return buffer;
}

function randomFill(buffer, offset, size, callback) {
  if (!pirIsAnyArrayBuffer(buffer) && !ArrayBuffer.isView(buffer)) {
    throw pirInvalidArgType("buf", "an instance of ArrayBuffer or ArrayBufferView", buffer);
  }
  const elementSize = buffer.BYTES_PER_ELEMENT || 1;
  const length = buffer.byteLength;
  if (typeof offset === "function") {
    callback = offset;
    offset = 0;
    size = buffer.length;
  } else if (typeof size === "function") {
    callback = size;
    size = (buffer.length !== undefined ? buffer.length : length) - offset;
  } else {
    pirValidateFunction(callback, "callback");
  }
  offset = pirAssertOffset(offset === undefined ? 0 : offset, elementSize, length);
  size = size === undefined
    ? length - offset
    : pirAssertSize(size, elementSize, offset, length);
  if (size === 0) {
    callback(null, buffer);
    return;
  }
  globalThis.__pirMicrotask(() => {
    try {
      pirFillWindow(buffer, offset, size).set(pirRandomBytes(size));
      callback(null, buffer);
    } catch (error) {
      callback(error);
    }
  });
}

// Unbiased `randomInt` via rejection sampling over the host entropy source.
function pirRandomIntValue(lo, hi) {
  const range = hi - lo;
  const bytes = range <= 0x100 ? 1 : range <= 0x10000 ? 2 : range <= 0x100000000 ? 4 : 6;
  const pow = bytes === 6 ? 281474976710656 : Math.pow(2, bytes * 8);
  const limit = pow - (pow % range);
  for (;;) {
    const buf = pirRandomBytes(bytes);
    let value = 0;
    for (let i = 0; i < bytes; i++) value = value * 256 + buf[i];
    if (value < limit) return lo + (value % range);
  }
}

function randomInt(min, max, callback) {
  // randomInt(max) / randomInt(max, callback): the second argument is the
  // callback, so `max` (above) becomes the bound and `min` is 0.
  const minNotSpecified = typeof max === "undefined" || typeof max === "function";
  if (minNotSpecified) {
    callback = max;
    max = min;
    min = 0;
  }
  const isSync = typeof callback === "undefined";
  if (!isSync) pirValidateFunction(callback, "callback");
  if (!Number.isSafeInteger(min)) throw pirInvalidArgType("min", "a safe integer", min);
  if (!Number.isSafeInteger(max)) throw pirInvalidArgType("max", "a safe integer", max);
  if (max <= min) {
    throw pirOutOfRange("max", "greater than the value of \"min\" (" + min + ")", max);
  }
  const range = max - min;
  // Node's RAND_MAX is 2^48 - 1.
  if (!(range <= 281474976710655)) {
    throw pirOutOfRange(minNotSpecified ? "max" : "max - min", "<= 281474976710655", range);
  }
  const value = pirRandomIntValue(min, max);
  if (isSync) return value;
  const deliver = () => callback(undefined, value);
  if (typeof globalThis.__pirNextTick === "function") globalThis.__pirNextTick(deliver);
  else globalThis.__pirMicrotask(deliver);
  return undefined;
}

  }
}

// UUID v7: 48-bit big-endian millisecond timestamp + version/variant + random
// tail (RFC 9562 layout; time-based ordering is best-effort, no monotonic
// counter is kept).
function randomUUIDv7(options) {
  pirRandomUuidOptions(options);
  const bytes = pirRandomBytes(16);
  const ms = BigInt(Date.now());
  for (let i = 0; i < 6; i++) {
    bytes[i] = Number((ms >> BigInt(8 * (5 - i))) & 0xffn);
  }
  bytes[6] = (bytes[6] & 0x0f) | 0x70;
  bytes[8] = (bytes[8] & 0x3f) | 0x80;
  let hex = "";
  for (let i = 0; i < 16; i++) hex += (bytes[i] < 16 ? "0" : "") + bytes[i].toString(16);
  return hex.slice(0, 8) + "-" + hex.slice(8, 12) + "-" + hex.slice(12, 16) + "-" + hex.slice(16, 20) + "-" + hex.slice(20);
}

// Every digest this runtime can actually compute: the host `crypto` op serves
// the whole set (`PIR_HOST_DIGESTS`), so advertising only the original four
// would make plugins feature-detect SHA-3 as missing and take a slow path.
// tests/node_shims.rs and tests/zlib_crypto_shims.rs pin this exact list.
function getHashes() {
  // The real host-backed set: the Rust `hash`/`HashState` table serves all of
  // these (SHAKE through the extendable-output path). Node also lists RSA-*
  // and other provider aliases, which this runtime has no engine for.
  return [
    "md5", "sha1", "sha224", "sha256", "sha384", "sha512",
    "sha3-224", "sha3-256", "sha3-384", "sha3-512",
    "shake128", "shake256",
  ];
}

function toBytes(value, name) {
  if (ArrayBuffer.isView(value)) return Buffer.from(value.buffer, value.byteOffset, value.byteLength);
  if (value instanceof ArrayBuffer ||
      (typeof SharedArrayBuffer === "function" && value instanceof SharedArrayBuffer)) {
    return new Uint8Array(value);
  }
  const error = new TypeError(
    "The \"" + name + "\" argument must be an instance of ArrayBuffer, Buffer, TypedArray, or DataView.");
  error.code = "ERR_INVALID_ARG_TYPE";
  throw error;
}

function timingSafeEqual(a, b) {
  const left = toBytes(a, "buf1");
  const right = toBytes(b, "buf2");
  if (left.length !== right.length) {
    const error = new RangeError("Input buffers must have the same byte length");
    error.code = "ERR_CRYPTO_TIMING_SAFE_EQUAL_LENGTH";
    throw error;
  }
  let diff = 0;
  for (let i = 0; i < left.length; i++) diff |= left[i] ^ right[i];
  return diff === 0;
}

  if (typeof DOMException === "function") return new DOMException(message, name);

function pirIsFloatTypedArray(value) {

/// WebCrypto `getRandomValues` (internal/crypto/random.js `getRandomValues`):
/// only integer-type TypedArrays are accepted (DataView, ArrayBuffer, plain
/// arrays and the float arrays are a TypeMismatchError), and the byte length
/// is capped at 64 KiB with a QuotaExceededError.
const webcryptoGetRandomValues = (array) => {
  if (!ArrayBuffer.isView(array) || array instanceof DataView || pirIsFloatTypedArray(array)) {
    throw pirDomException("The data argument must be an integer-type TypedArray", "TypeMismatchError");
  }
  if (array.byteLength > 65536) {
    throw pirDomException("The requested length exceeds 65,536 bytes", "QuotaExceededError");
  }
  if (array.byteLength > 0) {
    new Uint8Array(array.buffer, array.byteOffset, array.byteLength).set(pirRandomBytes(array.byteLength));
  }
  return array;
};

// The cipher family the host backs; the JS classes below compose these into
// Node's streaming surface (one host round trip per update/final).
const CIPHER_MODES = ["cbc", "ctr", "gcm"];
const CIPHER_SIZES = [128, 192, 256];

function pirCipherInfo(name) {
  const match = /^aes-(128|192|256)-(cbc|ctr|gcm)$/.exec(String(name).toLowerCase());
  if (!match) return undefined;
  return {
    name: match[0],
    nid: undefined,
    blockSize: 16,
    ivLength: match[2] === "gcm" ? 12 : 16,
    keyLength: Number(match[1]) / 8,
    mode: match[2],
  };
}

function getCiphers() {
  const out = [];
  for (const size of CIPHER_SIZES) {
    for (const mode of CIPHER_MODES) out.push("aes-" + size + "-" + mode);
  }
  return out;
}

class PirCipher {
  constructor(algorithm, key, iv, encrypt) {
    if (typeof algorithm !== "string") {
      throw pirInvalidArgType("cipher", "of type string", algorithm);
    }
    const info = pirCipherInfo(algorithm);
    if (info === undefined) throw pirUnknownCipher();
    this._alg = info.name;
    this._info = info;
    this._encrypt = encrypt;
    const keyBytes = pirKeyBytes(key, "key");
    if (keyBytes.length !== info.keyLength) throw pirInvalidKeyLength();
    // CBC/CTR need a full block; GCM accepts any non-empty IV (Node's
    // `AES-128-GCM` setup rejects only a zero-length one).
    if (iv === undefined || iv === null) throw pirInvalidIv();
    const ivBytes = toKeyBytes(iv, "iv");
    if (ivBytes.length === 0 || (info.mode !== "gcm" && ivBytes.length !== info.blockSize)) {
      throw pirInvalidIv();
    }
    this._keyB64 = keyBytes.toString("base64");
    this._ivB64 = ivBytes.toString("base64");
    this._state = null;
    this._autoPadding = true;
    this._aadB64 = null;
    this._authTagB64 = null;
    this._finalTag = null;
    this._ended = false;
    this._outputEncoding = undefined;
  }
  _dataBytes(data, encoding) {
    if (typeof data === "string") return Buffer.from(data, encoding === undefined ? "utf8" : encoding);
    if (ArrayBuffer.isView(data)) {
      return Buffer.from(data.buffer, data.byteOffset, data.byteLength);
    }
    if (data instanceof ArrayBuffer ||
        (typeof SharedArrayBuffer === "function" && data instanceof SharedArrayBuffer)) {
      return Buffer.from(new Uint8Array(data));
    }
    throw pirInvalidArgType("data", PIR_INPUT_EXPECTED, data);
  }
  _run(data, isFinal) {
    if (this._ended) {
      // Node's C++ layer: update()/final() after final().
      throw new Error("Trying to add data in unsupported state");
    }
    const payload = {
      alg: this._alg,
      dir: this._encrypt ? "encrypt" : "decrypt",
      key: this._keyB64,
      iv: this._ivB64,
      data: data.toString("base64"),
      final: isFinal,
      autoPadding: this._autoPadding,
      state: this._state,
    };
    if (this._aadB64 !== null) payload.aad = this._aadB64;
    if (!this._encrypt && this._authTagB64 !== null) payload.authTag = this._authTagB64;
    let result;
    try {
      result = call("crypto", "cipheriv", payload);
    } catch (error) {
      // Node surfaces a failed GCM tag comparison as this plain Error (no
      // code); the host's openssl-style spelling is remapped here.
      if (error !== null && typeof error === "object" && error.code === "ERR_OSSL_EVP_BAD_DECRYPT") {
        throw new Error("Unsupported state or unable to authenticate data");
      }
      throw error;
    }
    this._state = result.state === null || result.state === undefined ? null : result.state;
    if (result.authTag) this._finalTag = Buffer.from(result.authTag, "base64");
    if (isFinal) this._ended = true;
    return Buffer.from(result.data, "base64");
  }
  update(data, inputEncoding, outputEncoding) {
    const encoding = this._checkOutputEncoding(outputEncoding);
    const out = this._run(this._dataBytes(data, inputEncoding), false);
    return encoding === undefined ? out : out.toString(encoding);
  }
  final(outputEncoding) {
    const encoding = this._checkOutputEncoding(outputEncoding);
    const out = this._run(Buffer.alloc(0), true);
    return encoding === undefined ? out : out.toString(encoding);
  }
  // Node's CipherBase pins one output encoding per cipher: a later update()/
  // final() with a different one raises ERR_INVALID_ARG_VALUE, and an unknown
  // name raises ERR_UNKNOWN_ENCODING (the unknown-encoding check runs first —
  // test-crypto-encoding-validation-error covers both orders).
  _checkOutputEncoding(outputEncoding) {
    if (outputEncoding === undefined || outputEncoding === null) return undefined;
    if (typeof outputEncoding !== "string" || !Buffer.isEncoding(outputEncoding)) {
      throw pirUnknownEncoding(outputEncoding);
    }
    // Node compares the StringDecoder's normalized label and reports that same
    // label in the error (`cannot be changed from 'utf8'`, not 'utf-8').
    const normalized = new StringDecoder(outputEncoding).encoding;
    if (this._outputEncoding === undefined) {
      this._outputEncoding = normalized;
    } else if (this._outputEncoding !== normalized) {
      const error = new TypeError(
        "The argument 'outputEncoding' cannot be changed from '" +
        this._outputEncoding + "'. Received '" + outputEncoding + "'");
      error.code = "ERR_INVALID_ARG_VALUE";
      throw error;
    }
    return outputEncoding;
  }
  setAutoPadding(value) {
    this._autoPadding = value === undefined ? true : !!value;
    return this;
  }
  setAAD(buffer, _options) {
    this._aadB64 = toBytes(buffer, "buffer").slice().toString("base64");
    return this;
  }
  getAuthTag() {
    if (this._finalTag === null) {
      const error = new Error("Invalid state for operation getAuthTag");
      error.code = "ERR_CRYPTO_INVALID_STATE";
      throw error;
    }
    return this._finalTag;
  }
  setAuthTag(tag) {
    const bytes = typeof tag === "string" ? Buffer.from(tag, "hex") : toBytes(tag, "tag");
    this._authTagB64 = Buffer.from(bytes).toString("base64");
    return this;
  }
}

// A minimal KeyObject: PEM material passes through to the sign/verify host
// ops; secret keys keep their raw bytes (base64) for export().
class KeyObject {
  constructor(type, pem) {
    this.type = type;
    this._pem = pem === undefined ? null : pem;
    this._secretB64 = null;
    this.asymmetricKeyType = type === "secret" ? undefined : (pem === null ? undefined : "rsa");
  }
  // `KeyObject.from(cryptoKey)`: the WebCrypto secret keys this shim produces
  // carry their raw bytes, so the wrapping is exact (asymmetric keys would need
  // the DER/PEM conversion the host does not host).
  static from(key) {
    if (key !== undefined && key !== null && key.type === "secret" && key._raw !== undefined) {
      const object = new KeyObject("secret", null);
      object._secretB64 = Buffer.from(key._raw).toString("base64");
      return object;
    }
    throw pirInvalidArgType("key", "an instance of CryptoKey", key);
  }
  export(options) {
    if (this.type === "secret") {
      return Buffer.from(this._secretB64 === null ? "" : this._secretB64, "base64");
    }
    return this._pem;
  }
  equals(other) {
    return other instanceof KeyObject && other.type === this.type &&
      other._pem === this._pem && other._secretB64 === this._secretB64;
  }
}

function pirKeyPem(key, name) {
  if (key instanceof KeyObject && typeof key._pem === "string") return key._pem;
  if (typeof key === "string") return key;
  if (key instanceof Uint8Array) return Buffer.from(key).toString("utf8");
  if (key instanceof ArrayBuffer) return Buffer.from(new Uint8Array(key)).toString("utf8");
  throw new TypeError("The \"" + name + "\" argument must be of type string or an instance of Buffer, TypedArray, or KeyObject");
}

function createPrivateKey(key) { return new KeyObject("private", pirKeyPem(key, "key")); }
function createPublicKey(key) { return new KeyObject("public", pirKeyPem(key, "key")); }
function createSecretKey(key, _encoding) {
  const object = new KeyObject("secret", null);
  object._secretB64 = toKeyBytes(key, "key").toString("base64");
  return object;
}

function normalizeSignAlgo(algorithm) {
  const compact = String(algorithm).toLowerCase().replace(/-/g, "");
  return compact.indexOf("rsa") === 0 ? compact.slice(3) : compact;
}

class Sign {
  constructor(algorithm) {
    this._algo = normalizeSignAlgo(algorithm);
    if (PIR_HOST_HMACS[this._algo] !== true) throw pirInvalidDigest(String(algorithm));
    this._chunks = [];
  }
  update(data, encoding) { pushData(this._chunks, data, encoding); return this; }
  write(data, encoding) { return this.update(data, encoding); }
  end(data, encoding) { if (data !== undefined) this.update(data, encoding); return this; }
  sign(key, encoding) {
    const data = Buffer.concat(this._chunks);
    const signature = Buffer.from(call("crypto", "sign", {
      algo: this._algo,
      key: pirKeyPem(key, "key"),
      data: data.toString("base64"),
    }), "base64");
    return encoding === undefined ? signature : signature.toString(encoding);
  }
}

class Verify {
  constructor(algorithm) {
    this._algo = normalizeSignAlgo(algorithm);
    if (PIR_HOST_HMACS[this._algo] !== true) throw pirInvalidDigest(String(algorithm));
    this._chunks = [];
  }
  update(data, encoding) { pushData(this._chunks, data, encoding); return this; }
  write(data, encoding) { return this.update(data, encoding); }
  end(data, encoding) { if (data !== undefined) this.update(data, encoding); return this; }
  verify(key, signature, signatureEncoding) {
    const data = Buffer.concat(this._chunks);
    let signatureBytes;
    if (typeof signature === "string") {
      signatureBytes = Buffer.from(signature, signatureEncoding === undefined ? "hex" : signatureEncoding);
    } else {
      signatureBytes = Buffer.from(signature instanceof ArrayBuffer ? new Uint8Array(signature) : signature);
    }
    return call("crypto", "verify", {
      algo: this._algo,
      key: pirKeyPem(key, "key"),
      data: data.toString("base64"),
      signature: signatureBytes.toString("base64"),
    }) === true;
  }
}

// internal/validators validateInteger: type, then integrality, then the
// inclusive window. Node prints the range with "&&" (not "and").
function pirValidateInteger(value, name, min, max) {
  if (typeof value !== "number") throw pirInvalidArgType(name, "of type number", value);
  if (!Number.isInteger(value)) throw pirOutOfRange(name, "an integer", value);
  if (value < min || value > max) {
    throw pirOutOfRange(name, ">= " + min + " && <= " + max, value);
  }
  return value;
}

function pirScryptParamsError(reason) {
  const error = new RangeError(reason === undefined ? "Invalid scrypt params" : reason);
  error.code = "ERR_CRYPTO_INVALID_SCRYPT_PARAMS";
  return error;
}

// Real PBKDF2 (RFC 2898) over the host's HMAC backends (SHA-1, SHA-2, SHA-3
// and MD5). Node validates password/salt, then iterations, keylen and digest.
function pbkdf2Sync(password, salt, iterations, keylen, digest) {
  const passwordBytes = toKeyBytes(password, "password");
  const saltBytes = toKeyBytes(salt, "salt");
  const rounds = pirValidateInteger(iterations, "iterations", 1, 2147483647);
  const length = pirValidateInteger(keylen, "keylen", 0, 2147483647);
  if (typeof digest !== "string") {
    throw pirInvalidArgType("digest", "of type string", digest);
  }
  const algo = normalizeDigest(digest);
  if (PIR_SHAKE_DEFAULTS[algo] !== undefined) {
    // OpenSSL's provider rejects XOF digests as PBKDF2 PRFs.
    throw new Error("error:1C8000B7:Provider routines::xof digests not allowed");
  }
  if (PIR_HOST_HMACS[algo] !== true) throw pirInvalidDigest(digest);
  return Buffer.from(call("crypto", "pbkdf2", {
    algo: algo,
    password: passwordBytes.toString("base64"),
    salt: saltBytes.toString("base64"),
    iterations: rounds,
    keylen: length,
  }), "base64");
}

function pbkdf2(password, salt, iterations, keylen, digest, callback) {
  // Node's overloads: pbkdf2(pw, salt, iterations, keylen, digest, cb) and
  // the digest-less form with the callback in the digest slot. Node validates
  // every argument synchronously (a bad digest throws even with a callback),
  // so the derivation runs before the callback check here as well.
  if (typeof digest === "function") { callback = digest; digest = undefined; }
  const value = pbkdf2Sync(password, salt, iterations, keylen, digest);
  if (typeof callback !== "function") throw pirInvalidArgType("callback", "of type function", callback);
  globalThis.__pirMicrotask(() => callback(null, value));
}

// Node's scrypt defaults: N=16384, r=8, p=1, maxmem=32 MiB. The option object
// itself is not type-checked (Node accepts a string and reads nothing).
function scryptOptions(options) {
  const opts = options === undefined || options === null ? {} : options;
  const n = opts.N === undefined ? 16384 : pirValidateInteger(opts.N, "N", 0, 4294967295);
  const r = opts.r === undefined ? 8 : pirValidateInteger(opts.r, "r", 0, 4294967295);
  const p = opts.p === undefined ? 1 : pirValidateInteger(opts.p, "p", 0, 4294967295);
  const maxmem = opts.maxmem === undefined
    ? 32 * 1024 * 1024
    : pirValidateInteger(opts.maxmem, "maxmem", 0, 9007199254740991);
  const logN = Math.log2(n);
  if (n < 2 || !Number.isInteger(logN)) throw pirScryptParamsError();
  // OpenSSL accepts r=0/p=0 but the Rust `scrypt` crate does not; report the
  // same error class/code as the other invalid parameter shapes.
  if (r === 0 || p === 0) throw pirScryptParamsError();
  // OpenSSL's guard: 128 * r * (N + p + 2) bytes.
  const memory = 128 * r * (n + p + 2);
  if (memory > maxmem) {
    throw pirScryptParamsError(
      "Invalid scrypt params: error:030000AC:digital envelope routines::memory limit exceeded",
    );
  }
  return { logN: logN, r: r, p: p, maxmem: maxmem };
}

function scryptSync(password, salt, keylen, options) {
  const passwordBytes = toKeyBytes(password, "password");
  const saltBytes = toKeyBytes(salt, "salt");
  const length = pirValidateInteger(keylen, "keylen", 0, 2147483647);
  const opts = scryptOptions(options);
  return Buffer.from(call("crypto", "scrypt", {
    password: passwordBytes.toString("base64"),
    salt: saltBytes.toString("base64"),
    logN: opts.logN,
    r: opts.r,
    p: opts.p,
    maxmem: opts.maxmem,
    keylen: length,
  }), "base64");
}

function scrypt(password, salt, keylen, options, callback) {
  if (typeof options === "function") { callback = options; options = undefined; }
  // Node validates and starts the derivation synchronously; bad arguments
  // throw even when a callback is supplied (test-crypto-scrypt's `bad` loop).
  const value = scryptSync(password, salt, keylen, options);
  if (typeof callback !== "function") throw pirInvalidArgType("callback", "of type function", callback);
  globalThis.__pirMicrotask(() => callback(null, value));
}

// Real HKDF (RFC 5869) composed from the real HMAC chain. Node returns an
// ArrayBuffer from hkdfSync and requires both salt and info.
function pirHkdf(digest, ikm, salt, info, keylen) {
  if (typeof digest !== "string") {
    throw pirInvalidArgType("digest", "of type string", digest);
  }
  // Node accepts a secret KeyObject as `ikm` (not for salt/info).
  let ikmBytes;
  if (KeyObject !== undefined && ikm instanceof KeyObject) {
    if (ikm.type !== "secret" || typeof ikm._secretB64 !== "string") {
      throw pirInvalidArgType("ikm", PIR_IKM_EXPECTED, ikm);
    }
    ikmBytes = Buffer.from(ikm._secretB64, "base64");
  } else {
    ikmBytes = toKeyBytes(ikm, "ikm", PIR_IKM_EXPECTED);
  }
  // Node's HKDF requires explicit salt/info byte strings (undefined and null
  // are ERR_INVALID_ARG_TYPE, unlike older Node releases), and caps info at
  // 1024 bytes.
  const saltBytes = toKeyBytes(salt, "salt", PIR_HKDF_BYTES_EXPECTED);
  const infoBytes = toKeyBytes(info, "info", PIR_HKDF_BYTES_EXPECTED);
  if (infoBytes.length > 1024) {
    throw pirOutOfRange("info", "must not contain more than 1024 bytes", infoBytes.length);
  }
  const length = pirValidateInteger(keylen, "length", 0, 9007199254740991);
  const algo = normalizeDigest(digest);
  if (PIR_SHAKE_DEFAULTS[algo] !== undefined || PIR_HOST_HMACS[algo] !== true) {
    throw pirInvalidDigest(digest);
  }
  const prk = createHmac(algo, saltBytes).update(ikmBytes).digest();
  const hashLen = prk.length;
  const blocks = Math.ceil(length / hashLen);
  if (blocks > 255) {
    const error = new RangeError("Invalid key length");
    error.code = "ERR_CRYPTO_INVALID_KEYLEN";
    throw error;
  }
  const out = [];
  let previous = Buffer.alloc(0);
  for (let i = 1; i <= blocks; i++) {
    const block = createHmac(algo, prk)
      .update(Buffer.concat([previous, infoBytes, Buffer.from([i])]))
      .digest();
    out.push(block);
    previous = block;
  }
  const joined = Buffer.concat(out);
  const exact = new Uint8Array(length);
  exact.set(joined.subarray(0, length));
  return exact.buffer;
}

function hkdfSync(digest, ikm, salt, info, keylen) {
  return pirHkdf(digest, ikm, salt, info, keylen);
}

function hkdf(digest, ikm, salt, info, keylen, callback) {
  // Argument validation is synchronous in Node: a bad digest throws even when
  // a callback is supplied (test-crypto-hkdf's `assert.throws` cases).
  const value = pirHkdf(digest, ikm, salt, info, keylen);
  if (typeof callback !== "function") throw pirInvalidArgType("callback", "of type function", callback);
  globalThis.__pirMicrotask(() => callback(null, value));
}

function pirWebcryptoHashName(hash) {
  const name = typeof hash === "string" ? hash : (hash === undefined || hash === null ? undefined : hash.name);
  if (typeof name !== "string") {
    throw new TypeError("Failed to normalize algorithm: passed algorithm must include a hash");
  }
  if (PIR_WEBCRYPTO_HASHES[name] === undefined) {
    const error = new Error("Unrecognized algorithm name");
    error.name = "NotSupportedError";
    error.code = 9;
    throw error;
  }
  return name;
}

// The normalized algorithm dictionary WebCrypto hands back from key.algorithm
// (Node: { name, length, hash: { name } }).
function pirWebcryptoAlgorithm(algorithm) {
  const source = typeof algorithm === "string" ? { name: algorithm } : (algorithm || {});
  const name = source.name;
  if (typeof name !== "string") {
    throw new TypeError("The \"algorithm\" argument must be a string or an object with a name");
  }
  if (name.toUpperCase() !== "HMAC") return { name: name };
  const normalized = { name: name };
  if (source.length !== undefined) normalized.length = source.length;
  normalized.hash = { name: pirWebcryptoHashName(source.hash) };
  return normalized;
}

  const error = new Error(api + ": algorithm " + name + " is not supported by the pir extension runtime");
  error.code = "ERR_NOT_SUPPORTED";

}
