import { EventEmitter } from "@node/events";
import { StringDecoder } from "@node/string_decoder";

// node:readline surface. A non-terminal Interface has the real Node consumer
// path: it decodes the input's `data` chunks and splits them into `line`
// events (CRLF pairs kept together across chunk boundaries), flushing the
// last unterminated line on `end` before emitting `close`
// (test-stream-preprocess reads a file through createReadStream + readline).
// The cursor helpers write the real ANSI sequences to the stream; no raw
// keypress decoding happens (the runtime has no terminal input host).

class Interface extends EventEmitter {
  constructor(input, output, options) {
    super();
    let opts = options === undefined || options === null ? {} : options;
    // Modern form: createInterface({ input, output, terminal, ... }). Node
    // unwraps the options object before validating the real stream.
    if (input !== undefined && input !== null && typeof input === "object" &&
        typeof input.on !== "function" && input.input !== undefined) {
      opts = input;
      output = input.output;
      input = input.input;
    }
    if (input !== undefined && input !== null && typeof input !== "object") {
      throw new TypeError("The \"input\" argument must be a readable stream");
    }
    this.input = input;
    this.output = output !== undefined && output !== null ? output : input;
    this.terminal = Boolean(opts.terminal);
    // Stored as `_prompt`: real Node keeps the prompt text off the method
    // name (`this.prompt` must remain the callable method).
    this._prompt = opts.prompt !== undefined ? String(opts.prompt) : "> ";
    this.historySize = typeof opts.historySize === "number" ? opts.historySize : 30;
    this.history = Array.isArray(opts.history) ? opts.history.slice() : [];
    this.line = "";
    this.cursor = 0;
    this.closed = false;
    // Non-terminal input consumption (Node: 'data' + 'end' listeners, utf8
    // StringDecoder, line-ending split with a CRLF pair spanning chunks).
    this._decoder = new StringDecoder("utf8");
    this._lineBuffer = "";
    this._sawCR = false;
    this._input = undefined;
    if (this.terminal || input === undefined || input === null ||
        typeof input.on !== "function") {
      return;
    }
    const self = this;
    this._onData = function (chunk) { self._normalWrite(chunk); };
    this._onEnd = function () {
      if (self._lineBuffer.length > 0) {
        const line = self._lineBuffer;
        self._lineBuffer = "";
        self.emit("line", line);
      }
      self.close();
    };
    this._input = input;
    input.on("data", this._onData);
    input.on("end", this._onEnd);
    this.once("close", function () {
      if (typeof input.removeListener === "function") {
        input.removeListener("data", self._onData);
        input.removeListener("end", self._onEnd);
      }
    });
  }
  /// Split one input chunk into complete lines (Node's kNormalWrite).
  _normalWrite(chunk) {
    if (chunk === undefined) return;
    let string = typeof chunk === "string" ? chunk : this._decoder.write(chunk);
    if (this._sawCR) {
      this._sawCR = false;
      if (string.charCodeAt(0) === 10) string = string.slice(1);
    }
    if (!string) return;
    const lines = string.split(/\r\n|\r|\n|\u2028|\u2029/);
    if (lines.length === 1) {
      this._lineBuffer += string;
      return;
    }
    if (string.charCodeAt(string.length - 1) === 13) this._sawCR = true;
    let first = lines[0];
    if (this._lineBuffer) first = this._lineBuffer + first;
    const last = lines.length - 1;
    this._lineBuffer = lines[last];
    this.emit("line", first);
    for (let i = 1; i < last; i++) this.emit("line", lines[i]);
  }
  getPrompt() {
    return this._prompt;
  }
  setPrompt(prompt) {
    this._prompt = String(prompt);
  }
  prompt(_preserveCursor) {
    if (this.closed) return;
    if (this.output && typeof this.output.write === "function") {
      this.output.write(this._prompt);
    }
  }
  question(query, callback) {
    if (this.closed) return;
    this.prompt();
    if (typeof callback === "function") {
      // No terminal-input host: the honest answer is the empty line.
      globalThis.__pirMicrotask(() => callback(""));
    }
  }
  pause() {
    if (this._input && typeof this._input.pause === "function") this._input.pause();
    return this;
  }
  resume() {
    if (this._input && typeof this._input.resume === "function") this._input.resume();
    return this;
  }
  write(data, _key) {
    if (this.closed) {
      const error = new Error("Cannot write to a closed readline interface");
      error.code = "ERR_USE_AFTER_CLOSE";
      throw error;
    }
    if (data === undefined) return;
    // Non-terminal: the line is fed through the same splitting path as input.
    if (!this.terminal) this._normalWrite(data);
  }
  close() {
    if (this.closed) return;
    this.closed = true;
    this.emit("close");
  }
  refreshLine() {}
  getCursorPos() {
    return { rows: 0, cols: this.cursor };
  }
}

function createInterface(input, output, options) {
  if (input !== undefined && input !== null && typeof input === "object" &&
      typeof input.on !== "function" && input.input !== undefined) {
    // `createInterface({ input, output, terminal, ... })` options object.
    return new Interface(input);
  }
  if (input !== undefined && input !== null && typeof input === "object" && options === undefined && output !== undefined && output !== null && typeof output === "object" && typeof output.read !== "function") {
    // (input, options) shape.
    return new Interface(input, undefined, output);
  }
  return new Interface(input, output, options);
}

// --- ANSI cursor helpers (real writes) -----------------------------------------

function cursorTo(stream, x, y, callback) {
  if (typeof y === "function") {
    callback = y;
    y = undefined;
  }
  if (stream && typeof stream.write === "function") {
    stream.write(y === undefined ? `\u001b[${x + 1}G` : `\u001b[${y + 1};${x + 1}H`);
  }
  if (typeof callback === "function") globalThis.__pirMicrotask(() => callback(undefined));
  return true;
}

function moveCursor(stream, dx, dy, callback) {
  if (stream && typeof stream.write === "function") {
    if (dx > 0) stream.write(`\u001b[${dx}C`);
    else if (dx < 0) stream.write(`\u001b[${-dx}D`);
    if (dy > 0) stream.write(`\u001b[${dy}B`);
    else if (dy < 0) stream.write(`\u001b[${-dy}A`);
  }
  if (typeof callback === "function") globalThis.__pirMicrotask(() => callback(undefined));
  return true;
}

function clearLine(stream, dir, callback) {
  if (stream && typeof stream.write === "function") {
    const dirCode = dir === -1 ? "\u001b[1K" : dir === 1 ? "\u001b[0K" : "\u001b[2K";
    stream.write(dirCode + "\r");
  }
  if (typeof callback === "function") globalThis.__pirMicrotask(() => callback(undefined));
  return true;
}

function clearScreenDown(stream, callback) {
  if (stream && typeof stream.write === "function") {
    stream.write("\u001b[0J");
  }
  if (typeof callback === "function") globalThis.__pirMicrotask(() => callback(undefined));
  return true;
}

function emitKeypressEvents(_stream, _interface) {
  // No raw terminal-input host: keypress decoding never fires here.
}

// Promise-API module: an Interface whose question resolves via Promise.
const promises = {
  Interface: Interface,
  createInterface(input, output, options) {
    if (input !== undefined && input !== null && typeof input === "object" && typeof input.read !== "function") {
      return new PromiseInterface(undefined, undefined, input);
    }
    return new PromiseInterface(input, output, options);
  },
};

class PromiseInterface extends Interface {
  question(query, options) {
    return new Promise((resolve) => {
      this.prompt();
      const answer = options && typeof options.signal === "object" && options.signal.aborted
        ? ""
        : "";
      globalThis.__pirMicrotask(() => resolve(answer));
    });
  }
}

const readline = {
  Interface: Interface,
  createInterface: createInterface,
  cursorTo: cursorTo,
  moveCursor: moveCursor,
  clearLine: clearLine,
  clearScreenDown: clearScreenDown,
  emitKeypressEvents: emitKeypressEvents,
  promises: promises,
};

export default readline;
export { Interface, createInterface, cursorTo, moveCursor, clearLine, clearScreenDown, emitKeypressEvents, promises };
