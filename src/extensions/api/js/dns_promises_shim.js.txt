
const call = globalThis.__pirNodeCall;

function pirDnsNotFound(host) {
  const error = new Error("getaddrinfo ENOTFOUND " + host);
  error.code = "ENOTFOUND";
  error.syscall = "getaddrinfo";
  error.hostname = host;
  return error;
}

function pirDnsUnsupported(api) {
  // The async faces reject (Node's dns/promises never throws synchronously);
  // one clear code/message per unsupported query surface.
  return function () {
    const error = new Error("dns.promises." + api + " is not supported by the pir extension runtime");
    error.code = "ERR_NOT_SUPPORTED";
    return Promise.reject(error);
  };
}

// Real lookup over the host resolver: single {address, family} (or the full
// array with `all: true`), family-filtered, ENOTFOUND when nothing matches.
function lookup(hostname, options) {
  const host = String(hostname);
  const opts = options === undefined || options === null ? {} : options;
  try {
    const resolved = call("dns", "lookup", { host: host });
    const wanted = opts.family === 4 || opts.family === 6 ? opts.family : 0;
    const addresses = (resolved.addresses || []).filter((entry) => wanted === 0 || entry.family === wanted);
    if (addresses.length === 0) throw pirDnsNotFound(host);
    if (opts.all) return Promise.resolve(addresses);
    return Promise.resolve({ address: addresses[0].address, family: addresses[0].family });
  } catch (error) {
    return Promise.reject(error);
  }
}

class Resolver {
  constructor() {
    // Construction throws synchronously: an inert Resolver instance would
    // silently swallow every later query.
    const error = new Error("dns.promises.Resolver is not supported by the pir extension runtime");
    error.code = "ERR_NOT_SUPPORTED";
    throw error;
  }
}

const resultOrderState = { order: "ipv4first" };

function getDefaultResultOrder() {
  return resultOrderState.order;
}

function setDefaultResultOrder(order) {
  if (order !== "ipv4first" && order !== "verbatim") {
    throw new TypeError("The \"order\" argument must be ipv4first or verbatim");
  }
  resultOrderState.order = order;
}
