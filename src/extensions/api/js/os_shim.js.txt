
const call = globalThis.__pirNodeCall;
const process = globalThis.process;
const pirPlatformValue = globalThis.__pirPlatform;
// Real node exposes os.platform as a function; the raw string stays
// available as the __pirPlatform engine global.
function platform() { return pirPlatformValue; }
const EOL = pirPlatformValue === "win32" ? "\r\n" : "\n";
const devNull = pirPlatformValue === "win32" ? "\\\\.\\nul" : "/dev/null";
const TOTAL_MEMORY = 17179869184;
const FREE_MEMORY = 8589934592;

function arch() { return process.arch; }

function type() {
  if (pirPlatformValue === "win32") return "Windows_NT";
  if (pirPlatformValue === "darwin") return "Darwin";
  return "Linux";
}

function release() { return ""; }

function hostname() { return String(call("os", "hostname")); }

function homedir() { return String(call("os", "homedir")); }

function tmpdir() { return String(call("os", "tmpdir")); }

function cpus() {
  const count = Number(call("os", "cpus_count")) || 1;
  const out = [];
  for (let i = 0; i < count; i++) out.push({ model: "unknown", speed: 0 });
  return out;
}

function totalmem() { return TOTAL_MEMORY; }
function freemem() { return FREE_MEMORY; }

function endianness() {
  const probe = new Uint8Array(new Uint16Array([1]).buffer);
  return probe[0] === 1 ? "LE" : "BE";
}

function userInfo() {
  const env = call("os", "env");
  return {
    uid: -1,
    gid: -1,
    username: env.USERNAME || env.USER || "",
    homedir: homedir(),
    shell: null,
  };
}

function networkInterfaces() { return {}; }

};

// Real fields where the host provides them; loadavg returns Node's win32
// convention (always zeros), uptime rides the process clock.
const availableParallelism = () => 1;

function uptime() {
  // The runtime has no boot-time host op; the clock starts at bootstrap.
  return Math.floor((Date.now() - (globalThis.__pirBootTime || (globalThis.__pirBootTime = Date.now()))) / 1000);
}

function loadavg() {
  return [0, 0, 0];
}

function machine() {
  return process.arch === "x64" ? "x86_64" : process.arch === "arm64" ? "arm64" : process.arch;
}

function version() {
  // The host exposes no OS product string; the honest value is empty.
  return "";
}

function getPriority(pid) {
  if (pid !== undefined && (typeof pid !== "number" || (pid | 0) !== pid)) {
    throw new TypeError("pid must be an integer");
  }
  return 0;
}

function setPriority(_pid, _priority) {
  // No scheduling control in the runtime; a valid call is a no-op.
}

};

const osWithConstants = Object.assign({}, os, {

  getPriority: getPriority,
  setPriority: setPriority,
});

// Node publishes `EOL`/`devNull` as non-writable (configurable) data
// properties: `os.EOL = 123` is a TypeError in strict mode, while
// `Object.defineProperties(os, { EOL: {...} })` still works. The object
// literal above made them plain writable properties.
for (const name of ["EOL", "devNull"]) {
  Object.defineProperty(osWithConstants, name, {
    value: name === "EOL" ? EOL : devNull,
    writable: false,
    enumerable: true,
    configurable: true,
  });
}

export default osWithConstants;
