
// TextEncoder / TextDecoder — engine globals (LLRT tier: real for the
// supported label set). Pure JS: the codecs are the handwritten state
// machines the buffer/string_decoder shims use (the QuickJS engine has no
// native TextDecoder/Intl), lifted here into standalone classes.
//
// Supported labels: utf-8 (with fatal + streaming + BOM handling), latin1
// (iso-8859-1/ascii/binary aliases — byte->U+00xx, NOT windows-1252, a
// documented deviation from the WHATWG label table) and utf-16le
// (utf-16/ucs-2/ucs2 aliases, little-endian only). Anything else throws a
// RangeError like the Web platform does for unknown labels.

(function () {
  // Host fast path (§6.5 host-first policy): when the pir extension runtime
  // registered its native codec (`__pirBufferCodec`, see
  // src/extensions/runtime/host_functions/buffer_ops.rs), the decode cores
  // below dispatch to it; `undefined` results (unmarshalable inputs like
  // lone surrogates) or thrown errors fall back to the JS state machines —
  // behavior never changes either way.
  const pirNativeCodec = typeof globalThis.__pirBufferCodec === "function" ? globalThis.__pirBufferCodec : null;

  function pirToBytes(input) {
    if (typeof ArrayBuffer !== "undefined" && input instanceof ArrayBuffer) {
      return new Uint8Array(input.slice(0));
    }
    if (ArrayBuffer.isView(input)) {
      return new Uint8Array(
        input.buffer.slice(input.byteOffset, input.byteOffset + input.byteLength)
      );
    }
    throw new TypeError("The provided value is not of type '(ArrayBuffer or ArrayBufferView)'");
  }

  function pirNormalizeLabel(label) {
    const text = String(label === undefined || label === null ? "utf-8" : label)
      .trim()
      .toLowerCase();
    if (text === "utf-8" || text === "utf8" || text === "unicode-1-1-utf-8") return "utf-8";
    if (text === "latin1" || text === "iso-8859-1" || text === "ascii" ||
        text === "us-ascii" || text === "binary" || text === "windows-1252" ||
        text === "iso-8859-1-windows-3-1-latin-1") return "latin1";
    if (text === "utf-16le" || text === "utf-16" || text === "ucs-2" || text === "ucs2") {
      return "utf-16le";
    }
    throw new RangeError("The encoding label provided ('" + label + "') is invalid.");
  }

  function pirEncodeIntoUtf8(text, target, targetStart) {
    // Returns {read, written}: `read` counts code units consumed, `written`
    // bytes. A character that would not fit is left for the next call; lone
    // surrogates encode as U+FFFD (WHATWG parity).
    let read = 0;
    let written = 0;
    let i = 0;
    const length = text.length;
    while (i < length) {
      let cp = text.codePointAt(i);
      let units = cp > 0xffff ? 2 : 1;
      let bytes;
      if (cp >= 0xd800 && cp <= 0xdfff) {
        cp = 0xfffd;
        bytes = 3;
      } else if (cp < 0x80) {
        bytes = 1;
      } else if (cp < 0x800) {
        bytes = 2;
      } else if (cp < 0x10000) {
        bytes = 3;
      } else {
        bytes = 4;
      }
      if (written + bytes > target.length - targetStart) break;
      if (bytes === 1) {
        target[targetStart + written++] = cp;
      } else if (bytes === 2) {
        target[targetStart + written++] = 0xc0 | (cp >> 6);
        target[targetStart + written++] = 0x80 | (cp & 63);
      } else if (bytes === 3) {
        target[targetStart + written++] = 0xe0 | (cp >> 12);
        target[targetStart + written++] = 0x80 | ((cp >> 6) & 63);
        target[targetStart + written++] = 0x80 | (cp & 63);
      } else {
        target[targetStart + written++] = 0xf0 | (cp >> 18);
        target[targetStart + written++] = 0x80 | ((cp >> 12) & 63);
        target[targetStart + written++] = 0x80 | ((cp >> 6) & 63);
        target[targetStart + written++] = 0x80 | (cp & 63);
      }
      i += units;
      read += units;
    }
    return { read: read, written: written };
  }

  function pirUtf8Bytes(text) {
    // Straight encoding (no target-size limit): one pass over code points.
    const out = [];
    for (let i = 0; i < text.length; i++) {
      let cp = text.codePointAt(i);
      if (cp > 0xffff) i++;
      if (cp >= 0xd800 && cp <= 0xdfff) cp = 0xfffd;
      if (cp < 0x80) out.push(cp);
      else if (cp < 0x800) out.push(0xc0 | (cp >> 6), 0x80 | (cp & 63));
      else if (cp < 0x10000) {
        out.push(0xe0 | (cp >> 12), 0x80 | ((cp >> 6) & 63), 0x80 | (cp & 63));
      } else {
        out.push(0xf0 | (cp >> 18), 0x80 | ((cp >> 12) & 63), 0x80 | ((cp >> 6) & 63), 0x80 | (cp & 63));
      }
    }
    return Uint8Array.from(out);
  }

  function pirDecodeUtf8Chunk(bytes, state) {
    // bytes: Uint8Array of the current chunk; state carries the incomplete
    // trailing sequence across streaming decode() calls.
    let all;
    if (state.pending.length > 0) {
      all = new Uint8Array(state.pending.length + bytes.length);
      all.set(state.pending, 0);
      all.set(bytes, state.pending.length);
      state.pending = new Uint8Array(0);
    } else {
      all = bytes;
    }
    let out = "";
    let i = 0;
    while (i < all.length) {
      const b0 = all[i];
      if (b0 < 0x80) {
        out += String.fromCharCode(b0);
        i += 1;
        continue;
      }
      let need;
      if (b0 >= 0xc2 && b0 <= 0xdf) need = 1;
      else if (b0 >= 0xe0 && b0 <= 0xef) need = 2;
      else if (b0 >= 0xf0 && b0 <= 0xf4) need = 3;
      else {
        if (state.fatal) throw new TypeError("The encoded data is not valid.");
        out += "\uFFFD";
        i += 1;
        continue;
      }
      if (all.length - i - 1 < need) {
        // Incomplete trailing sequence: carried to the next chunk when
        // streaming, replaced (or fatal) when the stream ends.
        state.pending = all.slice(i);
        break;
      }
      let valid = true;
      for (let k = 1; k <= need; k++) {
        const cont = all[i + k];
        if (cont < 0x80 || cont > 0xbf) {
          valid = false;
          break;
        }
      }
      if (!valid) {
        if (state.fatal) throw new TypeError("The encoded data is not valid.");
        out += "\uFFFD";
        i += 1;
        continue;
      }
      let cp;
      if (need === 1) cp = ((b0 & 0x1f) << 6) | (all[i + 1] & 0x3f);
      else if (need === 2) {
        cp = ((b0 & 0x0f) << 12) | ((all[i + 1] & 0x3f) << 6) | (all[i + 2] & 0x3f);
      } else {
        cp = ((b0 & 0x07) << 18) | ((all[i + 1] & 0x3f) << 12) |
          ((all[i + 2] & 0x3f) << 6) | (all[i + 3] & 0x3f);
      }
      out += String.fromCodePoint(cp);
      i += need + 1;
    }
    return out;
  }

  class TextEncoder {
    constructor() {
      this.encoding = "utf-8";
    }
    encode(input) {
      const text = input === undefined || input === null ? "" : String(input);
      // Native core (rquickjs C-ABI, installed by __pirInstallTextCodec):
      // returns a Uint8Array, or undefined when the string cannot cross the
      // host bridge (lone surrogates) — the JS state machine is the fallback.
      const native = TextEncoder.prototype.__pirNativeEncode;
      if (native) {
        try {
          const out = native(text);
          if (out !== undefined) return out;
        } catch (error) { /* fall through to the JS encoder */ }
      }
      return pirUtf8Bytes(text);
    }
    encodeInto(source, target) {
      if (!(target instanceof Uint8Array)) {
        throw new TypeError("The provided value is not of type '(ArrayBuffer or ArrayBufferView)'");
      }
      const text = source === undefined || source === null ? "" : String(source);
      const native = TextEncoder.prototype.__pirNativeEncodeInto;
      if (native) {
        try {
          const out = native(text, target);
          if (out !== undefined) return out;
        } catch (error) { /* fall through to the JS encoder */ }
      }
      return pirEncodeIntoUtf8(text, target, 0);
    }
  }

  class TextDecoder {
    constructor(label, options) {
      this.encoding = pirNormalizeLabel(label);
      options = options || {};
      this.fatal = options.fatal === true;
      this.ignoreBOM = options.ignoreBOM === true;
      this.__pending = new Uint8Array(0);
      this.__first = true;
    }
    decode(input, options) {
      const stream = !!(options && options.stream);
      let bytes = input === undefined || input === null ? new Uint8Array(0) : pirToBytes(input);
      if (this.__first && !this.ignoreBOM && this.encoding === "utf-8" &&
          bytes.length >= 3 && bytes[0] === 0xef && bytes[1] === 0xbb && bytes[2] === 0xbf) {
        bytes = bytes.slice(3);
      }
      if (this.__first && !this.ignoreBOM && this.encoding === "utf-16le" &&
          bytes.length >= 2 && bytes[0] === 0xff && bytes[1] === 0xfe) {
        bytes = bytes.slice(2);
      }
      this.__first = false;
      let out;
      if (this.encoding === "utf-8") {
        if (pirNativeCodec) {
          // Native WHATWG utf-8 core: we merge the pending bytes into one
          // buffer, hand it over, and get {text, consumed} back — an incomplete trailing sequence stays
          // unconsumed while streaming and becomes U+FFFD (or a fatal
          // TypeError) at end of stream. Bytes never fail to marshal, so no
          // fallback path is needed here.
          let all = bytes;
          if (this.__pending.length > 0) {
            all = new Uint8Array(this.__pending.length + bytes.length);
            all.set(this.__pending, 0);
            all.set(bytes, this.__pending.length);
          }
          const result = pirNativeCodec("utf8DecodeChunk", all, !stream, this.fatal);
          this.__pending = result.consumed < all.length ? all.slice(result.consumed) : new Uint8Array(0);
          return result.text;
        }
        const state = { pending: this.__pending, fatal: this.fatal };
        out = pirDecodeUtf8Chunk(bytes, state);
        this.__pending = state.pending;
        if (!stream && this.__pending.length > 0) {
          // End of the stream: an incomplete trailing sequence becomes
          // U+FFFD (or a fatal TypeError).
          if (this.fatal) throw new TypeError("The encoded data is not valid.");
          out += "\uFFFD";
          this.__pending = new Uint8Array(0);
        }
        return out;
      }
      if (this.encoding === "latin1") {
        if (pirNativeCodec) {
          try {
            const latin = pirNativeCodec("latin1Decode", bytes);
            if (typeof latin === "string") return latin;
          } catch (error) { /* fall through to the JS loop */ }
        }
        let latin = "";
        for (let i = 0; i < bytes.length; i++) latin += String.fromCharCode(bytes[i] & 0xff);
        return latin;
      }
      // utf-16le: pair bytes; an odd trailing byte is carried when streaming
      // and dropped at the end (Node parity).
      const all = this.__pending.length > 0
        ? Uint8Array.from(Array.from(this.__pending).concat(Array.from(bytes)))
        : bytes;
      this.__pending = new Uint8Array(0);
      if (pirNativeCodec) {
        try {
          const result = pirNativeCodec("utf16leDecodeChunk", all);
          if (typeof result === "object" && result !== null) {
            if (stream && result.consumed < all.length) this.__pending = all.slice(result.consumed);
            return result.text;
          }
          // Unpaired surrogates (unmarshalable): fall through to the JS loop.
        } catch (error) { /* fall through to the JS loop */ }
      }
      let text = "";
      let i = 0;
      for (; i + 1 < all.length; i += 2) {
        text += String.fromCharCode(all[i] | (all[i + 1] << 8));
      }
      if (i < all.length) {
        if (stream) this.__pending = all.slice(i);
      }
      return text;
    }
  }

  globalThis.TextEncoder ?? (globalThis.TextEncoder = TextEncoder);
  globalThis.TextDecoder ?? (globalThis.TextDecoder = TextDecoder);

  // Host fast path: native encode/encodeInto cores onto the prototype (see
  // buffer_ops.rs). The class methods stay the entry points so the decoder's
  // per-instance streaming state remains here; the native hooks are advisory
  // (undefined result ⇒ JS state machine). A failed install changes nothing.
  if (typeof globalThis.__pirInstallTextCodec === "function") {
    try {
      globalThis.__pirInstallTextCodec(TextEncoder.prototype, TextDecoder.prototype);
    } catch (error) {
      /* native install failed: the JS implementations above stay active */
    }
    delete globalThis.__pirInstallTextCodec;
  }
})();
