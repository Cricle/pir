// ---------------------------------------------------------------------------
// Node-shaped argument errors
// ---------------------------------------------------------------------------

// `util.inspect`-shaped scalar rendering for the `Received …` tail of
// ERR_OUT_OF_RANGE (numbers print bare, strings quoted) — the forms the zlib
// option validators below need.
function pirShow(value) {
  if (typeof value === "string") return "'" + value + "'";
  return String(value);
}

// Node's `determineSpecificType` tail for ERR_INVALID_ARG_TYPE.
function pirTypeTail(value) {
  if (value === null) return "null";
  const type = typeof value;
  if (type === "undefined") return "undefined";
  if (type === "string") return "type string (" + pirShow(value) + ")";
  if (type === "number") return "type number (" + String(value) + ")";
  if (type === "boolean") return "type boolean (" + String(value) + ")";
  if (type === "bigint") return "type bigint (" + String(value) + ")";
  if (type === "symbol") return "type symbol (" + String(value) + ")";
  if (type === "function") return "function " + (value.name === undefined ? "" : value.name);
  if (Buffer.isBuffer(value)) return "an instance of Buffer";
  if (value instanceof ArrayBuffer) return "an instance of ArrayBuffer";
  if (ArrayBuffer.isView(value)) {
    return "an instance of " + (value.constructor && value.constructor.name ? value.constructor.name : "TypedArray");
  }
  const ctor = value.constructor;
  return "an instance of " + (ctor && ctor.name ? ctor.name : "Object");
}

function pirInvalidArgType(name, expected, value) {
  const kind = name.indexOf(".") === -1 ? "argument" : "property";
  const error = new TypeError(
    "The \"" + name + "\" " + kind + " must be " + expected +
    ". Received " + pirTypeTail(value),
  );
  error.code = "ERR_INVALID_ARG_TYPE";
  return error;
}

function pirOutOfRange(name, range, value) {
  const error = new RangeError(
    "The value of \"" + name + "\" is out of range. It must be " + range +
    ". Received " + pirShow(value),
  );
  error.code = "ERR_OUT_OF_RANGE";
  return error;
}

function zlibBytes(data, encoding) {
  if (typeof data === "string") return Buffer.from(data, encoding === undefined ? "utf8" : encoding);
  if (data instanceof ArrayBuffer) return Buffer.from(new Uint8Array(data));
  // Every ArrayBuffer view kind is a valid chunk in node (test-zlib-convenience-methods
  // feeds Int8Array..DataView), so the check is the view brand, not Uint8Array.
  if (ArrayBuffer.isView(data)) {
    return Buffer.from(new Uint8Array(data.buffer, data.byteOffset, data.byteLength));
  }
  throw pirInvalidArgType(
    "chunk",
    "of type string or an instance of Buffer, TypedArray, DataView, or ArrayBuffer",
    data,
  );
}

// ---------------------------------------------------------------------------
// Option validation (zlib option ranges are the only guard the convenience
// methods get before the host codec runs; Node validates them up front).
// ---------------------------------------------------------------------------

/// One `options.<name>` numeric range check in Node's order: type first, then
/// finiteness, then the [min, max] window (`max === Infinity` collapses the
/// range text to `>= min`, exactly like `checkRangesOrGetDefault`).
function pirZlibOption(options, name, min, max) {
  if (options === undefined || options === null) return undefined;
  const value = options[name];
  if (value === undefined) return undefined;
  if (typeof value !== "number") {
    throw pirInvalidArgType("options." + name, "of type number", value);
  }
  if (!Number.isFinite(value)) {
    throw pirOutOfRange("options." + name, "a finite number", value);
  }
  if (value < min || value > max) {
    throw pirOutOfRange(
      "options." + name,
      ">= " + min + (max === Infinity ? "" : " and <= " + max),
      value,
    );
  }
  return value;
}

/// Node's `zlib` dictionary accepts every ArrayBuffer view kind plus a raw
/// ArrayBuffer.
function pirZlibDictionary(options) {
  if (options === undefined || options === null) return undefined;
  const dictionary = options.dictionary;
  if (dictionary === undefined) return undefined;
  if (dictionary instanceof ArrayBuffer || ArrayBuffer.isView(dictionary)) return dictionary;
  throw pirInvalidArgType(
    "options.dictionary",
    "an instance of Buffer, TypedArray, DataView, or ArrayBuffer",
    dictionary,
  );
}

/// The option surface shared by every deflate/inflate family constructor.
function pirZlibValidateOptions(options) {
  const opts = options === undefined || options === null ? {} : options;
  pirZlibOption(opts, "chunkSize", 64, Infinity);
  pirZlibOption(opts, "windowBits", 8, 15);
  pirZlibOption(opts, "level", -1, 9);
  pirZlibOption(opts, "memLevel", 1, 9);
  pirZlibOption(opts, "strategy", 0, 4);
  pirZlibDictionary(opts);
  return opts;
}

/// `zlib.<Family>#params(level, strategy)` — Node's ranges for the deflate
/// family (zlib.h's Z_MIN_LEVEL..Z_MAX_LEVEL / Z_DEFAULT_STRATEGY..Z_FIXED).
function pirZlibParams(self, level, strategy) {
  if (typeof level !== "number") {
    throw pirInvalidArgType("level", "of type number", level);
  }
  if (!Number.isFinite(level)) throw pirOutOfRange("level", "a finite number", level);
  if (level < -1 || level > 9) throw pirOutOfRange("level", ">= -1 and <= 9", level);
  if (strategy !== undefined) {
    if (typeof strategy !== "number") {
      throw pirInvalidArgType("strategy", "of type number", strategy);
    }
    if (!Number.isFinite(strategy)) throw pirOutOfRange("strategy", "a finite number", strategy);
    if (strategy < 0 || strategy > 4) throw pirOutOfRange("strategy", ">= 0 and <= 4", strategy);
  }
  self._level = level;
  self._strategy = strategy === undefined ? 0 : strategy;
  return undefined;
}

// Host op names as routed by __pirNodeCall: the gzip/zlib-wrapped family
// lives on the legacy nodeExt table, the raw-DEFLATE pair on the primary
// table (see js/node_call_helper.js.txt). The brotli/zstd ops have no entry
// here — they keep their own name and fall through to the legacy table
// (`CODEC_OPS[resolved] || resolved`), where child_fns.rs::zlib_op serves them.
const CODEC_OPS = {
  gzip: "gzip",
  gunzip: "gunzip",
  deflate: "deflate",
  inflate: "inflate",
  deflateRaw: "deflateRawSync",
  inflateRaw: "inflateRawSync",
};

/// The option window of the modern codecs (brotli/zstd): node validates
/// `chunkSize` (Z_MIN_CHUNK..Z_MAX_CHUNK) and the two operation knobs
/// (`flush`/`finishFlush`, 0..=BROTLI_OPERATION_EMIT_METADATA for brotli,
/// 0..=ZSTD_e_end for zstd). The deflate-family window
/// (level/windowBits/memLevel/strategy) does NOT apply — real node ignores
/// those keys here, so they must not be rejected.
function pirZlibModernOptions(op, options) {
  const opts = options === undefined || options === null ? {} : options;
  const isZstd = op.indexOf("zstd") === 0;
  const flushMax = isZstd ? 2 : 3;
  pirZlibOption(opts, "chunkSize", 64, Infinity);
  pirZlibOption(opts, "flush", 0, flushMax);
  pirZlibOption(opts, "finishFlush", 0, flushMax);
  return opts;
}

/// The encoder knobs the host ops understand, read from `options.params` by
/// node's integer keys on a plain object (a Map is ignored, as in node):
/// BROTLI_PARAM_QUALITY(1) / BROTLI_PARAM_LGWIN(2) and
/// ZSTD_c_compressionLevel(100). `options.level` is deliberately not read —
/// probed on node v24.19.0, `zstdCompressSync(_, { level: 1 | 19 })` produces
/// byte-identical output (only `params` reaches the encoder), and brotli does
/// not have a `level` key at all.
function pirCodecKnobs(op, options) {
  const params = options === undefined || options === null ? undefined : options.params;
  if (params === undefined || params === null || typeof params !== "object") return undefined;
  if (op === "brotliCompress") {
    const knobs = {};
    if (typeof params[1] === "number") knobs.quality = params[1];
    if (typeof params[2] === "number") knobs.lgwin = params[2];
    return knobs;
  }
  if (op === "zstdCompress" && typeof params[100] === "number") {
    return { level: params[100] };
  }
  return undefined;
}

function zlibCodec(op, data, encoding, options) {
  const bytes = zlibBytes(data, encoding);
  const resolved = pirResolveCodecOp(op, bytes);
  const payload = { data: bytes.toString("base64") };
  const knobs = pirCodecKnobs(resolved, options);
  if (knobs !== undefined) {
    for (const key of Object.keys(knobs)) payload[key] = knobs[key];
  }
  return Buffer.from(call("zlib", CODEC_OPS[resolved] || resolved, payload), "base64");
}

// `unzip` sniffs the container instead of assuming one: the host decoders are
// single-format (flate2's GzDecoder rejects a zlib header and vice versa),
// while Node's unzip (windowBits 15+32) accepts either. The gzip magic number
// is the whole test — the zlib header has no fixed signature beyond the
// 0x??8x CMF/FLG pair, and every non-gzip input Node accepts is zlib-wrapped.
// `gunzip` stays strict (Node: gunzipSync(deflateSync(x)) throws Z_DATA_ERROR),
// so unzip is its own binding and not the former `unzip = gunzip` alias.
// tests/node_shims.rs still pins that alias (`unzip must alias gunzip`) and
// needs its one-line expectation flipped — see scripts/node_tests/REPORT.md.
function pirResolveCodecOp(op, bytes) {
  if (op !== "unzip") return op;
  return bytes.length >= 2 && bytes[0] === 0x1f && bytes[1] === 0x8b ? "gunzip" : "inflate";
}

/// `options.info` wraps the result as `{ buffer, engine }` with a fresh
/// transform of the family Node would have driven (test-zlib-convenience-methods
/// checks `engine instanceof zlib.<Family>`).
function zlibInfoWrap(op, buffer, options) {
  if (options === undefined || options === null || !options.info) return buffer;
  const Ctor = INFO_ENGINE[op];
  return { buffer: buffer, engine: Ctor === undefined ? undefined : new Ctor(options) };
}

// Codec-family option validation, run synchronously before the host call:
// node constructs the transform eagerly, so an out-of-range `chunkSize` throws
// even from the callback form.
const MODERN_CODEC_OPS = {
  brotliCompress: true,
  brotliDecompress: true,
  zstdCompress: true,
  zstdDecompress: true,
};

function pirValidateCodecOptions(op, options) {
  return MODERN_CODEC_OPS[op] ? pirZlibModernOptions(op, options) : options;
}

function zlibSync(op, data, options) {
  const validated = pirValidateCodecOptions(op, options);
  const encoding = validated === undefined || validated === null ? undefined : validated.encoding;
  return zlibInfoWrap(op, zlibCodec(op, data, encoding, validated), validated);
}

function zlibCallback(op, data, options, callback) {
  let encoding;
  if (typeof options === "function") { callback = options; options = undefined; }
  else if (options !== undefined && options !== null) encoding = options.encoding;
  const done = typeof callback === "function" ? callback : function () {};
  // Node requires the callback: gzip('abc') with no callback is an
  // ERR_INVALID_ARG_TYPE, not a silent no-op.
  if (typeof callback !== "function") {
    throw pirInvalidArgType("callback", "of type function", callback);
  }
  options = pirValidateCodecOptions(op, options);
  globalThis.__pirMicrotask(() => {
    let out;
    try {
      out = zlibInfoWrap(op, zlibCodec(op, data, encoding, options), options);
    } catch (error) {
      done(error);
      return;
    }
    done(null, out);
  });
}

// ---------------------------------------------------------------------------
// Transform-style stream classes (buffered whole-input minimal streams)
// ---------------------------------------------------------------------------

// LIMIT (documented): each class buffers every `write()` and runs one host
// codec call at `end()`, then emits the whole output as a single `data`
// chunk — not byte-faithful streaming (see fs ReadStream for the same
// posture), but real compression with working write/end/pipe plumbing.
class PirZlibTransform extends EventEmitter {
  constructor(op, options, validate) {
    super();
    this._op = op;
    this._options = (validate === undefined ? pirZlibValidateOptions : validate)(options);
    this._chunks = [];
    this._ended = false;
    this._level = this._options.level === undefined ? -1 : this._options.level;
    this._strategy = this._options.strategy === undefined ? 0 : this._options.strategy;
    this.bytesWritten = 0;
    this.bytesRead = 0;
  }
  write(chunk, encoding, callback) {
    if (typeof encoding === "function") { callback = encoding; encoding = undefined; }
    if (this._ended) {
      const error = new Error("write after end");
      error.code = "ERR_STREAM_WRITE_AFTER_END";
      if (typeof callback === "function") { globalThis.__pirMicrotask(() => callback(error)); return false; }
      throw error;
    }
    const bytes = zlibBytes(chunk, encoding);
    this._chunks.push(bytes);
    this.bytesWritten += bytes.length;
    if (typeof callback === "function") globalThis.__pirMicrotask(callback);
    return true;
  }
  end(chunk, encoding, callback) {
    if (typeof chunk === "function") { callback = chunk; chunk = undefined; encoding = undefined; }
    else if (typeof encoding === "function") { callback = encoding; encoding = undefined; }
    if (chunk !== undefined && chunk !== null) this.write(chunk, encoding);
    this._ended = true;
    globalThis.__pirMicrotask(() => {
      let out;
      try {
        out = zlibCodec(this._op, Buffer.concat(this._chunks), undefined, this._options);
      } catch (error) {
        this.emit("error", error);
        return;
      }
      this.bytesRead = out.length;
      if (out.length > 0) this.emit("data", out);
      this.emit("end");
      this.emit("finish");
      if (typeof callback === "function") callback();
    });
    return this;
  }
  flush(callback) {
    if (typeof callback === "function") globalThis.__pirMicrotask(callback);
    return this;
  }
  pipe(destination, options) {
    this.on("data", (chunk) => destination.write(chunk));
    this.on("end", () => {
      if (!options || options.end !== false) destination.end();
    });
    return destination;
  }
  close(callback) {
    if (typeof callback === "function") globalThis.__pirMicrotask(callback);
    return this;
  }
  destroy(error) {
    this._ended = true;
    if (error) this.emit("error", error);
    this.emit("close");
    return this;
  }
}

// Node's `params(level, strategy)` lives on the deflate family (`Zlib`) and on
// the brotli pair; the zstd engines have no such method (probed on v24.19.0:
// `ZstdCompress.prototype.params === undefined`). The ranges come from zlib.h
// (Z_MIN_LEVEL..Z_MAX_LEVEL, Z_DEFAULT_STRATEGY..Z_FIXED).
const PARAMS_ENGINES = [Gzip, Gunzip, Deflate, DeflateRaw, Inflate, InflateRaw, Unzip,
                        BrotliCompress, BrotliDecompress];
for (const impl of PARAMS_ENGINES) {
  Object.defineProperty(impl.prototype, "params", {
    value: function (level, strategy) { return pirZlibParams(this, level, strategy); },
    writable: true,
    enumerable: false,
    configurable: true,
  });
}

// node's zlib stream constructors are callable without `new` (`Deflate()`
// instanceof Deflate — test-zlib-deflate-constructors.js), and its brotli pair
// too; a `class` is not, so the exported bindings are these thin callable forms
// that share the class prototype (instanceof / `constructor` unchanged). The
// zstd pair stays a plain class: node's throws when called without `new`.
function pirCallableCtor(impl) {
  const callable = function (...args) {
    return new.target === undefined ? new impl(...args) : Reflect.construct(impl, args, new.target);
  };
  callable.prototype = impl.prototype;
  impl.prototype.constructor = callable;
  Object.defineProperty(callable, "name", { value: impl.name, configurable: true });
  Object.defineProperty(callable, "length", { value: impl.length, configurable: true });
  return callable;
}

const CRC_TABLE = (() => {
  const table = new Int32Array(256);
  for (let n = 0; n < 256; n++) {
    let c = n;
    for (let k = 0; k < 8; k++) c = (c & 1) ? (0xedb88320 ^ (c >>> 1)) : (c >>> 1);
    table[n] = c;
  }
  return table;
})();

// Real CRC-32 (the zlib/pkzip polynomial), unsigned 32-bit result; `value`
// is the running CRC for incremental hashing (Node semantics). Both arguments
// are validated like Node: `data` takes string/Buffer/TypedArray/DataView (a
// bare ArrayBuffer is rejected), `value` must be a number.
function crc32(data, value) {
  let bytes;
  if (typeof data === "string") {
    bytes = Buffer.from(data, "utf8");
  } else if (Buffer.isBuffer(data) || ArrayBuffer.isView(data)) {
    bytes = Buffer.from(new Uint8Array(data.buffer, data.byteOffset, data.byteLength));
  } else {
    throw pirInvalidArgType(
      "data",
      "of type string or an instance of Buffer, TypedArray, or DataView",
      data,
    );
  }
  if (value !== undefined && typeof value !== "number") {
    throw pirInvalidArgType("value", "of type number", value);
  }
  let crc = ((value === undefined ? 0 : value) | 0) ^ -1;
  for (let i = 0; i < bytes.length; i++) {
    crc = (crc >>> 8) ^ CRC_TABLE[(crc ^ bytes[i]) & 0xff];
  }
  return (crc ^ -1) >>> 0;
}

// Node's zlib tables: `constants` is a null-prototype object whose entries are
// all non-writable, non-configurable data properties (a strict-mode write is a
// TypeError — test-zlib-const), and `codes` is that plus Object.freeze.
function pirDefineConstants(entries) {
  const table = Object.create(null);
  for (const key of Object.keys(entries)) {
    Object.defineProperty(table, key, {
      value: entries[key],
      writable: false,
      enumerable: true,
      configurable: false,
    });
  }
  return table;
}

// Node defines `codes` and `constants` on the zlib module object itself as
// non-writable, non-configurable data properties (lib/zlib.js uses
// ObjectDefineProperty), so a strict-mode `zlib.codes = …` is a TypeError.
// The generic builtin-require path materializes a plain copy whose
// descriptors are all writable (see node_cjs_runtime.js.txt), which would
// flatten that attribute — the same observable-property-attribute reason
// `fs`/`os` hand out their module object. zlib therefore registers its own
// module object in the shared builtin-require cache: every `require("zlib")`
// then returns this exact object, which is Node's `module.exports` semantics
// (one object identity for every require, no `default` key).
Object.defineProperty(zlibApi, "codes", {
  value: codes, writable: false, enumerable: true, configurable: false,
});
Object.defineProperty(zlibApi, "constants", {
  value: constants, writable: false, enumerable: true, configurable: false,
});
if (globalThis.__pirBuiltinRequireCache !== undefined &&
    !globalThis.__pirBuiltinRequireCache.has("zlib")) {
  globalThis.__pirBuiltinRequireCache.set("zlib", zlibApi);
}
