// node:stream/consumers — Node's lib/stream/consumers.js semantics over the
// runtime's Blob and the stream pad. Consumers accept ReadableStream
// (async-iterable), node Readable, and any (async) iterable; they drain with
// `for await` and let Blob perform the BlobPart conversion (strings and
// ArrayBufferViews keep their bytes, everything else becomes its USVString,
// which is what object-mode streams produce: "[object Object]").
import { Blob } from "@node/buffer";

function pirInvalidChunk(chunk) {
  const name = chunk === null ? "null" :
    typeof chunk === "object" && chunk.constructor && typeof chunk.constructor.name === "string"
      ? "an instance of " + chunk.constructor.name
      : "type " + typeof chunk;
  const error = new TypeError(
    "The \"chunk\" argument must be of type string or an instance of Buffer or Uint8Array. Received " +
    name);
  error.code = "ERR_INVALID_ARG_TYPE";
  return error;
}

// The source shapes Node's consumers accept. Blob/ArrayBuffer/Buffer are
// accepted by this pad as single-part sources.
function pirSource(source) {
  if (source !== null && source !== undefined &&
      (typeof source[Symbol.asyncIterator] === "function" ||
       typeof source[Symbol.iterator] === "function")) return source;
  if (source instanceof Blob || source instanceof ArrayBuffer ||
      ArrayBuffer.isView(source)) return [source];
  throw new TypeError("The \"stream\" argument must be an instance of AsyncIterable, ReadableStream, or Readable");
}

async function blob(source) {
  if (source instanceof Blob) {
    return new Blob([await source.arrayBuffer()], { type: source.type });
  }
  const chunks = [];
  for await (const chunk of pirSource(source)) chunks.push(chunk);
  return new Blob(chunks);
}

async function arrayBuffer(source) {
  const result = await blob(source);
  return result.arrayBuffer();
}

async function bufferValue(source) {
  return globalThis.Buffer.from(await arrayBuffer(source));
}

async function bytes(source) {
  return new Uint8Array(await arrayBuffer(source));
}

async function text(source) {
  let str = "";
  let pending = [];
  const flush = () => {
    if (pending.length === 0) return;
    str += globalThis.Buffer.concat(pending).toString("utf8");
    pending = [];
  };
  for await (const chunk of pirSource(source)) {
    if (typeof chunk === "string") {
      flush();
      str += chunk;
    } else if (chunk instanceof ArrayBuffer) {
      pending.push(globalThis.Buffer.from(new Uint8Array(chunk)));
    } else if (ArrayBuffer.isView(chunk)) {
      pending.push(globalThis.Buffer.from(chunk.buffer, chunk.byteOffset, chunk.byteLength));
    } else {
      // Node's TextDecoder.decode(chunk) rejects these with
      // ERR_INVALID_ARG_TYPE (the text()/json() object-mode cases).
      throw pirInvalidChunk(chunk);
    }
  }
  flush();
  return str;
}

async function json(source) {
  return JSON.parse(await text(source));
}

const streamConsumers = {
  arrayBuffer: arrayBuffer,
  blob: blob,
  buffer: bufferValue,
  bytes: bytes,
  text: text,
  json: json,
};

export default streamConsumers;
export { arrayBuffer, blob, bufferValue as buffer, bytes, text, json };
