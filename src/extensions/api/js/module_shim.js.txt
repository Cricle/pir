
import { fileURLToPath } from "@node/url";
import path from "@node/path";

// Host platform constant (the same `globalThis.__pirPlatform` the other node
// shims read): `_nodeModulePaths` / `_initPaths` / `_resolveLookupPaths` are
// separator- and platform-sensitive.
const __platform = typeof globalThis.__pirPlatform === "string" ? globalThis.__pirPlatform : "linux";
const isWindows = __platform === "win32";

function requireDirOf(input) {
  if (input && typeof input === "object" && typeof input.href === "string") {
    return path.dirname(fileURLToPath(input));
  }
  if (typeof input === "string") {
    return path.dirname(input.startsWith("file:") ? fileURLToPath(input) : input);
  }
  throw new TypeError("createRequire: input must be a path string, URL, or file URL string");
}

export function createRequire(input) {
  return globalThis.__pirMakeRequire(requireDirOf(input));
}

// Real Node exposes the scheme-only builtins exclusively as `node:test` and
// friends: `builtinModules` lists them WITH the scheme and `isBuiltin('test')`
// is false while `isBuiltin('node:test')` is true (the vendored
// `test-module-isBuiltin.js` / `test-module-builtin.js` pin exactly that).
// The host's `__pirBuiltinModuleNames` is a flat key table with the `node:`
// prefix already stripped (`src/extensions/api/node_shims.rs`), so the two
// public surfaces are rebuilt here rather than changed at the host boundary.
const SCHEME_ONLY_BUILTINS = ["sea", "sqlite", "test", "test/reporters"];

const BUILTIN_NAMES = globalThis.__pirBuiltinModuleNames || [];

export const builtinModules = BUILTIN_NAMES
  .filter((name) => SCHEME_ONLY_BUILTINS.indexOf(name) === -1)
  .concat(SCHEME_ONLY_BUILTINS.map((name) => "node:" + name));

export function isBuiltin(spec) {
  if (typeof spec !== "string") return false;
  if (spec.startsWith("node:")) {
    // With the scheme every known builtin qualifies — including the
    // scheme-only ones — while an unknown `node:foo` stays false.
    return BUILTIN_NAMES.indexOf(spec.slice(5)) !== -1;
  }
  return (
    BUILTIN_NAMES.indexOf(spec) !== -1 &&
    SCHEME_ONLY_BUILTINS.indexOf(spec) === -1
  );
}

/// `NODE_PATH`-style search roots, seeded by `_initPaths` and read by
/// `_resolveLookupPaths` for bare specifiers (Node's internal `modulePaths`).
let pirSearchPaths = [];

Module._initPaths = function () {
  // Port of Node's `Module._initPaths()` (lib/internal/modules/cjs/loader.js):
  // NODE_PATH entries are taken VERBATIM (a trailing separator survives —
  // the vendored `test-module-globalpaths-nodepath.js` asserts
  // `globalPaths.includes('C:\\Program Files (x86)\\nodejs\\')`), empty
  // segments are dropped, and the two home-directory entries sit between the
  // NODE_PATH block and the install's `lib/node`.
  const homeDir = isWindows ? process.env.USERPROFILE : process.env.HOME;
  const nodePath = process.env.NODE_PATH;

  // process.execPath is $PREFIX/bin/node except on Windows, where it is
  // $PREFIX\node.exe with $PREFIX the Node.js installation root.
  const prefixDir = isWindows
    ? path.resolve(process.execPath, "..")
    : path.resolve(process.execPath, "..", "..");

  const paths = [path.resolve(prefixDir, "lib", "node")];

  if (homeDir) {
    paths.unshift(path.resolve(homeDir, ".node_libraries"));
    paths.unshift(path.resolve(homeDir, ".node_modules"));
  }

  if (nodePath) {
    paths.unshift(...String(nodePath).split(path.delimiter).filter(Boolean));
  }

  pirSearchPaths = paths;
  // Node assigns a fresh array here (`ArrayPrototypeSlice`); the exported
  // `globalPaths` handles in this shim are captured once (a plain export
  // object is not Node's live class property), so the update happens in
  // place and both stay in sync — `mod._initPaths(); mod.globalPaths`
  // reflects the new NODE_PATH, which is what Node's API promises.
  Module.globalPaths.length = 0;
  for (const entry of pirSearchPaths) Module.globalPaths.push(entry);
};

// Node runs `_initPaths()` during its bootstrap, so `globalPaths` is already
// populated by the time user code reads it. Guarded: the prelude can evaluate
// this shim while `process` is still being assembled.
if (typeof process === "object" && process !== null) {
  Module._initPaths();
}
Module._nodeModulePaths = function (from) {
  // Port of Node's walk-up algorithm: every ancestor directory contributes a
  // `<dir>/node_modules` entry, a path segment that already ends in
  // `node_modules` is skipped, and `D:\` maps to `D:\node_modules`. Node
  // builds the entries with the host separator (its own `path.sep`), which is
  // what the vendored `test-module-nodemodulepaths.js` expects.
  const sep = isWindows ? "\\" : "/";
  from = path.resolve(String(from));

  // Return root node_modules when the path is a drive root ('D:\').
  if (
    from.length >= 2 &&
    from.charCodeAt(from.length - 1) === (isWindows ? 92 : 47) &&
    from.charCodeAt(from.length - 2) === 58 /* : */
  ) {
    return [from + "node_modules"];
  }
  // The POSIX filesystem root: a single '/node_modules'.
  if (from === "/") {
    return ["/node_modules"];
  }

  const nmChars = [
    115, 101, 108, 117, 100, 111, 109, 95, 101, 100, 111, 110,
  ]; // 'node_modules' reversed
  const nmLen = nmChars.length;
  const paths = [];
  for (let i = from.length - 1, p = 0, last = from.length; i >= 0; --i) {
    const code = from.charCodeAt(i);
    // '\' and '/' separate path segments; ':' is an extra condition so a
    // drive root ('C:\node_modules') needs no drive-name parsing.
    if (code === 92 /* \ */ || code === 47 /* / */ || code === 58 /* : */) {
      if (p !== nmLen) {
        paths.push(from.slice(0, last) + sep + "node_modules");
      }
      last = i;
      p = 0;
    } else if (p !== -1) {
      if (nmChars[p] === code) {
        ++p;
      } else {
        p = -1;
      }
    }
  }
  // The last boundary: a POSIX absolute path reaches the filesystem root at
  // index 0 (a Windows drive root is emitted by the ':' branch above), and
  // `/<root>` is always part of the walk-up chain.
  if (!isWindows && from.charAt(0) === "/" && paths[paths.length - 1] !== "/node_modules") {
    paths.push("/node_modules");
  }
  return paths;
};

Module._resolveLookupPaths = function (request, parent) {
  // Port of Node's helper. A builtin needs no lookup (`null`); a relative
  // request resolves against the parent's directory (`.` when there is no
  // parent — the REPL/`-e` shape the vendored
  // `test-module-relative-lookup.js` pins); a bare request searches the
  // `NODE_PATH`/home roots `_initPaths` collected.
  if (isBuiltin(request)) return null;

  if (
    request.charAt(0) !== "." ||
    (request.length > 1 &&
      request.charAt(1) !== "." &&
      request.charAt(1) !== "/" &&
      (!isWindows || request.charAt(1) !== "\\"))
  ) {
    let paths;
    if (parent && parent.paths && parent.paths.length) {
      paths = pirSearchPaths.slice();
      paths.unshift(...parent.paths);
    } else {
      paths = pirSearchPaths;
    }
    return paths.length > 0 ? paths.slice() : null;
  }

  // In the REPL, parent.filename is null: './path/to/foo' still has to work.
  if (!parent || !parent.id || !parent.filename) {
    return ["."];
  }
  return [path.dirname(parent.filename)];
};

Module.findPackageJSON = function (specifier, base) {
  // Walk up from the module path looking for package.json via the host fs.
  const fsSync = globalThis.__pirNodeModules && globalThis.__pirNodeModules.fs;
  if (!fsSync || typeof fsSync.existsSync !== "function") return undefined;
  let start = String(specifier);
  if (base !== undefined && base !== null) {
    const baseStr = String(base);
    const baseDir = typeof baseStr === "string" && baseStr.startsWith("file:") ? fileURLToPath(baseStr) : baseStr;
    start = baseDir.replace(/[\\/][^\\/]*\.ts$/, "").replace(/[\\/][^\\/]*\.js$/, "");
  }
  const stat = fsSync.statSync(start);
  let dir = stat && stat.isFile ? path.dirname(start) : start;
  for (let i = 0; i < 64; i++) {
    const candidate = path.join(dir, "package.json");
    if (fsSync.existsSync(candidate)) return candidate;
    const parent = path.dirname(dir);
    if (parent === dir) return undefined;
    dir = parent;
  }
  return undefined;
};

// Node's argument-type contract for the source-map switches: a boolean
// `enabled`/`options.nodeModules`/`options.generatedCode` is required and
// anything else raises ERR_INVALID_ARG_TYPE (the vendored
// `test-module-setsourcemapssupport.js` validates the whole rejection table).
// The flags are recorded rather than applied: this transpiler has no
// source-map store to rewire, and the load-compatible surface is what
// callers observe.
let sourceMapsSupport = { enabled: false, nodeModules: false, generatedCode: false };

function describeValue(value) {
  if (value === null) return "null";
  if (value === undefined) return "undefined";
  const kind = typeof value;
  if (kind === "function") return "function " + (value.name || "");
  if (kind === "object") {
    const name = (value.constructor && value.constructor.name) || "Object";
    return "an instance of " + name;
  }
  return "type " + kind + " (" + String(value) + ")";
}

function invalidArgType(name, kind, value) {
  const error = new TypeError(
    `The "${name}" ${kind} must be of type boolean. Received ${describeValue(value)}`,
  );
  error.code = "ERR_INVALID_ARG_TYPE";
  // Node's internal ERR_* errors stringify with the code in brackets
  // (`TypeError [ERR_INVALID_ARG_TYPE]: …`) even though `name` stays
  // `TypeError`; `String(err)` and `assert.throws(fn, /ERR_INVALID_ARG_TYPE/)`
  // observe that form (pir's assert matches the stringified error).
  error.toString = function () {
    return `${this.name} [${this.code}]: ${this.message}`;
  };
  return error;
}

function validateBooleanOption(value, name, kind) {
  if (typeof value !== "boolean") throw invalidArgType(name, kind, value);
  return value;
}

Module.getSourceMapsSupport = function () {
  return {
    enabled: sourceMapsSupport.enabled,
    nodeModules: sourceMapsSupport.nodeModules,
    generatedCode: sourceMapsSupport.generatedCode,
  };
};
Module.setSourceMapsSupport = function (enabled, options) {
  validateBooleanOption(enabled, "enabled", "argument");
  const opts = options === undefined ? {} : options;
  if (opts === null || typeof opts !== "object") {
    // Node validates `options` as an object (kEmptyObject default).
    const error = new TypeError(
      `The "options" argument must be of type object. Received ${describeValue(opts)}`,
    );
    error.code = "ERR_INVALID_ARG_TYPE";
    throw error;
  }
  const nodeModules = opts.nodeModules === undefined
    ? false
    : validateBooleanOption(opts.nodeModules, "options.nodeModules", "property");
  const generatedCode = opts.generatedCode === undefined
    ? false
    : validateBooleanOption(opts.generatedCode, "options.generatedCode", "property");
  sourceMapsSupport = {
    enabled: enabled,
    nodeModules: nodeModules,
    generatedCode: generatedCode,
  };
};
