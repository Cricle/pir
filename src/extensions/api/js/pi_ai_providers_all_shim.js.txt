
const CATALOG_UNAVAILABLE =
  "the generated built-in model catalog is not bundled in the pir runtime";

function catalogUnavailable(what) {
  throw new Error(what + " is not available: " + CATALOG_UNAVAILABLE);
}

// Placeholder for the built-in provider objects: any access or call fails
// loudly (no fake provider is ever handed out).
function unavailableExport(what) {
  const message = what + " is not available: built-in provider implementations are not bundled in the pir runtime";
  return new Proxy(function () { throw new Error(message); }, {
    get() { throw new Error(message); },
    apply() { throw new Error(message); },
    construct() { throw new Error(message); },
  });
}

// Upstream `getBuiltinProviders(): BuiltinProvider[]` returns `Object.keys(MODELS)`
// — the catalog provider **id strings**, not provider objects. The ids are
// exactly the manifest's catalog file names without the `.json` suffix (the
// same 42 built-in catalog ids the Rust host registers), so this is a real
// read, not a placeholder. Provider *objects* stay unavailable: the
// implementations live in the Rust host (`builtinProviders()` below throws).
export function getBuiltinProviders() {
  return Object.keys(getBuiltinModelDataManifest().files).map((file) => file.replace(/\.json$/, ""));
}

// FNV-1a 64 content fingerprint of every embedded catalog resource
// (`builtin_catalog_version()`, same value as `getBuiltinModelDataVersion`).
const PIR_BUILTIN_MODEL_DATA_VERSION = "c1625812d2db0645";

/** Upstream `getBuiltinModelDataGeneratedAt`: `generatedAt` in Unix ms. */
export function getBuiltinModelDataGeneratedAt() {
  const generatedAt = Date.parse(JSON.parse(PIR_BUILTIN_MODEL_DATA_MANIFEST_JSON).generatedAt);
  return Number.isNaN(generatedAt) ? undefined : generatedAt;
}

/**
 * The catalog generation manifest, in the upstream `ModelDataManifest` shape
 * (`schemaVersion` / `generatedAt` / `structureHash` / `files`). A fresh object
 * is returned so callers cannot mutate the pinned metadata.
 */
export function getBuiltinModelDataManifest() {
  return JSON.parse(PIR_BUILTIN_MODEL_DATA_MANIFEST_JSON);
}

/** pir's catalog content version (`builtin_catalog_version`, lowercase hex). */
export function getBuiltinModelDataVersion() { return PIR_BUILTIN_MODEL_DATA_VERSION; }
