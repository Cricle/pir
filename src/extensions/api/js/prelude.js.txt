// Abort plumbing: host-set aborts are polled via __pir_host.isAborted();
// JS-initiated aborts (ctx.abort / ctx.shutdown) additionally fire listeners.
const abortState = { local: false, listeners: [] };

function isAborted() {
  return abortState.local || __pir_host.isAborted();
}

function notifyAbort() {
  abortState.local = true;
  const listeners = abortState.listeners.splice(0);
  for (const listener of listeners) {
    try { listener(); } catch { /* listener errors are isolated */ }
  }
}

globalThis.__pir_clear_abort = () => {
  abortState.local = false;
  abortState.listeners.length = 0;
  return "ok";
};

// Console shim (P2-14): the QuickJS runtime provides no `console`, but real
// extensions call console.log/warn/error. Route the output to the host
// notify path — log/info/debug at "debug" level (invisible in the TUI by
// default), warn/error at their own levels.
globalThis.console = (() => {
  const format = (value) => {
    if (typeof value === "string") return value;
    if (value instanceof Error) return value.stack ?? String(value);
    try { return JSON.stringify(value); } catch { return String(value); }
  };
  const emit = (level) => (...args) => {
    try { __pir_host.notify(args.map(format).join(" "), level); } catch { /* best-effort */ }
  };
  return {
    log: emit("debug"),
    info: emit("debug"),
    debug: emit("debug"),
    warn: emit("warning"),
    error: emit("error"),
  };
})();

function makeSignal() {
  return {
    get aborted() { return isAborted(); },
    get reason() { return isAborted() ? new Error("The operation was aborted") : undefined; },
    throwIfAborted() { if (isAborted()) throw new Error("The operation was aborted"); },
    addEventListener(type, callback) {
      if (type !== "abort" || typeof callback !== "function") return;
      if (isAborted()) {
        try { callback(); } catch { /* listener errors are isolated */ }
        return;
      }
      abortState.listeners.push(callback);
    },
    removeEventListener(type, callback) {
      if (type !== "abort") return;
      const index = abortState.listeners.indexOf(callback);
      if (index >= 0) abortState.listeners.splice(index, 1);
    },
  };
}

function extRecord(extId) {
  return state.extensions[extId];
}

function assertActive(extId) {
  const ext = extRecord(extId);
  // "loading" is allowed: the factory registers hooks/tools while running.
  if (!ext) {
    throw new Error("Extension is not registered");
  }
  if (ext.status === "failed") {
    throw new Error(`Extension ${ext.path} failed to load`);
  }
  if (!__pir_host.isActive(extId)) {
    throw new Error(
      "The extension runtime was invalidated. Capture of the `pi` API object across sessions/reloads is not supported.",
    );
  }
}

// ---------------------------------------------------------------------------
// JSON-Schema subset validator (upstream TypeBox Compile/Value).
//
// Supported: type (string | array), properties, required, enum, const,
// anyOf / oneOf / allOf, not, items (schema | tuple), additionalProperties,
// patternProperties, contains (with TypeBox's minContains/maxContains bounds),
// minLength/maxLength, minimum/maximum/exclusiveMinimum/exclusiveMaximum,
// pattern, minItems/maxItems, and the basic `format` keywords. Unknown
// keywords are ignored (never fail), matching permissive TypeBox behavior.
// ---------------------------------------------------------------------------

const FORMAT_TESTS = {
  email: (v) => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v),
  hostname: (v) => /^[a-zA-Z0-9]([a-zA-Z0-9-]*[a-zA-Z0-9])?(\.[a-zA-Z0-9]([a-zA-Z0-9-]*[a-zA-Z0-9])?)*$/.test(v) && v.length <= 253,
  uuid: (v) => /^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$/.test(v),
  date: (v) => /^\d{4}-\d{2}-\d{2}$/.test(v) && !Number.isNaN(Date.parse(v)),
  "date-time": (v) => !Number.isNaN(Date.parse(v)),
  time: (v) => /^\d{2}:\d{2}(:\d{2}(\.\d+)?)?([zZ]|[+-]\d{2}:\d{2})?$/.test(v),
  // RFC 3986 URI: an absolute URI requires `scheme:` followed by the
  // hier-part/query/fragment, with no unencoded whitespace. (`new URL` does
  // not exist in QuickJS, so the previous implementation always failed.)
  uri: (v) => /^[a-zA-Z][a-zA-Z0-9+.-]*:(\/\/)?[^\s?#]*(\?[^#]*)?(#.*)?$/.test(v),
  "ipv4": (v) => { const parts = v.split("."); return parts.length === 4 && parts.every((p) => /^\d{1,3}$/.test(p) && Number(p) <= 255); },
  // Real IPv6 check: full or ::-compressed hexadecimal groups with at most
  // one "::" and an optional embedded IPv4 tail ("::::" and ":" fail).
  "ipv6": (v) => {
    if (typeof v !== "string" || !v.includes(":") || !/^[0-9a-fA-F:.]+$/.test(v)) return false;
    let text = v;
    let tailGroups = 0;
    const v4Match = /(?:^|:)(\d{1,3}(?:\.\d{1,3}){3})$/.exec(text);
    if (v4Match) {
      const octets = v4Match[1].split(".").map(Number);
      if (octets.some((octet) => octet > 255)) return false;
      text = text.slice(0, v4Match.index);
      tailGroups = 2;
    }
    const isGroupList = (list) => list.every((group) => /^[0-9a-fA-F]{1,4}$/.test(group));
    if (text.includes("::")) {
      const parts = text.split("::");
      if (parts.length !== 2) return false;
      const head = parts[0] === "" ? [] : parts[0].split(":");
      const tail = parts[1] === "" ? [] : parts[1].split(":");
      if (head.length + tail.length + tailGroups > 8) return false;
      return isGroupList(head) && isGroupList(tail);
    }
    const groups = text === "" ? [] : text.split(":");
    return groups.length + tailGroups === 8 && isGroupList(groups);
  },
};

function jsonEquals(a, b) {
  if (a === b) return true;
  if (typeof a !== typeof b) return false;
  if (a === null || b === null || typeof a !== "object") return false;
  if (Array.isArray(a) !== Array.isArray(b)) return false;
  if (Array.isArray(a)) {
    if (a.length !== b.length) return false;
    return a.every((item, index) => jsonEquals(item, b[index]));
  }
  if (isPlainObject(a) && isPlainObject(b)) {
    const keysA = Object.keys(a);
    const keysB = Object.keys(b);
    if (keysA.length !== keysB.length) return false;
    return keysA.every((key) => Object.prototype.hasOwnProperty.call(b, key) && jsonEquals(a[key], b[key]));
  }
  return false;
}

function isPlainObject(value) {
  return value !== null && typeof value === "object" && !Array.isArray(value);
}

function typeMatches(value, type) {
  switch (type) {
    case "string": return typeof value === "string";
    case "number": return typeof value === "number" && Number.isFinite(value);
    case "integer": return typeof value === "number" && Number.isFinite(value) && Number.isInteger(value);
    case "boolean": return typeof value === "boolean";
    case "object": return isPlainObject(value);
    case "array": return Array.isArray(value);
    case "null": return value === null;
    default: return true; // unknown type keywords never fail
  }
}

function validateInto(schema, value, path, errors) {
  if (schema === undefined || schema === null || schema === true) return;
  if (schema === false) {
    errors.push({ path, message: "schema forbids any value" });
    return;
  }
  if (!isPlainObject(schema)) return;

  // const / enum
  if (schema.const !== undefined) {
    if (!jsonEquals(value, schema.const)) {
      errors.push({ path, message: `must equal ${JSON.stringify(schema.const)}` });
    }
  }
  if (Array.isArray(schema.enum) && !schema.enum.some((option) => jsonEquals(value, option))) {
    errors.push({ path, message: `must be one of ${JSON.stringify(schema.enum)}` });
  }

  // combinators
  if (Array.isArray(schema.allOf)) {
    for (const sub of schema.allOf) validateInto(sub, value, path, errors);
  }
  if (Array.isArray(schema.anyOf)) {
    const probe = [];
    if (!schema.anyOf.some((sub) => { const e = []; validateInto(sub, value, path, e); return e.length === 0; })) {
      for (const sub of schema.anyOf) validateInto(sub, value, path, probe);
      errors.push({ path, message: "does not match any of the anyOf schemas" });
    }
  }
  if (Array.isArray(schema.oneOf)) {
    const matches = schema.oneOf.filter((sub) => { const e = []; validateInto(sub, value, path, e); return e.length === 0; }).length;
    if (matches !== 1) {
      errors.push({ path, message: `must match exactly one oneOf schema (matched ${matches})` });
    }
  }
  // `not`: the value must FAIL the subschema (evaluated before the `type` bail-out,
  // which returns early). TypeBox maps a literal `false` schema to `{ not: {} }`,
  // so a `not` whose subschema accepts everything rejects every value.
  if (schema.not !== undefined) {
    const probe = [];
    validateInto(schema.not, value, path, probe);
    if (probe.length === 0) {
      errors.push({ path, message: "must not match the schema" });
    }
  }

  // type (string | array of strings)
  if (schema.type !== undefined) {
    const types = Array.isArray(schema.type) ? schema.type : [schema.type];
    if (!types.some((t) => typeMatches(value, t))) {
      errors.push({ path, message: `expected ${types.join(" | ")}` });
      return; // further checks are meaningless for a type mismatch
    }
  }

  if (typeof value === "string") {
    if (typeof schema.minLength === "number" && value.length < schema.minLength) {
      errors.push({ path, message: `length must be >= ${schema.minLength}` });
    }
    if (typeof schema.maxLength === "number" && value.length > schema.maxLength) {
      errors.push({ path, message: `length must be <= ${schema.maxLength}` });
    }
    if (typeof schema.pattern === "string" && !new RegExp(schema.pattern).test(value)) {
      errors.push({ path, message: `must match pattern ${schema.pattern}` });
    }
  }
  if (typeof value === "number" && Number.isFinite(value)) {
    if (typeof schema.minimum === "number" && value < schema.minimum) {
      errors.push({ path, message: `must be >= ${schema.minimum}` });
    }
    if (typeof schema.maximum === "number" && value > schema.maximum) {
      errors.push({ path, message: `must be <= ${schema.maximum}` });
    }
    if (typeof schema.exclusiveMinimum === "number" && value <= schema.exclusiveMinimum) {
      errors.push({ path, message: `must be > ${schema.exclusiveMinimum}` });
    }
    if (typeof schema.exclusiveMaximum === "number" && value >= schema.exclusiveMaximum) {
      errors.push({ path, message: `must be < ${schema.exclusiveMaximum}` });
    }
  }
  if (Array.isArray(value)) {
    if (typeof schema.minItems === "number" && value.length < schema.minItems) {
      errors.push({ path, message: `must have >= ${schema.minItems} items` });
    }
    if (typeof schema.maxItems === "number" && value.length > schema.maxItems) {
      errors.push({ path, message: `must have <= ${schema.maxItems} items` });
    }
    if (isPlainObject(schema.items)) {
      value.forEach((item, index) => validateInto(schema.items, item, `${path}[${index}]`, errors));
    } else if (Array.isArray(schema.items)) {
      schema.items.forEach((sub, index) => validateInto(sub, value[index], `${path}[${index}]`, errors));
    }
    // `contains`: at least `minContains` (default 1) items must match the subschema,
    // and at most `maxContains` (default unbounded) may — the bounds TypeBox
    // implements for the 2019-09 revision of the keyword.
    if (schema.contains !== undefined) {
      const minContains = schema.minContains ?? 1;
      const maxContains = schema.maxContains ?? Infinity;
      let matches = 0;
      for (const item of value) {
        const probe = [];
        validateInto(schema.contains, item, path, probe);
        if (probe.length === 0) matches += 1;
      }
      if (matches < minContains) {
        errors.push({ path, message: `must contain at least ${minContains} matching item(s)` });
      }
      if (matches > maxContains) {
        errors.push({ path, message: `must contain at most ${maxContains} matching item(s)` });
      }
    }
  }
  if (isPlainObject(value)) {
    const required = Array.isArray(schema.required) ? schema.required : [];
    for (const key of required) {
      if (!(key in value)) {
        const child = path ? `${path}.${key}` : key;
        errors.push({ path: child, message: "is required" });
      }
    }
    const properties = isPlainObject(schema.properties) ? schema.properties : {};
    // `patternProperties` (what `Type.Record` emits): every name matching a
    // pattern is validated against that pattern's subschema — all matching
    // patterns apply — and such a name is never an `additionalProperties`
    // candidate, the draft-07 rule TypeBox implements. A pattern that does not
    // compile constrains nothing (permissive, like every unsupported keyword).
    const patternProperties = isPlainObject(schema.patternProperties) ? schema.patternProperties : undefined;
    let matchers;
    if (patternProperties !== undefined) {
      matchers = [];
      for (const [pattern, subschema] of Object.entries(patternProperties)) {
        try {
          matchers.push([new RegExp(pattern), subschema]);
        } catch { /* an uncompilable pattern matches nothing */ }
      }
    }
    for (const [key, item] of Object.entries(value)) {
      const child = path ? `${path}.${key}` : key;
      const declared = key in properties;
      if (declared) {
        validateInto(properties[key], item, child, errors);
      }
      let patterned = false;
      if (matchers !== undefined) {
        for (const [matcher, subschema] of matchers) {
          if (matcher.test(key)) {
            patterned = true;
            validateInto(subschema, item, child, errors);
          }
        }
      }
      if (declared || patterned) continue;
      if (schema.additionalProperties === false) {
        errors.push({ path: child, message: "is not an allowed property" });
      } else if (isPlainObject(schema.additionalProperties)) {
        validateInto(schema.additionalProperties, item, child, errors);
      }
    }
  }

  // Basic format checks (strings only, permissive on unknown formats).
  if (typeof value === "string" && typeof schema.format === "string" && FORMAT_TESTS[schema.format]) {
    if (!FORMAT_TESTS[schema.format](value)) {
      errors.push({ path, message: `must be a valid ${schema.format}` });
    }
  }
}

globalThis.__pir_schemaValidate = (schema, value) => {
  const errors = [];
  validateInto(schema, value, "", errors);
  return { valid: errors.length === 0, errors };
};

// Defensive host-mode normalization: exactly "tui" | "rpc" | "json" | "print".
function normalizeMode(mode) {
  switch (mode) {
    case "tui":
    case "rpc":
    case "json":
    case "print":
      return mode;
    case "interactive":
      return "tui";
    default:
      return "print";
  }
}

// ui_prompt_start / ui_prompt_end nesting (upstream runner.ts withUIPrompt):
// only the outermost prompt emits the events; a depth counter tracks nesting.
const uiPromptState = { depth: 0, active: undefined };

async function dispatchFireAndForget(eventName, payload) {
  try {
    await __pir_dispatchEvent(eventName, JSON.stringify(payload));
  } catch { /* lifecycle events are best-effort */ }
}

async function withUIPrompt(kind, title, run) {
  const outer = uiPromptState.depth++ === 0;
  if (outer) {
    uiPromptState.active = { kind, title };
    await dispatchFireAndForget("ui_prompt_start", {
      type: "ui_prompt_start",
      reason: "ui_prompt",
      kind,
      // Upstream runner.ts: truthiness (an empty title is omitted), not a
      // null check — keep the payload byte-identical to pi's.
      ...(title ? { title } : {}),
    });
  }
  try {
    return await run();
  } finally {
    uiPromptState.depth = Math.max(0, uiPromptState.depth - 1);
    if (uiPromptState.depth === 0) {
      const prompt = uiPromptState.active ?? { kind, title };
      uiPromptState.active = undefined;
      await dispatchFireAndForget("ui_prompt_end", {
        type: "ui_prompt_end",
        reason: "ui_prompt",
        kind: prompt.kind,
        // Same upstream truthiness rule as the start event.
        ...(prompt.title ? { title: prompt.title } : {}),
      });
    }
  }
}

function parseHostJson(text) {
  if (text == null) return undefined;
  try { return JSON.parse(text); } catch { return undefined; }
}

// (extensions/types.ts:1134-1139). The host's payload historically carried the
// previous model as an id string; resolve a string reference against the
// pushed model catalog so handlers get the upstream shape. An unknown id stays
// the string it was (never fabricated).
function resolveModelRef(ref) {
  if (ref == null || typeof ref === "object") return ref;
  if (typeof ref !== "string") return ref;
  const registry = parseHostJson(__pir_host.getModelsJson()) ?? {};
  const models = Array.isArray(registry.models) ? registry.models : [];
  return models.find((model) => model && model.id === ref) ?? ref;
}

// `ctx.compact` callback settling (upstream CompactOptions.onComplete /
// onError, extensions/types.ts:314-318). The host's compaction events are the
// settlement signal; the internal handlers exist only so the host's
// `has_handlers` gate emits them for a session whose extensions registered no
// listener of their own. They are registered once per extension and stay
// (they resolve an empty pending list once none remain).
function ensureCompactSettler(extId) {
  const ext = extRecord(extId);
  if (!ext || ext.compactSettler) return;
  ext.compactSettler = true;
  const step = (eventName) => async (payload) => {
    settleCompactCallbacks(eventName, payload);
    return undefined;
  };
  (ext.handlers["session_compact"] ??= []).push(step("session_compact"));
  (ext.handlers["session_compact_failed"] ??= []).push(step("session_compact_failed"));
  markHandlersDirty();
}

function settleCompactCallbacks(eventName, payload) {
  if (state.pendingCompactCallbacks.length === 0) return;
  const pending = state.pendingCompactCallbacks.splice(0);
  const result = payload && (payload.compaction ?? payload.result);
  for (const entry of pending) {
    try {
      if (eventName === "session_compact") {
        if (typeof entry.onComplete === "function") entry.onComplete(result ?? {});
      } else if (typeof entry.onError === "function") {
        const message =
          (payload && (payload.errorMessage ?? payload.error)) ?? "compaction failed";
        entry.onError(new Error(String(message)));
      }
    } catch {
      /* a callback's own failure must not abort the event dispatch */
    }
  }
}

// Render a mutable `BuildSystemPromptOptions` object through the host's
// ordered section builder (upstream `buildSystemPrompt`, system-prompt.ts:208).
// The renderer lives host-side so a JS-mutated prompt is byte-identical to one
// the session itself derives from the same inputs; `""` means the bridge is
// unavailable (a host-less prelude evaluation) and callers fall back to the
// last pushed prompt text.
function renderSystemPromptFromOptions(options) {
  if (typeof __pir_host?.renderSystemPromptOptionsJson !== "function") return "";
  try {
    const rendered = __pir_host.renderSystemPromptOptionsJson(JSON.stringify(options ?? {}));
    return typeof rendered === "string" ? rendered : "";
  } catch {
    return "";
  }
}

// `ctx.newSession/fork/switchSession` `withSession` callbacks: the function
// cannot cross the JSON host-fn bridge, so it is stored under a token the
// recorded action carries and invoked by the host through
// `__pir_callSessionCallback` after the replacement session is live.
// Bounded: an action a host never applies (a cancelled switch, a storage-less
// host) leaves its callback here, so the oldest entry is evicted past the cap
// instead of growing the map for the runtime's lifetime.
const SESSION_CALLBACK_LIMIT = 64;

function registerSessionCallback(fn) {
  const token = state.nextSessionCallbackToken + 1;
  state.nextSessionCallbackToken = token;
  const tokens = Object.keys(state.sessionCallbacks);
  if (tokens.length >= SESSION_CALLBACK_LIMIT) {
    delete state.sessionCallbacks[tokens[0]];
  }
  state.sessionCallbacks[token] = fn;
  return token;
}

// Validate and register an options bag's `withSession`. Returns the token to
// put on the action payload, or `undefined` when the caller passed none.
// A non-function is a caller bug and throws (upstream would fail when calling
// it; failing at the call site names the mistake).
function takeWithSessionToken(surface, options) {
  const callback = options?.withSession;
  if (callback === undefined || callback === null) return undefined;
  if (typeof callback !== "function") {
    throw new TypeError(`${surface}: options.withSession must be a function`);
  }
  return registerSessionCallback(callback);
}

function makeCtx(desc, getSystemPromptOverride, ownerExtId) {
  const hasUI = desc.hasUI === true;
  const ctxtMode = normalizeMode(desc.mode);
  // Lazy `ctx.tools` cache (see the getter at its declaration): a ctx is
  // constructed for EVERY dispatched event, and `getAllToolsJson` is a host
  // call that serializes every tool definition — paying it eagerly made each
  // per-delta `message_update` dispatch cost a registry-sized host call for a
  // field a streaming handler almost never reads.
  let toolsCache;
  const ui = {
    notify: (message, level) => { __pir_host.notify(String(message), String(level ?? "info")); },
    setStatus: (key, text) => { __pir_host.setStatus(String(key), text == null ? null : String(text)); },

        typeof option === "string" ? { value: option, label: option } : option);
      const out = await __pir_host.uiDialog("select", JSON.stringify({
        title,
        options: choices,
        ...(opts ?? {}),
      }));
      const parsed = parseHostJson(out);
      return parsed === undefined || parsed === null ? undefined : parsed;
    }),
    confirm: (title, message, opts) => withUIPrompt("confirm", title, async () => {
      if (!hasUI) return undefined;
      const out = await __pir_host.uiDialog("confirm", JSON.stringify({ title, message, ...(opts ?? {}) }));
      const parsed = parseHostJson(out);
      return typeof parsed === "boolean" ? parsed : undefined;
    }),
    input: (title, placeholder, opts) => withUIPrompt("input", title, async () => {

      const parsed = parseHostJson(out);

    }),

      const parsed = parseHostJson(out);
      return typeof parsed === "string" ? parsed : undefined;
    }),
    custom: (factory, options) => withUIPrompt("custom", undefined, async () => {
      if (!hasUI) return undefined;
      if (typeof factory !== "function") {
        // Upstream's signature requires a component factory; calling anything
        // else throws there too. This is a caller bug, not a silent no-op.
        throw new TypeError(
          "ctx.ui.custom requires a component factory function (tui, theme, keybindings, done) => Component",
        );
      }
      const key = `custom:${++state.nextCustomOverlayId}`;
      const sendOverlayOp = (op, extra) => {
        try {
          __pir_host.setStatus(
            "__pir_custom_overlay",
            JSON.stringify({ component: key, op, ...(extra ?? {}) }),
          );
        } catch { /* host-directed signals are best-effort */ }
      };
      let doneCalled = false;
      let doneResult;
      const done = (result) => {
        if (doneCalled) return;
        doneCalled = true;
        doneResult = result;
        // Completion rides the reserved status key; the host settles the
        // overlay through its one funnel (dispose, promote the queue head,
        // route this value back to the pending uiDialog await).
        sendOverlayOp("done", { result: result === undefined ? null : result });
      };
      // Upstream hands the factory a `tui` handle; the parts a component
      // actually uses are requestRender (repaint) — the rest is inert.
      const tui = {
        requestRender: () => sendOverlayOp("repaint"),
      };
      // Upstream `OverlayHandle` (packages/tui/src/tui.ts:309-321). Visibility
      // is real (hidden overlays paint nothing and release the keyboard to the
      // editor); `focus()`/`unfocus()` map onto show/hide. `bounds` is the
      // last terminal-relative rectangle the host reported with a render
      // request (upstream `entry.bounds`, set by `compositeOverlays`), so
      // `getBounds()` returns real geometry once the first frame rendered.
      const handleState = { hidden: false, permanentlyHidden: false, bounds: undefined };
      const applyHidden = (hidden) => {
        if (hidden === handleState.hidden) return;
        handleState.hidden = hidden;
        sendOverlayOp("hidden", { hidden });
      };
      const handle = {
        hide: () => {
          handleState.permanentlyHidden = true;
          applyHidden(true);
        },
        setHidden: (hidden) => {
          if (hidden) applyHidden(true);
          else if (!handleState.permanentlyHidden) applyHidden(false);
        },
        isHidden: () => handleState.hidden,
        focus: () => {
          if (!handleState.permanentlyHidden) applyHidden(false);
        },
        unfocus: () => applyHidden(true),
        isFocused: () => !handleState.hidden,
        getBounds: () => (handleState.bounds ? { ...handleState.bounds } : undefined),
      };
      let component;
      try {
        component = await factory(tui, pirWidgetTheme(), customOverlayKeybindings(), done);
      } catch (error) {
        // Upstream rejects the promise when the factory throws and never shows
        // an overlay; propagating keeps that contract and leaks nothing (the
        // host was never told to open anything).
        throw error;
      }
      if (!component || typeof component.render !== "function") {
        warnCustomOverlayUnsupported(
          `ctx.ui.custom: the factory must return a component with a render(width) method; got ${describeOverlayValue(component)}`,
        );
        return undefined;
      }
      const entry = { component, disposed: false, handleState };
      // A factory that completed synchronously needs no overlay at all.
      if (doneCalled) {
        try { component.dispose?.(); } catch { /* dispose errors are isolated */ }
        return doneResult;
      }
      state.customOverlays[key] = entry;
      try {
        // `onHandle` runs once the component exists, before the request is
        // handed to the host; a visibility change it makes rides the opening
        // payload so the overlay never flashes visible first.
        if (options && typeof options.onHandle === "function") {
          try {
            options.onHandle(handle);
          } catch (error) {
            try {
              console.warn(
                `ctx.ui.custom: options.onHandle threw: ${error && error.message ? error.message : String(error)}`,
              );
            } catch { /* the warning is best-effort */ }
          }
        }
        const payload = { component: key };
        if (handleState.hidden) payload.hidden = true;
        // Upstream `showExtensionCustom`: `overlay: true` shows the component
        // on the floating overlay stack, positioned by `overlayOptions`
        // (tui.ts:254-290). Without `overlay` the component keeps the
        // editor-slot path. A function-valued `overlayOptions` is resolved
        // here (upstream re-evaluates it per render; the host receives the
        // opening geometry). Function fields such as `visible` are dropped by
        // JSON.stringify — the documented host limitation.
        if (options && options.overlay === true) {
          payload.overlay = true;
          let overlayOptions = options.overlayOptions;
          if (typeof overlayOptions === "function") {
            try { overlayOptions = overlayOptions(); } catch { overlayOptions = undefined; }
          }
          if (overlayOptions && typeof overlayOptions === "object") {
            payload.overlayOptions = overlayOptions;
          } else if (typeof component.width === "number") {
            // Upstream falls back to the component's declared width.
            payload.overlayOptions = { width: component.width };
          }
        }
        if (options && typeof options.timeout === "number" && options.timeout > 0) {
          payload.timeout = options.timeout;
        }
        const out = await __pir_host.uiDialog("custom", JSON.stringify(payload));
        const parsed = parseHostJson(out);
        // The host echoes the `done(result)` value (or null for a dismissal,
        // which upstream resolves as `undefined`).
        return parsed === undefined || parsed === null ? undefined : parsed;
      } finally {
        delete state.customOverlays[key];
        if (!entry.disposed) {
          entry.disposed = true;
          try { component.dispose?.(); } catch { /* dispose errors are isolated */ }
        }
      }
    }),
    setTitle: (title) => { __pir_host.setTitle(String(title)); },
    // Upstream `setWidget(key, content: string[] | factory, options?)`
    // (types.ts:188-193): content is an array of lines or a component
    // factory, `undefined` removes the widget. The TUI host wire shape is
    // `{text, placement}` for static content and `{factory, placement}` for
    // component factories — an upstream-style array call would be silently
    // dropped, so it is normalized HERE (host side; the TUI is not touched):
    // array -> {text: lines.join("\n")}. RPC hosts consume the upstream wire
    // shape directly (rpc.rs forwards `widgetLines` arrays, matching
    // rpc-mode.ts setWidget), so that mode keeps the raw array and ignores
    // factories. The placement option follows the upstream default
    // "aboveEditor" (mapped to the TUI's "above_transcript";
    // "belowEditor" -> "below_transcript").
    setWidget: (id, content, options) => {
      if (content == null) {
        // Removal also retires a factory-backed registration (upstream
        // setWidget(key, undefined) removes the widget entirely).
        if (state.rendererMeta.widgetRenderers[String(id)] !== undefined) {
          delete state.rendererMeta.widgetRenderers[String(id)];
          markRegistryDirty();
        }
        __pir_host.setWidgetJson(String(id), null);
        return;
      }
      if (ctxtMode === "rpc") {
        // Upstream rpc-mode ignores component factories (no TUI access) and
        // forwards only string arrays.
        if (Array.isArray(content)) {
          __pir_host.setWidgetJson(String(id), JSON.stringify(content));
        }
        return;
      }
      if (typeof content === "function") {
        // Upstream overload 2 (types.ts:188-193): the factory is called once
        // and returns a component ({render(width) → string[]/string,
        // invalidate()}); the TUI re-renders the component on every frame
        // refresh. The component cannot cross into QuickJS, so pir registers
        // the factory in the JS-side `widgetRenderers` registry and the TUI
        // renders through the `__pir_callRenderer` widget round-trip (the
        // same actor bridge as the registerMessageRenderer family): each
        // refresh re-invokes `component.render(width)` and consumes the
        // returned lines. The factory receives a minimal `tui` stub
        // (`requestRender` → repaint signal) and the live theme snapshot —
        // upstream's rpiv-todo overlay captures exactly these two.
        const normalizedPlacement =
          options?.placement === "belowEditor" || options?.placement === "below_transcript"
            ? "below_transcript"
            : "above_transcript";
        state.rendererMeta.widgetRenderers[String(id)] = {
          key: String(id),
          // The exported-defs registry check reads `customType` (the same
          // key field the message/entry buckets use).
          customType: String(id),
          options: content,
          placement: normalizedPlacement,
          component: undefined,
        };
        markRegistryDirty();

        // factory-backed (rendered on demand) instead of static text.
        __pir_host.setWidgetJson(
          String(id),
          JSON.stringify({ factory: true, placement: normalizedPlacement }),
        );
        return;
      }
      let text;
      let placement = options?.placement;
      if (Array.isArray(content)) {
        text = content.map((line) => String(line)).join("\n");
      } else if (typeof content === "string") {
        text = content;
      } else if (typeof content === "object" && typeof content.text === "string") {
        // Already-normalized (legacy pir) shape: {text, placement?}.
        text = content.text;
        if (placement === undefined) placement = content.placement;
      } else {
        throw new TypeError(
          "ctx.ui.setWidget content must be a string[], a component factory, or undefined to remove the widget",
        );
      }
      // A static (re-)registration replaces a factory-backed one: retire the
      // JS factory so the TUI's static text wins.
      if (state.rendererMeta.widgetRenderers[String(id)] !== undefined) {
        delete state.rendererMeta.widgetRenderers[String(id)];
        markRegistryDirty();
      }
      const normalizedPlacement =
        placement === "belowEditor" || placement === "below_transcript"
          ? "below_transcript"
          : "above_transcript";
      __pir_host.setWidgetJson(
        String(id),
        JSON.stringify({ text, placement: normalizedPlacement }),
      );
    },
    // Upstream `setFooter(factory?)` (types.ts:195-204): the factory receives
    // `(tui, theme, footerData)` and its component's `render(width)` lines
    // REPLACE the built-in footer; `undefined` restores the built-in footer.
    // The component is created ONCE per registration (upstream calls the
    // factory once and keeps the component) and re-rendered by the TUI through
    // the shared renderer round-trip; `dispose()` runs on replacement/removal
    // (upstream disposes the previous custom footer).
    setFooter: (factory) => {
      if (factory != null) {
        if (typeof factory !== "function") {
          throw new TypeError("ctx.ui.setFooter requires a component factory or undefined");
        }
        // Replacing a registered custom footer disposes the previous
        // component (upstream `setExtensionFooter` disposes `customFooter`).
        const previous = state.footerFactory;
        if (previous) {
          try { previous.component?.dispose?.(); } catch { /* dispose errors are isolated */ }
        }

        state.footerFactory = {
          factory,
          component: undefined,
          footerApi: footerData.api,
          footerApply: footerData.apply,
        };
        // Export the slot registration so the host (native factory
        // precedence/removal) can see that a JS footer is still registered.
        markRegistryDirty();
        __pir_host.setFooterJson(true);
        return;
      }
      const previous = state.footerFactory;
      state.footerFactory = null;
      if (previous) {
        try { previous.component?.dispose?.(); } catch { /* dispose errors are isolated */ }
      }
      markRegistryDirty();
      __pir_host.setFooterJson(false);
    },
    // Upstream `setHeader(factory?)` (types.ts:206-208): shown above the
    // chat; `undefined` restores the built-in startup header. Same factory
    // pipeline as the footer (minus the footerData argument).

      if (factory != null) {
        if (typeof factory !== "function") {
          throw new TypeError("ctx.ui.setHeader requires a component factory or undefined");
        }
        // Replacing a registered custom header disposes the previous
        // component (upstream `setExtensionHeader` disposes `customHeader`).
        const previous = state.headerFactory;
        if (previous) {
          try { previous.component?.dispose?.(); } catch { /* dispose errors are isolated */ }
        }
        state.headerFactory = { factory, component: undefined };
        markRegistryDirty();
        __pir_host.setHeaderJson(true);
        return;
      }
      const previous = state.headerFactory;
      state.headerFactory = null;
      if (previous) {
        try { previous.component?.dispose?.(); } catch { /* dispose errors are isolated */ }
      }
      markRegistryDirty();
      __pir_host.setHeaderJson(false);
    },
    // Upstream `setWorkingIndicator(options?)` (types.ts:176-184): omit the
    // argument to restore the default animated spinner; `frames: ["●"]` is a
    // static indicator; `frames: []` hides the indicator entirely; custom
    // frames render verbatim (the extension owns their colors). The TUI host
    // applies frames/intervalMs to its live spinner.
    setWorkingIndicator: (options) => {
      if (options == null) {
        __pir_host.setWorkingIndicatorJson(null);
        return;
      }

        throw new TypeError("ctx.ui.setWorkingIndicator requires an options object or undefined");

        payload.frames = options.frames.map((frame) => String(frame));
      }

        typeof options.intervalMs === "number"
        && Number.isFinite(options.intervalMs)

      }

    },
    // Upstream `setWorkingMessage(message?)` (types.ts:174-175): no argument
    // restores the default label. The host channel encodes "restore default"
    // as the empty message.
    setWorkingMessage: (message) => {
      __pir_host.setWorkingMessage(message == null ? null : String(message));
    },
    // Upstream `setWorkingVisible(visible)` (types.ts:171-172): show or hide
    // the built-in working loader row. This rides its own host fn and
    // notification — it must NOT be encoded as the working-message reset
    // signal, because an empty working message means "streaming" to the TUI
    // (working_message.is_some()), which misjudges the run state when the
    // extension only asked to hide the row.
    setWorkingVisible: (visible) => {
      __pir_host.setWorkingVisible(!!visible);
    },
    // Editor plumbing: TUI hosts apply through the live editor (notifications
    // + a pushed text snapshot); headless hosts ignore the writes and read
    // "" (upstream no-ops when no UI is attached too).
    setEditorText: (text) => { __pir_host.setEditorText(String(text)); },
    getEditorText: () => __pir_host.getEditorTextValue(),
    pasteToEditor: (text) => { __pir_host.pasteToEditor(String(text)); },
    // Upstream `setHiddenThinkingLabel(label?)` (types.ts:185): sets the label

      __pir_host.setHiddenThinkingLabel(label == null ? null : String(label));

    // (types.ts:250-278): the core editor component factory slot. pir's editor

      if (factory != null && typeof factory !== "function") {
        throw new TypeError("ctx.ui.setEditorComponent requires a factory function or undefined");

      if (factory != null) {

            "ctx.ui.setEditorComponent: pir keeps its built-in editor; the custom editor factory is stored but never rendered",

        } catch { /* the warning is best-effort */ }

    // Upstream `getEditorComponent()` returns the factory configured through
    // `setEditorComponent`, or undefined when the default editor is in use.
    getEditorComponent: () => state.editorComponentFactory,
    // Raw terminal input (upstream types.ts:163, interactive mode only).
    // Handlers run through the host's `terminal_input` bridge event: the TUI

    },
    // Autocomplete provider: pir simplification of the upstream factory
    // wrapping — the callback receives the serialized completion context
    // ({lines, cursorLine, cursorCol, prefix}) and returns an item array
    // (or null/undefined to pass). First registered non-null result wins.
    addAutocompleteProvider: (factory) => {
      if (typeof factory !== "function") {
        throw new TypeError("addAutocompleteProvider requires a function");
      }
      state.autocompleteSuggesters.push(factory);
      __pir_host.autocompleteProviderRegistered();
      // Upstream factory-chain semantics (types.ts:142 AutocompleteProviderFactory,
      // interactive-mode.ts setupAutocompleteProvider): each factory WRAPS the
      // previous provider (`factory(current)`), and the last registration is
      // outermost. pir's suggesters list is therefore collapsed to exactly one
      // entry: the outermost wrapper. The first factory receives a `current`
      // shim whose getSuggestions returns null — the built-in provider lives
      // host-side and has already run when the JS chain is consulted (bridge
      // limitation: the host-side provider cannot cross into JS).
      const current = state.autocompleteChain;
      const currentShim = current
        ? { getSuggestions: async (context) => current(context) }
        : { getSuggestions: async () => null };
      const provider = factory(currentShim);
      const suggest =
        typeof provider === "function"
          ? provider
          : provider && typeof provider.getSuggestions === "function"
            ? (context) => provider.getSuggestions(context)
            : null;
      if (!suggest) {
        throw new TypeError(
          "addAutocompleteProvider factory must return a provider with getSuggestions",
        );
      }
      state.autocompleteChain = async (context) => suggest(context);
      state.autocompleteSuggesters = [state.autocompleteChain];
    },
  };
  // Upstream `SessionManager.getTree()` answers with nested `SessionTreeNode`
  // objects (`{entry, children, label, labelTimestamp}`, session-manager.ts:
  // 199-207), and `getEntry()`/`getLeafEntry()` resolve entries by id across
  // the whole forest. Walk the node forest iteratively (deep linear chains
  // must not overflow the JS stack, matching the host-side builder); legacy
  // flat snapshots (plain entry objects without an `entry`/`children`
  // wrapper) still resolve.
  const sessionTreeNodes = (parsed) => {
    const nodes = [];
    const stack = Array.isArray(parsed.tree) ? parsed.tree.slice() : [];
    while (stack.length > 0) {
      const node = stack.pop();
      if (!node || typeof node !== "object") continue;
      nodes.push(node);
      if (Array.isArray(node.children)) {
        for (const child of node.children) stack.push(child);
      }
    }
    return nodes;
  };
  const findSessionEntry = (parsed, id) => {
    const node = sessionTreeNodes(parsed).find(
      (candidate) =>
        candidate.id === id ||
        (candidate.entry !== undefined && candidate.entry.id === id),
    );
    if (node === undefined) return undefined;
    // Nested nodes wrap the entry (`node.entry`); legacy flat snapshots ARE
    // the entry.
    return node.entry !== undefined ? node.entry : node;
  };
  const sessionManager = {
    getEntries: () => {
      const parsed = JSON.parse(__pir_host.getSessionManagerJson());
      return parsed.entries ?? [];
    },
    // Scalar getters read the host's live snapshot fields directly:
    // parsing `getSessionManagerJson` for one string meant serializing the
    // whole context projection, tree text and session listing on every
    // `getSessionId()`/`getCwd()`/... call. Return shapes are unchanged
    // (`null`/absent -> `undefined`).
    getSessionId: () => __pir_host.getSessionIdJson() ?? undefined,
    getSessionPath: () => __pir_host.getSessionPathJson() ?? undefined,
    getSessionName: () => __pir_host.getSessionName(),
    getSessions: () => {
      const parsed = JSON.parse(__pir_host.getSessionManagerJson());
      return parsed.sessions ?? [];
    },
    getCwd: () => __pir_host.getSessionCwdJson() ?? undefined,
    getSessionFile: () => __pir_host.getSessionPathJson() ?? undefined,
    getHeader: () => {
      const header = parseHostJson(__pir_host.getSessionHeaderJson());
      return header ?? undefined;
    },
    getLeafId: () => __pir_host.getSessionLeafIdJson() ?? undefined,
    getLeafEntry: () => {
      const parsed = JSON.parse(__pir_host.getSessionManagerJson());
      const leafId = parsed.leafId;
      if (leafId == null) return undefined;
      return findSessionEntry(parsed, leafId);
    },
    getEntry: (id) => {
      const parsed = JSON.parse(__pir_host.getSessionManagerJson());
      return findSessionEntry(parsed, id);
    },
    getBranch: () => {
      const parsed = JSON.parse(__pir_host.getSessionManagerJson());
      return parsed.branch ?? [];
    },
    getTree: () => {
      const parsed = JSON.parse(__pir_host.getSessionManagerJson());
      return parsed.tree ?? [];
    },
    getLabel: (id) => __pir_host.getSessionLabelJson(String(id)) ?? undefined,
    buildContextEntries: () => {
      const parsed = JSON.parse(__pir_host.getSessionManagerJson());
      return parsed.entries ?? [];
    },
    buildSessionProjection: () => {
      const parsed = JSON.parse(__pir_host.getSessionManagerJson());
      return parsed.projection ?? { entries: [], messages: [] };
    },
    // Upstream `buildSessionContext()` (session-manager.ts:576-581,
    // 1497-1500) finalizes the model context:
    // `{ messages, thinkingLevel, model }`. The host snapshot already
    // carries the compaction-aware projection (`session_ctx.rs` pushes
    // `buildSessionProjection()`), so the messages ARE the projection's
    // messages — serialized pir `Message` objects that keep the
    // `toolsAdded`/`toolsRemoved` arrays on system messages (the dynamic
    // tool-activation contract pi-subagents relies on). Upstream
    // consumers (tool-activation.js, sdk.ts, agent-session.ts) read
    // `messages` only; thinkingLevel/model ride `ctx.thinkingLevel` /
    // `ctx.model`, so the context carries just the consumed field.
    // Without a projection snapshot (headless / --no-session hosts) the
    // `entries` snapshot IS the message list (session hosts push
    // `buildContextEntries()`, print hosts the raw history), so it
    // degrades to the same shape.
    buildSessionContext: () => {
      const parsed = JSON.parse(__pir_host.getSessionManagerJson());
      const projection = parsed.projection;
      if (projection && typeof projection === "object" && Array.isArray(projection.messages)) {
        return { messages: projection.messages };
      }
      return { messages: Array.isArray(parsed.entries) ? parsed.entries : [] };
    },
  };
  // Upstream `ctx.modelRegistry` is the core `ModelRegistry`
  // (core/model-registry.ts:48-244): the example extensions call
  // `find(provider, modelId)` / `hasConfiguredAuth(model)` /
  // `complete(model, context, options)` (model-registry.ts:72-74, 85-87,
  // 142-148) and the npm packages read the provider/auth surface. Backed by
  // the host-pushed registry snapshot (`getModelsJson`) and the host-pushed
  // authenticated-provider list; the credential-bearing methods additionally
  // read the credential sources the extension runtime can actually reach (see
  // the credential note below).
  const modelRegistrySnapshot = () => {
    const parsed = parseHostJson(__pir_host.getModelsJson()) ?? {};
    return {
      models: Array.isArray(parsed.models) ? parsed.models : [],
      providers: Array.isArray(parsed.providers) ? parsed.providers : [],
    };
  };
  // Upstream `hasConfiguredAuth(providerId)` (model-runtime.ts:544-546) reads
  // the resolved credential snapshot; the host's answer is its
  // authenticated-provider list under the permissive-empty convention
  // (context_fns.rs `hasAuthenticatedProvider`): an empty list means the host
  // cannot check (script/headless hosts), a non-empty list is authoritative.
  const providerHasConfiguredAuth = (providerId) => {
    if (typeof providerId !== "string" || !providerId) return false;
    return __pir_host.hasAuthenticatedProvider(providerId) === true;
  };
  // Accepts a provider id, a `Model` (`provider`) or a `Provider` (`id`) —
  // the registry methods below are documented per-model or per-provider and
  // real consumers pass both.
  const providerIdOf = (value) => {
    if (typeof value === "string") return value || undefined;
    if (value && typeof value === "object") {
      if (typeof value.provider === "string" && value.provider) return value.provider;
      if (typeof value.id === "string" && value.id) return value.id;
    }
    return undefined;
  };

  // -------------------------------------------------------------------
  // Credential access for the registry methods
  //
  // The host exposes exactly one credential answer (`hasAuthenticatedProvider`)
  // — credential VALUES never cross a host op. The extension runtime can still
  // read two of the three sources the host itself resolves:
  //  - the stored credential store (`<agentDir>/auth.json`) through the fs
  //    bridge (extension reads are global, node_fns.rs `read_path`), and
  //  - the provider's standard environment variable, delegated to the pi-ai
  //    shim's shipped env-key table (`getEnvApiKey`, the env-api-keys.ts port)
  //    so the ~40-provider map is not duplicated here.
  // Provider credentials that live only in models.json (`apiKey`/`command`
  // config) stay host-side and degrade to "configured, not readable" — never a
  // fabricated value.
  // -------------------------------------------------------------------

  const readAuthJsonEntry = (providerId) => {
    try {
      const agentDir = String(__pir_host.getAgentDir() ?? "").replace(/[\\/]+$/, "");
      if (!agentDir || !providerId) return undefined;
      const text = Buffer.from(
        __pirNodeCall("fs", "readFile", { path: agentDir + "/auth.json" }),
        "base64",
      ).toString("utf8");
      const data = JSON.parse(text.replace(/^\uFEFF/, ""));
      const credential = data ? data[providerId] : undefined;
      return credential && typeof credential === "object" ? credential : undefined;
    } catch {
      // Missing / unreadable / malformed store: no readable credential.
      return undefined;
    }
  };

  // Provider env-var keys through the pi-ai shim (resolved once, cached). An
  // unavailable shim degrades to "no env key".
  let envApiKeyLookup; // undefined = unresolved, null = unavailable
  let envApiKeyLookupPending;
  const lookupEnvApiKey = async (providerId) => {
    if (envApiKeyLookup === undefined) {
      envApiKeyLookupPending ??= import("@earendil-works/pi-ai").then(
        (mod) => {
          envApiKeyLookup = typeof mod.getEnvApiKey === "function" ? mod.getEnvApiKey : null;
        },
        () => { envApiKeyLookup = null; },
      );
      await envApiKeyLookupPending;
    }
    if (envApiKeyLookup === null) return undefined;
    try { return envApiKeyLookup(providerId) ?? undefined; } catch { return undefined; }
  };

  // Resolve the upstream `AuthResult` shape
  // (`{auth: {apiKey?, headers?, baseUrl?}, env?, source?}`, packages/ai/src/
  // auth/types.ts:104-110) for one provider. Returns `undefined` when nothing
  // resolves (upstream semantic) and `{auth: {}}` when the provider IS
  // configured but its credential value is host-side only. Precedence follows
  // upstream `resolveProviderAuth` (packages/ai/src/auth/resolve.ts:52-88): a
  // stored credential owns the provider; the registered provider config and
  // the environment are consulted only without one.
  const resolveProviderAuth = async (providerId) => {
    const stored = readAuthJsonEntry(providerId);
    if (stored) {
      if (stored.type === "api_key") {
        const auth = {};
        if (typeof stored.key === "string" && stored.key) auth.apiKey = stored.key;
        if (stored.env && typeof stored.env === "object") return { auth, env: stored.env };
        return { auth };
      }
      if (stored.type === "oauth") {
        // OAuth: hand out the access token only while it is comfortably valid.
        // Upstream refreshes inside `getAuth` through the provider's
        // `refresh`/`toAuth` (host-side for builtin providers); without that
        // the honest answers are the token or no request auth at all — never a
        // stale one. Upstream's builtin `toAuth` is `{apiKey: access}` for the
        // providers using this shape (anthropic/openai-codex/xai/radius).
        if (
          typeof stored.access === "string" && stored.access
          && Number(stored.expires) > Date.now() + 5 * 60 * 1000
        ) {
          return { auth: { apiKey: stored.access } };
        }
        return { auth: {} };
      }
      return { auth: {} };
    }
    // Full `Provider` registration (upstream `Provider.auth.apiKey.resolve`,
    // packages/ai/src/auth/types.ts:174-212). A stored credential owns the
    // provider (upstream precedence); without one the provider's own resolver
    // merges its ambient sources (env vars, profile files) into request auth.
    const resolver = state.providerApiKeyResolvers[providerId];
    if (resolver && resolver.auth && typeof resolver.auth.resolve === "function") {
      const authContext = {
        env: async (name) => {
          try { return process.env ? process.env[name] : undefined; } catch { return undefined; }
        },
        fileExists: async (path) => {
          try {
            return __pirNodeCall("fs", "exists", { path: String(path) }) === true;
          } catch {
            return false;
          }
        },
      };
      try {
        const resolved = await resolver.auth.resolve({
          ctx: authContext,
          ...(stored && stored.type === "api_key" ? { credential: stored } : {}),
          signal: makeSignal(),
        });
        if (resolved && typeof resolved === "object" && resolved.auth) return resolved;
        return undefined;
      } catch {
        // A resolver failure is "not configured" for request auth, never a
        // fabricated key (upstream getAuth propagates; the registry methods
        // that use this answer with their unauthenticated fallback).
        return undefined;
      }
    }
    const registered = state.providerConfigs[providerId];
    if (registered && typeof registered.apiKey === "string" && registered.apiKey) {
      return { auth: { apiKey: registered.apiKey } };
    }
    const envKey = await lookupEnvApiKey(providerId);
    if (typeof envKey === "string" && envKey) return { auth: { apiKey: envKey } };
    if (providerHasConfiguredAuth(providerId)) return { auth: {} };
    return undefined;
  };

  // -------------------------------------------------------------------
  // Provider stream dispatch (stream / streamSimple / complete)
  //
  // Upstream routes `ModelRegistry.stream/streamSimple/complete` through the
  // provider's api implementation with request-time auth (model-registry.ts:
  // 129-148). For providers an extension registered with
  // `pi.registerProvider(name, {api, apiKey, models, streamSimple})` the
  // implementation IS the JS callback held in `state.providerStreamHooks` —
  // the same callback the host drives per request through `js_stream.rs` — so
  // the registry dispatches to it directly and the call is REAL. Providers
  // executed host-side (builtin catalog / models.json providers with no
  // extension callback) have no such callback here: `stream`/`streamSimple`
  // therefore fail closed with a clear error instead of a fabricated stream.
  // `complete` additionally has a host path (`modelCompleteJson`, backed by
  // the session host's `AgentSession::complete_fn_for`) — see its method.
  // -------------------------------------------------------------------

  const dispatchProviderStream = (model, context, options) => {
    if (
      !model || typeof model !== "object"
      || typeof model.provider !== "string" || !model.provider
      || typeof model.id !== "string" || !model.id
    ) {
      throw new TypeError("modelRegistry stream calls require a model ({provider, id, ...})");
    }
    const hook = state.providerStreamHooks[model.provider];
    if (!hook || typeof hook.fn !== "function") {
      throw new Error(
        `Provider "${model.provider}" cannot be streamed from an extension: this pir extension ` +
        `runtime does not expose the host model runtime, so only providers an extension ` +
        `registers itself (pi.registerProvider(name, {api, apiKey, models, streamSimple})) have ` +
        `a stream implementation here.`,
      );
    }
    const registered = state.providerConfigs[model.provider];
    if (
      registered && typeof registered.api === "string" && registered.api
      && typeof model.api === "string" && model.api && registered.api !== model.api
    ) {
      // Upstream compat wrapper guard (`Mismatched api: <got> expected <want>`).
      throw new Error(`Mismatched api: ${model.api} expected ${registered.api}`);
    }
    const requestContext = context && typeof context === "object" ? context : { messages: [] };
    const requestOptions = options && typeof options === "object" ? { ...options } : {};
    // Request-time auth (upstream resolves the provider credential and hands
    // it to the implementation): the apiKey from the registration config
    // rides along when the caller did not pass one.
    if (
      requestOptions.apiKey === undefined
      && registered && typeof registered.apiKey === "string" && registered.apiKey
    ) {
      requestOptions.apiKey = registered.apiKey;
    }
    return hook.fn(model, requestContext, requestOptions);
  };

  const streamTerminalMessage = (event) => {
    if (!event || typeof event !== "object") return undefined;
    if (event.type === "done") return event.message;
    if (event.type === "error") return event.error;
    return undefined;
  };

  // Drain whatever a provider callback produced to its final assistant
  // message. Everything the host's JS-stream bridge accepts is handled here
  // (js_stream.rs wire contract): a live `AssistantMessageEventStream`
  // (`result()`), a batch event array / `{events: [...]}`, an async iterable
  // (async generator) or a raw `{next()}` iterator. An async callback (a
  // Promise of any of those) is awaited first.
  const drainProviderStream = async (produced) => {
    const value = await produced;
    if (value && typeof value.result === "function") return await value.result();
    const events = Array.isArray(value)
      ? value
      : value && Array.isArray(value.events) ? value.events : undefined;
    if (events !== undefined) {
      for (let index = events.length - 1; index >= 0; index--) {
        const message = streamTerminalMessage(events[index]);
        if (message !== undefined) return message;
      }
      return undefined;
    }
    if (value && typeof value[Symbol.asyncIterator] === "function") {
      let message;
      for await (const event of value) {
        const terminal = streamTerminalMessage(event);
        if (terminal !== undefined) message = terminal;
      }
      return message;
    }
    if (value && typeof value.next === "function") {
      let message;
      for (;;) {
        const step = await value.next();
        if (!step || step.done) break;
        const terminal = streamTerminalMessage(step.value);
        if (terminal !== undefined) message = terminal;
      }
      return message;
    }
    return undefined;
  };

  // `stream`/`streamSimple` hand back what the provider callback produced. A
  // live stream — `result()` plus async iteration, i.e. the shim's
  // `AssistantMessageEventStream`, which a consumer may both iterate and take
  // the result from — passes through unchanged (upstream returns that shape);
  // everything else (an async callback's promise, a batch event array, a bare
  // async generator or `{next()}` iterator) is adapted to it. The adapter
  // records the terminal message while iterating, so the upstream
  // "iterate, then `await stream.result()`" pattern works on adapted streams;
  // the underlying single-pass source is consumed at most once.
  const adaptProviderStream = (produced) => {
    if (
      produced && typeof produced === "object"
      && typeof produced.result === "function"
      && typeof produced[Symbol.asyncIterator] === "function"
    ) {
      return produced;
    }
    let lastTerminal;
    const events = async function* () {
      const value = await produced;
      const batch = Array.isArray(value)
        ? value
        : value && Array.isArray(value.events) ? value.events : undefined;
      if (batch !== undefined) {
        for (const event of batch) {
          const terminal = streamTerminalMessage(event);
          if (terminal !== undefined) lastTerminal = terminal;
          yield event;
        }
        return;
      }
      if (value && typeof value[Symbol.asyncIterator] === "function") {
        for await (const event of value) {
          const terminal = streamTerminalMessage(event);
          if (terminal !== undefined) lastTerminal = terminal;
          yield event;
        }
        return;
      }
      if (value && typeof value.next === "function") {
        for (;;) {
          const step = await value.next();
          if (!step || step.done) return;
          const terminal = streamTerminalMessage(step.value);
          if (terminal !== undefined) lastTerminal = terminal;
          yield step.value;
        }
      }
    };
    return {
      result: async () => {
        if (lastTerminal === undefined) {
          // Drain (records the terminal message); an already-consumed source
          // still answers from the recorded one.
          for await (const _event of events()) { /* draining */ }
        }
        if (lastTerminal === undefined) {
          throw new Error("modelRegistry stream: the provider stream ended without a final assistant message");
        }
        return lastTerminal;
      },
      [Symbol.asyncIterator]: () => events(),
    };
  };

  // Upstream `ModelRegistry.runtime` is the PRIVATE `ModelRuntime` field
  // (model-registry.ts:49) — TypeScript consumers cannot reach it, so nothing
  // legitimate reads it. It is emulated lazily as the readable ModelRuntime
  // subset (provider/model/auth reads plus `stream`/`streamSimple`/`complete`);
  // the mutation surfaces (registerProvider / refresh / credential operations)
  // live host-side and stay absent rather than pretending to work.
  let modelRuntimeShim;

  // Upstream `Models.getAvailable()` applies each provider's `filterModels`
  // policy after auth is confirmed (models.ts:198-213). The hook is a plain
  // JS callback from a full `Provider` registration; a throwing hook keeps the
  // unfiltered list (availability is best-effort, never fatal).
  const applyProviderModelFilters = (models) => {
    let filtered = models;
    for (const [providerId, hook] of Object.entries(state.providerFilterHooks)) {
      if (!hook || typeof hook.fn !== "function") continue;
      const forProvider = filtered.filter((model) => providerIdOf(model) === providerId);
      if (forProvider.length === 0) continue;
      let keep;
      try {
        keep = hook.fn(forProvider, undefined);
      } catch {
        continue;
      }
      if (!Array.isArray(keep)) continue;
      const keepSet = new Set(keep);
      filtered = filtered.filter(
        (model) => providerIdOf(model) !== providerId || keepSet.has(model),
      );
    }
    return filtered;
  };

  // The extension API object for a ctx-level registry mutation. The registry
  // is process-wide; any ACTIVE extension may perform the write (upstream's
  // `ModelRegistry` is not tied to one extension and its methods do not assert
  // an owner).
  const activePiApi = () => {
    const ids = Object.keys(state.extensions).sort((a, b) => Number(a) - Number(b));
    for (const id of ids) {
      const ext = state.extensions[id];
      if (ext && ext.status === "active" && state.piApis[id]) return state.piApis[id];
    }
    return undefined;
  };
  const requirePiApi = () => {
    const api = activePiApi();
    if (!api) throw new Error("No active extension can mutate the model registry");
    return api;
  };

  const modelRegistry = {
    listModels: () => modelRegistrySnapshot().models,
    // Upstream `ModelRegistry.find(provider, modelId)`
    // (model-registry.ts:72-74).
    find: (provider, modelId) =>
      modelRegistrySnapshot().models.find((m) => m.provider === provider && m.id === modelId),
    // Upstream runtime call signature is `getModel(provider, modelId)` and
    // this is the CANONICAL order: the first lookup below is the aligned one.
    // The legacy pir signature was reversed — `getModel(id, provider)` — and
    // is still resolved by the fallback lookup for compatibility (both call
    // styles coexist; new code should prefer (provider, modelId) or `find`).
    getModel: (provider, modelId) => {
      const models = modelRegistrySnapshot().models;
      const aligned = models.find((m) => m.provider === provider && m.id === modelId);
      if (aligned !== undefined) return aligned;
      return models.find(
        (m) => m.id === provider && (modelId === undefined || m.provider === modelId),
      );
    },
    // Upstream `ModelRegistry.getAll()`: every registered model.
    getAll: () => modelRegistrySnapshot().models,
    // Upstream `ModelRegistry.getAvailable()` (model-registry.ts:68-70):
    // every model whose provider holds working credentials. pir derives it
    // from the authenticated-provider list; the permissive-empty convention
    // above applies, so without a pushed list this equals `getAll()`.
    getAvailable: () =>
      applyProviderModelFilters(
        modelRegistrySnapshot().models.filter((model) => providerHasConfiguredAuth(providerIdOf(model))),
      ),
    // Upstream `ModelRegistry.hasConfiguredAuth(model)`: whether the model's
    // provider holds a usable credential. The host answer is permissive when
    // the authenticated-provider list was never pushed (host cannot check —
    // same convention as `pi.setModel`); accepts a Model or a provider id.
    hasConfiguredAuth: (model) => providerHasConfiguredAuth(providerIdOf(model)),
    listProviders: () => modelRegistrySnapshot().providers,
    // Upstream `ModelRegistry.getProviderDisplayName(provider)`
    // (model-registry.ts:190-192): the provider's display name, falling back to
    // the id (the host snapshot carries `{id, name}` per provider).
    getProviderDisplayName: (provider) => {
      const providerId = providerIdOf(provider);
      if (!providerId) return "";
      const entry = modelRegistrySnapshot().providers.find(
        (candidate) => candidate && candidate.id === providerId,
      );
      return entry && typeof entry.name === "string" && entry.name ? entry.name : providerId;
    },
    // Upstream `ModelRegistry.getProviderAuthStatus(provider)`
    // (model-registry.ts:120-122) answers `AuthStatus`
    // (`{configured, source?, label?}`, provider-composer.ts:110-114). pir
    // knows configured/not-configured only: `source`/`label` describe the
    // resolved credential, which the host does not expose (see the credential
    // note above), so they are omitted rather than guessed.
    getProviderAuthStatus: (provider) => {
      const providerId = providerIdOf(provider);
      return { configured: providerHasConfiguredAuth(providerId) };
    },
    // Upstream `ModelRegistry.getProviderAuth(provider)`
    // (model-registry.ts:194-196): `Promise<AuthResult | undefined>`.
    getProviderAuth: async (provider) => {
      const providerId = providerIdOf(provider);
      if (!providerId) return undefined;
      return await resolveProviderAuth(providerId);
    },
    // Upstream `ModelRegistry.getApiKeyAndHeaders(model)`
    // (model-registry.ts:89-118): `{ok: true, apiKey?, headers?, baseUrl?,
    // env?}` or `{ok: false, error}` — "configured but not readable" answers
    // with the upstream `No API key found for "<provider>"` error so consumers
    // take their unauthenticated fallback instead of sending an empty key.
    getApiKeyAndHeaders: async (model) => {
      const providerId = providerIdOf(model);
      if (!providerId) {
        return { ok: false, error: "No API key found for the requested model" };
      }
      let resolution;
      try {
        resolution = await resolveProviderAuth(providerId);
      } catch (error) {
        return { ok: false, error: error && error.message ? error.message : String(error) };
      }
      const auth = resolution && resolution.auth ? resolution.auth : undefined;
      if (!auth || typeof auth.apiKey !== "string" || !auth.apiKey) {
        return { ok: false, error: `No API key found for "${providerId}"` };
      }
      return {
        ok: true,
        apiKey: auth.apiKey,
        ...(auth.headers !== undefined ? { headers: auth.headers } : {}),
        ...(auth.baseUrl !== undefined ? { baseUrl: auth.baseUrl } : {}),
        ...(resolution.env !== undefined ? { env: resolution.env } : {}),
      };
    },
    // Upstream `ModelRegistry.getApiKeyForProvider(provider)`
    // (model-registry.ts:198-204): the resolved api key, or undefined — never
    // rejects (upstream catches too).
    getApiKeyForProvider: async (provider) => {
      try {
        const resolution = await modelRegistry.getProviderAuth(provider);
        const apiKey = resolution && resolution.auth ? resolution.auth.apiKey : undefined;
        return typeof apiKey === "string" && apiKey ? apiKey : undefined;
      } catch {
        return undefined;
      }
    },
    // Upstream `ModelRegistry.isUsingOAuth(model)` (model-registry.ts:206-208):
    // whether the provider's credential is an OAuth one. The stored store is
    // the readable source here, so a host-side (models.json/env) credential
    // reports false.
    isUsingOAuth: (model) => {
      const providerId = providerIdOf(model);
      if (!providerId) return false;
      const stored = readAuthJsonEntry(providerId);
      return stored !== undefined && stored.type === "oauth";
    },
    // Upstream `ModelRegistry.stream/streamSimple` (model-registry.ts:129-140)
    // hand the model to the provider's api implementation — here the
    // extension-registered `streamSimple` callback.
    stream: (model, context, options) =>
      adaptProviderStream(dispatchProviderStream(model, context, options)),
    streamSimple: (model, context, options) =>
      adaptProviderStream(dispatchProviderStream(model, context, options)),
    // Upstream `ModelRegistry.complete(...)`: `Promise<AssistantMessage>`.
    // Two real dispatch paths, in upstream precedence order:
    //  1. a provider an extension registered with a `streamSimple` callback —
    //     that callback IS the provider's api implementation, so the request
    //     runs here and is drained to its final assistant message
    //     (`AssistantMessageEventStream.result()` resolves `event.message` on
    //     `done` / `event.error` on `error` — the upstream terminal contract);
    //  2. the host model runtime through `__pir_host.modelCompleteJson` (the
    //     `AgentSession::complete_fn_for` capability the session host pushes),
    //     which is how builtin/models.json providers complete.
    // Without either, the call fails closed with the clear boundary error
    // `dispatchProviderStream` raises (never a fabricated message).
    complete: async (model, context, options) => {
      const hook =
        model && typeof model === "object" ? state.providerStreamHooks[model.provider] : undefined;
      if (hook && typeof hook.fn === "function") {
        const message = await drainProviderStream(dispatchProviderStream(model, context, options));
        if (message === undefined) {
          throw new Error(
            "modelRegistry.complete: the provider stream ended without a final assistant message",
          );
        }
        return message;
      }
      if (typeof __pir_host.modelCompleteJson === "function") {
        let out;
        try {
          out = await __pir_host.modelCompleteJson(
            JSON.stringify(model ?? {}),
            JSON.stringify(context ?? {}),
            JSON.stringify(options ?? {}),
          );
        } catch (error) {
          const detail = error && error.message ? error.message : String(error);
          throw new Error(`modelRegistry.complete: the host completion failed: ${detail}`);
        }
        const parsed = parseHostJson(out);
        if (parsed && typeof parsed === "object") {
          if (parsed.message !== undefined && parsed.message !== null) return parsed.message;
          if (typeof parsed.error === "string" && parsed.error) throw new Error(parsed.error);
        }
        throw new Error(
          "modelRegistry.complete: the host completion returned no assistant message",
        );
      }
      // No host capability: dispatchProviderStream raises the clear
      // "cannot be streamed from an extension" boundary error.
      const message = await drainProviderStream(dispatchProviderStream(model, context, options));
      if (message === undefined) {
        throw new Error(
          "modelRegistry.complete: the provider stream ended without a final assistant message",
        );
      }
      return message;
    },
    // Upstream `ModelRegistry.registerProvider` / `unregisterProvider`
    // (model-registry.ts:210-225): the same registration surface `pi` exposes,
    // reachable from a command/tool ctx too. The actual mutation runs through
    // the active extension API object (one registry, any owner).
    registerProvider: (providerOrName, config) => {
      const api = requirePiApi();
      if (typeof providerOrName === "string" && config === undefined) {
        throw new Error("Provider config is required when registering by name");
      }
      api.registerProvider(providerOrName, config);
    },
    unregisterProvider: (name) => {
      requirePiApi().unregisterProvider(name);
    },
    registerVirtualModel: (definition) => {
      requirePiApi().registerVirtualModel(definition);
    },
    unregisterVirtualModel: (provider, id) => {
      requirePiApi().unregisterVirtualModel(provider, id);
    },
    // Upstream `ModelRegistry.refresh(options?)` (models.ts:106-118):
    // `{aborted, errors: ReadonlyMap<providerId, Error>}`. Every registered
    // `refreshModels` callback runs; a native provider's `publish()` already
    // applied its in-memory catalog, and the host re-syncs the configured
    // model list on its next prompt.
    refresh: async (options) => {
      const errors = new Map();
      const wanted = options && Array.isArray(options.providers) ? new Set(options.providers) : null;
      for (const [providerId, hook] of Object.entries(state.providerRefreshHooks)) {
        if (wanted !== null && !wanted.has(providerId)) continue;
        if (!hook || typeof hook.fn !== "function") continue;
        try {
          const result = await hook.fn({
            allowNetwork: options?.allowNetwork !== false,
            force: options?.force === true,
            ...(options?.stored ? { stored: options.stored } : {}),
            publish: async (publication) => {
              if (publication && typeof publication.update === "function") {
                publication.update();
                return true;
              }
              return false;
            },
            signal: options?.signal ?? makeSignal(),
          });
          void result;
        } catch (error) {
          errors.set(providerId, error instanceof Error ? error : new Error(String(error)));
        }
      }
      return { aborted: false, errors };
    },
    // Upstream `ModelRegistry.generateImages` / `classify` route to the
    // provider implementation for the model's api (models.ts:234-246); a full
    // `Provider` registration's `images` / `classifiers` maps are those
    // implementations. Missing implementations reject with the upstream
    // "no implementation for <api>" wording rather than fabricating a result.
    generateImages: async (model, context, options) => {
      const providerId = providerIdOf(model);
      const entry = providerId ? state.providerImageHooks[providerId] : undefined;
      const impl = entry && entry.impl ? entry.impl[model?.api] : undefined;
      if (!impl || typeof impl.generateImages !== "function") {
        throw new Error(`Provider ${providerId ?? "<unknown>"} has no image implementation for "${model?.api}"`);
      }
      return await impl.generateImages(model, context, options);
    },
    classify: async (model, context, options) => {
      const providerId = providerIdOf(model);
      const entry = providerId ? state.providerClassifierHooks[providerId] : undefined;
      const impl = entry && entry.impl ? entry.impl[model?.api] : undefined;
      if (!impl || typeof impl.classify !== "function") {
        throw new Error(`Provider ${providerId ?? "<unknown>"} has no classifier implementation for "${model?.api}"`);
      }
      return await impl.classify(model, context, options);
    },
    // Readable ModelRuntime subset (see the note above).
    get runtime() {
      return (modelRuntimeShim ??= {
        getProviders: () => modelRegistry.listProviders(),
        getProvider: (provider) => {
          const providerId = providerIdOf(provider);
          return modelRegistry.listProviders().find((candidate) => candidate.id === providerId);
        },
        getModels: () => modelRegistry.getAll(),
        getAllModels: () => modelRegistry.getAll(),
        getModel: (provider, id) => modelRegistry.find(provider, id),
        getAvailable: async () => modelRegistry.getAvailable(),
        hasConfiguredAuth: (provider) => providerHasConfiguredAuth(providerIdOf(provider)),
        getProviderAuthStatus: (provider) => modelRegistry.getProviderAuthStatus(provider),
        getAuth: (provider) => modelRegistry.getProviderAuth(provider),
        isUsingOAuth: (provider) => modelRegistry.isUsingOAuth(provider),
        stream: (model, context, options) => modelRegistry.stream(model, context, options),
        streamSimple: (model, context, options) => modelRegistry.streamSimple(model, context, options),
        complete: (model, context, options) => modelRegistry.complete(model, context, options),
        registerProvider: (providerOrName, config) =>
          modelRegistry.registerProvider(providerOrName, config),
        unregisterProvider: (provider) => modelRegistry.unregisterProvider(provider),
        registerVirtualModel: (definition) => modelRegistry.registerVirtualModel(definition),
        unregisterVirtualModel: (provider, id) => modelRegistry.unregisterVirtualModel(provider, id),
        refreshModels: (options) => modelRegistry.refresh(options),
        generateImages: (model, context, options) =>
          modelRegistry.generateImages(model, context, options),
        classify: (model, context, options) => modelRegistry.classify(model, context, options),
      });
    },
  };
  const toolCallId = desc.toolCallId ?? "";
  const ctx = {
    cwd: desc.cwd,
    // Exactly "tui" | "rpc" | "json" | "print" (host-normalized; defensive
    // JS-side fallback kept for direct runtime construction).
    mode: ctxtMode,
    hasUI,
    model: desc.model ?? undefined,
    thinkingLevel: desc.thinkingLevel ?? undefined,
    scopedModels: desc.scopedModels ?? [],
    sessionManager,
    modelRegistry,
    signal: makeSignal(),
    ui,
    isIdle: () => __pir_host.isIdle(),
    // Upstream `ExtensionCommandContext.waitForIdle` (types.ts:405-406):
    // resolve when no prompt run is in flight. The host fn answers against
    // the session-pushed streaming flag and is deadline-bounded.
    waitForIdle: async () => {
      await __pir_host.waitForIdle();
    },
    hasPendingMessages: () => __pir_host.hasPendingMessages(),
    isProjectTrusted: () => __pir_host.isProjectTrusted(),
    abort: () => { __pir_host.abort(); notifyAbort(); },
    // Upstream `ExtensionContext.shutdown` (`agent-session.ts:3510-3512`):
    // routes to the host's shutdown handler. It is NOT an abort — the host
    // exits at its idle / agent_settled boundary and a run in flight settles
    // normally, so the local abort signal is deliberately left untouched.
    shutdown: () => { __pir_host.shutdown(); },
    // `getSystemPrompt` sees the chained before_agent_start override
    // (upstream renderCurrentSystemPrompt); without one the host value.
    getSystemPrompt: () => {
      const override =
        typeof getSystemPromptOverride === "function" ? getSystemPromptOverride() : undefined;
      return override !== undefined ? override : __pir_host.getSystemPrompt();
    },
    getContextUsage: () => {
      const parsed = JSON.parse(__pir_host.getContextUsageJson());
      return parsed.usage ?? undefined;
    },
    // Upstream `ExtensionContext.compact(options)` (`types.ts:314-318`):
    // `onComplete`/`onError` fire when the compaction the request started
    // settles. The callbacks stay in JS state; the internal settlers below
    // observe the host's `session_compact` / `session_compact_failed` events
    // (the same events every extension sees) and resolve the pending entries.
    compact: (options) => {
      const onComplete = typeof options?.onComplete === "function" ? options.onComplete : null;
      const onError = typeof options?.onError === "function" ? options.onError : null;
      if (onComplete !== null || onError !== null) {
        state.pendingCompactCallbacks.push({ extId, onComplete, onError });
        ensureCompactSettler(extId);
      }
      const accepted = __pir_host.requestCompact(
        options?.customInstructions == null ? null : String(options.customInstructions),
      );
      if (accepted === false) {
        // The host refused (a prompt run is in flight; upstream's manual
        // compact cannot start while streaming). Resolve the callbacks now
        // instead of leaving them pending for a compaction that never ran.
        settleCompactCallbacks("session_compact_failed", {
          errorMessage: "Cannot compact while the agent is streaming",
        });
      }
    },
    // Tool context surface (upstream ExtensionToolContext): `tools` lists the
    // callable tool definitions; `executeTool` runs one through the nested
    // gate/scheduler path with `parentToolCallId` semantics. `reachesTools`
    // hooks are re-attached live (upstream ToolDefinition.reachesTools).
    //
    // `options.background` (TOOL-BACKGROUND-SCOPE.md §4): the scheduler's
    // background request. It travels as an EXPLICIT flag on the host call,
    // never merged into `args` — the args are validated against the tool's
    // schema and must reach the tool untouched. With `background: true` the
    // awaited envelope is the PLACEHOLDER (isError false, `durationMs: null`,
    // `result.details = { background: true, callId, toolName }`) and the call
    // keeps running; the real result arrives LATER as a `tool_result` event
    // whose `toolCallId` is that same `callId` (the envelope's
    // `toolCall.id`), so a `pi.on("tool_result", ...)` handler correlates the
    // completion.
    // `options.signal` / `options.onUpdate` ride the shared abort flag and
    // toolUpdate path.
    //
    // LAZY: the list is built on first access and cached per ctx. The host call
    // (`getAllToolsJson`) serializes every tool definition, and a ctx is built
    // for every dispatched event (including each `message_update` delta, whose
    // handlers read the live message, never the tool list): building it eagerly
    // made the per-delta path pay a registry-sized host call per dispatch — the
    // host-call pin in `message_update_live_object_matches_the_reference_at_every_delta`
    // holds the boundary at zero calls between dispatches. Semantics are
    // unchanged: the same value, built the same way, on the same ctx object.
    get tools() {
      if (toolsCache === undefined) {
        toolsCache = attachToolHooks(JSON.parse(__pir_host.getAllToolsJson()));
      }
      return toolsCache;
    },
    executeTool: async (name, args, options) => {
      const out = await __pir_host.executeTool(
        String(toolCallId),
        String(name),
        JSON.stringify(args ?? {}),
        options?.background === true,
      );
      const outcome = JSON.parse(out);
      if (outcome && outcome.error !== undefined) {
        // Never rejects for tool failures; only a missing runtime capability
        // (executeTool outside a running tool call) throws.
        throw new Error(outcome.error);
      }
      return outcome;
    },
    // Real values (upstream types.ts:403 BuildSystemPromptOptions). The host
    // pushes the session's custom system-prompt override (--system-prompt /
    // SYSTEM.md; empty = default preamble) into runtime state; cwd comes
    // from the runtime config. pir tracks no other BuildSystemPromptOptions
    // decision inputs (selectedTools/appendSystemPrompt/sections/... are not
    // held as session state), so this is the minimal correct subset.
    getSystemPromptOptions: () => {
      const parsed = parseHostJson(__pir_host.getSystemPromptOptionsJson());
      return parsed ?? { cwd: desc.cwd };
    },
    // -------------------------------------------------------------------
    // Command-context session actions (upstream
    // `ExtensionCommandContextActions`, extensions/types.ts:2208-2228;
    // runner.ts:1006-1025 mounts them on the command ctx). The print/rpc
    // hosts bind real implementations (print-mode.ts:78-100 /
    // rpc-mode.ts:322-344); pir routes every call through the `commandAction`
    // host fn, which records the action for the host to apply against its
    // session-control plane after the command handler settles. The host fn
    // answers with the runner-DEFAULT result (`{cancelled: false}`,
    // accepted); a `{error}` reply is rethrown here so the dispatch reports
    // the failure. `withSession` (upstream ReplacedSessionContext,
    // extensions/types.ts:437-452) cannot cross the JSON bridge as a
    // function: the callback is registered under a token that rides the
    // action, and a host that can replace sessions calls it back through
    // `__pir_callSessionCallback` with a FRESH command context bound to the
    // replacement session. A storage-only host (print mode) reports that it
    // cannot run the callback when it applies the action.
    // -------------------------------------------------------------------
    newSession: async (options) => {
      const withSession = takeWithSessionToken("newSession", options);
      const out = await __pir_host.commandAction("newSession", JSON.stringify({
        ...(options?.parentSession != null ? { parentSession: String(options.parentSession) } : {}),
        ...(withSession !== undefined ? { withSession } : {}),
      }));
      const parsed = parseHostJson(out);
      if (parsed && parsed.error !== undefined) {
        // The host refused to record the action: the callback will never run.
        if (withSession !== undefined) delete state.sessionCallbacks[withSession];
        throw new Error(parsed.error);
      }
      return parsed ?? { cancelled: false };
    },
    fork: async (entryId, options) => {
      // `options.position` rides through to the host (upstream
      // `fork(entryId, {position})`, agent-session-runtime.ts:263-289:
      // default "before"); `options.withSession` rides as a callback token
      // (see the block above). The host applies the fork after the command
      // handler settles, so the reply is the runner-default
      // `{cancelled: false}` (selectedText is reported by the RPC fork
      // bridge, not here).
      const withSession = takeWithSessionToken("fork", options);
      if (entryId != null && (typeof entryId !== "string" || !entryId)) {
        if (withSession !== undefined) delete state.sessionCallbacks[withSession];
        throw new TypeError("fork requires an entry id string");
      }
      const out = await __pir_host.commandAction("fork", JSON.stringify({
        ...(entryId != null ? { entryId: String(entryId) } : {}),
        ...(options?.position != null ? { position: String(options.position) } : {}),
        ...(withSession !== undefined ? { withSession } : {}),
      }));
      const parsed = parseHostJson(out);
      if (parsed && parsed.error !== undefined) {
        if (withSession !== undefined) delete state.sessionCallbacks[withSession];
        throw new Error(parsed.error);
      }
      return parsed ?? { cancelled: false };
    },
    navigateTree: async (targetId, options) => {
      if (typeof targetId !== "string" || !targetId) {
        throw new TypeError("navigateTree requires a target entry id");
      }
      const out = await __pir_host.commandAction("navigateTree", JSON.stringify({
        targetId,
        summarize: options?.summarize === true,
        ...(options?.customInstructions != null ? { customInstructions: String(options.customInstructions) } : {}),
        replaceInstructions: options?.replaceInstructions === true,
        ...(options?.label != null ? { label: String(options.label) } : {}),
      }));
      const parsed = parseHostJson(out);
      if (parsed && parsed.error !== undefined) throw new Error(parsed.error);
      return parsed ?? { cancelled: false };
    },
    switchSession: async (sessionPath, options) => {
      const withSession = takeWithSessionToken("switchSession", options);
      if (typeof sessionPath !== "string" || !sessionPath) {
        if (withSession !== undefined) delete state.sessionCallbacks[withSession];
        throw new TypeError("switchSession requires a session path");
      }
      const out = await __pir_host.commandAction("switchSession", JSON.stringify({
        sessionPath,
        ...(withSession !== undefined ? { withSession } : {}),
      }));
      const parsed = parseHostJson(out);
      if (parsed && parsed.error !== undefined) {
        if (withSession !== undefined) delete state.sessionCallbacks[withSession];
        throw new Error(parsed.error);
      }
      return parsed ?? { cancelled: false };
    },
    // ctx.reload(): reload extensions, native plugins, tools, skills, prompts
    // and themes in place (upstream `session.reload`, types.ts:433-434). The
    // host applies it after the command handler settles; the session and
    // transcript stay put. A host with no resource surface answers with the
    // explicit error rethrown here instead of pretending.
    reload: async () => {
      const out = await __pir_host.commandAction("reload", "{}");
      const parsed = parseHostJson(out);
      if (parsed && parsed.error !== undefined) throw new Error(parsed.error);
    },
  };
  // Upstream's tool context carries the extension API as `ctx.pi` — the very
  // object the extension's factory received (`pi.on/registerTool/events/...`).
  // Tools that emit or resolve through it (`ctx.pi.events.emit(...)`,
  // pi-subagents' `finishSpawnResult`) fail with a TypeError without it. Set
  // only when the owning extension is known (tool execution); contexts without
  // an owner keep the previous shape.
  const ownerPi = ownerExtId === undefined ? undefined : state.piApis[ownerExtId];
  if (ownerPi !== undefined) ctx.pi = ownerPi;
  return ctx;
}

// Handler-table dirty signal: every mutation of an extension's event-handler
// table (registration and unregistration) bumps a host-side generation
// counter through `__pir_host.handlersDirty`. The Rust fast-path gate
// (`ExtensionHandle::has_handlers`) decides from a snapshot of
// `handlerCounts()` whether an event emission is worth the JS round-trip; the
// snapshot is only re-published at the host's refresh points, so a `pi.on(...)`
// running with NO command in flight (a `setTimeout`/promise continuation
// resumed by the actor's drive loop) would otherwise stay invisible and its
// event skipped. The bump makes the gate treat the snapshot as possibly stale
// (conservative `true`) until the next refresh re-publishes it.
//
// Guarded: the prelude is also evaluated with no host bridge, where the signal
// is simply unavailable (`__pir_handlerCountsJson` still exports the counts).
function markHandlersDirty() {
  if (typeof __pir_host !== "undefined" && typeof __pir_host.handlersDirty === "function") {
    __pir_host.handlersDirty();
  }
}

// Publish the live `ctx.ui.onTerminalInput` handler count to the host. The
// Rust fast-path gate (`ExtensionHandle::has_terminal_input_handlers`) reads it
// as one atomic load per physical input, so a session with no handlers never
// pays a runtime-state lock on the typing path. Called from the
// register/unregister sites together with `markHandlersDirty` (the count and
// the handler table must move together: the TUI's per-input gate is the count,
// `has_handlers("terminal_input")` is the table).
function publishTerminalInputHandlers() {
  if (
    typeof __pir_host !== "undefined" &&
    typeof __pir_host.terminalInputHandlers === "function"
  ) {
    try {
      __pir_host.terminalInputHandlers(state.terminalInputHandlers.length);
    } catch {
      // The count is an optimization; registration semantics do not depend on it.
    }
  }
}

// Registry-table dirty signal: every mutation of an exported registry table (a
// tool/command/shortcut/renderer registration or unregistration, a
// `setWidget` factory registration/retirement, the load-discard cleanup)
// bumps a host-side generation counter through `__pir_host.registryDirty`.
// The host's `refresh_registry` runs after every dispatch (and after load /
// command / shortcut) and re-serializes the WHOLE registry
// (`__pir_exportRegistryJson`: every tool schema, command, shortcut and
// renderer) plus a Rust re-parse; the generation lets it early-out when
// nothing changed — the common per-delta `message_update` case. A registration
// that DOES happen still bumps, so the next refresh re-exports and the
// registration is visible to the host registry before the dispatch returns.
//
// Guarded like `markHandlersDirty`: with no host bridge the signal is simply
// unavailable (`__pir_exportRegistryJson` still exports the registry).
function markRegistryDirty() {
  if (typeof __pir_host !== "undefined" && typeof __pir_host.registryDirty === "function") {
    __pir_host.registryDirty();
  }
}

function makePi(extId) {
  const pi = {
    on(event, handler) {
      assertActive(extId);
      const ext = extRecord(extId);
      if (typeof handler !== "function") throw new TypeError("handler must be a function");
      const list = (ext.handlers[event] ??= []);
      list.push(handler);
      markHandlersDirty();
      let removed = false;
      return () => {
        if (removed) return;
        removed = true;
        const idx = ext.handlers[event]?.indexOf(handler);
        if (idx !== undefined && idx >= 0) {
          ext.handlers[event].splice(idx, 1);
          markHandlersDirty();
        }
      };
    },

    registerTool(tool) {
      assertActive(extId);
      if (!tool || typeof tool !== "object") throw new TypeError("registerTool requires a tool definition");
      if (typeof tool.name !== "string" || !tool.name) throw new TypeError("tool.name must be a non-empty string");
      if (tool.name.startsWith("__pir_")) {
        // Reserved: the session multiplexes its streamSimple/route channels
        // through `__pir_`-prefixed keys (see __pir_callPrepareLoadout).
        throw new TypeError("tool.name must not use the reserved __pir_ prefix");
      }
      if (!tool.parameters || tool.parameters.type !== "object") {
        throw new TypeError(`Tool ${tool.name}: parameters must be a TypeBox object schema`);
      }
      if (typeof tool.execute !== "function") {
        throw new TypeError(`Tool ${tool.name}: execute must be a function`);
      }
      const ext = extRecord(extId);
      ext.tools[tool.name] = tool;
      // The exported registry reads `toolMeta` AND the live `toolReachHooks`
      // (the reach snapshot), so even a duplicate-name registration that does
      // not change `toolMeta` is a registry mutation.
      markRegistryDirty();
      if (typeof tool.prepareLoadout === "function") {
        // The hook function stays inside QuickJS; the session invokes it by
        // name through __pir_callPrepareLoadout when it builds the loadout.
        state.toolLoadoutHooks[tool.name] = tool.prepareLoadout;
        state.toolLoadoutOwners[tool.name] = extId;
      } else {
        delete state.toolLoadoutHooks[tool.name];
        delete state.toolLoadoutOwners[tool.name];
      }
      if (typeof tool.reachesTools === "function") {
        // Upstream `ToolDefinition.reachesTools`: which tools a call with the
        // given input may reach. Other extensions call it through
        // `pi.getAllTools()`; the live function stays here and the exported
        // metadata carries only a marker.
        state.toolReachHooks[tool.name] = tool.reachesTools;
      } else {
        delete state.toolReachHooks[tool.name];
      }
      // First registration per name wins (upstream getAllRegisteredTools
      // resolution; __pir_callTool dispatches against the first extension
      // that owns the name). Cross-extension duplicates stay registered on
      // their own extensions for conflict diagnostics. The surviving entry
      // records its owner so a later failing extension's discard cannot
      // delete another extension's metadata.
      if (state.toolMeta[tool.name] === undefined) {
        state.toolMeta[tool.name] = {
          extId,
          name: tool.name,
          label: tool.label ?? undefined,
          description: tool.description ?? "",
          promptSnippet: tool.promptSnippet ?? undefined,
          promptGuidelines: tool.promptGuidelines ?? [],
          parameters: sanitizeSchema(tool.parameters),
          outputSchema: tool.outputSchema ? sanitizeSchema(tool.outputSchema) : undefined,
          sourceInfo: { path: ext.path },
          // Upstream ToolDefinition.executionMode is an OPTIONAL override of
          // the session's default execution mode (`toolExecution: "parallel"`,
          // agent.ts:261; only `"sequential"` forces one-at-a-time batches,
          // agent-loop.ts:527). An omitted field therefore means "parallel",
          // not "sequential": the old default demoted every whole batch to
          // serial whenever one extension tool was declared.
          executionMode: tool.executionMode ?? "parallel",
          exposure: tool.exposure ?? "direct",
          namespace: tool.namespace ?? undefined,
          annotations: tool.annotations ?? undefined,
          defaultActive: tool.defaultActive ?? true,
          prepareLoadout: typeof tool.prepareLoadout === "function",
          // Upstream `ToolDefinition.constrainedSampling` / `renderShell`
          // (types.ts:590-594, 641-647): declarative provider-side constrained
          // sampling request and self-framing choice. They travel in the

    registerProvider(nameOrProvider, maybeConfig) {
      assertActive(extId);
      let name;
      let config;
      if (typeof nameOrProvider === "string") {
        name = nameOrProvider;
        config = maybeConfig;
      } else if (nameOrProvider && typeof nameOrProvider === "object") {
        // One-arg Provider form: the object carries id/name + config fields.
        name = nameOrProvider.id ?? nameOrProvider.name;
        config = nameOrProvider;
      }
      if (typeof name !== "string" || !name) {
        throw new TypeError("registerProvider requires a provider name");
      }
      if (!config || typeof config !== "object") {
        throw new TypeError(`registerProvider(${name}): config must be an object`);
      }
      // Full `Provider` object form (upstream `Provider`,
      // packages/ai/src/models.ts:158-247; extensions/loader.ts:459-467 routes
      // it to `registerNativeProvider`). Its catalog is a `getModels()`
      // FUNCTION and auth/filtering/image/classifier support are callbacks.
      // Normalize the catalog into the JSON definition NOW (so the host
      // registers real models) and keep the callbacks in JS state:
      //   auth.apiKey.resolve  -> state.providerApiKeyResolvers
      //   filterModels         -> state.providerFilterHooks
      //   images/classifiers   -> state.providerImageHooks / providerClassifierHooks
      //   stream/streamSimple  -> state.providerStreamHooks (same channel as
      //                           ProviderConfig.streamSimple)
      const isNativeProvider =
        typeof config.getModels === "function" || typeof config.getAllModels === "function";
      if (isNativeProvider) {
        let nativeModels;
        try {
          nativeModels =
            typeof config.getAllModels === "function" ? config.getAllModels() : config.getModels();
        } catch (error) {
          throw new TypeError(
            `registerProvider(${name}): getModels() failed: ${error && error.message ? error.message : String(error)}`,
          );
        }
        config = {
          ...config,
          models: Array.isArray(nativeModels) ? nativeModels : [],
          nativeProvider: true,
        };
        if (config.auth && typeof config.auth === "object") {
          const apiKeyAuth = config.auth.apiKey;
          if (apiKeyAuth && typeof apiKeyAuth.resolve === "function") {
            state.providerApiKeyResolvers[name] = { extId, auth: apiKeyAuth };
          }
          if (config.auth.oauth && typeof config.auth.oauth === "object") {
            // Canonical `auth.oauth` (login(interaction)/refresh/toAuth)
            // cannot ride the legacy ProviderConfig.oauth bridge
            // (login(callbacks)); keep it observable instead of silently
            // losing the declaration and report the wiring gap once.
            state.providerNativeOauth[name] = { extId, auth: config.auth.oauth };
            try {
              console.warn(
                `[pir] registerProvider(${name}): canonical auth.oauth flows are not wired into /login yet`,
              );
            } catch { /* diagnostics are best-effort */ }
          }
        }
        if (typeof config.filterModels === "function") {
          state.providerFilterHooks[name] = { extId, fn: config.filterModels };
        }
        if (config.images && typeof config.images === "object") {
          state.providerImageHooks[name] = { extId, impl: config.images };
        }
        if (config.classifiers && typeof config.classifiers === "object") {
          state.providerClassifierHooks[name] = { extId, impl: config.classifiers };
        }
        const nativeStream =
          typeof config.streamSimple === "function"
            ? config.streamSimple
            : typeof config.stream === "function"
              ? config.stream
              : null;
        if (nativeStream !== null) {
          state.providerStreamHooks[name] = { extId, fn: nativeStream };
          // Same marker discipline as ProviderConfig.streamSimple: the host
          // builds a JS-stream provider for this id.
          config = { ...config, streamSimple: true };
        }
      }
      if (config.models !== undefined && !Array.isArray(config.models)) {
        throw new TypeError(`registerProvider(${name}): models must be an array`);
      }
      // A native provider's models carry their own baseUrl; only the config
      // form requires the provider-level URL.
      if (!isNativeProvider && Array.isArray(config.models) && config.models.length > 0 && !config.baseUrl) {
        throw new TypeError(`registerProvider(${name}): baseUrl is required when defining models`);
      }
      __pir_host.registerProviderJson(JSON.stringify({ id: name, config: config ?? {} }));
      // Config values the registry reads back (upstream resolves provider auth
      // from this config): string `apiKey` / `api` make
      // `modelRegistry.getProviderAuth` / `getApiKeyAndHeaders` real for an
      // extension-registered provider, and `api` pins the stream-dispatch
      // mismatch guard.
      state.providerConfigs[name] = {
        api: typeof config.api === "string" ? config.api : undefined,
        apiKey: typeof config.apiKey === "string" ? config.apiKey : undefined,
        baseUrl: typeof config.baseUrl === "string" ? config.baseUrl : undefined,
      };
      // streamSimple (upstream ProviderConfig.streamSimple): the callback
      // cannot cross the JSON host-fn bridge, so it stays in JS state and the
      // host invokes it per request through the reserved `__pir_stream:` hook
      // channel. The JSON definition re-registers with a `streamSimple: true`
      // marker (upserted by id) so the host builds a JS-stream provider.
      if (typeof config.streamSimple === "function") {
        state.providerStreamHooks[name] = { extId, fn: config.streamSimple };
        __pir_host.registerProviderJson(
          JSON.stringify({ id: name, config: { ...config, streamSimple: true } }),
        );
      }
      // Upstream ProviderConfig.oauth (types.ts:1943-1958): a custom OAuth
      // login flow for /login support. The callbacks cannot cross the JSON
      // host-fn bridge, so they stay in JS state and the host drives them
      // through `__pir_providerOauthStart/Poll/Cancel`; the JSON definition
      // carries boolean markers in the oauth segment (same discipline as the
      // streamSimple marker).
      if (config.oauth !== undefined) {
        if (typeof config.oauth !== "object" || config.oauth === null) {
          throw new TypeError(`registerProvider(${name}): oauth must be an object`);
        }
        if (typeof config.oauth.login !== "function") {
          throw new TypeError(`registerProvider(${name}): oauth.login must be a function`);
        }
        state.providerOauthHooks[name] = {
          extId,
          login: config.oauth.login,
          refreshToken: typeof config.oauth.refreshToken === "function" ? config.oauth.refreshToken : null,
          getApiKey: typeof config.oauth.getApiKey === "function" ? config.oauth.getApiKey : null,
          // Upstream ProviderConfig.oauth.modifyModels (types.ts:1989): a
          // legacy synchronous credential-dependent chat-model projection.
          // Kept with the other OAuth callbacks and reported as a marker; the
          // host has no credential value to project with, so the projection is
          // exposed to extensions (modelRegistry) rather than applied to the
          // registered catalog.
          modifyModels: typeof config.oauth.modifyModels === "function" ? config.oauth.modifyModels : null,
        };
      }
      // Upstream ProviderConfig.refreshModels (types.ts:1941): a directory
      // refresh callback whose returned model list replaces the
      // extension-provided models. Rides the `provider:refreshModels`
      // synthetic event (see __pir_dispatchEvent); the host triggers it when
      // it consumes the marker.
      if (typeof config.refreshModels === "function") {
        state.providerRefreshHooks[name] = { extId, fn: config.refreshModels };
        const ext = extRecord(extId);
        const handlers = (ext.handlers["provider:refreshModels"] ??= []);
        handlers.push(async (payload) => {
          if (!payload || payload.providerId !== name) return undefined;
          // Upstream RefreshModelsContext (packages/ai/src/models.ts:80-97):
          // {credential?, stored?, publish(publication): Promise<boolean>,
          //  allowNetwork, force?, signal}. Data sources available here:
          // - stored: the host-pushed `models-store.json` entry for this
          //   provider (`payload.stored`, a fresh immutable copy in the
          //   upstream ModelsStoreEntry shape), falling back to the live
          //   registry snapshot filtered to this provider; omitted when
          //   neither holds models.
          // - allowNetwork: pir has no offline/cache-only mode, and the host
          //   only fires this synthetic event when it wants a refresh, so
          //   `true` (upstream: "False during offline/cache-only init").
          // - credential: pir tracks authenticated provider IDS, not
          //   credentials, so none can be passed — upstream allows undefined.
          // - publish: `update()` mutates the provider's private in-memory
          //   catalog synchronously; `persist` is handed back to the host on
          //   the event outcome, which writes it to `models-store.json`
          //   (`null` deletes the entry). The publication's `models` are used
          //   as the refreshed list when the callback returns none.
          const registry = parseHostJson(__pir_host.getModelsJson()) ?? {};
          const registryModels = (registry.models ?? []).filter(
            (model) => model && model.provider === name,
          );
          // The host-pushed persisted entry (`payload.stored`, the upstream
          // camelCase ModelsStoreEntry) is the provider-scoped catalog from
          // `models-store.json`; prefer it over the live registry snapshot so
          // a fresh process restores the last downloaded catalog.
          const hostStored =
            payload && payload.stored && typeof payload.stored === "object"
              ? payload.stored
              : undefined;
          const storedModels =
            hostStored && Array.isArray(hostStored.models)
              ? hostStored.models.filter(
                  (model) =>
                    model &&
                    (model.provider == null ||
                      model.provider === "" ||
                      model.provider === name),
                )
              : registryModels;
          let publishedModels;
          let persistedPublication;
          let hasPersistedPublication = false;
          const publish = async (publication) => {
            if (!publication || typeof publication !== "object") return false;
            let applied = false;
            try {
              if (typeof publication.update === "function") {
                publication.update();
                applied = true;
              }
            } catch (error) {
              throw error instanceof Error ? error : new Error(String(error));
            }
            const persisted = publication.persist;
            if (persisted === null) {
              // Upstream deletes the provider's stored catalog.
              persistedPublication = null;
              hasPersistedPublication = true;
              applied = true;
            } else if (persisted && typeof persisted === "object") {
              if (Array.isArray(persisted.models)) {
                publishedModels = persisted.models;
              }
              persistedPublication = persisted;
              hasPersistedPublication = true;
              applied = true;
            } else if (Array.isArray(persisted)) {
              // Legacy pir shape: `persist` was the model array itself.
              publishedModels = persisted;
              persistedPublication = { models: persisted };
              hasPersistedPublication = true;
              applied = true;
            }
            return applied;
          };
          let models = await config.refreshModels({
            allowNetwork: true,
            ...(hostStored
              ? { stored: { ...hostStored, models: storedModels } }
              : storedModels.length > 0
                ? { stored: { models: storedModels } }
                : {}),
            publish,
            signal: makeSignal(),
          });
          if (!Array.isArray(models)) {
            // Native `Provider.refreshModels` returns void and publishes the
            // new catalog through `getModels()`/`publish`. Read the live
            // catalog after the callback so the host replaces its list with
            // whatever the provider now serves.
            if (isNativeProvider && typeof config.getModels === "function") {
              try { models = config.getModels(); } catch { models = undefined; }
            }
            if (!Array.isArray(models) && publishedModels !== undefined) {
              models = publishedModels;
            }
          }
          const result = {
            providerId: name,
            models: Array.isArray(models) ? models : [],
          };
          if (hasPersistedPublication) result.persist = persistedPublication;
          return result;
        });
        // The one handler-table row this file registers without `pi.on`: same
        // dirty signal, so a snapshot published before the row existed cannot
        // make the `provider:refreshModels` gate skip the synthetic event.
        markHandlersDirty();
      }
      // Re-register with the FULL marker set: the blocks above each spread the
      // raw config, which would drop a sibling marker when one provider
      // combines streamSimple with refreshModels and/or oauth hooks.
      const hasStreamSimple = state.providerStreamHooks[name] != null;
      const hasOauth = state.providerOauthHooks[name] != null;
      const hasRefresh = state.providerRefreshHooks[name] != null;
      if (hasStreamSimple || hasOauth || hasRefresh) {
        const markers = {};
        if (hasStreamSimple) markers.streamSimple = true;
        if (hasOauth) {
          markers.oauth = {
            ...config.oauth,
            login: true,
            refreshToken: true,
            getApiKey: true,
            ...(state.providerOauthHooks[name]?.modifyModels
              ? { modifyModels: true }
              : {}),
          };
        }
        if (hasRefresh) markers.refreshModels = true;
        __pir_host.registerProviderJson(
          JSON.stringify({ id: name, config: { ...config, ...markers } }),
        );
      }
    },

      // The event cannot reach the host worker directly, so it is relayed per
      // call as `execAbort(<call id>)`; the host also polls the shared ctx
      // abort flag while the call runs, because host-initiated cancellation
      // aborts ctx.signal without firing JS-side listeners.
      const signal = options?.signal ?? null;
      let callId = null;
      let onAbort = null;
      if (signal) {
        callId = `exec-${Date.now().toString(36)}-${Math.random().toString(36).slice(2)}`;
        if (signal.aborted) {
          __pir_host.execAbort(callId);
        } else {
          onAbort = () => { try { __pir_host.execAbort(callId); } catch { /* best-effort */ } };
          try { signal.addEventListener?.("abort", onAbort); } catch { /* non-signal object: stay defensive vs upstream's throw */ }
        }
      }
      try {
        const out = await __pir_host.exec(
          String(command),
          JSON.stringify(argv),
          options?.cwd ?? null,
          options?.timeout == null ? null : Number(options.timeout),
          callId,
        );
        return JSON.parse(out);
      } finally {
        if (onAbort) {
          try { signal.removeEventListener?.("abort", onAbort); } catch { /* best-effort */ }
        }
        if (callId != null) {
          try { __pir_host.execAbortDone(callId); } catch { /* best-effort */ }
        }
      }
    },

    getActiveTools() { return JSON.parse(__pir_host.getActiveToolsJson()); },
    // Upstream `getAllTools()`: ToolInfo entries, so the `reachesTools` hooks
    // the host cannot serialize are re-attached live here.
    getAllTools() { return attachToolHooks(JSON.parse(__pir_host.getAllToolsJson())); },
    setActiveTools(toolNames) {
      assertActive(extId);
      __pir_host.setActiveToolsJson(JSON.stringify((toolNames ?? []).map(String)));
    },
    getCommands() { return JSON.parse(__pir_host.getCommandsJson()); },

    async setModel(model) {
      assertActive(extId);
      // `model` may be a full Model object or a `{provider, id}` reference.
      const provider = model && typeof model === "object" ? (model.provider ?? null) : null;
      const id = model && typeof model === "object" ? (model.id ?? String(model)) : String(model);
      return __pir_host.setModel(provider == null ? null : String(provider), String(id));
    },
    getThinkingLevel() { return __pir_host.getThinkingLevel(); },
    setThinkingLevel(level) {
      assertActive(extId);
      __pir_host.setThinkingLevel(String(level));
    },

    events: {
      emit(channel, data) {
        const subs = state.busSubs[channel] ?? [];
        for (const sub of subs) {
          // Subscriptions of failed/invalidated extensions are torn down
          // implicitly: their ext record is gone (discard) or no longer
          // active (invalidate), so their handlers never fire again.
          const ext = state.extensions[sub.extId];
          if (!ext || ext.status !== "active" || !__pir_host.isActive(sub.extId)) continue;
          try { sub.handler(data); } catch { /* swallowed like upstream */ }
        }
      },
      on(channel, handler) {
        assertActive(extId);
        const sub = { extId, handler };
        (state.busSubs[channel] ??= []).push(sub);
        let removed = false;
        return () => {
          if (removed) return;
          removed = true;
          const idx = state.busSubs[channel]?.indexOf(sub);
          if (idx !== undefined && idx >= 0) state.busSubs[channel].splice(idx, 1);
        };
      },
    },
  };
  return pi;
}

function sanitizeSchema(value) {
  if (Array.isArray(value)) return value.map(sanitizeSchema);
  if (value && typeof value === "object") {
    const out = {};
    for (const [k, v] of Object.entries(value)) {
      if (k === "__pir_optional__") continue;
      out[k] = sanitizeSchema(v);
    }
    return out;
  }
  return value;
}

// Re-attach the live tool hooks the host's JSON cannot carry to a tool-info
// list (`pi.getAllTools()` / `ctx.tools`). Upstream `ToolInfo` carries the
// `reachesTools` FUNCTION (ToolDefinition.reachesTools), so the host exports a
// marker and the function is found again here by tool name — the same
// discipline `prepareLoadout` uses on the session side.
function attachToolHooks(tools) {
  if (!Array.isArray(tools)) return tools;
  for (const tool of tools) {
    if (!tool || typeof tool.name !== "string") continue;
    const reach = state.toolReachHooks[tool.name];
    if (typeof reach === "function") tool.reachesTools = reach;
    // The remaining ToolDefinition members are FUNCTIONS (or opaque objects)
    // the host JSON cannot carry: re-attach them from the owning extension's
    // live tool object, which is the same source `pi.getAllTools()` mirrors
    // upstream (`getAllRegisteredTools` returns the definitions themselves).
    const meta = state.toolMeta[tool.name];
    const def = meta ? state.extensions[meta.extId]?.tools?.[tool.name] : undefined;
    if (!def) continue;
    if (typeof def.prepareArguments === "function") tool.prepareArguments = def.prepareArguments;
    if (typeof def.renderCall === "function") tool.renderCall = def.renderCall;
    if (typeof def.renderResult === "function") tool.renderResult = def.renderResult;
    if (typeof def.prepareLoadout === "function") tool.prepareLoadout = def.prepareLoadout;
    if (def.constrainedSampling !== undefined) tool.constrainedSampling = def.constrainedSampling;
    if (def.renderShell !== undefined) tool.renderShell = def.renderShell;
  }
  return tools;
}

// The renderers a registered tool itself declares (`ToolRenderers`,
// upstream types.ts:652-664): the terminal value of a `registerToolRenderer`
// `next()` chain. `undefined` when the tool declares none.
function registeredToolRenderers(toolName) {
  const meta = state.toolMeta[toolName];
  const def = meta ? state.extensions[meta.extId]?.tools?.[toolName] : undefined;
  if (!def) return undefined;
  const renderers = {};
  if (def.renderShell !== undefined) renderers.renderShell = def.renderShell;
  if (typeof def.renderCall === "function") renderers.renderCall = def.renderCall;
  if (typeof def.renderResult === "function") renderers.renderResult = def.renderResult;
  return Object.keys(renderers).length > 0 ? renderers : undefined;
}

// Dispatch one registration's tool renderer (upstream ToolDefinition.renderCall
// / renderResult, extensions/types.ts:659-667). The TUI's tool block is the
// consumer: it asks through the runtime for `"call"` (arguments render) or
// `"result"` (final result render) with the full ToolRenderContext wire.
// Returns `{ok, lines, text, shell}`; `lines: null` means "no custom
// renderer" (the caller keeps its default drawing) and `shell` reports the
// tool's `renderShell` choice ("default" | "self").
globalThis.__pir_callToolRenderer = async (toolName, kind, requestJson) => {
  const meta = state.toolMeta[toolName];
  const def = meta ? state.extensions[meta.extId]?.tools?.[toolName] : undefined;
  if (!def) return JSON.stringify({ ok: false, error: `Unknown extension tool renderer: ${toolName}` });
  const request = parseHostJson(requestJson) ?? {};
  const renderOptions = request.renderOptions ?? {};
  const theme = request.theme ?? rendererTheme();
  const args = request.args ?? request.arguments ?? {};
  const stateForTool = request.state ?? {};
  const context = {
    args,
    toolCallId: String(request.toolCallId ?? ""),
    invalidate: () => {
      try {
        __pir_host.setStatus(
          "__pir_request_render",
          JSON.stringify({ kind: "tool", toolCallId: String(request.toolCallId ?? "") }),
        );
      } catch { /* the repaint signal is best-effort */ }
    },
    lastComponent: request.lastComponent,
    state: stateForTool,
    cwd: String(request.cwd ?? ""),
    executionStarted: request.executionStarted === true,
    argsComplete: request.argsComplete !== false,
    isPartial: request.isPartial === true,
    expanded: request.expanded === true,
    showImages: request.showImages === true,
    isError: request.isError === true,
    durationMs: typeof request.durationMs === "number" ? request.durationMs : undefined,
    outputPad: typeof request.outputPad === "number" ? request.outputPad : 0,
  };
  try {
    let result;
    if (kind === "result") {
      if (typeof def.renderResult !== "function") {
        return JSON.stringify({ ok: true, lines: null, text: null, shell: def.renderShell ?? "default" });
      }
      result = await def.renderResult(
        request.result ?? {},
        { expanded: context.expanded, isPartial: context.isPartial },
        theme,
        context,
      );
    } else {
      if (typeof def.renderCall !== "function") {
        return JSON.stringify({ ok: true, lines: null, text: null, shell: def.renderShell ?? "default" });
      }
      result = await def.renderCall(args, theme, context);
    }
    const lines = await rendererLines(result, renderOptions);
    return JSON.stringify({
      ok: true,
      lines,
      text: null,
      shell: def.renderShell ?? "default",
    });
  } catch (error) {
    return JSON.stringify({
      ok: false,
      error: error && error.message ? error.message : String(error),
      shell: def.renderShell ?? "default",
    });
  }
};

// ---------------------------------------------------------------------------
// Load lifecycle: factory → commit | discard (spec §4.2)
// ---------------------------------------------------------------------------

globalThis.__pir_loadExtension = async (extIdRaw, moduleName, path) => {
  const extId = Number(extIdRaw);
  const ext = {
    path,
    status: "loading",
    handlers: {},
    tools: {},
    commands: {},
    flags: {},
    shortcuts: {},
    virtualModels: new Set(),
    eventUnsubs: [],
  };
  state.extensions[extId] = ext;
  __pir_host.registerExtId(extId);
  try {
    const mod = await import(moduleName);
    const factory = mod.default;
    if (typeof factory !== "function") {
      throw new TypeError("Extension module has no default export function");
    }
    const pi = makePi(extId);
    // Keep the factory's API object for this extension's tool contexts
    // (`ctx.pi`, see makeCtx): one object per extension, same capabilities.
    state.piApis[extId] = pi;
    await factory(pi);
    // commit
    ext.status = "active";
    for (const flag of Object.values(ext.flags)) {
      __pir_host.registerFlagJson(JSON.stringify({
        extId,
        name: flag.name,
        type: flag.type,
        description: flag.description,
        default: flag.default,
      }));
    }
    return JSON.stringify({ ok: true });
  } catch (error) {
    // discard
    ext.status = "failed";
    // The cleanup below removes this (failed) extension's exported registry
    // entries (first-wins tool meta, commands, shortcuts, renderers); mark the
    // JS registry dirty so the load-tail refresh re-exports the cleaned
    // registry instead of early-outing.
    markRegistryDirty();
    // Only delete metadata this extension owns: a later failing extension
    // must not remove a surviving (first-wins) registration of another
    // extension (P2-6).
    for (const name of Object.keys(ext.tools)) {
      if (state.toolMeta[name] !== undefined && state.toolMeta[name].extId === extId) {
        delete state.toolMeta[name];
      }
      if (state.toolLoadoutOwners[name] === extId) {
        delete state.toolLoadoutHooks[name];
        delete state.toolLoadoutOwners[name];
      }
    }
    for (const key of ext.virtualModels) {
      const slash = key.indexOf("/");
      if (slash > 0) {
        __pir_host.unregisterVirtualModel(key.slice(0, slash), key.slice(slash + 1));
      }
    }
    // JS-owned callbacks of the failed extension are discarded too (parity
    // with the toolLoadoutHooks cleanup above).
    for (const name of Object.keys(state.providerStreamHooks)) {
      if (state.providerStreamHooks[name].extId === extId) delete state.providerStreamHooks[name];
    }
    for (const name of Object.keys(state.providerOauthHooks)) {
      if (state.providerOauthHooks[name].extId === extId) delete state.providerOauthHooks[name];
    }
    for (const name of Object.keys(state.providerRefreshHooks)) {
      if (state.providerRefreshHooks[name].extId === extId) delete state.providerRefreshHooks[name];
    }
    for (const key of Object.keys(state.virtualRouteHooks)) {
      if (state.virtualRouteHooks[key].extId === extId) delete state.virtualRouteHooks[key];
    }
    for (const name of Object.keys(ext.commands)) {
      state.commandMeta[name] = (state.commandMeta[name] ?? []).filter((c) => c.extId !== extId);
      if (state.commandMeta[name].length === 0) delete state.commandMeta[name];
    }
    // Only delete shortcut metadata this extension owns: shortcut resolution
    // is last-wins, so a later registration may own the key even when the
    // failed extension registered it earlier.
    for (const key of Object.keys(ext.shortcuts)) {
      if (state.shortcutMeta[key] !== undefined && state.shortcutMeta[key].extId === extId) {
        delete state.shortcutMeta[key];
      }
    }
    dropExtensionRenderers(extId);
    for (const unsub of ext.eventUnsubs) { try { unsub(); } catch {} }
    ext.handlers = {};
    const message = error && error.message ? error.message : String(error);
    return JSON.stringify({ ok: false, error: message });
  }
};

// Remove renderer registrations owned by an extension (load discard + reload
// teardown); re-registrations by the fresh runtime restore them.
function dropExtensionRenderers(extId) {
  for (const kind of ["messageRenderers", "entryRenderers", "markdownTransformers", "toolRenderers"]) {
    const bucket = state.rendererMeta[kind];
    for (const key of Object.keys(bucket)) {
      const value = bucket[key];
      if (Array.isArray(value)) {
        // toolRenderers: one bucket per tool name holds the resolver chain.
        const kept = value.filter((entry) => entry.extId !== extId);
        if (kept.length === 0) delete bucket[key];
        else bucket[key] = kept;
      } else if (value.extId === extId) delete bucket[key];
    }
  }
}

function isUserBashResult(result) {
  // Ported from upstream runner.ts isUserBashEventResult (runner.ts:140-164):
  // exactly one of { operations } or { result }, with the full shape check.
  if (!result || typeof result !== "object" || Array.isArray(result)) return false;
  const hasOperations = result.operations !== undefined;
  const hasResult = result.result !== undefined;
  if (hasOperations === hasResult) return false; // exactly one of the two
  if (hasOperations) {
    const operations = result.operations;
    if (!operations || typeof operations !== "object" || Array.isArray(operations)) return false;
    return typeof operations.exec === "function";
  }
  const bashResult = result.result;
  if (!bashResult || typeof bashResult !== "object" || Array.isArray(bashResult)) return false;
  return (
    typeof bashResult.output === "string" &&
    "exitCode" in bashResult &&
    (bashResult.exitCode === undefined || typeof bashResult.exitCode === "number") &&
    typeof bashResult.cancelled === "boolean" &&
    typeof bashResult.truncated === "boolean" &&
    (bashResult.fullOutputPath === undefined || typeof bashResult.fullOutputPath === "string")
  );
}

// Export a handler's `{operations}` result across the JSON bridge. The
// `BashOperations.exec` callback is a JS closure (upstream
// extensions/types.ts:1466-1476; the interface is `{exec(command, cwd,
// {onData, signal, timeout, env})}`), so it stays in JS state under a token and
// the returned object carries that token. The host executes the command
// through `__pir_runBashOperations` and streams chunks back through
// `__pir_host.bashOperationData`.
function exportUserBashResult(result) {
  if (result.operations === undefined) return result;
  const exec = result.operations && result.operations.exec;
  if (typeof exec !== "function") return result;
  // Runtime-unique: the host's sink registry is process-wide, so two
  // concurrent runtimes must never mint the same token.
  const token = `bash-op-${Math.random().toString(36).slice(2)}-${++state.nextBashOperationsId}`;
  state.bashOperationsHooks[token] = { exec };
  return { operations: { __pirBashOperations: token } };
}

// Execute one exported `BashOperations` (the host calls this after a
// `user_bash` handler returned `{operations}`). `requestJson` carries
// `{command, cwd, timeout?, env?}`; the reply is `{exitCode}` (upstream
// `Promise<{exitCode: number | null}>`) or `{error}`. Chunks are pushed
// synchronously through the `bashOperationData` host fn (a plain sink
// append — no re-entry into the extension actor), so `onData` keeps its
// streaming contract.
globalThis.__pir_runBashOperations = async (token, requestJson) => {
  const entry = state.bashOperationsHooks[token];
  if (!entry || typeof entry.exec !== "function") {
    return JSON.stringify({ error: `No custom bash operations registered for token ${token}` });
  }
  let request;
  try {
    request = JSON.parse(requestJson);
  } catch (error) {
    delete state.bashOperationsHooks[token];
    return JSON.stringify({ error: `invalid bash operation request: ${error && error.message ? error.message : String(error)}` });
  }
  const onData = (chunk) => {
    let text;
    try {
      text = typeof Buffer !== "undefined" && Buffer.isBuffer && Buffer.isBuffer(chunk)
        ? chunk.toString("utf8")
        : Buffer.from(chunk ?? "").toString("utf8");
    } catch {
      text = String(chunk ?? "");
    }
    try {
      __pir_host.bashOperationData(token, text);
    } catch {
      /* the host sink is best-effort (a released token drops the chunk) */
    }
  };
  try {
    const out = await entry.exec(String(request.command ?? ""), String(request.cwd ?? ""), {
      onData,
      signal: makeSignal(),
      ...(request.timeout != null ? { timeout: Number(request.timeout) } : {}),
      ...(request.env != null && typeof request.env === "object" ? { env: request.env } : {}),
    });
    const exitCode = out && typeof out.exitCode === "number" ? out.exitCode : null;
    return JSON.stringify({ exitCode });
  } catch (error) {
    return JSON.stringify({ error: error && error.message ? error.message : String(error) });
  } finally {
    delete state.bashOperationsHooks[token];
  }
};

// Release an exported operations token without running it (the host aborted
// before execution): the closure must not stay reachable for the runtime's
// lifetime.
globalThis.__pir_releaseBashOperations = (token) => {
  delete state.bashOperationsHooks[token];
  return "";
};

// One handler invocation, timed for `PI_TIMING=1` and tagged with its
// dispatch epoch. The elapsed time includes every `await` inside the handler
// (a handler waiting on a background promise shows that wait). The epoch
// enters the host's epoch stack BEFORE the handler starts and leaves it when
// the handler settles, so a continuation that resumes after the dispatch
// returned still tags its side effects with the (possibly abandoned) epoch —
// the host drops those instead of applying them to a later run.
async function invokeTimedHandler(extId, handler, event, ctx, eventName, dispatchEpoch) {
  const epoch = typeof dispatchEpoch === "number" && dispatchEpoch > 0 ? dispatchEpoch : 0;
  const trackEpoch =
    epoch !== 0 &&
    typeof __pir_host !== "undefined" &&
    typeof __pir_host.enterDispatch === "function";
  if (trackEpoch) {
    try {
      __pir_host.enterDispatch(epoch);
    } catch {
      // Side-effect tagging is best-effort; dispatch behavior is not.
    }
  }
  const started = PIR_TIMING_ENABLED ? Date.now() : 0;
  let ok = false;
  try {
    const result = await handler(event, ctx);
    ok = true;
    return result;
  } finally {
    if (PIR_TIMING_ENABLED) {
      try {
        __pir_host.handlerTiming(Number(extId), String(eventName), Date.now() - started, ok);
      } catch {
        // Timing must never change dispatch behavior.
      }
    }
    if (trackEpoch) {
      try {
        __pir_host.exitDispatch(epoch);
      } catch {
        // Best-effort, same as the enter call.
      }
    }
  }
}

// Upstream snapshotEventHandlers (runner.ts:269-271): the handler list is
// snapshotted before dispatch so a handler registering during the round does
// not join it (P2-12).
function snapshotEventHandlers(eventName) {
  const handlers = [];
  for (const extId of Object.keys(state.extensions).sort((a, b) => Number(a) - Number(b))) {
    const ext = state.extensions[extId];
    if (ext.status !== "active") continue;
    for (const handler of (ext.handlers[eventName] ?? []).slice()) handlers.push([extId, handler]);
  }
  return handlers;
}

// Shallow element-identity compare (upstream sameMessages): detects in-place
// edits of a handler-owned message list (P2-11).
function sameMessagesShallow(left, right) {
  return left.length === right.length && left.every((message, index) => message === right[index]);
}

// Raw terminal input (upstream `TerminalInputHandler`): handlers receive the
// raw chunk as a string (not `(event, ctx)`) and may return
// `{consume?: boolean, data?: string}`. This is deliberately a dedicated path
// instead of a `pi.on` handler loop: the API's registration list lives on
// `state.terminalInputHandlers` (upstream keeps it outside the per-extension
// event tables) and the return shape is the raw-input one.
//
// Upstream semantics (pi-tui `handleTerminalInput`): listeners run in
// registration order; `consume: true` returns immediately (later listeners and
// the focused component never see the chunk); `data` replaces the chunk for
// later listeners; a final empty string consumes the chunk. Errors are
// isolated per handler and fall through (the input is never lost).
function dispatchTerminalInputToHandlers(payload) {
  const errors = [];
  let data = isPlainObject(payload) && typeof payload.data === "string" ? payload.data : "";
  const original = data;
  for (const handler of state.terminalInputHandlers.slice()) {
    let result;
    try {
      // Upstream calls the listener synchronously and inspects the returned
      // object; an async handler's Promise has no `consume`/`data` fields and
      // is therefore ignored (same observable behavior as upstream).
      result = handler(data);
    } catch (error) {
      const message = error && error.message ? error.message : String(error);
      errors.push({ extId: 0, event: "terminal_input", error: message });
      continue;
    }
    if (!result || typeof result !== "object") continue;
    if (result.consume === true) {
      return {
        outcome: { consume: true, data },
        payload: { ...payload, data },
        errors,
      };
    }
    if (result.data !== undefined) data = String(result.data);
  }
  const next = isPlainObject(payload) ? { ...payload, data } : { data };
  if (data.length === 0 && original.length > 0) {
    // Upstream: `if (current.length === 0) return;` — consumed.
    return { outcome: { consume: true, data }, payload: next, errors };
  }
  return {
    outcome: data === original ? null : { data },
    payload: next,
    errors,
  };
}

// Event dispatch core: runs the registered handler list for `payload` and
// returns the merged `{outcome, payload, errors}` triple. Split out of
// `__pir_dispatchEvent` so the streaming read-only entry point below runs the
// exact same handler semantics without paying for the JSON reply (see
// `__pir_dispatchEventReadOnly`).
async function dispatchEventToHandlers(eventName, payload, dispatchEpoch) {
  if (eventName === "terminal_input") {
    return dispatchTerminalInputToHandlers(payload);
  }
  // `ctx.compact` callbacks settle on the compaction events (upstream invokes
  // them from the compact promise, not from a handler).
  if (eventName === "session_compact" || eventName === "session_compact_failed") {
    settleCompactCallbacks(eventName, payload);
  }
  const desc = JSON.parse(__pir_host.getContextJson());
  const errors = [];
  let outcome = null;
  // The timing/epoch wrapper is only needed when the PI_TIMING gate is on or
  // this dispatch carries an epoch. A dispatch without an epoch (0, timing
  // off) calls the handler directly; the reply-carrying entry always carries
  // one, and the read-only entry does too (the host allocates it for
  // abandoned-dispatch tracking).
  const timedDispatch =
    PIR_TIMING_ENABLED ||
    (typeof dispatchEpoch === "number" && dispatchEpoch > 0);

  // Chaining state per event family (upstream runner.ts keeps the current
  // value across handlers so each handler observes earlier replacements).
  let currentMessages = Array.isArray(payload.messages) ? payload.messages : undefined;
  let currentMessage = payload.message !== undefined ? payload.message : undefined;
  let currentProviderPayload =
    isPlainObject(payload) && payload.payload !== undefined ? payload.payload : undefined;
  let contextTouched = false;
  let messageEndTouched = false;
  // Boundary events (turn_end / agent_before_settle): entries drafts and the
  // continue decision chain across handlers (upstream emitBoundary).
  let boundaryEntries = Array.isArray(payload?.entries) ? payload.entries : [];
  let boundaryContinue = payload?.continue === true;
  // before_agent_start: every handler's message is accumulated (upstream
  // collects `messages`) and a forced system prompt chains into both the
  // event and ctx.getSystemPrompt for later handlers (P2-10).
  const agentStartMessages = [];
  let forcedSystemPrompt;
  // The MUTABLE BuildSystemPromptOptions object shared by every handler
  // (upstream runner.ts:1431: one `currentOptions` object across the whole
  // dispatch). A handler mutates `event.systemPromptOptions` (or replaces it)
  // and later handlers observe the change in `event.systemPrompt`, which is
  // re-rendered through the host's ordered section builder exactly like
  // upstream's `renderCurrentSystemPrompt` getter (runner.ts:1429-1446).
  let currentPromptOptions =
    eventName === "before_agent_start" && isPlainObject(payload.systemPromptOptions)
      ? payload.systemPromptOptions
      : undefined;
  const renderCurrentSystemPrompt = () => {
    if (forcedSystemPrompt !== undefined) return forcedSystemPrompt;
    if (currentPromptOptions !== undefined) {
      const rendered = renderSystemPromptFromOptions(currentPromptOptions);
      if (typeof rendered === "string" && rendered !== "") return rendered;
    }
    return payload.systemPrompt;
  };
  // tool_result merge state (upstream emitToolResult): handlers chain, and
  // replacing `content` without `structuredContent` drops the structured
  // content (it may no longer match).
  let toolResultState = null;

  const handlers = snapshotEventHandlers(eventName);
  // The ctx factory sees the chained system-prompt override (upstream
  // renderCurrentSystemPrompt; the before_agent_start ctx re-renders the
  // mutable options, every other context keeps the host prompt).
  const ctx = makeCtx(
    desc,
    () => (eventName === "before_agent_start" ? renderCurrentSystemPrompt() : forcedSystemPrompt),
  );

  handlerLoop: for (const [extId, handler] of handlers) {
    try {
      let event = payload;
      const messagesSnapshot =
        (eventName === "context" || eventName === "context_with_system") && currentMessages !== undefined
          ? currentMessages.slice()
          : undefined;
      if ((eventName === "context" || eventName === "context_with_system") && currentMessages !== undefined) {
        event = { ...payload, messages: currentMessages };
      } else if (eventName === "message_end" && currentMessage !== undefined) {
        event = { ...payload, message: currentMessage };
      } else if (eventName === "before_provider_request" && currentProviderPayload !== undefined) {
        event = { ...payload, payload: currentProviderPayload };
      } else if (eventName === "before_agent_start") {
        // The event exposes the current (possibly forced) system prompt
        // exactly like upstream's getter (runner.ts:1444-1446), and it is a
        // union on `messageType` (upstream `BeforeAgentStartEvent`): "user"
        // runs carry `prompt`/`images`, "custom" runs the starting `message`
        // and no `prompt`. A host that emits the pre-union payload shape (no
        // `messageType`) is a user run, so extensions that branch on the field
        // behave like upstream's.
        event = {
          ...payload,
          messageType: payload.messageType ?? "user",
          systemPrompt: renderCurrentSystemPrompt(),
          ...(currentPromptOptions !== undefined
            ? { systemPromptOptions: currentPromptOptions }
            : {}),
        };
      } else if (eventName === "model_select") {
        // Upstream hands Model OBJECTS for `model`/`previousModel`
        // (extensions/types.ts:1134-1139); the host payload may name the
        // previous model by id. Resolve it against the pushed catalog.
        event = {
          ...payload,
          model: resolveModelRef(payload.model),
          ...(payload.previousModel !== undefined
            ? { previousModel: resolveModelRef(payload.previousModel) }
            : {}),
        };
      } else if (eventName === "turn_end" || eventName === "agent_before_settle") {
        // Later handlers see the earlier handlers' drafts/decision
        // (upstream emitBoundary threads entries + continue, P2-8).
        event = { ...payload, entries: boundaryEntries, continue: boundaryContinue };
      } else if (eventName === "tool_result" && toolResultState !== null) {
        // Upstream emitToolResult threads `currentEvent` across handlers
        // (runner.ts:1183-1196): later handlers observe the replacements
        // earlier handlers made (content/details/structuredContent/isError/
        // usage). `undefined` means "field dropped" (replacing content
        // without structuredContent deletes it upstream), so the key is
        // removed rather than set to undefined.
        event = { ...payload };
        for (const [field, value] of Object.entries(toolResultState.values)) {
          if (value === undefined) delete event[field];
          else event[field] = value;
        }
      }
      const result = timedDispatch
        ? await invokeTimedHandler(extId, handler, event, ctx, eventName, dispatchEpoch)
        : await handler(event, ctx);

      // Context chaining: a returned replacement wins; an in-place edit of
      // event.messages (detected against the pre-handler snapshot) becomes
      // the new currentMessages too (P2-11).
      if ((eventName === "context" || eventName === "context_with_system") && currentMessages !== undefined) {
        if (result !== undefined && result !== null && result.messages !== undefined) {
          currentMessages = result.messages;
          contextTouched = true;
        } else if (
          event.messages !== undefined &&
          !sameMessagesShallow(messagesSnapshot, event.messages)
        ) {
          currentMessages = event.messages;
          contextTouched = true;
        }
      }

      // before_agent_start: a handler may REPLACE `event.systemPromptOptions`
      // (not only mutate it in place). The replacement becomes the shared
      // object for later handlers and rides the returned payload home, where
      // the host renders the run's prompt from it.
      if (eventName === "before_agent_start" && event.systemPromptOptions !== undefined) {
        currentPromptOptions = event.systemPromptOptions;
        payload.systemPromptOptions = currentPromptOptions;
      }

      if (result === undefined || result === null) continue;
      switch (eventName) {
        case "tool_call":
          if (result.block === true) {
            outcome = { block: true, reason: result.reason ?? null, terminate: result.terminate === true };
          }
          break;
        case "input":
          if (result.action === "handled") {
            outcome = { action: "handled" };
          } else if (result.action === "transform") {
            payload.text = result.text ?? "";
            if (result.images !== undefined) payload.images = result.images;
            outcome = { action: "transform", text: payload.text, images: payload.images };
          }
          break;
        case "context_with_system": {
          // Chained like `context` but with the leading system message present;
          // the final replacement flows back to the host. (The chaining itself
          // runs above; nothing else to merge here.)
          break;
        }
        case "before_agent_start": {
          if (result.systemPrompt !== undefined) {
            forcedSystemPrompt = result.systemPrompt;
            // Upstream stores the override ON the shared options object
            // (`currentOptions.forceSystemPrompt = result.systemPrompt`,
            // runner.ts:1450-1452): later handlers' `event.systemPrompt` and
            // the host's final render both take the forced text, even if a
            // later handler mutates other option fields.
            if (isPlainObject(currentPromptOptions)) {
              currentPromptOptions.forceSystemPrompt = result.systemPrompt;
              payload.systemPromptOptions = currentPromptOptions;
            }
            outcome = { ...(outcome ?? {}), systemPrompt: result.systemPrompt };
          }
          if (result.message !== undefined) {
            agentStartMessages.push(result.message);
            outcome = {
              ...(outcome ?? {}),
              message: result.message,
              messages: [...agentStartMessages],
            };
          }
          break;
        }
        case "user_bash": {
          if (!isUserBashResult(result)) {
            const message = "Invalid user_bash handler result: return undefined for local execution or exactly one valid { operations } or { result } object";
            errors.push({ extId: Number(extId), event: eventName, error: message });
            // Upstream rethrows past the remaining handlers and the host does
            // NOT fall back to local execution (P1-1).
            outcome = { aborted: true };
            break handlerLoop;
          }
          // First valid result wins (upstream returns immediately). Custom
          // operations carry only a token across the bridge (the `exec`
          // closure stays in JS state); the host executes them for real.
          outcome = exportUserBashResult(result);
          break handlerLoop;
        }
        case "before_provider_request": {
          // Upstream chains: the next handler sees the previous replacement.
          currentProviderPayload = result;
          outcome = result;
          break;
        }
        case "before_provider_headers": {
          // Handlers mutate payload.headers in place; return values ignored.
          // The mutated payload is returned to the host below.
          break;
        }
        case "project_trust": {
          // The first decided (trusted !== "undecided") result wins and stops
          // the dispatch (upstream emitProjectTrustEvent, P2-9).
          if (result.trusted === undefined || result.trusted === "undecided") break;
          outcome = {
            trusted: result.trusted,
            ...(result.remember !== undefined ? { remember: result.remember } : {}),
          };
          break handlerLoop;
        }
        case "resources_discover": {
          // Accumulate path lists across handlers (runner.ts emitResourcesDiscover).
          const next = { ...(outcome ?? {}) };
          for (const key of ["skillPaths", "promptPaths", "themePaths"]) {
            if (Array.isArray(result[key]) && result[key].length > 0) {
              next[key] = [...(next[key] ?? []), ...result[key].map(String)];
            }
          }
          outcome = next;
          break;
        }
        case "session_before_switch":
          if (result.cancel === true) outcome = { cancel: true };
          break;
        case "session_before_fork":
          // Any session_before_* cancel short-circuits the whole dispatch
          // (upstream emit, runner.ts:1089-1118, P1-2); later handlers can no
          // longer overwrite an earlier cancel decision.
          if (result.cancel === true) { outcome = { cancel: true }; break handlerLoop; }
          if (result.cancel !== undefined || result.skipConversationRestore !== undefined) {
            outcome = { ...(outcome ?? {}), cancel: result.cancel === true, skipConversationRestore: result.skipConversationRestore === true };
          }
          break;
        case "session_before_compact":
          if (result.cancel === true) { outcome = { cancel: true }; break handlerLoop; }
          if (result.cancel !== undefined || result.compaction !== undefined) {
            outcome = { ...(outcome ?? {}), cancel: result.cancel, compaction: result.compaction };
          }
          break;
        case "session_before_tree":
          if (result.cancel === true) { outcome = { cancel: true }; break handlerLoop; }
          if (result.cancel !== undefined || result.summary !== undefined || result.customInstructions !== undefined
            || result.replaceInstructions !== undefined || result.label !== undefined) {
            outcome = { ...(outcome ?? {}), ...result };
          }
          break;
        case "turn_end":
        case "agent_before_settle": {
          // Boundary events (upstream emitBoundary): handlers chain `entries`
          // drafts and the `continue` decision; the merged result flows back
          // to the host which commits the drafts and honors the continuation.
          if (result.entries !== undefined || result.continue !== undefined) {
            const merged = { ...(outcome ?? {}) };
            if (result.entries !== undefined) {
              boundaryEntries = result.entries;
              merged.entries = result.entries;
            }
            if (result.continue !== undefined) {
              boundaryContinue = result.continue;
              merged.continue = result.continue;
            }
            outcome = merged;
          }
          break;
        }
        case "message_end": {
          // Chained with role checks (runner.ts emitMessageEnd): a falsy
          // message (null included) means "no replacement" (P2-7); a
          // replacement must keep the original message role or the handler's
          // result is dropped with an isolated error.
          if (!result.message) break;
          if (currentMessage !== undefined && result.message.role !== currentMessage.role) {
            errors.push({
              extId: Number(extId),
              event: eventName,
              error: "message_end handlers must return a message with the same role",
            });
            break;
          }
          currentMessage = result.message;
          messageEndTouched = true;
          break;
        }
        case "tool_result": {
          if (toolResultState === null) {
            toolResultState = {
              values: {
                content: payload.content,
                details: payload.details,
                structuredContent: payload.structuredContent,
                isError: payload.isError,
                usage: payload.usage,
              },
              modified: {},
            };
          }
          const mark = (field, value) => {
            toolResultState.values[field] = value;
            toolResultState.modified[field] = true;
          };
          if (result.content !== undefined) {
            mark("content", result.content);
            // Structured content that is not replaced along with the content may
            // no longer match it (upstream drops it in that case).
            if (result.structuredContent === undefined) {
              toolResultState.values.structuredContent = undefined;
              toolResultState.modified.structuredContent = false;
            }
          }
          if (result.details !== undefined) mark("details", result.details);
          if (result.structuredContent !== undefined) mark("structuredContent", result.structuredContent);
          if (result.isError !== undefined) mark("isError", result.isError);
          if (result.usage !== undefined) mark("usage", result.usage);
          // Only the fields handlers set are reported (absent = unchanged);
          // `structuredContent: false` in the modified map means "dropped".
          const merged = {};
          for (const [field, wasModified] of Object.entries(toolResultState.modified)) {
            if (wasModified) merged[field] = toolResultState.values[field];
          }
          if (Object.keys(merged).length > 0) outcome = merged;
          break;
        }
        case "provider:refreshModels": {
          // Synthetic host->JS channel for registerProvider refreshModels
          // (upstream ModelRuntime availability refresh calling
          // `ProviderConfig.refreshModels`): the matching provider's hook
          // result rides the outcome; handlers of other providers pass.
          if (
            result &&
            result.providerId === payload.providerId &&
            Array.isArray(result.models)
          ) {
            outcome = { providerId: result.providerId, models: result.models };
            // `publish({persist})` decision handed to the host, which owns
            // models-store.json (`null` = delete). `undefined` means "leave
            // storage unchanged" and must not appear on the wire.
            if (Object.prototype.hasOwnProperty.call(result, "persist")) {
              outcome.persist = result.persist;
            }
          }
          break;
        }
        case "cache_warming_decision": {
          // Upstream emitCacheWarmingDecision (runner.ts:1121): each handler's
          // {action} result overrides the running decision; the LAST override
          // wins and the final value decides whether the warm refresh is
          // sent. Only the two valid actions are honored — anything else
          // leaves pi's own decision untouched.
          if (result.action === "warm" || result.action === "stop") {
            outcome = { ...(outcome ?? {}), action: result.action };
          }
          break;
        }
        default:
          // Informational events (agent_*, turn_*, message_start/update,
          // tool_execution_*, model_select, thinking_level_select,
          // session_info_changed, session_compact*, session_tree,
          // session_shutdown, after_provider_response, ui_prompt_*): return
          // values are ignored, matching upstream's generic emit().
          break;
      }
    } catch (error) {
      const message = error && error.message ? error.message : String(error);
      errors.push({
        extId: Number(extId),
        event: eventName,
        error: message,
      });
      // Upstream treats a throwing tool_call handler as fail-closed: the
      // tool must not run. A user_bash handler that throws or returns an
      // invalid shape aborts the dispatch entirely (runner.ts rethrows and
      // the interactive mode does not fall back to local execution, P1-1).
      if (eventName === "tool_call") {
        outcome = { block: true, reason: `extension handler failed: ${message}` };
        break;
      }
      if (eventName === "user_bash") {
        outcome = { aborted: true };
        break handlerLoop;
      }
      continue;
    }
    if (outcome !== null && (eventName === "tool_call" || eventName === "input"
        || eventName === "session_before_switch")
        && (outcome.block === true || outcome.action === "handled" || outcome.cancel === true)) {
      break;
    }
  }

  if ((eventName === "context" || eventName === "context_with_system") && contextTouched) {
    outcome = { messages: currentMessages };
  }
  if (eventName === "message_end" && messageEndTouched) {
    outcome = { message: currentMessage };
  }
  if (eventName === "before_provider_headers") {
    outcome = isPlainObject(payload.headers) ? { headers: payload.headers } : null;
  }

  return {
    outcome,
    payload: isPlainObject(payload) ? payload : undefined,
    errors,
  };
}

// Standard dispatch entry: the handler outcome, the (possibly mutated) payload
// and the per-handler errors flow back to the host as one JSON string — the
// host seams that apply in-place handler mutations read them.
globalThis.__pir_dispatchEvent = async (eventName, payloadJson, dispatchEpochJson) =>
  JSON.stringify(await dispatchEventToHandlers(eventName, JSON.parse(payloadJson), Number(dispatchEpochJson)));

// Streaming read-only dispatch entry: the per-delta `message_update` hook (and
// any other hook whose return value the host does NOT consume). Handlers run
// through the same core — same registration order, same chaining, same error
// isolation — but no reply is built, so a slow JS side no longer pays a
// reply-sized `JSON.stringify` per delta for a value nobody reads (the agent
// loop ignores `message_update`'s return: upstream parity gap, see
// `MessageUpdateDispatcher` in agent_loop.rs).
//
// ---------------------------------------------------------------------------
// The live message (no cumulative snapshot crosses the boundary)
// ---------------------------------------------------------------------------
//
// Upstream hands a `message_update` handler the LIVE `AgentMessage` object: the
// provider mutates it in place as chunks arrive, so a handler's reads are plain
// property reads and cost nothing. pir cannot send a live object across the
// Rust/QuickJS boundary, and shipping the cumulative snapshot per delta is
// O(reply) per delta / O(reply^2) per stream — enough for one defensive
// handler (reading `event.message.role`, or `.content.length`) to dominate a
// whole run's wall clock and memory. So the HOST classifies every event
// (`message_update_stream_json`) and ships one of two wires:
//
//   - a DELTA LINE (marked `__pirDelta: true`, all per-delta event types): the
//     message's O(1) scalars, never its accumulated `content`, plus the delta
//     itself — with the authoritative `block` helper (or the O(delta)
//     `arguments` `patch`) for every event whose block this accumulator cannot
//     derive from the delta alone. The host's delta set and the type set
//     `applyMessageUpdateDelta` below knows MUST stay identical: a shape this
//     side cannot apply must be classified as the full payload by the host,
//     never shipped as a delta line;
//   - the FULL REFERENCE PAYLOAD (no `__pirDelta`; today the terminal
//     `done`/`error`, which have no `partial` and no derivable content): the
//     old `{type, message, assistantMessageEvent}` construction, byte for byte
//     what a handler observed before this lane. A non-terminal full payload
//     means the host could not express the event as a delta line: it is synced
//     into the live object in place and reported loudly (see below), never
//     silently replaced.
//
// The per-delta accumulator is keyed by the wire's stream id (`__pirStream`,
// assigned by the host per assistant-message stream), so two interleaved
// streams sharing one runtime (subagent children reuse the parent handle)
// keep separate live objects. ONE object per stream, created from the stream's
// first dispatched event's wire message — the wire's object BECOMES the live
// object, so its key order, its scalar values and its `content` array are the
// host's, and no message shape is mirrored in JS:
//
//   - updated in place per delta: `text_delta` / `thinking_delta` append the
//     chunk to the block at `content_index` (the provider's own `push_str`),
//     every other per-delta event ships the AUTHORITATIVE block the provider
//     put into the partial (the internal `block` helper, deleted before any
//     handler runs) and it replaces the block — `toolcall_delta` ships either
//     that block or the O(delta) `patch` (`arguments[key] += tail`, a rope
//     append), because the provider mirrors PARSED arguments into the block;
//   - every non-`content` field rides the host's view verbatim and the key set
//     is reconciled against the wire (a field that disappears is dropped, and
//     so is a key a handler added), so `role`, `timestamp`, `usage`,
//     `stopReason`, ... cannot drift;
//   - `done` / `error`: the terminal event's own final message supplies the
//     final scalars, then the accumulator is dropped — nothing accumulates
//     across turns.
//
// `payload.message` and `assistantMessageEvent.partial` are handed that SAME
// object (upstream's live object in both slots; the previous wire aliased the
// two as well). Per-delta work is O(delta): a string append or a block
// replace, no host call, no snapshot parse. The object is self-contained plain
// data, so a handler that keeps it (pi-phone's `liveAssistantMessage`) still
// reads it after the dispatch returns.
//
// The read-only contract is kept even though the object is shared: the
// accumulator appends on its OWN copy of each block string (`stream.texts`) and
// writes that copy back into the block before every delta, so a handler's
// in-place edit of `content[...]` (or of the array itself) can never leak into
// what a LATER handler reads, and a corrupted array takes the fallback path
// instead of drifting. The handler's return value stays ignored entirely (the
// read-only entry builds no reply).
//
// NO SILENT DRIFT: a delta the accumulator cannot apply exactly — a delta for a
// block it never saw, a `content_index` gap, a malformed `patch` — is reported
// once per stream (console.error -> host notification, plus the
// `__pir_messageUpdateFallbacks` counter) and the accumulator keeps the
// authoritative blocks it does have; the next `start` re-anchors it. KNOWN
// RESIDUAL (documented, not hidden): such an event has no authoritative source
// to re-sync from — that is exactly the reply-sized host call this lane removed
// — so the live object stays partial for that stream. The host ships an
// authoritative `block`/`patch` for every event whose value cannot be derived
// from the delta, so this path requires a provider projection that emits
// block indices the reference content does not have.
//
// Terminal events without a `partial` (the historical `{role: "assistant"}`
// message shape) keep the reference payload; the wire for those IS the
// reference construction (`agent_loop::message_update_payload`).

// The wire key carrying one event's authoritative content block (the provider's
// value at `content_index`). Must match `MESSAGE_UPDATE_BLOCK_KEY` in
// `src/extensions/runtime/dispatch.rs`; removed from the event before any
// handler runs, so `assistantMessageEvent` keeps the reference keys exactly.
const MESSAGE_UPDATE_BLOCK_KEY = "block";

// The wire key carrying one `toolcall_delta`'s O(delta) `arguments` patch
// (`{key, tail}`). Must match `MESSAGE_UPDATE_PATCH_KEY` in dispatch.rs.
const MESSAGE_UPDATE_PATCH_KEY = "patch";

// The wire key marking a delta line. Must match `MESSAGE_UPDATE_DELTA_KEY` in
// dispatch.rs.
const MESSAGE_UPDATE_DELTA_KEY = "__pirDelta";

// The wire key carrying the stream id the dispatch belongs to. Must match
// `MESSAGE_UPDATE_STREAM_KEY` in dispatch.rs.
const MESSAGE_UPDATE_STREAM_KEY = "__pirStream";

// Live objects kept at once. Streams are short-lived and sequential per run;
// the bound only bites for streams that never sent their terminal event (an
// aborted run), where the oldest entry is the one whose deltas stopped first.
const MESSAGE_UPDATE_STREAM_LIMIT = 8;

// One accumulator fallback: counted always (probeable), reported once per
// stream and reason — a repeated unrecognised shape must not flood the host's
// notification stream.
function messageUpdateFallbackDiag(stream, reason) {
  globalThis.__pir_messageUpdateFallbacks =
    (globalThis.__pir_messageUpdateFallbacks ?? 0) + 1;
  if (stream !== null && stream.reported.has(reason)) return;
  if (stream !== null) stream.reported.add(reason);
  try {
    console.error(`[pir] message_update accumulator fallback: ${reason}`);
  } catch {
    /* best-effort (a host-less prelude evaluation has no host to notify) */
  }
}

// The live object of one stream, or null when the stream has none (its `start`
// has not been dispatched yet, or it already ended).
function getLiveMessageUpdate(key) {
  const stream = state.liveMessageUpdates.get(key);
  return stream === undefined ? null : stream;
}

function setLiveMessageUpdate(key, stream) {
  if (stream === null) {
    state.liveMessageUpdates.delete(key);
    return;
  }
  if (!state.liveMessageUpdates.has(key)
    && state.liveMessageUpdates.size >= MESSAGE_UPDATE_STREAM_LIMIT) {
    const oldest = state.liveMessageUpdates.keys().next();
    if (!oldest.done) state.liveMessageUpdates.delete(oldest.value);
  }
  state.liveMessageUpdates.set(key, stream);
}

// The accumulator stream a payload belongs to. The host always stamps the id;
// a missing one (a hand-built payload in a host-less harness) shares one slot.
function messageUpdateStreamKey(payload) {
  const id = payload[MESSAGE_UPDATE_STREAM_KEY];
  return typeof id === "number" || typeof id === "string" ? id : "default";
}

// Apply the host's view of the message (everything but the accumulated
// `content` array) to the live object. The key set is reconciled by SET: every
// key the live object carries that this wire does not is dropped — a field
// that went back to absent (`skip_serializing_if`), a stale key from an
// equal-length field swap (one field leaving while another arrives), and a key
// a handler added itself. Comparing key COUNTS (the previous shape) missed all
// three, leaving a live object that drifted from the reference for the rest of
// the stream.
function applyMessageUpdateScalars(stream, wireMessage) {
  const live = stream.message;
  const keys = Object.keys(wireMessage);
  const present = new Set(keys);
  const liveKeys = Object.keys(live);
  for (let index = 0; index < liveKeys.length; index++) {
    const key = liveKeys[index];
    if (key !== "content" && !present.has(key)) delete live[key];
  }
  for (let index = 0; index < keys.length; index++) {
    const key = keys[index];
    if (key !== "content") live[key] = wireMessage[key];
  }
  stream.keys = keys;
}

// Seed the accumulator's private copy of every block string (`text` /
// `thinking`) from the blocks the live object currently carries. The copy is
// what a delta appends to, and it is written back into the block before every
// append: a handler's in-place edit of the live object (the read-only contract
// says those are ignored) is therefore stomped by the next delta instead of
// leaking into what later handlers read.
function rememberMessageUpdateBlocks(stream) {
  stream.texts = [];
  const content = stream.message.content;
  if (!Array.isArray(content)) return;
  for (let index = 0; index < content.length; index++) {
    const block = content[index];
    if (!isPlainObject(block)) continue;
    if (typeof block.text === "string") stream.texts[index] = block.text;
    else if (typeof block.thinking === "string") stream.texts[index] = block.thinking;
  }
}

// Place the authoritative block at `index`. `mode` is "start" for a
// block-opening event, "delta" for a delta that carries its block, "finalize"
// for an end event:
// - "start": the provider appended the block at the end of the partial's
//   content right before emitting it, so the reference content ends exactly at
//   `index + 1`; anything else (a gap, or an index the accumulator already
//   passed) means the accumulator and the reference disagree about the block
//   layout — a fallback, never a silent truncation that would drop blocks the
//   reference still carries;
// - "finalize": the block must already be accumulated — anything else is an
//   accumulator that lost state, and must fall back;
// - "delta": the block may open one too (a provider that emitted no `*_start`),
//   but never beyond the end.
function setAccumulatedBlock(stream, index, block, mode) {
  const content = stream.message.content;
  if (!Array.isArray(content)) {
    return "live message without a content array";
  }
  if (!Number.isInteger(index) || index < 0) {
    return `invalid content_index ${index}`;
  }
  if (index > content.length) {
    return `content_index ${index} beyond the accumulated ${content.length} block(s)`;
  }
  if (mode === "start") {
    if (index !== content.length) {
      return `block start at content_index ${index} with ${content.length} accumulated block(s)`;
    }
    content.push(block);
    rememberBlockText(stream, index, block);
    return null;
  }
  if (index === content.length) {
    if (mode === "finalize") {
      return `finalized a block at content_index ${index} the accumulator never saw`;
    }
    content.push(block);
    rememberBlockText(stream, index, block);
    return null;
  }
  content[index] = block;
  rememberBlockText(stream, index, block);
  return null;
}

// Remember the authoritative string of the block placed at `index` (the
// accumulator's append base), dropping the entry for blocks that carry none
// (e.g. a tool call).
function rememberBlockText(stream, index, block) {
  const field = block.type === "thinking" ? "thinking" : "text";
  if (typeof block[field] === "string") stream.texts[index] = block[field];
  else delete stream.texts[index];
}

// Apply one O(delta) `toolcall_delta` patch: the block's `arguments[key]`
// gained `tail`. Rope append (O(1) in QuickJS), and the block is left in place —
// no per-delta parse of the accumulated arguments JSON.
function applyToolArgumentPatch(stream, index, patch) {
  const content = stream.message.content;
  if (!Array.isArray(content)) {
    return "live message without a content array";
  }
  if (!Number.isInteger(index) || index < 0 || index >= content.length) {
    return `toolcall_delta patch for an unknown block at content_index ${index}`;
  }
  const block = content[index];
  if (!isPlainObject(block) || !isPlainObject(block.arguments)) {
    return `toolcall_delta patch for a block without arguments at content_index ${index}`;
  }
  const key = patch.key;
  const tail = patch.tail;
  if (typeof key !== "string" || typeof tail !== "string") {
    return "toolcall_delta with a malformed arguments patch";
  }
  const current = block.arguments[key];
  if (typeof current !== "string") {
    return `toolcall_delta patch for arguments key ${JSON.stringify(key)} the block does not carry as a string`;
  }
  block.arguments[key] = current + tail;
  return null;
}

// Apply one per-delta event to the accumulated content. Returns null when it
// was applied exactly, else the reason the fallback path must take over.
function applyMessageUpdateDelta(stream, event, type) {
  const content = stream.message.content;
  if (!Array.isArray(content)) return "live message without a content array";
  const index = event.content_index;
  const block = event[MESSAGE_UPDATE_BLOCK_KEY];
  switch (type) {
    case "start":
      // The wire message carried the initial partial's content itself.
      return null;
    case "text_start":
    case "thinking_start":
    case "toolcall_start":
      if (!isPlainObject(block)) return `${type} without an authoritative block`;
      return setAccumulatedBlock(stream, index, block, "start");
    case "toolcall_delta":
      // The host ships the O(delta) patch when exactly one string argument
      // gained a suffix and nothing else changed (host-side
      // `tool_argument_patch`), the whole parsed block otherwise.
      if (isPlainObject(event[MESSAGE_UPDATE_PATCH_KEY])) {
        return applyToolArgumentPatch(stream, index, event[MESSAGE_UPDATE_PATCH_KEY]);
      }
      if (!isPlainObject(block)) return `${type} without an authoritative block`;
      return setAccumulatedBlock(stream, index, block, "finalize");
    case "text_end":
    case "thinking_end":
    case "toolcall_end":
      // The host ships the provider's own block for every event the delta
      // cannot express (finalized blocks, tool-call argument deltas).
      if (!isPlainObject(block)) return `${type} without an authoritative block`;
      return setAccumulatedBlock(stream, index, block, "finalize");
    case "text_delta":
    case "thinking_delta": {
      const field = type === "text_delta" ? "text" : "thinking";
      if (isPlainObject(block)) return setAccumulatedBlock(stream, index, block, "delta");
      if (!isPlainObject(content[index]) || typeof content[index][field] !== "string") {
        return `${type} for an unknown block at content_index ${index}`;
      }
      // Append on the accumulator's private copy and write it back: the block
      // never carries a handler's edit into the next delta.
      const accumulated = `${stream.texts[index] ?? content[index][field]}${
        typeof event.delta === "string" ? event.delta : ""}`;
      stream.texts[index] = accumulated;
      content[index][field] = accumulated;
      return null;
    }
    default:
      return `unrecognised assistantMessageEvent type ${JSON.stringify(type)}`;
  }
}

// Apply the scalars of a final (terminal) message to the live object in place,
// leaving the accumulated `content` array alone. The reference payload's own
// `message` slot is the historical `{role: "assistant"}` stub, so the terminal
// values come from the event's final message/error.
function syncLiveMessageUpdateScalars(stream, final) {
  const live = stream.message;
  const keys = Object.keys(final);
  for (let index = 0; index < keys.length; index++) {
    const key = keys[index];
    if (key !== "content") live[key] = final[key];
  }
}

// The full-payload route: replace the live object's fields IN PLACE from the
// reference `message` — identity is preserved (a handler holding the object
// sees the authoritative values) and the content array the accumulator owns is
// refilled rather than swapped.
function syncMessageUpdateFromSnapshot(stream, snapshot) {
  const live = stream.message;
  for (const key of Object.keys(live)) {
    if (!(key in snapshot)) delete live[key];
  }
  for (const key of Object.keys(snapshot)) {
    if (key === "content") continue;
    live[key] = snapshot[key];
  }
  let content = live.content;
  if (!Array.isArray(content)) {
    // A handler deleted or replaced the array (the read-only contract forbids
    // it, but a broken handler must not poison the stream): re-attach a fresh
    // array, otherwise every later delta keeps failing against a value no
    // handler should ever see.
    content = [];
    live.content = content;
  }
  content.length = 0;
  if (Array.isArray(snapshot.content)) {
    for (const block of snapshot.content) content.push(block);
  }
  stream.keys = Object.keys(live);
  rememberMessageUpdateBlocks(stream);
}

// Install the live message on one read-only `message_update` payload, before
// any handler runs. Two routes (see the section docs above): a delta line
// (`__pirDelta`) feeds the per-stream accumulator, the full reference payload
// syncs the live object from its own `message` value.
function applyLiveMessageUpdate(payload) {
  const assistantEvent = isPlainObject(payload.assistantMessageEvent)
    ? payload.assistantMessageEvent
    : null;
  const streamKey = messageUpdateStreamKey(payload);
  delete payload[MESSAGE_UPDATE_STREAM_KEY];
  const deltaLine = payload[MESSAGE_UPDATE_DELTA_KEY] === true;
  delete payload[MESSAGE_UPDATE_DELTA_KEY];
  if (assistantEvent === null) return;
  const type = assistantEvent.type;
  if (type === "done" || type === "error") {
    // Terminal event: the host ships the reference payload (the final message
    // rides the event itself; the payload's `message` slot is the historical
    // stub), so its values stay exactly what they were. A handler that kept
    // the live object takes the final scalars, then the accumulator is
    // dropped.
    const stream = getLiveMessageUpdate(streamKey);
    const final = type === "done" ? assistantEvent.message : assistantEvent.error;
    if (stream !== null) {
      if (isPlainObject(final)) syncLiveMessageUpdateScalars(stream, final);
      setLiveMessageUpdate(streamKey, null);
    }
    delete assistantEvent[MESSAGE_UPDATE_BLOCK_KEY];
    delete assistantEvent[MESSAGE_UPDATE_PATCH_KEY];
    return;
  }
  const wireMessage = isPlainObject(payload.message) ? payload.message : null;
  if (!deltaLine) {
    // The full reference payload for a non-terminal event: a shape the host
    // could not express as a delta line. No accumulator source can rebuild a
    // delta from it, so this is the loud path — sync the live object in place
    // from the reference `message` (the value the reference construction
    // produced) and report once per stream. Without an accumulator the
    // reference payload is already exactly what a handler must see.
    const stream = getLiveMessageUpdate(streamKey);
    const reason = `full message_update payload for type ${JSON.stringify(type)}`;
    if (stream !== null && wireMessage !== null) {
      syncMessageUpdateFromSnapshot(stream, wireMessage);
      payload.message = stream.message;
      assistantEvent.partial = stream.message;
    }
    messageUpdateFallbackDiag(stream, reason);
    delete assistantEvent[MESSAGE_UPDATE_BLOCK_KEY];
    delete assistantEvent[MESSAGE_UPDATE_PATCH_KEY];
    return;
  }
  if (wireMessage === null) {
    // The host always ships the scalar view on a delta line; a missing one is
    // a broken wire. Report it and keep the payload's own values (nothing to
    // accumulate), with the internal helpers stripped either way.
    messageUpdateFallbackDiag(null, "message_update delta line without a wire message");
    delete assistantEvent[MESSAGE_UPDATE_BLOCK_KEY];
    delete assistantEvent[MESSAGE_UPDATE_PATCH_KEY];
    return;
  }
  let stream = getLiveMessageUpdate(streamKey);
  if (type === "start" || stream === null) {
    // The wire object becomes the live object: a stream's `start` re-anchors
    // the accumulator (its wire message carries the initial content), so a
    // stale accumulator from a stream whose start/settle was never dispatched
    // cannot leak into this one. Each stream reports its own fallbacks.
    stream = {
      message: wireMessage,
      keys: Object.keys(wireMessage),
      texts: [],
      reported: new Set(),
    };
    setLiveMessageUpdate(streamKey, stream);
    rememberMessageUpdateBlocks(stream);
  } else {
    applyMessageUpdateScalars(stream, wireMessage);
  }
  const reason = applyMessageUpdateDelta(stream, assistantEvent, type);
  if (reason !== null) messageUpdateFallbackDiag(stream, reason);
  delete assistantEvent[MESSAGE_UPDATE_BLOCK_KEY];
  delete assistantEvent[MESSAGE_UPDATE_PATCH_KEY];
  payload.message = stream.message;
  assistantEvent.partial = stream.message;
}

globalThis.__pir_dispatchEventReadOnly = async (eventName, payloadJson, dispatchEpochJson) => {
  const payload = JSON.parse(payloadJson);
  if (eventName === "message_update" && isPlainObject(payload)) {
    applyLiveMessageUpdate(payload);
  }
  await dispatchEventToHandlers(eventName, payload, Number(dispatchEpochJson));
  // No reply: the caller ignores it (see the entry's docs).
  return "";
};

// ---------------------------------------------------------------------------
// Tool execution (arguments validated against the declared JSON schema,
// fail-closed like upstream's tool_call handling)
// ---------------------------------------------------------------------------

globalThis.__pir_callTool = async (toolName, toolCallId, paramsJson) => {
  let params = JSON.parse(paramsJson);
  const desc = JSON.parse(__pir_host.getContextJson());
  // The ctx created for this execution binds `executeTool` to this call's id
  // (nested calls get `<parent>/<n>` child ids), and carries the OWNING
  // extension's API object as `ctx.pi` (see makeCtx).
  desc.toolCallId = String(toolCallId);
  let def = null;
  let ownerExtId;
  for (const extId of Object.keys(state.extensions).sort((a, b) => Number(a) - Number(b))) {
    const ext = state.extensions[extId];
    if (ext.status === "active" && ext.tools[toolName]) {
      def = ext.tools[toolName];
      ownerExtId = extId;
      break;
    }
  }
  const ctx = makeCtx(desc, undefined, ownerExtId);
  if (!def) {
    return JSON.stringify({ ok: false, error: `Unknown extension tool: ${toolName}` });
  }
  if (__pir_host.isInvalidated()) {
    return JSON.stringify({
      ok: false,
      error: "The extension runtime was invalidated. Tool execution is no longer available.",
    });
  }
  // Upstream `ToolDefinition.prepareArguments` (types.ts:596-598): a
  // compatibility shim that repairs raw tool-call arguments BEFORE schema
  // validation. Awaited so an async shim works too (the reference type is
  // synchronous; awaiting a non-promise is free). A shim that throws makes the
  // call fail closed — the un-repaired arguments must never reach execute().
  if (typeof def.prepareArguments === "function") {
    try {
      const prepared = await def.prepareArguments(params);
      if (prepared !== undefined) params = prepared;
    } catch (error) {
      return JSON.stringify({
        ok: false,
        error: `prepareArguments for tool ${toolName} failed: ${error && error.message ? error.message : String(error)}`,
      });
    }
  }
  // Fail-closed argument validation against the declared schema.
  if (def.parameters) {
    const check = __pir_schemaValidate(sanitizeSchema(def.parameters), params);
    if (!check.valid) {
      const detail = check.errors
        .map((e) => `${e.path || "(root)"}: ${e.message}`)
        .join("; ");
      return JSON.stringify({
        ok: false,
        error: `Invalid arguments for tool ${toolName}: ${detail}`,
      });
    }
  }
  const signal = makeSignal();
  const onUpdate = (partial) => {
    __pir_host.toolUpdate(
      String(toolCallId),
      toolName,
      JSON.stringify(params ?? {}),
      partial == null ? null : JSON.stringify(partial),
    );
  };
  try {
    const result = await def.execute(String(toolCallId), params, signal, onUpdate, ctx);
    const structuredContent = result?.structuredContent ?? null;
    // NOTE: structuredContent is NOT validated against `outputSchema`.
    // Upstream treats outputSchema as purely declarative (no validation in
    // packages/agent or packages/coding-agent); the pir-only fail-closed
    // check was removed so tools return their structuredContent freely.
    return JSON.stringify({
      ok: true,
      content: Array.isArray(result?.content) ? result.content : [],
      details: result?.details ?? null,
      structuredContent,
      terminate: result?.terminate === true,
    });
  } catch (error) {
    return JSON.stringify({
      ok: false,
      error: error && error.message ? error.message : String(error),
    });
  }
};

globalThis.__pir_dispatchCommand = async (name, args) => {
  let entry = null;
  let extId = null;
  for (const id of Object.keys(state.extensions).sort((a, b) => Number(a) - Number(b))) {

  try {
    await entry.handler(args ?? "", ctx);
    return JSON.stringify({ ok: true });
  } catch (error) {
    const message = error && error.message ? error.message : String(error);
    return JSON.stringify({ ok: false, error: message, extId: Number(extId) });
  }
};

// ---------------------------------------------------------------------------
// `withSession` callback invocation (upstream ReplacedSessionContext,
// extensions/types.ts:437-452). A host that replaced the session
// (newSession/fork/switchSession) calls back with the token the recorded action
// carried. The callback runs with a FRESH command context built from the
// host's pushed snapshot of the replacement session — the same `makeCtx` every
// command dispatch uses, including the ReplacedSessionContext methods
// (`sendMessage` / `sendUserMessage`), which ride the normal host functions.
// One-shot: the token is consumed before the call. Unknown/expired tokens are
// an explicit error, never a silent no-op.
// ---------------------------------------------------------------------------

globalThis.__pir_callSessionCallback = async (tokenJson) => {
  const token = Number(tokenJson);
  const fn = state.sessionCallbacks[token];
  if (typeof fn !== "function") {
    return JSON.stringify({
      ok: false,
      error: `withSession callback token ${tokenJson} is not registered (already ran, or the extension runtime was reloaded)`,
    });
  }
  delete state.sessionCallbacks[token];
  const desc = JSON.parse(__pir_host.getContextJson());
  const ctx = makeCtx(desc);
  // Upstream `ReplacedSessionContext` (types.ts:442-452) adds message
  // injection to the command context. The callback's ctx gets it; a plain
  // command ctx keeps the upstream `ExtensionCommandContext` shape. The host
  // fn queues the message through the same action queue every other
  // `pi.send*` call uses (the session host applies it on its next drain).
  ctx.sendMessage = async (message, options) => {
    __pir_host.sendMessageJson(JSON.stringify({ message: message ?? null, options: options ?? null }));
  };
  ctx.sendUserMessage = async (content, options) => {
    __pir_host.sendUserMessageJson(JSON.stringify({ content: content ?? null, options: options ?? null }));
  };
  try {
    await fn(ctx);
    return JSON.stringify({ ok: true });
  } catch (error) {
    const message = error && error.message ? error.message : String(error);
    return JSON.stringify({ ok: false, error: message });
  }
};

// Renderer + completion callbacks (TUI consumption of registerMessageRenderer

// Renderer function resolution: upstream passes the renderer function itself
// as the second argument of registerMessageRenderer/registerEntryRenderer; an
// object with a `render` function is also accepted.
function rendererFn(entry) {
  const options = entry?.options;
  if (typeof options === "function") return options;
  if (options && typeof options.render === "function") return options.render;
  return null;
}

// Normalize one renderer result into transcript lines (upstream MessageRenderer
// returns a Component — adapted: an array of strings, a multi-line string, or
// a component-like object with render(width) are all accepted).
async function rendererLines(result, renderOptions) {
  if (result == null) return null;
  if (typeof result === "string") return result.split("\n");
  if (Array.isArray(result)) return result.map((line) => String(line));
  if (result && typeof result.render === "function") {
    const lines = await result.render(renderOptions.width ?? 80);
    if (lines == null) return null;
    return (Array.isArray(lines) ? lines : [lines]).map((line) => String(line));
  }
  return null;
}

function rendererTheme() {
  return parseHostJson(__pir_host.getThemeJson()) ?? {};
}

// Upstream `ReadonlyFooterDataProvider` (footer-data-provider.ts:385-387):
// the data not otherwise accessible to extensions — `getGitBranch`,
// `getExtensionStatuses`, `getAvailableProviderCount`, `onBranchChange`.
// The TUI owns the git-branch watcher and the extension status line, so it
// pushes a fresh snapshot with every footer render request; this object keeps
// the last snapshot live for the footer component (the same object is handed
// to the factory once, upstream semantics) and fires `onBranchChange`
// listeners when the pushed branch differs from the previous one.
function createFooterData() {
  const live = {
    branch: null,
    statuses: new Map(),
    providerCount: 0,
    listeners: new Set(),
  };
  const api = {
    getGitBranch: () => live.branch,
    getExtensionStatuses: () => live.statuses,
    getAvailableProviderCount: () => live.providerCount,
    onBranchChange: (callback) => {
      if (typeof callback !== "function") {
        throw new TypeError("footerData.onBranchChange requires a callback");
      }
      live.listeners.add(callback);
      return () => { live.listeners.delete(callback); };
    },
  };
  const apply = (data) => {
    if (!data || typeof data !== "object") return;
    const nextBranch = typeof data.gitBranch === "string" ? data.gitBranch : null;
    const branchChanged = nextBranch !== live.branch;
    live.branch = nextBranch;
    const statuses = data.extensionStatuses;
    live.statuses.clear();
    if (statuses && typeof statuses === "object") {
      for (const [key, value] of Object.entries(statuses)) {
        if (value != null) live.statuses.set(String(key), String(value));
      }
    }
    if (typeof data.availableProviderCount === "number") {
      live.providerCount = data.availableProviderCount;
    }
    if (branchChanged) {
      // Upstream notifies branch-change listeners and the footer component
      // re-renders on the following frame (a listener usually calls
      // `tui.requestRender()`); listener errors are isolated.
      for (const callback of Array.from(live.listeners)) {
        try { callback(); } catch { /* listener errors are isolated */ }
      }
    }
  };
  return { api, apply };
}

// Upstream `ctx.ui.theme` (a Theme instance, theme.ts): widget component
// factories (rpiv-todo's overlay) and custom dialogs style through
// `theme.fg(color, text)` / `theme.bold` / `theme.strikethrough`. Built from
// the host theme snapshot's color map (hex → truecolor ANSI); unknown tokens
// pass the text through unchanged (upstream throws on unknown tokens — pir
// degrades so a mis-keyed widget still renders its text). Fresh snapshot per
// call, so a live `ctx.ui.setTheme` is picked up on the next render.
function pirWidgetTheme() {
  const snapshot = parseHostJson(__pir_host.getThemeJson());
  const colors = snapshot && typeof snapshot === "object"
    ? (snapshot.colors && typeof snapshot.colors === "object" ? snapshot.colors : snapshot)
    : {};
  const ansi = {};
  for (const token of Object.keys(colors ?? {})) {
    const value = colors[token];
    if (typeof value !== "string") continue;
    const match = /^#([0-9a-fA-F]{6})$/.exec(value);
    if (!match) continue;
    const int = parseInt(match[1], 16);
    ansi[token] = `\u001b[38;2;${(int >> 16) & 255};${(int >> 8) & 255};${int & 255}m`;
  }
  return {
    // Upstream Theme carries its resolved name (theme.ts); the snapshot is
    // the authoritative current-theme name (the live swap is applied by the
    // TUI host, so the snapshot IS the current theme).
    name: snapshot && typeof snapshot === "object" ? snapshot.name : undefined,
    fg: (color, text) => {
      const prefix = ansi[color];
      return prefix ? prefix + String(text) + "\u001b[39m" : String(text);
    },
    bg: (color, text) => {
      const prefix = ansi[color];
      return prefix ? prefix + String(text) + "\u001b[49m" : String(text);
    },
    bold: (text) => "\u001b[1m" + String(text) + "\u001b[22m",
    italic: (text) => "\u001b[3m" + String(text) + "\u001b[23m",
    underline: (text) => "\u001b[4m" + String(text) + "\u001b[24m",
    inverse: (text) => "\u001b[7m" + String(text) + "\u001b[27m",
    strikethrough: (text) => "\u001b[9m" + String(text) + "\u001b[29m",
  };
}

// One-shot warning for a genuinely unsupported `ctx.ui.custom` shape. Keyed by
// the exact message so different gaps each report once; a supported overlay
// never calls this (the old "custom is unsupported" latch is gone).
function warnCustomOverlayUnsupported(message) {
  if (state.customOverlayWarnings[message]) return;
  state.customOverlayWarnings[message] = true;
  try { console.warn(message); } catch { /* the warning is best-effort */ }
}

// Human-readable shape of whatever a custom factory returned, for the
// unsupported-component message (an object with no `render` is the common
// case; `null` and primitives are caller bugs).
function describeOverlayValue(value) {
  if (value === null) return "null";
  if (Array.isArray(value)) return "an array";
  return typeof value === "object" ? "an object without render(width)" : typeof value;
}

// The `keybindings` argument upstream hands a custom factory. pir does not
// expose a JS KeybindingsManager; components that match the raw input chunk
// (the pi-tui `matchesKey(data, …)` pattern, which every upstream example
// uses) are unaffected. A component that calls `matches` gets `false` and an
// honest one-shot warning instead of a silent wrong answer.
function customOverlayKeybindings() {
  return {
    matches: () => {
      warnCustomOverlayUnsupported(
        "ctx.ui.custom: keybindings.matches is not supported in pir; match the raw input chunk (handleInput(data)) instead",
      );
      return false;
    },
    getKeys: () => [],
    getBinding: () => undefined,
  };
}

globalThis.__pir_callRenderer = async (requestJson) => {
  let request;
  try {
    request = JSON.parse(requestJson);
  } catch {
    return JSON.stringify({ ok: false, error: "invalid renderer request" });
  }
  try {
    const theme = rendererTheme();
    const renderOptions = request.options ?? {};

    // A registered tool's own renderers (upstream ToolDefinition.renderCall /
    // renderResult). The TUI's tool block asks through this kind; the reply is
    // the same `{ok, lines, text}` envelope every renderer path uses.
    if (request.kind === "toolDefinition") {
      return await __pir_callToolRenderer(
        String(request.key ?? ""),
        String(request.renderKind ?? "call"),
        JSON.stringify(request.payload ?? {}),
      );
    }

    // Footer / header component factories (`ctx.ui.setFooter` / `setHeader`):
    // one slot each (last registration wins, `undefined` restores the built-in
    // component). The factory is created ONCE and every refresh re-invokes
    // `component.render(width)`; the TUI renders the returned lines as its
    // bottom bar (footer) / top bar (header). The footer additionally receives
    // the live `ReadonlyFooterDataProvider` built from the snapshot the TUI
    // pushes with each render request.
    if (request.kind === "footerRenderers" || request.kind === "headerRenderers") {
      const isFooter = request.kind === "footerRenderers";
      const entry = isFooter ? state.footerFactory : state.headerFactory;
      if (!entry || typeof entry.factory !== "function") {
        return JSON.stringify({ ok: true, lines: null, text: null });
      }
      const width = Math.max(1, Math.floor(renderOptions.width ?? 80));
      if (isFooter) {
        entry.footerApply(renderOptions.footerData);
      }
      if (!entry.component) {
        const tui = {
          // Upstream TUI.requestRender(): the component's content changed; the
          // host repaints on the next tick. Rides the reserved status key the
          // TUI drain intercepts (never shown in the status line).
          requestRender: () => {
            try {
              // W10-D: carry the owning slot so the TUI dirties ONLY this
              // component, never every factory widget on the screen.
              __pir_host.setStatus(
                "__pir_request_render",
                JSON.stringify({ kind: isFooter ? "footer" : "header" }),
              );
            } catch { /* the repaint signal is best-effort */ }
          },
        };
        entry.component = isFooter
          ? entry.factory(tui, pirWidgetTheme(), entry.footerApi)
          : entry.factory(tui, pirWidgetTheme());
      }
      if (!entry.component || typeof entry.component.render !== "function") {
        return JSON.stringify({ ok: true, lines: null, text: null });
      }
      const lines = await entry.component.render(width);
      if (lines == null) return JSON.stringify({ ok: true, lines: null, text: null });
      return JSON.stringify({
        ok: true,
        lines: (Array.isArray(lines) ? lines : [lines]).map((line) => String(line)),
        text: null,
      });
    }

    if (request.kind === "customOverlays") {
      // `ctx.ui.custom`'s component (upstream `showExtensionCustom`): the
      // object stays in the JS registry and the TUI drives it through this
      // one bridge — `op: "render"` returns the frame lines, `op: "input"`
      // delivers the focused component's `handleInput(data)`, `op: "dispose"`
      // releases it. The same bounded round trip as the widget factories, so a
      // hung component degrades to the TUI's error row instead of the frame.
      // Its registry is `state.customOverlays`, NOT a `rendererMeta` bucket.
      const entry = state.customOverlays[request.key];
      if (!entry || entry.disposed) return JSON.stringify({ ok: true, lines: null, text: null });
      const op = request.payload?.op;
      if (op === "input") {
        if (typeof entry.component.handleInput === "function") {
          try { entry.component.handleInput(String(request.payload.data ?? "")); } catch { /* isolated */ }
        }
        return JSON.stringify({ ok: true, lines: null, text: null });
      }
      if (op === "dispose") {
        entry.disposed = true;
        try { entry.component.dispose?.(); } catch { /* dispose errors are isolated */ }
        return JSON.stringify({ ok: true, lines: null, text: null });
      }
      const width = Math.max(1, Math.floor(renderOptions.width ?? 80));
      // Host-reported geometry for `OverlayHandle.getBounds()` (floating
      // overlays only; the editor-slot path sends none).
      if (renderOptions.bounds && typeof renderOptions.bounds === "object") {
        entry.handleState.bounds = {
          row: Number(renderOptions.bounds.row),
          col: Number(renderOptions.bounds.col),
          width: Number(renderOptions.bounds.width),
          height: Number(renderOptions.bounds.height),
        };
      }
      const lines = await entry.component.render(width);
      if (lines == null) return JSON.stringify({ ok: true, lines: null, text: null });
      return JSON.stringify({
        ok: true,
        lines: (Array.isArray(lines) ? lines : [lines]).map((line) => String(line)),
        text: null,
      });
    }

    const bucket = state.rendererMeta[request.kind];
    if (!bucket) return JSON.stringify({ ok: true, lines: null, text: null });
    if (request.kind === "widgetRenderers") {
      // Widget component factories (`ctx.ui.setWidget(key, factory)`,
      // upstream types.ts:188-193 overload 2). The factory is created ONCE
      // per registration (upstream calls it once and keeps the component);
      // each refresh re-invokes `component.render(width)` and returns the
      // lines. rpiv-todo's overlay is the reference consumer: it captures
      // the `tui` stub (requestRender) and the theme, then renders its todo
      // panel per width. A factory/component failure rejects out of this
      // try-block into the shared `{ok:false,error}` reply — the TUI host
      // isolates the failure to that one widget (it renders nothing) and
      // the frame survives.
      const entry = bucket[request.key];
      if (!entry || typeof entry.options !== "function") {
        return JSON.stringify({ ok: true, lines: null, text: null });
      }
      if (!entry.component) {
        const tui = {
          // Upstream TUI.requestRender(): the widget's content changed; the
          // host repaints on the next tick. Rides the reserved status key
          // the TUI drain intercepts (never shown in the status line).
          requestRender: () => {
            try {
              // W10-D: carry the widget key so the TUI dirties ONLY this
              // widget (a 500 ms requestRender stream from one fleet-status
              // widget must not mark every factory slot dirty).
              __pir_host.setStatus(
                "__pir_request_render",
                JSON.stringify({ kind: "widget", key: String(request.key) }),
              );
            } catch { /* the repaint signal is best-effort */ }
          },
        };
        entry.component = entry.options(tui, pirWidgetTheme());
      }
      if (!entry.component || typeof entry.component.render !== "function") {
        return JSON.stringify({ ok: true, lines: null, text: null });
      }
      const width = Math.max(1, Math.floor(renderOptions.width ?? 80));
      const lines = await entry.component.render(width);
      if (lines == null) return JSON.stringify({ ok: true, lines: null, text: null });
      return JSON.stringify({
        ok: true,
        lines: (Array.isArray(lines) ? lines : [lines]).map((line) => String(line)),
        text: null,
      });
    }

    if (request.kind === "markdownTransformers") {
      // All registered transformers compose in registration order: each
      // output feeds the next (upstream getMarkdownTransformers returns one
      // transformer per extension in load order and applyMarkdownTransformers
      // chains them — a transformer that throws or returns a non-string keeps
      // the current markdown and the chain continues,
      // markdown-transform.ts:22-36).
      let markdown = request.payload?.markdown ?? "";
      const entries = Object.values(bucket).sort((a, b) => a.extId - b.extId);
      for (const entry of entries) {
        const fn = rendererFn(entry);
        if (!fn) continue;
        try {
          const out = await fn(markdown, renderOptions);
          if (typeof out === "string") markdown = out;
        } catch { /* keep the current markdown, continue (upstream) */ }
      }
      return JSON.stringify({ ok: true, lines: null, text: markdown });
    }

    if (request.kind === "toolRenderers") {
      // Resolver chain in extension load order (upstream registerToolRenderer
      // resolvers run in load order; pir keys registrations per tool name and
      // chains every registration for that name, sorted by extId with the
      // per-name registration order preserved within one extension).
      //
      // CONTRACT (upstream types.ts:678-681): the resolver is
      // `(toolName, next) => ToolRenderers | undefined` and `next()` is
      // SYNCHRONOUS — it calls the remaining resolvers and returns their
      // renderers directly. The chain below therefore resolves synchronously
      // (`resolve`) and only the materialized renderer OUTPUT may be a promise
      // (an async resolver's promise is awaited by this entry point, which is
      // async).
      // Wildcard (`*`) resolvers registered with the bare
      // `registerToolRenderer(resolver)` form apply to EVERY tool name
      // (upstream stores one resolver list, not a per-name map); per-name
      // registrations chain alongside them.
      const entries = Object.values(bucket)
        .flat()
        .filter((entry) => entry.toolName === request.key || entry.toolName === "*")
        .sort((a, b) => a.extId - b.extId);
      const resolve = (index) => {
        // Upstream `next()` returns "the renderers the remaining resolvers,
        // then the registered tool, would use" (types.ts:678-681): at the end
        // of the resolver chain the registered tool's own renderers answer.
        if (index >= entries.length) return registeredToolRenderers(request.key);
        const options = entries[index].options;
        if (typeof options === "function") {
          // Upstream resolver shape: (tool, next) => renderers | undefined.
          return options(request.payload, () => resolve(index + 1));
        }
        if (options && typeof options === "object") {
          const fn = options.render ?? options.renderResult ?? options.renderCall;
          return typeof fn === "function" ? fn(request.payload, renderOptions, theme) : undefined;
        }
        return undefined;
      };
      let resolved = resolve(0);
      if (resolved && typeof resolved.then === "function") resolved = await resolved;
      let rendered;
      // Upstream `ToolRenderers.renderShell` (types.ts:672): the resolved
      // renderer object may carry the shell choice the tool/host must paint
      // with. Legacy resolver shapes (a bare render fn, a string/array) carry
      // none, and the host keeps its bare-line rendering for them.
      let shell;
      if (resolved == null) {
        rendered = undefined;
      } else if (typeof resolved === "string" || Array.isArray(resolved)) {
        rendered = resolved;
      } else if (typeof resolved === "object") {
        if (resolved.renderShell === "self" || resolved.renderShell === "default") {
          shell = resolved.renderShell;
        }
        const fn = resolved.render ?? resolved.renderResult ?? resolved.renderCall;
        rendered = typeof fn === "function" ? fn(request.payload, renderOptions, theme) : undefined;
      }
      const lines = await rendererLines(rendered, renderOptions);
      return JSON.stringify({
        ok: true,
        lines,
        text: null,
        ...(shell !== undefined ? { shell } : {}),
      });
    }

    // messageRenderers / entryRenderers: keyed by customType, first-wins.
    const entry = bucket[request.key];
    if (!entry) return JSON.stringify({ ok: true, lines: null, text: null });
    const fn = rendererFn(entry);
    if (!fn) return JSON.stringify({ ok: true, lines: null, text: null });
    const lines = await rendererLines(
      await fn(request.payload, renderOptions, theme),
      renderOptions,
    );
    return JSON.stringify({ ok: true, lines, text: null });
  } catch (error) {
    const message = error && error.message ? error.message : String(error);
    return JSON.stringify({ ok: false, error: message });
  }
};

// Argument completions for a slash command (upstream
// RegisteredCommand.getArgumentCompletions; first extension in load order
// that registered the hook handles the command).
globalThis.__pir_callArgCompletions = async (commandName, prefixJson) => {
  let entry = null;
  for (const id of Object.keys(state.extensions).sort((a, b) => Number(a) - Number(b))) {
    const ext = state.extensions[id];
    const def = ext.status === "active" ? ext.commands[commandName] : undefined;
    if (def && typeof def.getArgumentCompletions === "function") { entry = def; break; }
  }
  if (!entry) return JSON.stringify({ ok: true, items: [] });
  try {
    const prefix = parseHostJson(prefixJson);
    const items = await entry.getArgumentCompletions(prefix ?? "");
    if (!Array.isArray(items)) return JSON.stringify({ ok: true, items: [] });
    return JSON.stringify({
      ok: true,
      items: items.map((item) => ({
        value: String(item?.value ?? ""),
        label: String(item?.label ?? item?.value ?? ""),
        description: item?.description == null ? null : String(item.description),
      })),
    });
  } catch (error) {
    const message = error && error.message ? error.message : String(error);
    return JSON.stringify({ ok: false, error: message });
  }
};

// ui.addAutocompleteProvider suggesters: first non-null item list wins

globalThis.__pir_callAutocompleteProviders = async (contextJson) => {

  } catch (error) {
    const message = error && error.message ? error.message : String(error);
    return JSON.stringify({ ok: false, error: message });
  }
};

// ---------------------------------------------------------------------------
// registerProvider OAuth bridge (upstream ProviderConfig.oauth, types.ts
// 1943-1958: extension-implemented login / refreshToken / getApiKey callbacks)
// ---------------------------------------------------------------------------
//
// The eval/eval_json channel cannot await, so the host runs each OAuth
// operation as a detached JS promise (`__pir_providerOauthStart`), then polls
// it to completion (`__pir_providerOauthPoll`); the runtime's resident drive
// loop keeps the op's pending continuations live between polls (the ops only
// ever await host fns such as `uiDialog`, which resolve from direct state
// writes).
// `__pir_providerOauthCancel` marks an op abandoned after the host-side
// deadline so the poll loop can give up deterministically; an abandoned op's
// eventual result is discarded (its remaining awaits are bounded by the
// dialog TTL, never a wedge).
//
// Ops:
// - "login": runs the stored `login(callbacks)` with JS-implemented
//   OAuthLoginCallbacks — onAuth/onDeviceCode/onProgress feed the op's event
//   log (the Rust login pump forwards them to the login UI's
//   AuthInteraction), onPrompt/onSelect/onManualCodeInput surface through the
//   existing uiDialog channel. Resolves with {credential, apiKey}.
// - "refresh": runs `refreshToken(credentials, signal)` then `getApiKey`.
// Returns `{ok: true, credential, apiKey}` or `{ok: false, error}`.

function providerOauthError(op, error) {
  const message = error && error.message ? error.message : String(error);
  return { ok: false, error: `provider oauth ${op}: ${message}` };
}

globalThis.__pir_providerOauthStart = (providerId, op, payloadJson, callId) => {
  try {
    const hook = state.providerOauthHooks[providerId];
    if (!hook) {
      state.providerOauthOps[callId] = {
        done: true,
        result: { ok: false, error: `provider ${providerId} has no oauth hooks registered` },
      };
      return "null";
    }
    const payload = parseHostJson(payloadJson) ?? {};
    const signal = {
      get aborted() { return isAborted(); },
      throwIfAborted() { if (isAborted()) throw new Error("The operation was aborted"); },
      addEventListener() {},
      removeEventListener() {},
    };
    const opState = { done: false, events: [], result: null };
    state.providerOauthOps[callId] = opState;
    (async () => {
      try {
        let result;
        if (op === "login") {
          if (typeof hook.login !== "function") throw new Error("oauth.login is not registered");
          const credential = await hook.login({
            onAuth: (info) => {
              opState.events.push({
                type: "auth_url",
                url: String(info?.url ?? ""),
                instructions: info?.instructions == null ? null : String(info.instructions),
              });
            },
            onDeviceCode: (info) => {
              opState.events.push({
                type: "device_code",
                userCode: String(info?.userCode ?? info?.user_code ?? ""),
                verificationUri: String(info?.verificationUri ?? info?.verification_uri ?? ""),
                intervalSeconds: info?.intervalSeconds ?? info?.interval_seconds ?? null,
                expiresInSeconds: info?.expiresIn ?? info?.expires_in ?? null,
              });
            },
            onProgress: (message) => {
              opState.events.push({ type: "progress", message: String(message ?? "") });
            },
            onPrompt: async (prompt) => {
              const out = await __pir_host.uiDialog(
                "input",
                JSON.stringify({
                  title: String(prompt?.message ?? "Input"),
                  placeholder: prompt?.placeholder == null ? null : String(prompt.placeholder),
                }),
              );
              return parseHostJson(out) ?? null;
            },
            onSelect: async (prompt) => {
              const options = (prompt?.options ?? []).map((option) =>
                typeof option === "string"
                  ? { value: option, label: option }
                  : { value: String(option?.id ?? option?.value ?? ""), label: String(option?.label ?? option?.value ?? "") });
              const out = await __pir_host.uiDialog(
                "select",
                JSON.stringify({ title: String(prompt?.message ?? "Select"), options }),
              );
              return parseHostJson(out) ?? null;
            },
            onManualCodeInput: async () => {
              const out = await __pir_host.uiDialog(
                "input",
                JSON.stringify({ title: "Paste the authorization code" }),
              );
              return parseHostJson(out) ?? null;
            },
            signal,
          });
          let apiKey = null;
          if (typeof hook.getApiKey === "function") {
            apiKey = await hook.getApiKey(credential);
          }
          result = { ok: true, credential, apiKey: apiKey == null ? null : String(apiKey) };
        } else if (op === "refresh") {
          if (typeof hook.refreshToken !== "function") {
            throw new Error("oauth.refreshToken is not registered");
          }
          const credential = payload.credential ?? null;
          const refreshed = await hook.refreshToken(credential, signal);
          let apiKey = null;
          if (typeof hook.getApiKey === "function") {
            apiKey = await hook.getApiKey(refreshed);
          }
          result = { ok: true, credential: refreshed, apiKey: apiKey == null ? null : String(apiKey) };
        } else {
          throw new Error(`unknown oauth op: ${op}`);
        }
        opState.result = result;
      } catch (error) {
        opState.result = providerOauthError(op, error);
      } finally {
        opState.done = true;
      }
    })();
    return "null";
  } catch (error) {
    state.providerOauthOps[callId] = { done: true, result: providerOauthError(op, error) };
    return "null";
  }
};

globalThis.__pir_providerOauthPoll = (callId) => {
  const op = state.providerOauthOps[callId];
  if (!op) return JSON.stringify({ done: true, result: { ok: false, error: "unknown oauth call id" } });
  const events = op.events.splice(0);
  if (!op.done) return JSON.stringify({ done: false, events });
  delete state.providerOauthOps[callId];
  return JSON.stringify({ done: true, events, result: op.result });
};

globalThis.__pir_providerOauthCancel = (callId) => {
  delete state.providerOauthOps[callId];
  return "null";
};

// Registry export for the Rust side (tools, commands, shortcuts, handler
// counts used by the fast-path `has_handlers` check)
// ---------------------------------------------------------------------------

function handlerCounts() {
  const counts = {};
  for (const ext of Object.values(state.extensions)) {
    if (ext.status !== "active") continue;
    for (const [event, handlers] of Object.entries(ext.handlers)) {
      if (handlers.length > 0) counts[event] = (counts[event] ?? 0) + handlers.length;
    }
  }
  // `ctx.ui.onTerminalInput` handlers live outside the per-extension event
  // tables (upstream keeps them as TUI input subscriptions), so they are not
  // in the loop above. Expose the same count under the bridge event name so
  // the host's `has_handlers("terminal_input")` answers correctly.
  if (state.terminalInputHandlers.length > 0) {
    counts.terminal_input = (counts.terminal_input ?? 0) + state.terminalInputHandlers.length;
  }
  return counts;
}

// Targeted publication of the same counts for the Rust fast-path snapshot
// (`RuntimeState::handler_counts`). The JS-execution entry points that are NOT
// event/command/shortcut dispatch — a tool `execute` body, the renderer and
// autocomplete callbacks — can register handlers through `pi.on(...)` too, and
// the host must see them before it next consults `has_handlers` (the gate
// skips an emission whose event looks unobserved). Those points refresh
// through this call instead of the full `__pir_exportRegistryJson` (which
// serializes every tool schema, command, shortcut and renderer), so the
// publication stays proportional to what the check actually reads. Semantics
// are identical: the counts are computed by the same `handlerCounts()`.
globalThis.__pir_handlerCountsJson = () => JSON.stringify(handlerCounts());

// The registry-export snapshot of one tool's `reachesTools` hook (upstream
// ToolDefinition.reachesTools). The export is a SYNCHRONOUS channel and the
// hook is a function, so the export probes it with the same empty input the
// Rust `ensureDiscoveryActive` probe uses (`reachesTools({})`) and serializes
// the result: `"all"`, the returned word list, or `true` when the hook
// declared a reach the probe could not serialize (it threw, or returned
// another type). The LIVE per-call value stays inside QuickJS and is pulled by
// the host through `__pir_callToolReach` at the tool-call wait seam.
function reachSnapshot(name) {
  const hook = state.toolReachHooks[name];
  if (typeof hook !== "function") return undefined;
  try {
    const reach = hook({});
    if (reach === "all") return "all";
    if (Array.isArray(reach)) return reachWords(reach);
  } catch {
    // A hook that rejects the probe input still DECLARES a reach: upstream
    // reads the hook's presence (`tool.reachesTools`) for activations.
  }
  return true;
}

// The word list of a `reachesTools` answer (upstream `readonly string[]`): the
// strings it contains. A non-string entry is not a name a namespace could
// match and is dropped (upstream would throw on it inside `word.includes`).
function reachWords(words) {
  return words.filter((word) => typeof word === "string");
}

// One `state.toolMeta` entry with the reach snapshot injected under the same
// `reachesTools` key the metadata already uses for the presence marker (the
// host parses it as the snapshot; `pi.getAllTools()` re-attaches the live
// function by name from `state.toolReachHooks`). The stored metadata object is
// never mutated.
function exportToolMeta(meta) {
  const snapshot = reachSnapshot(meta.name);
  return snapshot === undefined ? meta : { ...meta, reachesTools: snapshot };
}

globalThis.__pir_exportRegistryJson = () => JSON.stringify({
  tools: Object.values(state.toolMeta).map(exportToolMeta),
  // Per-tool wire extras keyed by name: the `reachesTools` PRESENCE marker
  // (upstream ToolDefinition.reachesTools; the `tools` entries above carry the
  // probe VALUE instead) and the raw `namespace` object with its `aliases`
  // (upstream ToolNamespace.aliases). Only tools that carry something are
  // listed; the host merges them back into `getAllToolsJson` output, where
  // `pi.getAllTools()` re-attaches the live reach function by name.
  toolExtras: Object.fromEntries(
    Object.entries(state.toolMeta)
      .map(([name, meta]) => [
        name,
        {
          reachesTools: meta.reachesTools === true,
          namespace: meta.namespace ?? null,
          // Declarative ToolDefinition fields the typed Rust `ToolDef` does
          // not model (upstream types.ts:590-667): host consumers read them
          // synchronously from this extras map (the TUI's tool-block shell
          // choice, constrained-sampling advertisement).
          ...(meta.renderShell !== undefined ? { renderShell: meta.renderShell } : {}),
          ...(meta.constrainedSampling !== undefined
            ? { constrainedSampling: meta.constrainedSampling }
            : {}),
          renderCall: meta.renderCall === true,
          renderResult: meta.renderResult === true,
        },
      ])
      .filter(
        ([, extra]) =>
          extra.reachesTools === true ||
          extra.namespace !== null ||
          extra.renderShell !== undefined ||
          extra.constrainedSampling !== undefined ||
          extra.renderCall === true ||
          extra.renderResult === true,
      ),
  ),
  commands: Object.values(state.commandMeta).flat(),
  shortcuts: Object.values(state.shortcutMeta),
  handlers: handlerCounts(),
  renderers: {
    messageRenderers: Object.values(state.rendererMeta.messageRenderers),
    entryRenderers: Object.values(state.rendererMeta.entryRenderers),
    markdownTransformers: Object.values(state.rendererMeta.markdownTransformers),
    // toolRenderers buckets hold per-name resolver chains; flatten so the
    // exported defs carry one flat entry list (the TUI registry check reads
    // `toolName` off each entry).
    toolRenderers: Object.values(state.rendererMeta.toolRenderers).flat(),
    // Widget component factories (key -> {key, placement}); the TUI registry
    // check reads `customType` off each entry.
    widgetRenderers: Object.values(state.rendererMeta.widgetRenderers),
    // Single-slot bar factories. Exported so the host can distinguish "the
    // JS face still owns the slot" from "no JS registration" when a native
    // `ctx.ui.setFooter/setHeader` registration is removed.
    footerRenderers: state.footerFactory ? [{ key: "__footer" }] : [],
    headerRenderers: state.headerFactory ? [{ key: "__header" }] : [],
  },
});

// The reach of ONE call for a JS tool (upstream the mcp `tool_call` handler's
// `const reach = tool.reachesTools?.(event.input)`): the host asks for it at
// the tool-call wait seam, so the MCP wait resolves the same value for JS and
// Rust tools. Returns a JSON string — `"all"`, the word array, `null` when the
// tool declares no reach (or the hook returned undefined), or `{error}` when
// the hook threw or answered with a Promise. Synchronous by contract: the host
// reads it through the runtime's synchronous eval channel (the upstream hook
// type is synchronous; a Promise answer falls back to the export snapshot).
globalThis.__pir_callToolReach = (toolName, inputJson) => {
  const hook = state.toolReachHooks[toolName];
  if (typeof hook !== "function") return JSON.stringify(null);
  let input;
  try { input = JSON.parse(inputJson); } catch { input = {}; }
  try {
    const reach = hook(input);
    if (reach && typeof reach.then === "function") {
      return JSON.stringify({ error: "reachesTools hook returned a Promise" });
    }
    if (reach === "all") return JSON.stringify("all");
    if (Array.isArray(reach)) return JSON.stringify(reachWords(reach));
    return JSON.stringify(null);
  } catch (error) {
    return JSON.stringify({ error: error && error.message ? error.message : String(error) });
  }
};

// Run a tool's `prepareLoadout` hook (upstream ToolDefinition.prepareLoadout):
// `loadoutJson` is the ToolLoadout wire shape; a hook that returns an object
// contributes {descriptions?, hiddenDeclarations?}. Returns "null" when the
// tool has no hook (or the hook returned undefined) and an error object when
// the hook throws.
globalThis.__pir_callPrepareLoadout = async (toolName, loadoutJson) => {
  // Reserved host bridge channels (extension-owned stream/route callbacks):
  // the Rust side reuses this actor command with a reserved key prefix. Real
  // tool names are validated identifiers and can never start with the prefix.
  if (typeof toolName === "string" && toolName.startsWith("__pir_stream:")) {
    return __pir_dispatchProviderStream(toolName.slice("__pir_stream:".length), loadoutJson);
  }
  if (typeof toolName === "string" && toolName.startsWith("__pir_route:")) {
    return __pir_dispatchVirtualRoute(toolName.slice("__pir_route:".length), loadoutJson);
  }
  const hook = state.toolLoadoutHooks[toolName];
  if (typeof hook !== "function") return JSON.stringify(null);
  try {
    const loadout = JSON.parse(loadoutJson);
    // Awaits BOTH sync and async hooks (upstream tools.ts awaits
    // prepareLoadout): serializing an un-awaited async hook would stringify
    // the Promise object as "{}" and defeat the per-call deadline, since the
    // host-side round trip would finish before the hook's real work.
    const changes = await hook(loadout);
    return JSON.stringify(changes ?? null);
  } catch (error) {
    return JSON.stringify({ error: error && error.message ? error.message : String(error) });
  }
};

// streamSimple dispatch (upstream ProviderConfig.streamSimple): run the
// provider's JS callback with (model, context, options) and serialize the
// produced assistant events. Upstream returns a live
// AssistantMessageEventStream; this slice collects the event array the
// callback returns (or resolves to). Errors are reported as `{error}` so the
// host terminates the request with a typed error event (fail-closed).
globalThis.__pir_dispatchProviderStream = async (providerId, requestJson) => {
  if (__pir_host.isInvalidated()) {
    return JSON.stringify({ error: "The extension runtime was invalidated." });
  }
  const hook = state.providerStreamHooks[providerId];
  if (!hook || typeof hook.fn !== "function") {
    return JSON.stringify({
      error: `streamSimple callback for provider ${providerId} is not registered`,
    });
  }
  let request;
  try {
    request = JSON.parse(requestJson);
  } catch (error) {
    return JSON.stringify({
      error: `invalid streamSimple request: ${error && error.message ? error.message : String(error)}`,
    });
  }
  try {
    const result = await hook.fn(
      request.model,
      {
        messages: request.messages ?? [],
        systemPrompt: request.systemPrompt,
        tools: request.tools,
      },
      request.options ?? {},
    );
    if (Array.isArray(result)) return JSON.stringify({ events: result });
    if (result && Array.isArray(result.events)) return JSON.stringify({ events: result.events });
    return JSON.stringify({
      error:
        "streamSimple callback must return an array of assistant events (or {events: [...]})",
    });
  } catch (error) {
    return JSON.stringify({ error: error && error.message ? error.message : String(error) });
  }
};

// Virtual-model route dispatch (upstream VirtualModelDefinition.route): run
// the registered route(request) callback and serialize the decision. Errors
// are reported as `{error}`; the host falls back to the static route.
globalThis.__pir_dispatchVirtualRoute = async (key, requestJson) => {
  if (__pir_host.isInvalidated()) {
    return JSON.stringify({ error: "The extension runtime was invalidated." });
  }
  const hook = state.virtualRouteHooks[key];
  if (!hook || typeof hook.fn !== "function") {
    return JSON.stringify({
      error: `route callback for virtual model ${key} is not registered`,
    });
  }
  let request;
  try {
    request = JSON.parse(requestJson);
  } catch (error) {
    return JSON.stringify({
      error: `invalid route request: ${error && error.message ? error.message : String(error)}`,
    });
  }
  try {
    const decision = await hook.fn(request);
    if (
      !decision || typeof decision !== "object" || !decision.model
      || typeof decision.model.provider !== "string" || !decision.model.provider
      || typeof decision.model.id !== "string" || !decision.model.id
    ) {
      return JSON.stringify({
        error: `route(${key}) must return {model: {provider, id}}`,
      });
    }
    const route = { physical: { provider: decision.model.provider, id: decision.model.id } };
    if (decision.thinkingLevel !== undefined) route.thinkingLevel = decision.thinkingLevel;
    if (decision.state !== undefined) route.state = decision.state;
    return JSON.stringify(route);
  } catch (error) {
    return JSON.stringify({ error: error && error.message ? error.message : String(error) });
  }
};

// ---------------------------------------------------------------------------
// Incremental streamSimple streaming (upstream live AssistantMessageEventStream).
// The Rust side starts a session (__pir_startProviderStream) and then pulls one
// event at a time (__pir_streamNext). The legacy batch forms — an event array
// or {events: [...]} — are answered whole in the start reply (byte-frozen
// compat); an async iterable (upstream AssistantMessageEventStream is one, as
// are async generators) or a raw {next()} iterator starts a pull session.
// Errors are reported as `{error}` (fail-closed); a host abort is reported as
// `{aborted: true}`.
// ---------------------------------------------------------------------------

function __pir_isEventStream(value) {
  if (!value || typeof value !== "object") return false;
  if (typeof value[Symbol.asyncIterator] === "function") return true;
  return typeof value.next === "function";
}

globalThis.__pir_startProviderStream = async (hookKey, requestJson) => {
  if (__pir_host.isInvalidated()) {
    return JSON.stringify({ error: "The extension runtime was invalidated." });
  }
  // The Rust side passes the full reserved hook key (`__pir_stream:<id>`),
  // same convention as the legacy `__pir_callPrepareLoadout` channel — strip
  // the prefix to reach the plain provider id the callback was registered
  // under.
  const providerId =
    typeof hookKey === "string" && hookKey.startsWith("__pir_stream:")
      ? hookKey.slice("__pir_stream:".length)
      : hookKey;
  const hook = state.providerStreamHooks[providerId];
  if (!hook || typeof hook.fn !== "function") {
    return JSON.stringify({
      error: `streamSimple callback for provider ${providerId} is not registered`,
    });
  }
  let request;
  try {
    request = JSON.parse(requestJson);
  } catch (error) {
    return JSON.stringify({
      error: `invalid streamSimple request: ${error && error.message ? error.message : String(error)}`,
    });
  }
  let result;
  try {
    let options = request.options ?? {};
    // streamSimple options hooks (upstream `SimpleStreamOptions.onPayload` /
    // `onResponse` / `onProviderStreamEvent`): when the host bound a hook
    // session the options gain callable wrappers firing the Rust-side
    // observers. Fire-and-forget: wrapper throws/serialization failures are
    // isolated and never interrupt the stream.
    if (typeof options.hookSession === "string" && options.hookSession) {
      const hookSession = options.hookSession;
      const emitHook = (kind, value) => {
        try {
          __pir_host.streamHookEvent(kind, hookSession, JSON.stringify(value === undefined ? null : value));
        } catch { /* isolated: a hook failure never breaks the stream */ }
      };
      options = {
        ...options,
        onPayload: (payload) => emitHook("payload", payload),
        onResponse: (status, headers) => emitHook("response", { status: status ?? 0, headers: headers ?? {} }),
        onProviderStreamEvent: (event) => emitHook("event", event),
      };
    }
    result = await hook.fn(
      request.model,
      {
        messages: request.messages ?? [],
        systemPrompt: request.systemPrompt,
        tools: request.tools,
      },
      options,
    );
  } catch (error) {
    return JSON.stringify({ error: error && error.message ? error.message : String(error) });
  }
  // Legacy batch forms (compat): the whole event array at once.
  if (Array.isArray(result)) return JSON.stringify({ mode: "batch", events: result });
  if (result && Array.isArray(result.events)) {
    return JSON.stringify({ mode: "batch", events: result.events });
  }
  // Incremental forms: an async iterable (upstream AssistantMessageEventStream,
  // async generators) or a raw async iterator ({next()}).
  let stream = result;
  if (result && !__pir_isEventStream(result) && result.events !== undefined) {
    stream = result.events;
  }
  if (__pir_isEventStream(stream)) {
    let iterator;
    try {
      iterator =
        typeof stream[Symbol.asyncIterator] === "function"
          ? stream[Symbol.asyncIterator]()
          : stream;
    } catch (error) {
      return JSON.stringify({
        error: `streamSimple iterator creation failed: ${error && error.message ? error.message : String(error)}`,
      });
    }
    const id = String(++state.nextProviderStreamId);
    state.providerStreamSessions[id] = { iterator };
    return JSON.stringify({ mode: "stream", id });
  }
  return JSON.stringify({
    error:
      "streamSimple callback must return an array of assistant events, {events: [...]}, or an async iterable of events",
  });
};

globalThis.__pir_streamNext = async (streamId) => {
  const session = state.providerStreamSessions[streamId];
  if (!session) {
    return JSON.stringify({ error: `provider stream session ${streamId} is not active` });
  }
  if (isAborted()) {
    delete state.providerStreamSessions[streamId];
    // Best-effort close (same as __pir_streamCancel): async generators run
    // their finally blocks via return(), releasing JS-provider resources
    // deterministically instead of waiting for GC.
    try {
      if (session.iterator && typeof session.iterator.return === "function") {
        Promise.resolve(session.iterator.return()).catch(() => {});
      }
    } catch { /* best-effort close */ }
    return JSON.stringify({ aborted: true });
  }
  try {
    const step = await session.iterator.next();
    if (step.done) {
      delete state.providerStreamSessions[streamId];
      return JSON.stringify({ done: true });
    }
    return JSON.stringify({ event: step.value });
  } catch (error) {
    delete state.providerStreamSessions[streamId];
    return JSON.stringify({ error: error && error.message ? error.message : String(error) });
  }
};

globalThis.__pir_streamCancel = (streamId) => {
  const session = state.providerStreamSessions[streamId];
  if (!session) return "ok";
  delete state.providerStreamSessions[streamId];
  // Best-effort close: async generators run their finally blocks via return().
  try {
    if (session.iterator && typeof session.iterator.return === "function") {
      Promise.resolve(session.iterator.return()).catch(() => {});
    }
  } catch { /* best-effort close */ }
  return "ok";
};

// ---------------------------------------------------------------------------
// Timer surface (Node parity): upstream extensions run in Node, where
// setTimeout/setInterval/clearTimeout/clearInterval are process globals. pir
// embeds QuickJS on the extension actor thread, so the timers ride the host
// sleep primitive (`__pir_host.sleep`) on the actor's tokio executor, whose
// resident drive loop drives suspended JS continuations whether or not a
// command is in flight. Awaited timers (the hook pattern:
// `new Promise(r => setTimeout(r, ms))`) resolve in-flight under the pending
// call, and a fire-and-forget timer fires on its own when its host sleep
// resolves — no command and no post-command job drain is needed (that sweep
// only covers continuations left queued as a resolving command task
// finishes).
// ---------------------------------------------------------------------------
let __pirTimerSeq = 0;
const __pirTimers = new Map();
const __pirTimerArm = (fn, ms, args, repeat) => {
  const id = ++__pirTimerSeq;
  const entry = { cancelled: false };
  __pirTimers.set(id, entry);
  (async () => {
    let delay = Number(ms);
    if (!Number.isFinite(delay) || delay < 0) delay = 0;
    do {
      try {
        await __pir_host.sleep(delay);
      } catch {
        break; // host sleep failed: the timer ends
      }
      if (entry.cancelled) break;
      if (typeof fn === "function") {
        try {
          fn(...args);
        } catch (error) {
          // Node delivers a throwing timer callback to `uncaughtException`
          // (and exits when nothing listens). Swallowing it here hid real
          // extension errors — same routing as the nextTick drain.
          const proc = globalThis.process;
          if (
            proc && typeof proc.listenerCount === "function" &&
            proc.listenerCount("uncaughtException") > 0
          ) {
            proc.emit("uncaughtException", error, "uncaughtException");
          } else {
            throw error; // rejects this closure -> host fatal path
          }
        }
      }
    } while (repeat);
    __pirTimers.delete(id);
  })();
  return id;
};
globalThis.setTimeout = (fn, ms, ...args) => __pirTimerArm(fn, ms, args, false);
globalThis.setInterval = (fn, ms, ...args) => __pirTimerArm(fn, ms, args, true);
globalThis.clearTimeout = (id) => {
  const entry = __pirTimers.get(id);
  if (entry) { entry.cancelled = true; __pirTimers.delete(id); }
};
globalThis.clearInterval = globalThis.clearTimeout;
// ---------------------------------------------------------------------------
// Error.captureStackTrace + CallSites (V8-style error metadata).
// quickjs-ng ships a V8-shaped Error.captureStackTrace with real structured
// CallSite objects, but its CallSite lacks several V8 methods — depd (the
// express/body-parser/http-errors dependency) calls `isEval()` and
// `getEvalOrigin()` unconditionally on every frame and dies with
// "not a function" without them. Reach the engine's CallSite prototype
// through a throwaway capture and fill in the missing V8 surface. When the
// engine has no captureStackTrace at all (other embedders), a minimal
// string-parsing implementation takes over below.
// ---------------------------------------------------------------------------
(() => {
  if (typeof Error !== "function") return;
  if (typeof Error.captureStackTrace === "function") {
    try {
      const prev = Error.prepareStackTrace;
      Error.prepareStackTrace = (_error, sites) => sites;
      const probe = {};
      Error.captureStackTrace(probe);
      Error.prepareStackTrace = prev;
      const first = probe.stack && probe.stack.length > 0 ? probe.stack[0] : null;
      const proto = first ? Object.getPrototypeOf(first) : null;
      if (proto && typeof proto.isEval !== "function") {
        proto.isEval = function () { return false; };
        proto.getEvalOrigin = function () { return undefined; };
        proto.getMethodName = function () { return null; };
        proto.getTypeName = function () { return null; };
        proto.isToplevel = function () {
          const name = this.getFunctionName();
          return name === null || name === undefined;
        };
        proto.toString = function () {
          const file = this.getFileName() || "<anonymous>";
          const line = this.getLineNumber();
          const column = this.getColumnNumber();
          const name = this.getFunctionName();
          if (line === null || line === undefined) {
            return name ? name + " (" + file + ")" : file;
          }
          const location =
            file + ":" + line + (column === null || column === undefined ? "" : ":" + column);
          return name ? name + " (" + location + ")" : location;
        };
      }
    } catch {
      /* augmentation is best-effort: depd-class consumers degrade on their own */
    }
    return;
  }

  const parseCallSite = (line) => {
    const match = /^\s*at\s+(.+?)\s*$/.exec(line);
    if (!match) return null;
    let rest = match[1];
    let fn = null;
    const wrapped = /^(.*?)\s*\((.*)\)$/.exec(rest);
    let location = rest;
    if (wrapped && !/:\d+(?::\d+)?$/.test(wrapped[1])) {
      fn = wrapped[1] === "" || wrapped[1] === "<anonymous>" ? null : wrapped[1];
      location = wrapped[2];
    } else if (rest === "<anonymous>") {
      location = "";
    }
    let file = null;
    let lineNo = null;
    let colNo = null;
    if (location) {
      const parts = /^(.*):(\d+)(?::(\d+))?$/.exec(location);
      if (parts) {
        file = parts[1] === "" ? null : parts[1];
        lineNo = Number(parts[2]);
        colNo = parts[3] !== undefined ? Number(parts[3]) : null;
      } else {
        file = location;
      }
    }
    return { fn, file, lineNo, colNo, native: /\[native code\]/.test(rest) };
  };

  const makeCallSite = (site) => ({
    getFunctionName: () => site.fn,
    getMethodName: () => null,
    getTypeName: () => null,
    getFileName: () => site.file,
    getLineNumber: () => site.lineNo,
    getColumnNumber: () => site.colNo,
    isNative: () => site.native,
    isEval: () => false,
    isConstructor: () => false,
    isToplevel: () => site.fn === null,
    toString: () =>
      (site.fn ? site.fn + " " : "") +
      (site.file === null && site.lineNo === null
        ? "<anonymous>"
        : "(" + (site.file ?? "") + (site.lineNo !== null ? ":" + site.lineNo : "") + (site.colNo !== null ? ":" + site.colNo : "") + ")"),
  });

  Error.captureStackTrace = function captureStackTrace(target, constructorOpt) {
    const limit = typeof Error.stackTraceLimit === "number" ? Math.max(0, Math.trunc(Error.stackTraceLimit)) : 10;
    const raw = new Error().stack || "";
    const frames = [];
    for (const line of raw.split("\n").slice(1)) {
      const site = parseCallSite(line);
      if (site) frames.push(site);
    }
    // Drop this helper's own frame(s), then — when a constructorOpt was
    // given — everything above it plus its own frame (Node semantics: the
    // stack starts just above the capture point's public boundary).
    let start = 0;
    while (start < frames.length && frames[start].fn === "captureStackTrace") start += 1;
    if (typeof constructorOpt === "function") {
      const name = constructorOpt.name;
      while (start < frames.length && frames[start].fn !== name) start += 1;
      if (start < frames.length) start += 1;
    }
    const sites = frames.slice(start, start + limit).map(makeCallSite);
    if (typeof Error.prepareStackTrace === "function") {
      target.stack = Error.prepareStackTrace(target, sites);
      return;
    }
    const header = typeof target.name === "string" && target.name !== "" ? target.name : "Error";
    const body = sites.map((site) => "    at " + site.toString()).join("\n");
    target.stack = body === "" ? header : header + "\n" + body;
  };
})();

// ---------------------------------------------------------------------------
// `error.stack` header: node/V8 put "Name: message" on the first line.
// quickjs-ng builds the frame list eagerly and stores it header-less, both in
// the Error [[ErrorData]] slot (read by the prototype accessor) and in the own
// "stack" property that `Error.captureStackTrace` installs on its target; the
// engine exposes no hook for the header, so the prototype accessor is wrapped
// here and the static is wrapped to re-header the own property it just wrote.
// V8 rules reproduced: `name` (undefined -> "Error") + ": " + `message`, with
// an empty part dropped, and a `prepareStackTrace` result never re-formatted
// (it owns the shape, arrays included).
// ---------------------------------------------------------------------------
(() => {
  if (typeof Error !== "function") return;
  const desc = Object.getOwnPropertyDescriptor(Error.prototype, "stack");
  if (!desc || typeof desc.get !== "function") return;
  const rawGet = desc.get;
  const rawSet = typeof desc.set === "function" ? desc.set : null;

  const headerFor = (target) => {
    let name = target.name;
    name = name === undefined ? "Error" : String(name);
    let message = target.message;
    message = message === undefined ? "" : String(message);
    if (name === "") return message;
    if (message === "") return name;
    return name + ": " + message;
  };
  // The engine's frame lists always start with the "    at " frame prefix (or
  // are empty when Error.stackTraceLimit is 0); anything else is a
  // prepareStackTrace-shaped string and is handed back verbatim.
  const withHeader = (target, raw) => {
    if (typeof raw !== "string") return raw;
    if (typeof Error.prepareStackTrace === "function") return raw;
    if (raw !== "" && raw.indexOf("    at ") !== 0) return raw;
    return raw === "" ? headerFor(target) : headerFor(target) + "\n" + raw;
  };

  Object.defineProperty(Error.prototype, "stack", {
    configurable: true,
    enumerable: desc.enumerable === true,
    get() { return withHeader(this, rawGet.call(this)); },
    set(value) { if (rawSet) rawSet.call(this, value); },
  });

  const rawCapture = Error.captureStackTrace;
  if (typeof rawCapture === "function") {
    // Declared with two parameters so `captureStackTrace.length === 2`, as in
    // node. `Error.captureStackTrace(target, ctorOpt)` writes the frame list
    // into an own "stack" property on the target (rawCapture above), which
    // shadows the prototype accessor — so the header has to be applied here.
    const captureStackTrace = function captureStackTrace(target, constructorOpt) {
      const result = rawCapture.call(Error, target, constructorOpt);
      if (target !== null && (typeof target === "object" || typeof target === "function")) {
        const own = Object.getOwnPropertyDescriptor(target, "stack");
        if (own !== undefined && typeof own.value === "string") {
          const formatted = withHeader(target, own.value);
          if (formatted !== own.value) {
            Object.defineProperty(target, "stack", {
              value: formatted,
              writable: true,
              enumerable: own.enumerable === true,
              configurable: true,
            });
          }
        }
      }
      return result;
    };
    Object.defineProperty(Error, "captureStackTrace", {
      value: captureStackTrace,
      writable: true,
      enumerable: false,
      configurable: true,
    });
  }
})();

// Real Node's `globalThis` sits behind a plain intermediate object: its
// [[Prototype]] is not Object.prototype directly
// (`Object.getPrototypeOf(globalThis) === Object.prototype` is false there),
// and the intermediate carries an own `constructor`. Bootstrap and polyfill
// loaders probe that shape; QuickJS's global object is a bare
// Object.prototype instance, so mirror Node's chain. The guard keeps this
// idempotent if the engine ever grows a global prototype of its own.
(() => {
  if (Object.getPrototypeOf(globalThis) !== Object.prototype) return;
  const intermediate = {};
  Object.defineProperty(intermediate, "constructor", {
    value: Object,
    writable: true,
    enumerable: false,
    configurable: true,
  });
  Object.setPrototypeOf(globalThis, intermediate);
})();

// ---------------------------------------------------------------------------
// Eval-source registry for `new Function` bodies.
//
// V8 keeps every script's source, so `assert.ok(expr)` inside a dynamically
// constructed function still reports the expression text (node's
// internal/errors/error_source reads it from the stack's script data). QuickJS
// drops the source after compilation, so the constructor is wrapped to record
// each created function's body text; the assert shim reads it back through the
// frame's function object. The proxy only intercepts construct/apply and
// forwards everything else, so Function's surface (length, name, prototype,
// instanceof, toString) is unchanged; the prototype's `constructor` back-link
// is re-pointed so it still equals `globalThis.Function` as in Node.
// ---------------------------------------------------------------------------
(() => {
  const OriginalFunction = globalThis.Function;
  if (typeof OriginalFunction !== "function") return;
  const sources = new WeakMap();
  globalThis.__pirEvalSources = sources;
  const record = (fn, args) => {
    if (typeof fn === "function" && args.length > 0) {
      const last = args[args.length - 1];
      if (typeof last === "string") {
        const params = [];
        for (let i = 0; i < args.length - 1; i += 1) params.push(String(args[i]));
        try {
          sources.set(fn, { params, body: last });
        } catch {
          // Non-object target (never a function here): nothing to record.
        }
      }
    }
    return fn;
  };
  const wrapped = new Proxy(OriginalFunction, {
    construct(target, args, newTarget) {
      return record(Reflect.construct(target, args, newTarget), args);
    },
    apply(target, thisArg, args) {
      return record(Reflect.apply(target, thisArg, args), args);
    },
  });
  Object.defineProperty(globalThis, "Function", {
    value: wrapped,
    writable: true,
    enumerable: false,
    configurable: true,
  });
  Object.defineProperty(OriginalFunction.prototype, "constructor", {
    value: wrapped,
    writable: true,
    enumerable: false,
    configurable: true,
  });
})();
