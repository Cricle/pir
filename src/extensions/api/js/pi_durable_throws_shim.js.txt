// @earendil-works/pi-durable — throws tier (images, testing, Cloudflare storage).
//
// Five upstream subpath entries have no counterpart in the pir extension host.
// Every exported value throws the runtime's standard unsupported error (unified
// wording + `code` = `ERR_NOT_SUPPORTED`) at first use, so an extension that
// imports one of these entries still links and fails with a nameable error
// instead of at load time. `/images`, `/images/node` and `/images/cloudflare`
// are here because pir already ships an image pipeline (`image` 0.25 +
// the read tool's image path): a second Photon-wasm implementation would be a
// duplicate, and `/images/node` needs a worker thread over
// `@silvia-odwyer/photon`'s `.wasm` plus `createRequire`. `/testing` is here
// because it is a vitest-bound conformance registrar (pir's counterpart is the
// Rust-side suite in tests/). `/storage/sqlite/cloudflare` is here because it
// targets Durable Object SQL bindings; pir is a local CLI with no Workers
// target. `createPhotonImages` is the one entry point whose parameter is the
// caller's own wasm bytes: it still throws, because the wasm itself
// (`@silvia-odwyer/photon`, Rust→wasm) is not part of this port.
//
// `durableUnsupported` mirrors the facade wording of
// `pi_durable_shim.js.txt`; the two files are independent modules.

/** The standard unsupported error of this runtime (unified wording + `code`). */
function durableUnsupported(api, why) {
  const error = new Error(
    api +
      " is not supported by the pir extension runtime (" +
      why +
      "); wave D4 ports the durable SDK thin surface (tools, env, tool-search, codemode, storage, session), not the harness engine (D5)",
  );
  error.code = "ERR_NOT_SUPPORTED";
  return error;
}

const IMAGES_WHY = "pir ships its own image pipeline (image 0.25 + the read tool's image path); the Photon wasm module and its worker-thread wrapper are not part of this port";

/** A value that throws on every use (property access, call, construction): data and class exports below. */
function durableFacade(api, why) {
  const fail = (what) => {
    throw durableUnsupported(what, why);
  };
  const target = function () {
    fail(api);
  };
  return new Proxy(target, {
    apply: () => fail(api),
    construct: () => fail(api),
    get(_target, property) {
      // Reading any member is first use too (`Harness.open`, and a data export
      // such as `STORAGE_MEMORY_SCALES.length`), so it throws with that name.
      if (property === "name") return api;
      if (property === "prototype") return target.prototype;
      if (property === Symbol.toPrimitive || property === "toString" || property === "valueOf") return () => api;
      fail(api + "." + String(property));
    },
  });
}
