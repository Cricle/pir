import * as dnsPromises from "@node/dns/promises";

// node:dns: the callback form over the promise surface. `lookup` is real
// (host resolver); the resolve* family throws one clear error each (no
// c-ares query surface); the error-code/flag tables mirror the real values.

function lookup(...args) {
  let hostname = args[0];
  let options = args[1];
  let callback = args[args.length - 1];
  if (typeof callback !== "function") {
    throw new TypeError("The \"callback\" argument must be of type function");
  }
  if (typeof options === "function") {
    options = {};
  }
  if (typeof options === "number") {
    options = { family: options };
  }
  if (options === undefined || options === null) options = {};
  Promise.resolve()
    .then(() => dnsPromises.lookup(hostname, options))
    .then(
      (value) => {
        if (options.all) callback(null, value);
        else callback(null, value.address, value.family);
      },
      (error) => callback(error)
    );
}

// Callback wrappers over the promise family (which rejects with the unified
// unsupported error).
function resolve(...args) {
  const callback = args[args.length - 1];
  if (typeof callback !== "function") {
    throw new TypeError("The \"callback\" argument must be of type function");
  }
  const rrtype = typeof args[1] === "string" ? args[1] : "A";
  const dispatch = {
    A: dnsPromises.resolve4, AAAA: dnsPromises.resolve6, ANY: dnsPromises.resolveAny,
    CAA: dnsPromises.resolveCaa, CNAME: dnsPromises.resolveCname, MX: dnsPromises.resolveMx,
    NAPTR: dnsPromises.resolveNaptr, NS: dnsPromises.resolveNs, PTR: dnsPromises.resolvePtr,
    SOA: dnsPromises.resolveSoa, SRV: dnsPromises.resolveSrv, TLSA: dnsPromises.resolveTlsa,
    TXT: dnsPromises.resolveTxt,
  };
  const handler = dispatch[rrtype] || dnsPromises.resolve;
  handler(args[0]).then((value) => callback(null, value), (error) => callback(error));
}

function setServers(...args) {
  const callback = args[args.length - 1];
  if (typeof callback === "function") {
    dnsPromises.setServers().catch((error) => callback(error));
    return;
  }
  throw dnsPromises.setServers();
}
function getServers() {
  return dnsPromises.getServers();
}
function setDefaultResultOrder(order) {
  return dnsPromises.setDefaultResultOrder(order);
}
function getDefaultResultOrder() {
  return dnsPromises.getDefaultResultOrder();
}

// Resolver: inert like the promises Resolver (construction throws).
class Resolver {
  constructor() {
    const error = new Error("dns.Resolver is not supported by the pir extension runtime");
    error.code = "ERR_NOT_SUPPORTED";
    throw error;
  }
}
