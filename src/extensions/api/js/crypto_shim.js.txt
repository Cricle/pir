// ---------------------------------------------------------------------------
// Node-shaped argument errors
// ---------------------------------------------------------------------------
// Same rendering the zlib shim builds (each shim is an independent module, so
// the two keep their own copy): the `Received …` tail is Node's
// `determineSpecificType`, the noun before it is `property` for a dotted name
// and `argument` otherwise.
function pirTypeTail(value) {
  if (value === null) return "null";
  const type = typeof value;
  if (type === "undefined") return "undefined";
  if (type === "string") return "type string ('" + value + "')";
  if (type === "number") return "type number (" + String(value) + ")";
  if (type === "boolean") return "type boolean (" + String(value) + ")";
  if (type === "bigint") return "type bigint (" + String(value) + ")";
  if (type === "symbol") return "type symbol (" + String(value) + ")";
  if (type === "function") return "function " + (value.name === undefined ? "" : value.name);
  if (Buffer.isBuffer(value)) return "an instance of Buffer";
  if (value instanceof ArrayBuffer) return "an instance of ArrayBuffer";
  if (ArrayBuffer.isView(value)) {
    return "an instance of " + (value.constructor && value.constructor.name ? value.constructor.name : "TypedArray");
  }
  const ctor = value.constructor;
  return "an instance of " + (ctor && ctor.name ? ctor.name : "Object");
}

function pirInvalidArgType(name, expected, value) {
  const kind = name.indexOf(".") === -1 ? "argument" : "property";
  const error = new TypeError(
    "The \"" + name + "\" " + kind + " must be " + expected +
    ". Received " + pirTypeTail(value),
  );
  error.code = "ERR_INVALID_ARG_TYPE";
  return error;
}

function pirUnknownEncoding(encoding) {
  const error = new TypeError("Unknown encoding: " + String(encoding));
  error.code = "ERR_UNKNOWN_ENCODING";
  return error;
}

function pushData(chunks, data, encoding) {
  if (typeof data === "string") chunks.push(Buffer.from(data, encoding === undefined ? "utf8" : encoding));
  else if (data instanceof Uint8Array) chunks.push(Buffer.from(data));
  else if (data instanceof ArrayBuffer) chunks.push(Buffer.from(new Uint8Array(data)));
  else chunks.push(Buffer.from(data));
}

// Node key-material coercion: string keys hash as UTF-8, views/ArrayBuffers
// pass through, everything else is the Node argument TypeError.
function toKeyBytes(value, name) {
  if (typeof value === "string") return Buffer.from(value, "utf8");
  if (value instanceof Uint8Array) return Buffer.from(value);
  if (value instanceof ArrayBuffer) return Buffer.from(new Uint8Array(value));
  throw new TypeError("The \"" + name + "\" argument must be of type string or an instance of Buffer, TypedArray, or DataView");
}

// The wider key surface HMAC/cipher accept: string / bytes / KeyObject (with
// the secret material) / CryptoKey, with Node's ERR_INVALID_ARG_TYPE otherwise.
function pirKeyBytes(value, name) {
  if (KeyObject !== undefined && value instanceof KeyObject) {
    if (value._secretB64 === null) {
      throw pirInvalidArgType(name, "of type string or an instance of ArrayBuffer, Buffer, TypedArray, DataView, KeyObject, or CryptoKey", value);
    }
    return Buffer.from(value._secretB64, "base64");
  }
  if (CryptoKey !== undefined && value instanceof CryptoKey) {
    return Buffer.from(value._raw);
  }
  if (typeof value === "string") return Buffer.from(value, "utf8");
  if (value instanceof Uint8Array) return Buffer.from(value);
  if (value instanceof ArrayBuffer) return Buffer.from(new Uint8Array(value));
  throw pirInvalidArgType(
    name,
    "of type string or an instance of ArrayBuffer, Buffer, TypedArray, DataView, KeyObject, or CryptoKey",
    value,
  );
}

// Node digest-name normalization ("SHA-256" -> "sha256") shared by every
// hash/hmac/hkdf/kdf entry point; `dss1` is OpenSSL's historical SHA-1 alias
// and Node resolves it to SHA-1 (test-crypto-hmac asserts the lowercase and
// uppercase spellings give the SHA-1 HMAC).
const PIR_DIGEST_ALIASES = { dss1: "sha1" };

function normalizeDigest(algo) {
  const name = String(algo === undefined || algo === null ? "" : algo).toLowerCase().replace(/-/g, "");
  return PIR_DIGEST_ALIASES[name] === undefined ? name : PIR_DIGEST_ALIASES[name];
}

function pirRotr32(value, bits) {
  return (value >>> bits) | (value << (32 - bits));
}

function pirSha2_256Family(bytes, is224) {
  const h = is224
    ? [0xc1059ed8, 0x367cd507, 0x3070dd17, 0xf70e5939, 0xffc00b31, 0x68581511, 0x64f98fa7, 0xbefa4fa4]
    : [0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a, 0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19];
  const length = bytes.length;
  const padded = new Uint8Array(Math.ceil((length + 9) / 64) * 64);
  padded.set(bytes);
  padded[length] = 0x80;
  const bits = length * 8;
  // 64-bit big-endian bit length in the last 8 bytes; JS string/buffer sizes
  // stay below 2^48 bits, so the top two bytes are zero.
  const tail = padded.length - 8;
  padded[tail + 2] = Math.floor(bits / 1099511627776) & 0xff;
  padded[tail + 3] = Math.floor(bits / 4294967296) & 0xff;
  padded[tail + 4] = Math.floor(bits / 16777216) & 0xff;
  padded[tail + 5] = Math.floor(bits / 65536) & 0xff;
  padded[tail + 6] = Math.floor(bits / 256) & 0xff;
  padded[tail + 7] = bits & 0xff;
  const w = new Uint32Array(64);
  for (let offset = 0; offset < padded.length; offset += 64) {
    for (let i = 0; i < 16; i++) {
      const p = offset + i * 4;
      w[i] = (padded[p] << 24) | (padded[p + 1] << 16) | (padded[p + 2] << 8) | padded[p + 3];
    }
    for (let i = 16; i < 64; i++) {
      const x = w[i - 15];
      const y = w[i - 2];
      const s0 = pirRotr32(x, 7) ^ pirRotr32(x, 18) ^ (x >>> 3);
      const s1 = pirRotr32(y, 17) ^ pirRotr32(y, 19) ^ (y >>> 10);
      w[i] = (w[i - 16] + s0 + w[i - 7] + s1) | 0;
    }
    let a = h[0], b = h[1], c = h[2], d = h[3];
    let e = h[4], f = h[5], g = h[6], hh = h[7];
    for (let i = 0; i < 64; i++) {
      const s1 = pirRotr32(e, 6) ^ pirRotr32(e, 11) ^ pirRotr32(e, 25);
      const ch = (e & f) ^ (~e & g);
      const t1 = (hh + s1 + ch + PIR_SHA256_K[i] + w[i]) | 0;
      const s0 = pirRotr32(a, 2) ^ pirRotr32(a, 13) ^ pirRotr32(a, 22);
      const maj = (a & b) ^ (a & c) ^ (b & c);
      const t2 = (s0 + maj) | 0;
      hh = g; g = f; f = e; e = (d + t1) | 0;
      d = c; c = b; b = a; a = (t1 + t2) | 0;
    }
    h[0] = (h[0] + a) | 0; h[1] = (h[1] + b) | 0; h[2] = (h[2] + c) | 0; h[3] = (h[3] + d) | 0;
    h[4] = (h[4] + e) | 0; h[5] = (h[5] + f) | 0; h[6] = (h[6] + g) | 0; h[7] = (h[7] + hh) | 0;
  }
  const words = is224 ? 7 : 8;
  const out = Buffer.alloc(words * 4);
  for (let i = 0; i < words; i++) {
    out[i * 4] = (h[i] >>> 24) & 0xff;
    out[i * 4 + 1] = (h[i] >>> 16) & 0xff;
    out[i * 4 + 2] = (h[i] >>> 8) & 0xff;
    out[i * 4 + 3] = h[i] & 0xff;
  }
  return out;
}

const PIR_MASK64 = (1n << 64n) - 1n;

function pirRotr64(value, bits) {
  return ((value >> bits) | (value << (64n - bits))) & PIR_MASK64;
}

function pirSha2_512Family(bytes, is384) {
  const h = is384
    ? [0xcbbb9d5dc1059ed8n, 0x629a292a367cd507n, 0x9159015a3070dd17n, 0x152fecd8f70e5939n,
      0x67332667ffc00b31n, 0x8eb44a8768581511n, 0xdb0c2e0d64f98fa7n, 0x47b5481dbefa4fa4n]
    : [0x6a09e667f3bcc908n, 0xbb67ae8584caa73bn, 0x3c6ef372fe94f82bn, 0xa54ff53a5f1d36f1n,
      0x510e527fade682d1n, 0x9b05688c2b3e6c1fn, 0x1f83d9abfb41bd6bn, 0x5be0cd19137e2179n];
  const length = bytes.length;
  const padded = new Uint8Array(Math.ceil((length + 17) / 128) * 128);
  padded.set(bytes);
  padded[length] = 0x80;
  // 128-bit big-endian bit length; JS inputs stay far below 2^64 bits, so the
  // high half is zero and only the low 64 bits are written.
  const bits = BigInt(length) * 8n;
  const tail = padded.length - 16;
  for (let i = 0; i < 8; i++) {
    padded[tail + 8 + i] = Number((bits >> BigInt(8 * (7 - i))) & 0xffn);
  }
  const w = new Array(80).fill(0n);
  for (let offset = 0; offset < padded.length; offset += 128) {
    for (let i = 0; i < 16; i++) {
      let word = 0n;
      for (let b = 0; b < 8; b++) word = (word << 8n) | BigInt(padded[offset + i * 8 + b]);
      w[i] = word;
    }
    for (let i = 16; i < 80; i++) {
      const x = w[i - 15];
      const y = w[i - 2];
      const s0 = pirRotr64(x, 1n) ^ pirRotr64(x, 8n) ^ (x >> 7n);
      const s1 = pirRotr64(y, 19n) ^ pirRotr64(y, 61n) ^ (y >> 6n);
      w[i] = (w[i - 16] + s0 + w[i - 7] + s1) & PIR_MASK64;
    }
    let a = h[0], b = h[1], c = h[2], d = h[3];
    let e = h[4], f = h[5], g = h[6], hh = h[7];
    for (let i = 0; i < 80; i++) {
      const s1 = pirRotr64(e, 14n) ^ pirRotr64(e, 18n) ^ pirRotr64(e, 41n);
      const ch = (e & f) ^ (~e & g & PIR_MASK64);
      const t1 = (hh + s1 + ch + PIR_SHA512_K[i] + w[i]) & PIR_MASK64;
      const s0 = pirRotr64(a, 28n) ^ pirRotr64(a, 34n) ^ pirRotr64(a, 39n);
      const maj = (a & b) ^ (a & c) ^ (b & c);
      const t2 = (s0 + maj) & PIR_MASK64;
      hh = g; g = f; f = e; e = (d + t1) & PIR_MASK64;
      d = c; c = b; b = a; a = (t1 + t2) & PIR_MASK64;
    }
    h[0] = (h[0] + a) & PIR_MASK64; h[1] = (h[1] + b) & PIR_MASK64;
    h[2] = (h[2] + c) & PIR_MASK64; h[3] = (h[3] + d) & PIR_MASK64;
    h[4] = (h[4] + e) & PIR_MASK64; h[5] = (h[5] + f) & PIR_MASK64;
    h[6] = (h[6] + g) & PIR_MASK64; h[7] = (h[7] + hh) & PIR_MASK64;
  }
  const words = is384 ? 6 : 8;
  const out = Buffer.alloc(words * 8);
  for (let i = 0; i < words; i++) {
    for (let b = 0; b < 8; b++) {
      out[i * 8 + b] = Number((h[i] >> BigInt(8 * (7 - b))) & 0xffn);
    }
  }
  return out;
}

// SHA-3 / Keccak-F[1600]: 24 rounds of the standard theta/rho/pi/chi/iota
// steps over 25 64-bit lanes (lane i = x + 5y, the tiny_sha3 layout).
const PIR_KECCAK_ROTC = [1, 3, 6, 10, 15, 21, 28, 36, 45, 55, 2, 14, 27, 41, 56, 8, 25, 43, 62, 18, 39, 61, 20, 44];
const PIR_KECCAK_PILN = [10, 7, 11, 17, 18, 3, 5, 16, 8, 21, 24, 4, 15, 23, 19, 13, 12, 2, 20, 14, 22, 9, 6, 1];
const PIR_KECCAK_RC = [
  0x0000000000000001n, 0x0000000000008082n, 0x800000000000808an, 0x8000000080008000n,
  0x000000000000808bn, 0x0000000080000001n, 0x8000000080008081n, 0x8000000000008009n,
  0x000000000000008an, 0x0000000000000088n, 0x0000000080008009n, 0x000000008000000an,
  0x000000008000808bn, 0x800000000000008bn, 0x8000000000008089n, 0x8000000000008003n,
  0x8000000000008002n, 0x8000000000000080n, 0x000000000000800an, 0x800000008000000an,
  0x8000000080008081n, 0x8000000000008080n, 0x0000000080000001n, 0x8000000080008008n,
];

function pirKeccakF(state) {
  const bc = new Array(5).fill(0n);
  for (let round = 0; round < 24; round++) {
    for (let i = 0; i < 5; i++) {
      bc[i] = state[i] ^ state[i + 5] ^ state[i + 10] ^ state[i + 15] ^ state[i + 20];
    }
    for (let i = 0; i < 5; i++) {
      const t = bc[(i + 4) % 5] ^ (((bc[(i + 1) % 5] << 1n) | (bc[(i + 1) % 5] >> 63n)) & PIR_MASK64);
      for (let j = 0; j < 25; j += 5) state[j + i] ^= t;
    }
    let t = state[1];
    for (let i = 0; i < 24; i++) {
      const j = PIR_KECCAK_PILN[i];
      const saved = state[j];
      state[j] = ((t << BigInt(PIR_KECCAK_ROTC[i])) | (t >> BigInt(64 - PIR_KECCAK_ROTC[i]))) & PIR_MASK64;
      t = saved;
    }
    for (let j = 0; j < 25; j += 5) {
      for (let i = 0; i < 5; i++) bc[i] = state[j + i];
      for (let i = 0; i < 5; i++) {
        state[j + i] = bc[i] ^ (~bc[(i + 1) % 5] & bc[(i + 2) % 5] & PIR_MASK64);
      }
    }
    state[0] ^= PIR_KECCAK_RC[round];
  }
}

function pirSha3(bytes, rate, outLength) {
  const state = new Array(25).fill(0n);
  const padded = new Uint8Array(Math.ceil((bytes.length + 1) / rate) * rate);
  padded.set(bytes);
  padded[bytes.length] ^= 0x06; // SHA-3 domain separation (pad10*1, bits 0b01)
  padded[padded.length - 1] ^= 0x80;
  for (let offset = 0; offset < padded.length; offset += rate) {
    for (let i = 0; i < rate / 8; i++) {
      let word = 0n;
      for (let b = 0; b < 8; b++) word |= BigInt(padded[offset + i * 8 + b]) << BigInt(8 * b);
      state[i] ^= word;
    }
    pirKeccakF(state);
  }
  const out = Buffer.alloc(outLength);
  let written = 0;
  while (written < outLength) {
    const take = Math.min(rate, outLength - written);
    for (let i = 0; i < take; i++) {
      out[written + i] = Number((state[(i / 8) | 0] >> BigInt(8 * (i % 8))) & 0xffn);
    }
    written += take;
    if (written < outLength) pirKeccakF(state);
  }
  return out;
}

// algo -> {hash, blockSize}; `hash` takes/returns byte arrays. `blockSize` is
// the HMAC pad length (RFC 2104 uses the digest's internal block size).
const PIR_JS_DIGESTS = {
  sha224: { hash: (bytes) => pirSha2_256Family(bytes, true), blockSize: 64 },
  sha384: { hash: (bytes) => pirSha2_512Family(bytes, true), blockSize: 128 },
  sha3512: { hash: (bytes) => pirSha3(bytes, 72, 64), blockSize: 72 },
  sha3256: { hash: (bytes) => pirSha3(bytes, 136, 32), blockSize: 136 },
  sha3384: { hash: (bytes) => pirSha3(bytes, 104, 48), blockSize: 104 },
};

// The digests the host op serves directly (RustCrypto).
const PIR_HOST_DIGESTS = { md5: true, sha1: true, sha256: true, sha512: true };

function pirInvalidDigest(algo) {
  const error = new TypeError("Invalid digest: " + algo);
  error.code = "ERR_CRYPTO_INVALID_DIGEST";
  return error;
}

function pirDigestKnown(algo) {
  return PIR_HOST_DIGESTS[algo] === true || PIR_JS_DIGESTS[algo] !== undefined;
}

function pirDigestBytes(algo, bytes) {
  if (PIR_HOST_DIGESTS[algo] === true) {
    return Buffer.from(call("crypto", "hash", { algo: algo, data: bytes.toString("base64") }), "base64");
  }
  const impl = PIR_JS_DIGESTS[algo];
  if (impl === undefined) throw pirInvalidDigest(algo);
  return impl.hash(bytes);
}

/// RFC 2104 HMAC over whichever digest backend owns `algo`. `key`/`data` are
/// raw byte arrays.
function pirHmacBytes(algo, key, data) {
  if (PIR_HOST_DIGESTS[algo] === true) {
    return Buffer.from(call("crypto", "hmac", {
      algo: algo,
      key: Buffer.from(key).toString("base64"),
      data: Buffer.from(data).toString("base64"),
    }), "base64");
  }
  const impl = PIR_JS_DIGESTS[algo];
  if (impl === undefined) throw pirInvalidDigest(algo);
  const blockSize = impl.blockSize;
  let keyBytes = Buffer.from(key);
  if (keyBytes.length > blockSize) keyBytes = impl.hash(keyBytes);
  const padded = Buffer.alloc(blockSize);
  keyBytes.copy(padded);
  const inner = Buffer.alloc(blockSize);
  const outer = Buffer.alloc(blockSize);
  for (let i = 0; i < blockSize; i++) {
    inner[i] = padded[i] ^ 0x36;
    outer[i] = padded[i] ^ 0x5c;
  }
  return impl.hash(Buffer.concat([outer, impl.hash(Buffer.concat([inner, data]))]));
}


// Node's `hash.digest(encoding)`: `undefined` and "buffer" give the raw bytes,
// anything else is a string encoding.
function pirDigestResult(bytes, encoding) {
  if (encoding === undefined || encoding === null || encoding === "buffer") return bytes;
  return bytes.toString(encoding);
}

// ---------------------------------------------------------------------------
// Hash / Hmac
// ---------------------------------------------------------------------------
// Both carry Node's HashBase surface: `update()…digest()` plus the
// Writable+Readable transform sides (`write`/`end` on the way in, `data`/`end`
// on the way out, `read()`/`setEncoding()` for consumers). The input is
// buffered and the digest computed once at `end()`, the same posture as the
// zlib transform pad — real digests, one host/JS call per stream.
class PirHashBase extends EventEmitter {
  constructor() {
    super();
    this._chunks = [];
    this._digestBytes = null;
    this._ended = false;
    this._digestCalled = false;
    this._encoding = undefined;
    this._readConsumed = false;
  }
  update(data, encoding) {
    pushData(this._chunks, data, encoding);
    return this;
  }
  // --- Writable side ------------------------------------------------------
  write(chunk, encoding, callback) {
    if (typeof encoding === "function") { callback = encoding; encoding = undefined; }
    if (this._ended) {
      const error = new Error("write after end");
      error.code = "ERR_STREAM_WRITE_AFTER_END";
      if (typeof callback === "function") { globalThis.__pirMicrotask(() => callback(error)); return false; }
      throw error;
    }
    pushData(this._chunks, chunk, encoding);
    if (typeof callback === "function") globalThis.__pirMicrotask(callback);
    return true;
  }
  end(chunk, encoding, callback) {
    if (typeof chunk === "function") { callback = chunk; chunk = undefined; encoding = undefined; }
    else if (typeof encoding === "function") { callback = encoding; encoding = undefined; }
    if (chunk !== undefined && chunk !== null) pushData(this._chunks, chunk, encoding);
    if (!this._ended) {
      this._ended = true;
      this._digestBytes = this._computeDigest();
      this.emit("data", this._encodedDigest());
      this.emit("end");
      this.emit("finish");
    }
    if (typeof callback === "function") callback();
    return this;
  }
  _encodedDigest() {
    return this._encoding === undefined
      ? this._digestBytes
      : this._digestBytes.toString(this._encoding);
  }
  // --- Readable side ------------------------------------------------------
  read(_size) {
    if (this._digestBytes === null || this._readConsumed) return null;
    this._readConsumed = true;
    return this._encodedDigest();
  }
  setEncoding(encoding) {
    this._encoding = encoding;
    return this;
  }
  pipe(destination, options) {
    this.on("data", (chunk) => destination.write(chunk));
    this.on("end", () => {
      if (!options || options.end !== false) destination.end();
    });
    return destination;
  }
  destroy(error) {
    this._ended = true;
    if (error) this.emit("error", error);
    this.emit("close");
    return this;
  }
  // --- digest() -----------------------------------------------------------
  digest(encoding) {
    // After the stream path finalized, Node hands back the digest the stream
    // produced instead of hashing a now-empty state
    // (test-crypto-hash-stream-pipe: `h.digest('hex')` inside the 'data'
    // handler must still return the stream's digest).
    if (this._ended) return pirDigestResult(this._digestBytes, encoding);
    if (this._digestCalled) return this._repeatDigest(encoding);
    this._digestCalled = true;
    const value = this._computeDigest();
    this._chunks = [];
    return pirDigestResult(value, encoding);
  }
  _repeatDigest(encoding) {
    return pirDigestResult(Buffer.alloc(0), encoding);
  }
}

class Hash extends PirHashBase {
  constructor(algo) {
    super();
    if (typeof algo !== "string") {
      throw pirInvalidArgType("algorithm", "of type string", algo);
    }
    // WebCrypto spells algorithms "SHA-256"; Node digests use "sha256".
    this._algo = normalizeDigest(algo);
    // Node reports an unknown digest at construction, not at digest().
    if (!pirDigestKnown(this._algo)) throw pirInvalidDigest(this._algo);
  }
  _computeDigest() {
    return pirDigestBytes(this._algo, Buffer.concat(this._chunks));
  }
  // Node's Hash throws on a second digest(); Hmac (below) returns empty.
  _repeatDigest() {
    const error = new Error("Digest already called");
    error.code = "ERR_CRYPTO_HASH_FINALIZED";
    throw error;
  }
  copy() {
    const clone = new Hash(this._algo);
    clone._chunks = this._chunks.slice();
    return clone;
  }
}

class Hmac extends PirHashBase {
  constructor(algo, key) {
    super();
    this._algo = normalizeDigest(algo);
    this._key = pirKeyBytes(key, "key");
  }
  _computeDigest() {
    return pirHmacBytes(this._algo, this._key, Buffer.concat(this._chunks));
  }
}

function createHash(algo) { return new Hash(algo); }

// Node's createHmac validates the algorithm *name* as a string first, then the
// key, then the digest name (ERR_CRYPTO_INVALID_DIGEST for an unknown one).
function createHmac(algo, key) {
  if (typeof algo !== "string") {
    throw pirInvalidArgType("hmac", "of type string", algo);
  }
  const hmac = new Hmac(algo, key);
  if (!pirDigestKnown(hmac._algo)) throw pirInvalidDigest(hmac._algo);
  return hmac;
}

// The exported `crypto.Hmac` constructor is deprecated (DEP0181). Node's
// emitWarning is deferred to the next tick and deduplicated per code, which is
// what lets test-crypto-hmac call the constructor at the top of the file and
// still have its `expectWarning` block (registered later in the same tick) see
// the warning. `createHmac` is the supported entry point and stays silent.
let pirHmacDeprecationSent = false;
function pirWarnHmacConstructor() {
  if (pirHmacDeprecationSent) return;
  pirHmacDeprecationSent = true;
  globalThis.__pirMicrotask(() => {
    const proc = globalThis.process;
    if (proc === undefined || typeof proc.emitWarning !== "function") return;
    proc.emitWarning("crypto.Hmac constructor is deprecated.", {
      type: "DeprecationWarning",
      code: "DEP0181",
    });
  });
}

// node's `crypto.Hash`/`Hmac`/`Sign`/`Verify`/`Cipheriv`/`Decipheriv` are
// callable without `new` (test-crypto-hmac.js line 12 builds an Hmac that way),
// while `KeyObject`/`X509Certificate` are classes that throw when called
// without it. Each exported binding below is the callable form, sharing the
// class prototype so instanceof and `constructor` are unchanged.
function pirCallableCtor(impl, before) {
  const callable = function (...args) {
    if (before !== undefined) before();
    return new.target === undefined ? new impl(...args) : Reflect.construct(impl, args, new.target);
  };
  callable.prototype = impl.prototype;
  impl.prototype.constructor = callable;
  Object.defineProperty(callable, "name", { value: impl.name, configurable: true });
  Object.defineProperty(callable, "length", { value: impl.length, configurable: true });
  return callable;
}

// One-shot `crypto.hash(algorithm, data, outputEncoding)` (Node 21+).
function hash(algorithm, data, outputEncoding) {
  const hasher = new Hash(algorithm).update(data);
  return hasher.digest(outputEncoding);
}

function randomBytes(size, callback) {
  const count = Math.trunc(Number(size));
  if (!Number.isFinite(count) || count < 0) {
    throw new TypeError("The value of \"size\" is out of range. It must be >= 0");
  }
  if (typeof callback === "function") {
    microtaskRandom(count, callback);
    return;
  }
  return Buffer.from(call("crypto", "randomBytes", { n: count }), "base64");
}

function microtaskRandom(count, callback) {
  globalThis.__pirMicrotask(() => {
    let value;
    try {
      value = Buffer.from(call("crypto", "randomBytes", { n: count }), "base64");
    } catch (error) {
      callback(error);
      return;
    }
    callback(null, value);
  });
}

function randomFillSync(buffer, offset, size) {
  if (!(buffer instanceof Uint8Array)) {
    throw new TypeError("The \"buf\" argument must be an instance of Buffer, TypedArray, or DataView");
  }
  const start = offset === undefined ? 0 : Math.trunc(offset);
  const length = size === undefined ? buffer.length - start : Math.trunc(size);
  if (length === 0) return buffer;
  const bytes = Buffer.from(call("crypto", "randomBytes", { n: length }), "base64");
  buffer.set(bytes, start);
  return buffer;
}

function randomFill(buffer, offset, size, callback) {
  if (typeof offset === "function") { callback = offset; offset = undefined; size = undefined; }
  else if (typeof size === "function") { callback = size; size = undefined; }
  const start = offset === undefined ? 0 : Math.trunc(offset);
  const length = size === undefined ? buffer.length - start : Math.trunc(size);
  globalThis.__pirMicrotask(() => {
    try {
      randomFillSync(buffer, start, length);
      if (typeof callback === "function") callback(null, buffer);
    } catch (error) {
      if (typeof callback === "function") callback(error);
      else throw error;
    }
  });
  if (typeof callback !== "function") return undefined;
}

// Unbiased `randomInt` via rejection sampling over the host entropy source.
function pirRandomIntValue(lo, hi) {
  const range = hi - lo;
  const bytes = range <= 0x100 ? 1 : range <= 0x10000 ? 2 : range <= 0x100000000 ? 4 : 6;
  const pow = bytes === 6 ? 281474976710656 : Math.pow(2, bytes * 8);
  const limit = pow - (pow % range);
  for (;;) {
    const buf = Buffer.from(call("crypto", "randomBytes", { n: bytes }), "base64");
    let value = 0;
    for (let i = 0; i < bytes; i++) value = value * 256 + buf[i];
    if (value < limit) return lo + (value % range);
  }
}

function randomInt(min, max, callback) {
  if (typeof max === "function") { callback = max; max = min; min = 0; }
  let lo = Math.trunc(Number(min));
  let hi = Math.trunc(Number(max));
  if (!Number.isFinite(lo) || !Number.isFinite(hi)) {
    throw new TypeError("The \"max\" argument must be an integer");
  }
  if (hi <= lo) {
    throw new RangeError("The value of \"max\" is out of range. It must be greater than \"min\". Received " + hi);
  }
  if (hi - lo > 281474976710656) {
    throw new RangeError("The value of \"max\" is out of range. It must be <= 2^48 - 1. Received " + hi);
  }
  if (typeof callback === "function") {
    globalThis.__pirMicrotask(() => callback(null, pirRandomIntValue(lo, hi)));
    return;
  }
  return pirRandomIntValue(lo, hi);
}

function randomUUID() { return String(call("crypto", "randomUUID")); }

// UUID v7: 48-bit big-endian millisecond timestamp + version/variant + random
// tail (RFC 9562 layout; time-based ordering is best-effort, no monotonic
// counter is kept).
function randomUUIDv7() {
  const bytes = Buffer.from(call("crypto", "randomBytes", { n: 16 }), "base64");
  const ms = BigInt(Date.now());
  for (let i = 0; i < 6; i++) {
    bytes[i] = Number((ms >> BigInt(8 * (5 - i))) & 0xffn);
  }
  bytes[6] = (bytes[6] & 0x0f) | 0x70;
  bytes[8] = (bytes[8] & 0x3f) | 0x80;
  let hex = "";
  for (let i = 0; i < 16; i++) hex += (bytes[i] < 16 ? "0" : "") + bytes[i].toString(16);
  return hex.slice(0, 8) + "-" + hex.slice(8, 12) + "-" + hex.slice(12, 16) + "-" + hex.slice(16, 20) + "-" + hex.slice(20);
}

function toBytes(value, name) {
  if (value instanceof Uint8Array) return value;
  if (value instanceof ArrayBuffer) return new Uint8Array(value);
  throw new TypeError("The \"" + name + "\" argument must be an instance of Buffer, TypedArray, or DataView");
}

function timingSafeEqual(a, b) {
  const left = toBytes(a, "buf1");
  const right = toBytes(b, "buf2");
  if (left.length !== right.length) {
    throw new RangeError("Input buffers must have the same byte length");
  }
  let diff = 0;
  for (let i = 0; i < left.length; i++) diff |= left[i] ^ right[i];
  return diff === 0;
}

const CIPHER_MODES = ["cbc", "ctr", "gcm"];

function pirCipherInfo(name) {
  const match = /^aes-(128|192|256)-(cbc|ctr|gcm)$/.exec(String(name).toLowerCase());
  if (!match) return undefined;
  return {
    name: match[0],
    nid: undefined,
    blockSize: 16,
    ivLength: match[2] === "gcm" ? 12 : 16,
    keyLength: Number(match[1]) / 8,
    mode: match[2],
  };
}

  for (const size of CIPHER_SIZES) {

  return out;
}

class PirCipher {
  constructor(algorithm, key, iv, encrypt) {
    this._alg = String(algorithm).toLowerCase();
    this._encrypt = encrypt;
    this._keyB64 = toKeyBytes(key, "key").toString("base64");
    this._ivB64 = iv === undefined || iv === null ? null : toKeyBytes(iv, "iv").toString("base64");
    this._state = null;
    this._autoPadding = true;
    this._aadB64 = null;
    this._authTagB64 = null;
    this._finalTag = null;
    this._ended = false;
    this._outputEncoding = undefined;
  }
  _dataBytes(data, encoding) {
    if (typeof data === "string") return Buffer.from(data, encoding === undefined ? "utf8" : encoding);
    if (data instanceof Uint8Array) return Buffer.from(data);
    if (data instanceof ArrayBuffer) return Buffer.from(new Uint8Array(data));
    throw new TypeError("The \"data\" argument must be of type string or an instance of Buffer, TypedArray, or DataView");
  }
  _run(data, isFinal) {
    if (this._ended) {
      const error = new Error("Invalid state: Cipher is already finalized");
      error.code = "ERR_CRYPTO_INVALID_STATE";
      throw error;
    }
    const payload = {
      alg: this._alg,
      dir: this._encrypt ? "encrypt" : "decrypt",
      key: this._keyB64,
      iv: this._ivB64,
      data: data.toString("base64"),
      final: isFinal,
      autoPadding: this._autoPadding,
      state: this._state,
    };
    if (this._aadB64 !== null) payload.aad = this._aadB64;
    if (!this._encrypt && this._authTagB64 !== null) payload.authTag = this._authTagB64;
    const result = call("crypto", "cipheriv", payload);
    this._state = result.state === null || result.state === undefined ? null : result.state;
    if (result.authTag) this._finalTag = Buffer.from(result.authTag, "base64");
    if (isFinal) this._ended = true;
    return Buffer.from(result.data, "base64");
  }
  update(data, inputEncoding, outputEncoding) {
    const encoding = this._checkOutputEncoding(outputEncoding);
    const out = this._run(this._dataBytes(data, inputEncoding), false);
    return encoding === undefined ? out : out.toString(encoding);
  }
  final(outputEncoding) {
    const encoding = this._checkOutputEncoding(outputEncoding);
    const out = this._run(Buffer.alloc(0), true);
    return encoding === undefined ? out : out.toString(encoding);
  }
  // Node's CipherBase pins one output encoding per cipher: a later update()/
  // final() with a different one raises `Cannot change encoding`, and an
  // unknown name raises ERR_UNKNOWN_ENCODING (the unknown-encoding check runs
  // first — test-crypto-encoding-validation-error covers both orders).
  _checkOutputEncoding(outputEncoding) {
    if (outputEncoding === undefined || outputEncoding === null) return undefined;
    if (typeof outputEncoding !== "string" || !Buffer.isEncoding(outputEncoding)) {
      throw pirUnknownEncoding(outputEncoding);
    }
    if (this._outputEncoding === undefined) {
      this._outputEncoding = outputEncoding;
    } else if (this._outputEncoding !== outputEncoding) {
      const error = new Error("Cannot change encoding");
      error.code = "ERR_INTERNAL_ASSERTION";
      throw error;
    }
    return outputEncoding;
  }
  setAutoPadding(value) {
    this._autoPadding = value === undefined ? true : !!value;
    return this;
  }
  setAAD(buffer, _options) {
    this._aadB64 = toBytes(buffer, "buffer").slice().toString("base64");
    return this;
  }
  getAuthTag() {
    if (this._finalTag === null) {
      throw new Error("The authentication tag has not been computed; call final() first");
    }
    return this._finalTag;
  }
  setAuthTag(tag) {
    const bytes = typeof tag === "string" ? Buffer.from(tag, "hex") : toBytes(tag, "tag");
    this._authTagB64 = Buffer.from(bytes).toString("base64");
    return this;
  }
}

  }

// A minimal KeyObject: PEM material passes through to the sign/verify host
// ops; secret keys keep their raw bytes (base64) for export().
class KeyObject {
  constructor(type, pem) {
    this.type = type;
    this._pem = pem === undefined ? null : pem;
    this._secretB64 = null;
    this.asymmetricKeyType = type === "secret" ? undefined : (pem === null ? undefined : "rsa");
  }
  // `KeyObject.from(cryptoKey)`: the WebCrypto secret keys this shim produces
  // carry their raw bytes, so the wrapping is exact (asymmetric keys would need
  // the DER/PEM conversion the host does not host).
  static from(key) {
    if (key !== undefined && key !== null && key.type === "secret" && key._raw !== undefined) {
      const object = new KeyObject("secret", null);
      object._secretB64 = Buffer.from(key._raw).toString("base64");
      return object;
    }
    throw pirInvalidArgType("key", "an instance of CryptoKey", key);
  }
  export(options) {
    if (this.type === "secret") {
      return Buffer.from(this._secretB64 === null ? "" : this._secretB64, "base64");
    }
    return this._pem;
  }
  equals(other) {
    return other instanceof KeyObject && other.type === this.type &&
      other._pem === this._pem && other._secretB64 === this._secretB64;
  }
}

function pirKeyPem(key, name) {
  if (key instanceof KeyObject && typeof key._pem === "string") return key._pem;
  if (typeof key === "string") return key;
  if (key instanceof Uint8Array) return Buffer.from(key).toString("utf8");
  if (key instanceof ArrayBuffer) return Buffer.from(new Uint8Array(key)).toString("utf8");
  throw new TypeError("The \"" + name + "\" argument must be of type string or an instance of Buffer, TypedArray, or KeyObject");
}

  return object;
}

function normalizeSignAlgo(algorithm) {
  const compact = String(algorithm).toLowerCase().replace(/-/g, "");
  return compact.indexOf("rsa") === 0 ? compact.slice(3) : compact;
}

class Sign {
  constructor(algorithm) {
    this._algo = normalizeSignAlgo(algorithm);
    this._chunks = [];
  }
  update(data, encoding) { pushData(this._chunks, data, encoding); return this; }
  write(data, encoding) { return this.update(data, encoding); }
  end(data, encoding) { if (data !== undefined) this.update(data, encoding); return this; }
  sign(key, encoding) {
    const data = Buffer.concat(this._chunks);
    const signature = Buffer.from(call("crypto", "sign", {
      algo: this._algo,
      key: pirKeyPem(key, "key"),
      data: data.toString("base64"),
    }), "base64");
    return encoding === undefined ? signature : signature.toString(encoding);
  }
}

class Verify {
  constructor(algorithm) {
    this._algo = normalizeSignAlgo(algorithm);
    this._chunks = [];
  }
  update(data, encoding) { pushData(this._chunks, data, encoding); return this; }
  write(data, encoding) { return this.update(data, encoding); }
  end(data, encoding) { if (data !== undefined) this.update(data, encoding); return this; }
  verify(key, signature, signatureEncoding) {
    const data = Buffer.concat(this._chunks);
    let signatureBytes;
    if (typeof signature === "string") {
      signatureBytes = Buffer.from(signature, signatureEncoding === undefined ? "hex" : signatureEncoding);
    } else {
      signatureBytes = Buffer.from(signature instanceof ArrayBuffer ? new Uint8Array(signature) : signature);
    }
    return call("crypto", "verify", {
      algo: this._algo,
      key: pirKeyPem(key, "key"),
      data: data.toString("base64"),
      signature: signatureBytes.toString("base64"),
    }) === true;
  }
}

// Real PBKDF2 (RFC 2898) over the host's Hmac<sha1|sha256|sha512>; Node
// argument shapes and error ordering (iterations/keylen validated first).
function pbkdf2Sync(password, salt, iterations, keylen, digest) {
  const passwordBytes = toKeyBytes(password, "password");
  const saltBytes = toKeyBytes(salt, "salt");
  const rounds = Number(iterations);
  if (!Number.isFinite(rounds) || rounds < 1) {
    throw new TypeError("The \"iterations\" argument must be a positive number");
  }
  const length = Number(keylen);
  if (!Number.isFinite(length) || length < 0) {
    throw new TypeError("The \"keylen\" argument must be a non-negative number");
  }
  const algo = normalizeDigest(digest);
  if (algo !== "sha1" && algo !== "sha256" && algo !== "sha512") {
    throw new Error("Digest method not supported: " + digest);
  }
  return Buffer.from(call("crypto", "pbkdf2", {
    algo: algo,
    password: passwordBytes.toString("base64"),
    salt: saltBytes.toString("base64"),
    iterations: Math.floor(rounds),
    keylen: Math.floor(length),
  }), "base64");
}

function pbkdf2(password, salt, iterations, keylen, digest, callback) {
  // Node's overloads: pbkdf2(pw, salt, iterations, keylen, digest, cb) and
  // the digest-less form with the callback in the digest slot.
  if (typeof digest === "function") { callback = digest; digest = undefined; }
  globalThis.__pirMicrotask(() => {
    let value;
    try {
      value = pbkdf2Sync(password, salt, iterations, keylen, digest);
    } catch (error) {
      if (typeof callback === "function") { callback(error); return; }
      throw error;
    }
    if (typeof callback === "function") callback(null, value);
  });
}

// Node's scryptSync defaults: N=16384, r=8, p=1, maxmem=32 MiB.
function scryptOptions(options) {
  const opts = options === undefined || options === null ? {} : options;
  const n = opts.N === undefined ? 16384 : Number(opts.N);
  const r = opts.r === undefined ? 8 : Number(opts.r);
  const p = opts.p === undefined ? 1 : Number(opts.p);
  const maxmem = opts.maxmem === undefined ? 32 * 1024 * 1024 : Number(opts.maxmem);
  const logN = Math.log2(n);
  if (!Number.isInteger(n) || n < 2 || !Number.isInteger(logN)) {
    throw new RangeError("The value of \"options.N\" is out of range. It must be a power of two > 1. Received " + n);
  }
  for (const [name, value, min] of [["r", r, 1], ["p", p, 1], ["maxmem", maxmem, 1]]) {
    if (!Number.isInteger(value) || value < min) {
      throw new RangeError("The value of \"options." + name + "\" is out of range. It must be an integer >= " + min + ". Received " + value);
    }
  }
  return { logN: logN, r: r, p: p, maxmem: maxmem };
}

function scryptSync(password, salt, keylen, options) {
  const passwordBytes = toKeyBytes(password, "password");
  const saltBytes = toKeyBytes(salt, "salt");
  const length = Number(keylen);
  if (!Number.isFinite(length) || length < 0) {
    throw new TypeError("The \"keylen\" argument must be a non-negative number");
  }
  const opts = scryptOptions(options);
  return Buffer.from(call("crypto", "scrypt", {
    password: passwordBytes.toString("base64"),
    salt: saltBytes.toString("base64"),
    logN: opts.logN,
    r: opts.r,
    p: opts.p,
    maxmem: opts.maxmem,
    keylen: Math.floor(length),
  }), "base64");
}

function scrypt(password, salt, keylen, options, callback) {
  if (typeof options === "function") { callback = options; options = undefined; }
  globalThis.__pirMicrotask(() => {
    let value;
    try {
      value = scryptSync(password, salt, keylen, options);
    } catch (error) {
      if (typeof callback === "function") { callback(error); return; }
      throw error;
    }
    if (typeof callback === "function") callback(null, value);
  });
}

// Real HKDF (RFC 5869) composed from the real HMAC chain. Node returns an
// ArrayBuffer from hkdfSync.
function pirHkdf(digest, ikm, salt, info, keylen) {
  const algo = normalizeDigest(digest);
  if (algo !== "sha1" && algo !== "sha256" && algo !== "sha384" && algo !== "sha512") {
    throw new Error("Digest method not supported: " + digest);
  }
  const ikmBytes = toKeyBytes(ikm, "ikm");
  const saltBytes = salt === undefined || salt === null ? Buffer.alloc(0) : toKeyBytes(salt, "salt");
  const infoBytes = info === undefined || info === null ? Buffer.alloc(0) : toKeyBytes(info, "info");
  const length = Number(keylen);
  if (!Number.isFinite(length) || length < 0) {
    throw new TypeError("The \"keylen\" argument must be a non-negative number");
  }
  const prk = createHmac(algo, saltBytes).update(ikmBytes).digest();
  const hashLen = prk.length;
  const blocks = Math.ceil(length / hashLen);
  if (blocks > 255) {
    throw new RangeError("Invalid key length");
  }
  const out = [];
  let previous = Buffer.alloc(0);
  for (let i = 1; i <= blocks; i++) {
    const block = createHmac(algo, prk)
      .update(Buffer.concat([previous, infoBytes, Buffer.from([i])]))
      .digest();
    out.push(block);
    previous = block;
  }
  const joined = Buffer.concat(out);
  const exact = new Uint8Array(length);
  exact.set(joined.subarray(0, length));
  return exact.buffer;
}

function hkdfSync(digest, ikm, salt, info, keylen) {
  return pirHkdf(digest, ikm, salt, info, keylen);
}

function hkdf(digest, ikm, salt, info, keylen, callback) {
  globalThis.__pirMicrotask(() => {
    let value;
    try {
      value = pirHkdf(digest, ikm, salt, info, keylen);
    } catch (error) {
      if (typeof callback === "function") { callback(error); return; }
      throw error;
    }
    if (typeof callback === "function") callback(null, value);
  });
}

// WebCrypto: CryptoKey + subtle.digest/importKey/generateKey

// The key material is an own property rather than an internal slot: the
// runtime has no slots, and own properties make two keys with different
// material — or a different algorithm / extractable flag / usage set —
// observably different to assert.deepStrictEqual, which is what Node's
// internal-slot comparison gives (test-assert-deep's Crypto case).
class CryptoKey {
  constructor(type, algorithm, extractable, usages, raw) {
    this.type = type;
    this.extractable = extractable;
    this.algorithm = algorithm;
    this.usages = usages;
    this._raw = raw;
  }
}

// WebCrypto hash names are the case-sensitive canonical strings ("SHA-256",
// "SHA3-512" — Node rejects "sha256" with NotSupportedError); each maps onto a
// digest this runtime really serves.
const PIR_WEBCRYPTO_HASHES = {
  "SHA-1": "sha1",
  "SHA-256": "sha256",
  "SHA-384": "sha384",
  "SHA-512": "sha512",
  "SHA3-256": "sha3256",
  "SHA3-384": "sha3384",
  "SHA3-512": "sha3512",
};

function pirWebcryptoHashName(hash) {
  const name = typeof hash === "string" ? hash : (hash === undefined || hash === null ? undefined : hash.name);
  if (typeof name !== "string") {
    throw new TypeError("Failed to normalize algorithm: passed algorithm must include a hash");
  }
  if (PIR_WEBCRYPTO_HASHES[name] === undefined) {
    const error = new Error("Unrecognized algorithm name");
    error.name = "NotSupportedError";
    error.code = 9;
    throw error;
  }
  return name;
}

// The normalized algorithm dictionary WebCrypto hands back from key.algorithm
// (Node: { name, length, hash: { name } }).
function pirWebcryptoAlgorithm(algorithm) {
  const source = typeof algorithm === "string" ? { name: algorithm } : (algorithm || {});
  const name = source.name;
  if (typeof name !== "string") {
    throw new TypeError("The \"algorithm\" argument must be a string or an object with a name");
  }
  if (name.toUpperCase() !== "HMAC") return { name: name };
  const normalized = { name: name };
  if (source.length !== undefined) normalized.length = source.length;
  normalized.hash = { name: pirWebcryptoHashName(source.hash) };
  return normalized;
}

function pirWebcryptoHmacOnly(api, name) {
  const error = new Error(api + ": algorithm " + name + " is not supported by the pir extension runtime");
  error.code = "ERR_NOT_SUPPORTED";
  return error;
}

function pirWebcryptoSecretKey(algorithm, raw, extractable, usages, bits) {
  const normalized = Object.assign({}, algorithm);
  normalized.length = bits;
  return new CryptoKey(
    "secret",
    normalized,
    extractable === true,
    Array.prototype.slice.call(usages === undefined ? [] : usages),
    Buffer.from(raw),
  );
}

const webcrypto = {
  subtle: {
    digest: async (algorithm, data) => {
      const name = typeof algorithm === "string" ? algorithm : (algorithm && algorithm.name);
      if (typeof name !== "string") {
        throw new TypeError("The \"algorithm\" argument must be a string or an object with a name");
      }
      const bytes = toBytes(data, "data");
      const result = pirDigestBytes(normalizeDigest(name), Buffer.from(bytes));
      const out = new Uint8Array(result.length);
      out.set(result);
      return out.buffer; // exact-length backing store: no extra bytes leak
    },
    importKey: async (format, keyData, algorithm, extractable, usages) => {
      const normalized = pirWebcryptoAlgorithm(algorithm);
      if (format !== "raw") {
        const error = new TypeError(
          "The \"format\" argument must be \"raw\" (only the raw key format is supported here)",
        );
        error.code = "ERR_INVALID_ARG_VALUE";
        throw error;
      }
      if (normalized.name.toUpperCase() !== "HMAC") {
        throw pirWebcryptoHmacOnly("subtle.importKey", normalized.name);
      }
      const raw = Buffer.from(toBytes(keyData, "keyData"));
      return pirWebcryptoSecretKey(normalized, raw, extractable, usages, raw.length * 8);
    },
    generateKey: async (algorithm, extractable, usages) => {
      const normalized = pirWebcryptoAlgorithm(algorithm);
      if (normalized.name.toUpperCase() !== "HMAC") {
        throw pirWebcryptoHmacOnly("subtle.generateKey", normalized.name);
      }
      const bits = normalized.length === undefined ? 512 : Number(normalized.length);
      const raw = Buffer.from(call("crypto", "randomBytes", { n: Math.ceil(bits / 8) }), "base64");
      return pirWebcryptoSecretKey(normalized, raw, extractable, usages, bits);
    },
  },
  getRandomValues: webcryptoGetRandomValues,
  randomUUID: () => randomUUID(),
};

// Host-inaccessible crypto surfaces: one clear coded error each, raised at
// call/construction time (never silently degraded).
function pirCryptoUnsupported(api) {
  const error = new Error(api + " is not supported by the pir extension runtime");
  error.code = "ERR_NOT_SUPPORTED";
  return error;
}

function pirCryptoThrower(api) {
  return function () { throw pirCryptoUnsupported(api); };
}

function pirCryptoClass(api) {
  return class {
    constructor(..._args) { throw pirCryptoUnsupported(api); }
  };
}
