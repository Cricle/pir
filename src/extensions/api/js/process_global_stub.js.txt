    const list = listenersByEvent.get(type);
    if (list === undefined) return this;
    for (let i = list.length - 1; i >= 0; i--) {
      if (list[i] === listener || list[i].listener === listener) {
        list.splice(i, 1);
        break;
      }
    }
    if (list.length === 0) listenersByEvent.delete(type);
    return this;
  };

      throw new TypeError("The \"listener\" argument must be of type function");
    }
    let wrapped = listener;
    if (once) {
      wrapped = function (...args) {
        removeProcessListener.call(this, type, wrapped);
        listener.apply(this, args);
      };
      wrapped.listener = listener;
    }
    if (!listenersByEvent.has(type)) listenersByEvent.set(type, []);
    listenersByEvent.get(type).push(wrapped);

    // first registration arms it, and every re-registration from an emission
    // handler re-arms a probe that stopped after the previous emission.
    if (type === "beforeExit" && armBeforeExitProbe !== null) armBeforeExitProbe(0);
    return this;

  // --- beforeExit (loop-idle emission) --------------------------------------
  // Node emits `beforeExit` (with `process.exitCode`) whenever the event
  // loop empties, which lets a listener keep the process alive by scheduling
  // more work; once that work runs out the event is emitted again, while a
  // nextTick-only handler drains inside the emission and does NOT keep the
  // loop alive — the official test's final `mustNotCall` listener depends on
  // exactly that.
  //
  // The host drains its loop (rquickjs jobs + the host timer wheel) with no
  // JS callback at the "loop is empty" point, so the shim detects idleness
  // itself: the prelude keeps every armed timer in its own map
  // (`__pirTimers`, a global lexical binding) and everything JS-visible above
  // the timers is the job queue, which has already drained by the time a
  // timer callback runs. A probe timer armed on the first `beforeExit`
  // listener is therefore a faithful idle detector: pending timers re-arm it,
  // an idle probe emits, and a probe that finds the loop still idle after an
  // emission stops — Node emits once per idle period (`uv_loop_alive()` is
  // false right after emitting, so its do-while exits).
  //
  // Liveness after an emission is deliberately NOT decided synchronously at
  // the end of the emit: Node re-checks `uv_loop_alive()` only after the
  // handler's own callbacks AND their microtask/nextTick drain have run
  // (src/api/embed_helpers.cc — `EmitProcessBeforeExit`, then
  // `uv_loop_alive()`). A handler that queues a nextTick whose callback arms
  // a setImmediate leaves nothing pending at that instant, and that work can
  // even run to completion before the next probe fires, so the pending count
  // alone cannot tell "the loop ran after the emission" (the next exhaustion
  // emits again) from "the loop is dead" (the process exits). The prelude's
  // timer sequence (`__pirTimerSeq`, bumped by every setTimeout/setInterval
  // arm and therefore by the immediate bridge) is the completion-proof
  // witness: any timer armed after the emission proves the loop was alive
  // after it. The probes this shim arms are subtracted by count, so an idle
  // probe still stops.
  //
  // Residual limits (documented, not hidden):
  //   * host-side work that is not a prelude timer (an in-flight
  //     fetch/child/watcher) is invisible here, so `beforeExit` can fire
  //     while such work is still pending; a fully faithful loop needs a drain
  //     hook in the runtime (js_mode/runtime.rs), which the shim cannot
  //     reach;
  //   * the probe polls (1 ms, backing off to 16 ms while other timers are
  //     pending), so the emission trails the loop emptying by up to that
  //     delay and a host without the prelude timer surface never emits.
  let beforeExitProbeArmed = false;
  let beforeExitEmitted = false;
  let beforeExitWaits = 0;
  // Ids of the probe timers this shim armed and that the prelude has not
  // retired yet. The prelude deletes a timer's map entry only after its
  // callback returns, so *two* of our entries can be in the map at once (the
  // running probe plus a probe armed from inside its own emission); counting
  // by id is the only correct way to subtract our own liveness from the
  // "pending work" total.
  const beforeExitProbeIds = new Set();
  // Window for the sequence witness: `beforeExitSeqBase`/`...SelfArmsAtBase`
  // are re-snapshotted when an emission ends and whenever a probe concludes
  // the loop was alive, so "timer armed since" counts exactly the arms after
  // the last such point. A self-arm can never be mistaken for foreign work:
  // the prelude assigns ids in increasing order, so every arm this shim makes
  // after the snapshot is one of the arms counted by the difference.
  let beforeExitSeqBase = -1;
  let beforeExitSelfArms = 0;
  let beforeExitSelfArmsAtBase = 0;
  const beforeExitTimerSeq = () =>
    (typeof __pirTimerSeq === "number" ? __pirTimerSeq : -1);
  const beforeExitRebase = () => {
    beforeExitSelfArmsAtBase = beforeExitSelfArms;
    beforeExitSeqBase = beforeExitTimerSeq();
  };
  const beforeExitArmedSinceBase = () => {
    const seq = beforeExitTimerSeq();
    if (seq < 0 || beforeExitSeqBase < 0) return 0; // no prelude timer surface
    return Math.max(seq - beforeExitSeqBase - (beforeExitSelfArms - beforeExitSelfArmsAtBase), 0);
  };
  const beforeExitPending = () => {
    // `typeof` on a global lexical binding is safe even when the prelude's
    // timer surface is absent.
    const map = typeof __pirTimers === "object" && __pirTimers !== null ? __pirTimers : null;
    if (map === null) return 0;
    let own = 0;
    for (const id of beforeExitProbeIds) {
      if (map.has(id)) own += 1;
      else beforeExitProbeIds.delete(id);
    }
    return Math.max(map.size - own, 0);
  };
  armBeforeExitProbe = (delay) => {
    if (beforeExitProbeArmed) return; // one probe at a time
    if (typeof globalThis.setTimeout !== "function") return; // no host timers
    if (proc.listenerCount("beforeExit") === 0) return; // nobody to emit to
    beforeExitProbeArmed = true;
    const id = globalThis.setTimeout(beforeExitProbe, delay);
    beforeExitSelfArms += 1;
    if (typeof id === "number") beforeExitProbeIds.add(id);
  };
  const beforeExitProbe = () => {
    const pending = beforeExitPending(); // still counts the running probe
    const armed = beforeExitArmedSinceBase();
    beforeExitProbeArmed = false;
    if (proc.listenerCount("beforeExit") === 0) return;
    if (pending > 0 || armed > 0) {
      // Work is outstanding, or was armed and ran to completion since the
      // last emission: the loop was alive, so the next time it empties is a
      // fresh idle period that emits again. The window restarts here, or the
      // same arms would keep resetting the latch forever.
      beforeExitEmitted = false;
      beforeExitRebase();
      beforeExitWaits += 1;
      armBeforeExitProbe(Math.min(1 << Math.min(beforeExitWaits, 4), 16));
      return;
    }
    beforeExitWaits = 0;
    if (beforeExitEmitted) return; // one emission per idle period
    beforeExitEmitted = true;
    try {
      proc.emit(
        "beforeExit",
        proc.exitCode === undefined || proc.exitCode === null ? 0 : proc.exitCode,
      );
    } finally {
      // Work the handlers armed synchronously is visible here and the loop is
      // alive again; a bare nextTick (and anything it may arm) is not — the
      // follow-up probe re-judges that against the sequence window. A
      // handler that scheduled nothing keeps the latch set and the process
      // exits, which is what stops a second emission.
      if (beforeExitPending() > 0) beforeExitEmitted = false;
      beforeExitRebase();
      if (proc.listenerCount("beforeExit") > 0) armBeforeExitProbe(0);
    }
  };
  return proc;
  })();
  // Real Node exposes `process` as an accessor whose setter stores the
  // assigned value, non-enumerable and configurable. A plain assignment
  // instead reads back as a writable data property, so
  // `getOwnPropertyDescriptor(globalThis, "process").writable` would be
  // `true` where Node reports `undefined`.
  let current = value;
  Object.defineProperty(globalThis, "process", {
    get() { return current; },
    set(next) { current = next; },
    enumerable: false,
    configurable: true,
  });
}
