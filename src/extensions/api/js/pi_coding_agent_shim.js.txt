export function isToolCallEventType(a, b) {
  // Upstream signature (types.ts): `isToolCallEventType(toolName, event)`.
  // pir historically exposed `(event, toolName)`; detect the argument order
  // from the types so both call shapes work.
  const toolName = typeof a === "string" ? a : b;
  const event = a != null && typeof a === "object" ? a : b;
  return event?.type === "tool_call" && (toolName === undefined || event.toolName === toolName);
}

const FILE_MUTATION_QUEUES = new Map();

// Upstream resolves + realpaths the queue key via node:fs; this runtime has no
// filesystem access, so the key is the path string exactly as given.
export async function withFileMutationQueue(filePath, fn) {
  const key = String(filePath);
  const previous = FILE_MUTATION_QUEUES.get(key) ?? Promise.resolve();
  let releaseNext;
  const nextQueue = new Promise((resolveQueue) => { releaseNext = resolveQueue; });
  const chainedQueue = previous.then(() => nextQueue);
  FILE_MUTATION_QUEUES.set(key, chainedQueue);
  await previous;
  try {
    return await fn();
  } finally {
    releaseNext();
    if (FILE_MUTATION_QUEUES.get(key) === chainedQueue) {
      FILE_MUTATION_QUEUES.delete(key);
    }
  }
}

// UPSTREAM SHAPE (packages/coding-agent core/session-manager.ts): projects the
// canonical entry list into { messages, thinkingLevel, model } by replaying
// message/compaction/custom entries. This is an APPROXIMATION: message-typed
// entries contribute their `message` as-is, compaction/branch summaries and
// the thinkingLevel/model projection are dropped — callers here read
// `messages` only (pi-phone's state snapshot and usage totals).
export function buildSessionContext(entriesOrContext, _leafId, _byId) {
  const list = Array.isArray(entriesOrContext) ? entriesOrContext : [];
  const messages = [];
  for (const entry of list) {
    if (entry === null || typeof entry !== "object") continue;
    let source;
    if (entry.type === "message" && entry.message && typeof entry.message === "object") {
      source = entry.message;
    } else if (typeof entry.role === "string") {
      // A plain message (no file-entry envelope) passes through directly.
      source = entry;
    } else {
      continue;
    }
    if (typeof source.role !== "string") continue;
    messages.push({ role: source.role, content: source.content });
  }
  return { messages };
}

// Host-backed `SessionManager` (upstream packages/coding-agent
// core/session-manager.ts): the pir runtime owns a byte-compatible session
// store (src/session_manager), exposed to the shim over the synchronous
// `__pir_host.pirSession*` ops (src/agent_session/ext_session_ops.rs). The
// implemented face is the EXTENSION CONSUMPTION SURFACE — the methods the
// ecosystem actually calls:
//   - pi-phone (phone-sessions.ts): open / getBranch / getTree / getLeafId /
//     createBranchedSession (session tree + branch panel);
//   - pi-subagents (fork-context.js): open(path, sessionDir) /
//     createBranchedSession (forked child transcripts);
//   - pi-subagents (child-session.js): open / create / inMemory as the
//     storage arg of createAgentSession.
// Read methods answer from a per-instance host snapshot; `createBranchedSession`
// forks root-to-leaf into a NEW session file and replaces the manager in place
// (upstream semantics: the same object continues on the new file). The write
// face (append*/branch/persist) is not in any extension's call surface and
// throws the standard unavailable error.
const PIR_SESSION_MANAGER_OP_PREFIX = "pirSession";

function pirSessionHostCall(op, args) {
  if (typeof __pir_host === "undefined" || typeof __pir_host[op] !== "function") {
    throw pirUnavailable("SessionManager", op);
  }
  return __pir_host[op](args);
}

function pirSessionParseEnvelope(payload, op) {
  let parsed;
  try {
    parsed = JSON.parse(payload);
  } catch (error) {
    throw new Error("SessionManager." + op + " host bridge returned invalid JSON");
  }
  if (!parsed || parsed.ok !== true) {
    throw new Error((parsed && parsed.error) || "SessionManager." + op + " failed");
  }
  return parsed;
}

function pirSessionHandleCall(manager, op, argsJson) {
  return pirSessionParseEnvelope(
    __pir_host[PIR_SESSION_MANAGER_OP_PREFIX + op + "Json"](manager._pirHandle, argsJson),
    op,
  );
}

function pirSessionSnapshotOf(manager) {
  if (manager._pirSnapshot) return manager._pirSnapshot;
  manager._pirSnapshot = pirSessionHandleCall(manager, "Snapshot").snapshot;
  return manager._pirSnapshot;
}

// Upstream `getBranch(fromId)` walks from ANY entry to the root; the host
// snapshot carries the current-leaf branch only, so a fromId argument walks
// the parent chain over the full entry list.
function pirSessionBranchOf(snapshot, fromId) {
  if (fromId === undefined || fromId === null) {
    return Array.isArray(snapshot.branch) ? snapshot.branch : [];
  }
  const byId = new Map();
  for (const entry of Array.isArray(snapshot.entries) ? snapshot.entries : []) {
    if (entry && entry.id !== undefined) byId.set(entry.id, entry);
  }
  const path = [];
  let current = byId.get(fromId);
  while (current) {
    path.push(current);
    current = current.parentId ? byId.get(current.parentId) : undefined;
  }
  path.reverse();
  return path;
}

// Upstream `getSessionName()`: the latest session_info entry's trimmed name
// (an empty name clears the title).
function pirSessionNameOf(snapshot) {
  const entries = Array.isArray(snapshot.entries) ? snapshot.entries : [];
  for (let i = entries.length - 1; i >= 0; i--) {
    const entry = entries[i];
    if (entry && entry.type === "session_info") {
      const name = entry.name;
      if (typeof name === "string" && name.trim()) return name.trim();
      return undefined;
    }
  }
  return undefined;
}

export class SessionManager {
  constructor(entriesOrOptions, _sessionDir, _cwdOverride) {
    // Upstream's constructor is private; extensions reach the class through
    // the static constructors. Inert storage keeps the construction shape.
    this._pirOptions = entriesOrOptions;
  }

  static _pirConstruct(kind, args) {
    const out = pirSessionParseEnvelope(
      pirSessionHostCall("pirSessionOpenJson", JSON.stringify(Object.assign({ kind: kind }, args))),
      kind,
    );
    const manager = new SessionManager(args);
    manager._pirHandle = out.handle;
    manager._pirSnapshot = out.snapshot;
    return manager;
  }

  /** Open a session file (upstream `open(path, sessionDir?, cwdOverride?)`). */
  static open(path, sessionDir, cwdOverride) {
    return SessionManager._pirConstruct("open", {
      path: String(path),
      sessionDir: sessionDir === undefined ? undefined : String(sessionDir),
      cwdOverride: cwdOverride === undefined ? undefined : String(cwdOverride),
    });
  }

  /** Create a session in `cwd` (upstream `create(cwd, sessionDir?, options?)`). */
  static create(cwd, sessionDir) {
    return SessionManager._pirConstruct("create", {
      cwd: cwd === undefined ? undefined : String(cwd),
      sessionDir: sessionDir === undefined ? undefined : String(sessionDir),
    });
  }

  /** In-memory session (upstream `inMemory(cwd, options?, entries?)`). */
  static inMemory(cwd) {
    return SessionManager._pirConstruct("inMemory", {
      cwd: cwd === undefined ? undefined : String(cwd),
    });
  }

  /** Continue the most recent session (upstream `continueRecent(cwd, sessionDir?)`). */
  static continueRecent(cwd, sessionDir) {
    return SessionManager._pirConstruct("continueRecent", {
      cwd: cwd === undefined ? undefined : String(cwd),
      sessionDir: sessionDir === undefined ? undefined : String(sessionDir),
    });
  }

  /** Fork a session into another project (upstream `forkFrom(source, targetCwd, sessionDir?)`). */
  static forkFrom(sourcePath, targetCwd, sessionDir) {
    return SessionManager._pirConstruct("forkFrom", {
      source: String(sourcePath),
      targetCwd: targetCwd === undefined ? undefined : String(targetCwd),
      sessionDir: sessionDir === undefined ? undefined : String(sessionDir),
    });
  }

  /** Exact-id discovery (upstream `findById(cwd, id, sessionDir?)`). */
  static findById(cwd, id, sessionDir) {
    const out = pirSessionParseEnvelope(
      pirSessionHostCall(
        "pirSessionFindByIdJson",
        JSON.stringify({
          cwd: cwd === undefined ? undefined : String(cwd),
          id: String(id),
          sessionDir: sessionDir === undefined ? undefined : String(sessionDir),
        }),
      ),
      "findById",
    );
    return out.path === null || out.path === undefined ? undefined : out.path;
  }

  /** Session discovery, most recently modified first (upstream `list`). */
  static async list(cwd, sessionDir) {
    const out = pirSessionParseEnvelope(
      pirSessionHostCall(
        "pirSessionListJson",
        JSON.stringify({
          kind: "list",
          cwd: cwd === undefined ? undefined : String(cwd),
          sessionDir: sessionDir === undefined ? undefined : String(sessionDir),
        }),
      ),
      "list",
    );
    return pirSessionInfos(out.sessions);
  }

  /** Discovery across every project directory (upstream `listAll`). */
  static async listAll(sessionDirOrProgress) {
    const out = pirSessionParseEnvelope(
      pirSessionHostCall(
        "pirSessionListJson",
        JSON.stringify({
          kind: "listAll",
          sessionDir: typeof sessionDirOrProgress === "string" ? String(sessionDirOrProgress) : undefined,
        }),
      ),
      "listAll",
    );
    return pirSessionInfos(out.sessions);
  }
}

// Upstream `SessionInfo` projection from the host discovery peek: `modified`
// arrives as epoch ms; `created` is not collected by the bounded scan and
// stays undefined; `allMessagesText` is not collected (upstream keeps it for
// fuzzy search only).
function pirSessionInfos(sessions) {
  const list = Array.isArray(sessions) ? sessions : [];
  return list.map((session) => ({
    path: session.path,
    id: session.id,
    cwd: session.cwd,
    name: session.name === null ? undefined : session.name,
    parentSessionPath: session.parentSessionPath === null ? undefined : session.parentSessionPath,
    created: undefined,
    modified: typeof session.modifiedMs === "number" ? new Date(session.modifiedMs) : undefined,
    messageCount: session.messageCount,
    firstMessage: session.firstMessage,
    allMessagesText: session.firstMessage,
  }));
}

const PIR_SESSION_MANAGER_READ_METHODS = [
  "getCwd", "getSessionDir", "getSessionId", "getSessionFile", "getSessionName",
  "getLeafId", "getLeafEntry", "getEntry", "getChildren", "getBranch", "getTree",
  "getLabel", "getEntries", "getHeader", "getEntryCount",
  "hasConversation", "isPersisted",
  "buildContextEntries", "buildSessionProjection", "buildSessionContext",
];

for (const name of PIR_SESSION_MANAGER_READ_METHODS) {
  if (SessionManager.prototype[name] !== undefined) continue;
  SessionManager.prototype[name] = function () {
    // A constructor-built instance has no host handle (upstream's constructor
    // is private; only the static constructors produce live managers).
    if (this._pirHandle === undefined) throw pirUnavailable("SessionManager", name);
    const snapshot = pirSessionSnapshotOf(this);
    switch (name) {
      case "getCwd": return snapshot.cwd === null ? undefined : snapshot.cwd;
      case "getSessionDir": return snapshot.sessionDir === null ? undefined : snapshot.sessionDir;
      case "getSessionId": return snapshot.sessionId === null ? undefined : snapshot.sessionId;
      case "getSessionFile": return snapshot.sessionFile === null ? undefined : snapshot.sessionFile;
      case "getSessionName": return pirSessionNameOf(snapshot);
      case "getLeafId": return snapshot.leafId === null ? null : snapshot.leafId;
      case "getLeafEntry": {
        const leafId = snapshot.leafId;
        if (leafId === null || leafId === undefined) return undefined;
        return this.getEntry(leafId);
      }
      case "getEntry": {
        const id = arguments[0];
        const entries = Array.isArray(snapshot.entries) ? snapshot.entries : [];
        const byId = new Map();
        for (const entry of entries) if (entry && entry.id !== undefined) byId.set(entry.id, entry);
        return byId.get(String(id));
      }
      case "getChildren": {
        const parentId = arguments[0] === undefined ? null : String(arguments[0]);
        const entries = Array.isArray(snapshot.entries) ? snapshot.entries : [];
        return entries.filter((entry) => entry && entry.parentId === parentId);
      }
      case "getBranch": return pirSessionBranchOf(snapshot, arguments[0]);
      case "getTree": return Array.isArray(snapshot.tree) ? snapshot.tree : [];
      case "getLabel": {
        const id = String(arguments[0]);
        const nodes = pirSessionTreeNodes(Array.isArray(snapshot.tree) ? snapshot.tree : []);
        for (const node of nodes) {
          const entry = node.entry !== undefined ? node.entry : node;
          if (entry.id === id) return node.label === undefined ? undefined : node.label;
        }
        return undefined;
      }
      case "getEntries": return Array.isArray(snapshot.entries) ? snapshot.entries : [];
      case "getHeader": return snapshot.header === null ? null : snapshot.header;
      case "getEntryCount": {
        const entries = Array.isArray(snapshot.entries) ? snapshot.entries : [];
        return entries.length;
      }
      case "hasConversation": {
        const entries = Array.isArray(snapshot.entries) ? snapshot.entries : [];
        return entries.some((entry) => {
          if (!entry || entry.type !== "message" || !entry.message) return false;
          return entry.message.role === "user" || entry.message.role === "assistant";
        });
      }
      case "isPersisted": return snapshot.isPersisted === true;
      case "buildContextEntries": return pirSessionBranchOf(snapshot, undefined);
      case "buildSessionProjection":
        return { entries: [], messages: pirSessionBranchOf(snapshot, undefined), thinkingLevel: "off", model: null };
      case "buildSessionContext":
        return buildSessionContext(pirSessionBranchOf(snapshot, undefined));
    }
  };
}

/**
 * Fork the root-to-leaf path into a NEW session file and continue the manager
 * on it (upstream `createBranchedSession(leafId)`). Returns the new session
 * file path, or undefined when not persisting. pi-phone's branch panel and
 * pi-subagents' fork-context both take the returned path as the session the
 * next run continues; the forked file lands in the manager's session dir
 * (for `open`ed managers that is the `sessionDir` override when one was given
 * — the nested "forks" directory pi-subagents passes).
 */
SessionManager.prototype.createBranchedSession = function (leafId) {
  if (this._pirHandle === undefined) {
    throw pirUnavailable("SessionManager", "createBranchedSession");
  }
  const out = pirSessionHandleCall(this, "CreateBranched", String(leafId));
  // The host replaced the manager state in place; drop the stale snapshot.
  this._pirSnapshot = undefined;
  return out.sessionFile === null || out.sessionFile === undefined ? undefined : out.sessionFile;
};

/**
 * Close the host handle early (pir addition; the upstream class relies on GC).
 * Every entry-point extension that opens panels repeatedly should close them,
 * so the bounded host registry never evicts a live handle.
 */
SessionManager.prototype.close = function () {
  if (this._pirHandle !== undefined && typeof __pir_host !== "undefined" && __pir_host.pirSessionClose) {
    __pir_host.pirSessionClose(this._pirHandle);
    this._pirHandle = undefined;
    this._pirSnapshot = undefined;
  }
};

function pirSessionTreeNodes(tree) {
  const nodes = [];
  const stack = Array.isArray(tree) ? tree.slice() : [];
  while (stack.length > 0) {
    const node = stack.pop();
    if (!node || typeof node !== "object") continue;
    nodes.push(node);
    if (Array.isArray(node.children)) {
      for (const child of node.children) stack.push(child);
    }
  }
  return nodes;
}

// Host theme snapshot colors ({role -> "#rrggbb"}, e.g. mdHeading); resolved
// lazily and cached. Headless runs carry no theme, so every lookup degrades to
// passthrough (documented degradation: no color, correct text).
let pirThemeColorsCache = undefined;
function pirThemeColors() {
  if (pirThemeColorsCache !== undefined) return pirThemeColorsCache;
  pirThemeColorsCache = null;
  try {
    const raw = typeof __pir_host !== "undefined" && __pir_host.getThemeJson ? __pir_host.getThemeJson() : null;
    if (raw) {
      const parsed = JSON.parse(raw);
      if (parsed && typeof parsed === "object" && parsed.colors && typeof parsed.colors === "object") {
        pirThemeColorsCache = parsed.colors;
      }
    }
  } catch { /* headless: passthrough */ }
  return pirThemeColorsCache;
}

// 24-bit SGR foreground from the host theme; passthrough when the role has no
// resolved color (upstream theme.fg falls back the same way for missing roles).
function pirAnsiFg(role, text) {
  const colors = pirThemeColors();
  const hex = colors && typeof colors[role] === "string" ? colors[role] : null;
  const match = hex ? /^#([0-9a-fA-F]{6})$/.exec(hex) : null;
  if (!match) return String(text);
  const value = parseInt(match[1], 16);
  return `\u001b[38;2;${(value >> 16) & 255};${(value >> 8) & 255};${value & 255}m${text}\u001b[39m`;
}

function pirStrikethrough(text) { return `\u001b[9m${text}\u001b[29m`; }

// Upstream formats through the live KeybindingsManager; pir keeps the default
// keys of the bindings extensions actually ask about, and returns "" for
// unknown names so callers take their documented fallback branch.
const PIR_DEFAULT_KEYBINDINGS = {
  "app.tools.expand": "ctrl+o",
  "app.message.copy": "ctrl+x",
  "tui.select.up": "up",
  "tui.select.down": "down",
  "tui.select.confirm": "enter",
  "tui.select.cancel": "escape/ctrl+c",
};

function pirFormatKeyText(key) {
  const isDarwin = typeof process !== "undefined" && process && process.platform === "darwin";
  return String(key ?? "")
    .split("/")
    .map((part) => part
      .split("+")
      .map((piece) => (isDarwin && piece.toLowerCase() === "alt" ? "option" : piece))
      .join("+"))
    .join("/");
}

export function keyText(keybinding) {
  const name = String(keybinding ?? "");
  // hasOwnProperty guard: names like "toString" must not hit the prototype.
  const keys = Object.prototype.hasOwnProperty.call(PIR_DEFAULT_KEYBINDINGS, name)
    ? PIR_DEFAULT_KEYBINDINGS[name]
    : undefined;
  return keys === undefined ? "" : pirFormatKeyText(keys);
}

export function keyHint(keybinding, description) {
  return pirAnsiFg("dim", keyText(keybinding)) + pirAnsiFg("muted", ` ${description}`);
}

export function rawKeyHint(key, description) {
  return pirAnsiFg("dim", pirFormatKeyText(key)) + pirAnsiFg("muted", ` ${description}`);
}

function pirReadOnlyParameters(properties, required) {

        path: { type: "string", description: "Directory to list." },
        limit: { type: "number", description: "Maximum number of entries." },
      }, []),
    },
  ];
}

// YAML-subset frontmatter parsing (upstream uses the real `yaml` package):
// scalar `key: value` pairs, quoted strings, booleans/null/numbers, inline
// [flow, lists] and `- item` block lists. Nested maps are not supported.
function pirParseScalar(raw) {
  const value = String(raw).trim();
  const quoted = /^"(.*)"$/.exec(value) || /^'(.*)'$/.exec(value);
  if (quoted) return quoted[1];
  if (value === "true") return true;
  if (value === "false") return false;
  if (value === "null" || value === "~") return null;
  if (/^-?\d+$/.test(value)) return parseInt(value, 10);
  if (/^-?\d+\.\d+$/.test(value)) return parseFloat(value);
  const flow = /^\[(.*)\]$/.exec(value);
  if (flow) {
    return flow[1]
      .split(",")
      .map((part) => part.trim())
      .filter((part) => part.length > 0)
      .map((part) => pirParseScalar(part));
  }
  return value;
}

function pirParseYamlSubset(text) {
  const result = {};
  let currentKey = null;
  for (const rawLine of String(text ?? "").split("\n")) {
    const line = rawLine.replace(/\s+$/, "");
    const trimmed = line.trim();
    if (trimmed === "" || trimmed.startsWith("#")) continue;
    const listItem = /^\s*-\s+(.*)$/.exec(line);
    if (listItem && currentKey) {
      if (!Array.isArray(result[currentKey])) result[currentKey] = [];
      result[currentKey].push(pirParseScalar(listItem[1]));
      continue;
    }
    const pair = /^([A-Za-z0-9_.\-]+):\s*(.*)$/.exec(line);
    if (!pair) continue;
    currentKey = pair[1];
    const value = pair[2].trim();
    result[currentKey] = value === "" ? [] : pirParseScalar(value);
  }
  return result;
}

export function parseFrontmatter(content) {
  const normalized = String(content ?? "").replace(/\r\n/g, "\n").replace(/\r/g, "\n").replace(/^\ufeff/, "");
  if (!normalized.startsWith("---")) {
    return { frontmatter: {}, body: normalized };
  }
  const endIndex = normalized.indexOf("\n---", 3);
  if (endIndex === -1) {
    return { frontmatter: {}, body: normalized };
  }
  return {
    frontmatter: pirParseYamlSubset(normalized.slice(4, endIndex)),
    body: normalized.slice(endIndex + 4).trim(),
  };
}

export function stripFrontmatter(content) {
  return parseFrontmatter(content).body;
}

// chars/4 heuristic (upstream compaction.ts): text/thinking/toolCall blocks
// counted exactly, images at the fixed upstream-estimated weight.
export function estimateTokens(message) {
  if (!message || typeof message !== "object") return 0;
  let chars = 0;
  const addContent = (content) => {
    if (typeof content === "string") {
      chars += content.length;
      return;
    }
    if (!Array.isArray(content)) return;
    for (const block of content) {
      if (!block || typeof block !== "object") continue;
      if (block.type === "text" && typeof block.text === "string") chars += block.text.length;
      else if (block.type === "thinking" && typeof block.thinking === "string") chars += block.thinking.length;
      else if (block.type === "toolCall") chars += String(block.name ?? "").length + JSON.stringify(block.arguments ?? {}).length;
      else if (block.type === "image") chars += 4800;
    }
  };
  if (message.role === "system") {
    addContent(message.content);
    if (message.sections) {
      for (const section of Object.values(message.sections)) {
        if (typeof section === "string") chars += section.length;
      }
    }
    return Math.ceil(chars / 4);
  }
  if (message.role === "bashExecution") {
    chars += String(message.command ?? "").length + String(message.output ?? "").length;
    return Math.ceil(chars / 4);
  }
  addContent(message.content);
  return Math.ceil(chars / 4);
}

export function calculateContextTokens(usage) {
  if (!usage || typeof usage !== "object") return 0;
  return usage.totalTokens || ((usage.input ?? 0) + (usage.output ?? 0) + (usage.cacheRead ?? 0) + (usage.cacheWrite ?? 0));
}

// The QuickJS engine ships no AbortController; extension code (pi-subagents
// constructs one during register) expects the Node/Web global. Installed
// here, guarded by typeof, so any module importing this package gets it
// before its own body evaluates.
if (typeof globalThis.AbortController !== "function") {
  class PirAbortSignal {
    constructor() {
      this.aborted = false;
      this.reason = undefined;
      this.onabort = null;
      this._listeners = [];
    }

    addEventListener(type, listener) {
      if (type === "abort" && typeof listener === "function") this._listeners.push(listener);
    }

    removeEventListener(type, listener) {
      if (type !== "abort") return;
      const index = this._listeners.indexOf(listener);
      if (index !== -1) this._listeners.splice(index, 1);
    }

    throwIfAborted() {
      if (!this.aborted) return;
      throw this.reason !== undefined ? this.reason : new Error("This operation was aborted");
    }

    pirDispatchAbort() {
      const event = { type: "abort", target: this, currentTarget: this };
      if (typeof this.onabort === "function") this.onabort(event);
      for (const listener of this._listeners.slice()) listener(event);
    }

    static abort(reason) {
      const signal = new PirAbortSignal();
      signal.aborted = true;
      signal.reason = reason !== undefined ? reason : new Error("This operation was aborted");
      return signal;
    }

    static any(signals) {
      const combined = new PirAbortSignal();
      const list = Array.isArray(signals) ? signals : [];
      for (const source of list) {
        if (!source) continue;
        if (source.aborted) {
          combined.aborted = true;
          combined.reason = source.reason;
          return combined;
        }
        source.addEventListener("abort", () => {
          if (combined.aborted) return;
          combined.aborted = true;
          combined.reason = source.reason;
          combined.pirDispatchAbort();
        });
      }
      return combined;
    }
  }

  class PirAbortController {
    constructor() {
      this.signal = new PirAbortSignal();
    }

    abort(reason) {
      if (this.signal.aborted) return;
      this.signal.aborted = true;
      this.signal.reason = reason !== undefined ? reason : new Error("This operation was aborted");
      this.signal.pirDispatchAbort();
    }
  }

  globalThis.AbortSignal = PirAbortSignal;
  globalThis.AbortController = PirAbortController;
}

function pirUtf8ByteLength(str) {
  let bytes = 0;
  for (let i = 0; i < str.length; i++) {
    const code = str.charCodeAt(i);
    if (code < 0x80) bytes += 1;
    else if (code < 0x800) bytes += 2;
    else if (code >= 0xd800 && code <= 0xdbff) { bytes += 4; i++; }
    else bytes += 3;
  }
  return bytes;
}

function pirStripBom(text) {
  return String(text ?? "").replace(/^\ufeff/, "");
}

function pirBase64DecodeBytes(b64) {
  const CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
  const clean = String(b64 ?? "").replace(/[^A-Za-z0-9+/]/g, "");
  const out = [];
  let buffer = 0;
  let bits = 0;
  for (let i = 0; i < clean.length; i++) {
    buffer = (buffer << 6) | CHARS.indexOf(clean[i]);
    bits += 6;
    if (bits >= 8) {
      bits -= 8;
      out.push((buffer >> bits) & 0xff);
    }
  }
  return out;
}

function pirUtf8Decode(bytes) {
  let out = "";
  for (let i = 0; i < bytes.length; ) {
    const b0 = bytes[i];
    let cp;
    if (b0 < 0x80) { cp = b0; i += 1; }
    else if (b0 < 0xe0) { cp = ((b0 & 0x1f) << 6) | (bytes[i + 1] & 0x3f); i += 2; }
    else if (b0 < 0xf0) { cp = ((b0 & 0x0f) << 12) | ((bytes[i + 1] & 0x3f) << 6) | (bytes[i + 2] & 0x3f); i += 3; }
    else { cp = ((b0 & 0x07) << 18) | ((bytes[i + 1] & 0x3f) << 12) | ((bytes[i + 2] & 0x3f) << 6) | (bytes[i + 3] & 0x3f); i += 4; }
    if (cp > 0xffff) {
      cp -= 0x10000;
      out += String.fromCharCode(0xd800 + (cp >> 10), 0xdc00 + (cp & 0x3ff));
    } else {
      out += String.fromCharCode(cp);
    }
  }
  return out;
}

// Synchronous host filesystem access through the shared `__pir_host.node`
// bridge (the node fs shims use the same op table). Every helper degrades to
// undefined on any failure: headless/test hosts without the bridge, paths
// outside the sandbox roots, or plain missing files.
function pirHostNodeCall(moduleName, op, payload) {
  if (typeof globalThis.__pirNodeCall === "function") {
    return globalThis.__pirNodeCall(moduleName, op, payload);
  }
  if (typeof __pir_host !== "undefined" && __pir_host && typeof __pir_host.node === "function") {
    const res = JSON.parse(__pir_host.node(moduleName, op, JSON.stringify(payload === undefined ? {} : payload)));
    if (res && res.ok === true) return res.value;
    throw new Error((res && res.message) || "host node call failed: " + moduleName + "." + op);
  }
  throw new Error("host node bridge is not available");
}

function pirFsReadBytes(path) {
  try {
    const value = pirHostNodeCall("fs", "readFile", { path: String(path) });
    return typeof value === "string" ? pirBase64DecodeBytes(value) : undefined;
  } catch {
    return undefined;
  }
}

function pirFsReadText(path) {
  const bytes = pirFsReadBytes(path);
  return bytes === undefined ? undefined : pirUtf8Decode(bytes);
}

  try {

    return undefined;
  }
}

// --- fs write face (settings persistence) ----------------------------------
//
// The host fs bridge takes base64 payloads, so the encoder lives here (the
// decoder above is the read-side counterpart). Writes go through the same
// `__pir_host.node` op table as every other shim, which means the host's
// sandbox roots apply unchanged: a settings path outside the pi config root /
// session cwd is rejected by the host, not silently written.
const PIR_BASE64_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

function pirUtf8Encode(str) {
  const bytes = [];
  const text = String(str ?? "");
  for (let i = 0; i < text.length; i++) {
    let code = text.charCodeAt(i);
    if (code >= 0xd800 && code <= 0xdbff && i + 1 < text.length) {
      const next = text.charCodeAt(i + 1);
      if (next >= 0xdc00 && next <= 0xdfff) {
        code = 0x10000 + ((code - 0xd800) << 10) + (next - 0xdc00);
        i++;
      }
    }
    if (code < 0x80) bytes.push(code);
    else if (code < 0x800) bytes.push(0xc0 | (code >> 6), 0x80 | (code & 0x3f));
    else if (code < 0x10000) bytes.push(0xe0 | (code >> 12), 0x80 | ((code >> 6) & 0x3f), 0x80 | (code & 0x3f));
    else bytes.push(
      0xf0 | (code >> 18),
      0x80 | ((code >> 12) & 0x3f),
      0x80 | ((code >> 6) & 0x3f),
      0x80 | (code & 0x3f),
    );
  }
  return bytes;
}

function pirBase64EncodeBytes(bytes) {
  let out = "";
  for (let i = 0; i < bytes.length; i += 3) {
    const b0 = bytes[i];
    const b1 = i + 1 < bytes.length ? bytes[i + 1] : undefined;
    const b2 = i + 2 < bytes.length ? bytes[i + 2] : undefined;
    out += PIR_BASE64_CHARS[b0 >> 2];
    out += PIR_BASE64_CHARS[((b0 & 0x03) << 4) | (b1 === undefined ? 0 : b1 >> 4)];
    out += b1 === undefined ? "=" : PIR_BASE64_CHARS[((b1 & 0x0f) << 2) | (b2 === undefined ? 0 : b2 >> 6)];
    out += b2 === undefined ? "=" : PIR_BASE64_CHARS[b2 & 0x3f];
  }
  return out;
}

function pirFsMkdir(path) {
  try {
    pirHostNodeCall("fs", "mkdir", { path: String(path), recursive: true });
    return true;
  } catch {
    return false;
  }
}

// Unpaired surrogate code unit anywhere in `text` — the fs shim's predicate
// (`pirHasLoneSurrogate` in fs_shim.js.txt), and the only thing that keeps a
// string write off the text channel here: this call site is always utf8.
function pirHasLoneSurrogate(text) {
  for (let i = 0; i < text.length; i++) {
    const code = text.charCodeAt(i);
    if (code >= 0xd800 && code <= 0xdbff) {
      const next = text.charCodeAt(i + 1);
      if (next >= 0xdc00 && next <= 0xdfff) {
        i++; // a well-formed pair encodes as one character
        continue;
      }
      return true;
    }
    if (code >= 0xdc00 && code <= 0xdfff) return true;
  }
  return false;
}

/** Write text atomically (temp sibling + rename). Throws the host error. */
function pirFsWriteText(path, text) {
  const target = String(path);
  const dir = pirDirname(target);
  if (dir) pirFsMkdir(dir);
  const temp = target + ".pir-tmp-" + Date.now().toString(36) + Math.floor(Math.random() * 0x10000).toString(36);
  // Text channel first, the fs shim's `pirWriteData` guard inlined (`dataText`
  // is the exact UTF-8 image of the string, no base64 round trip on either
  // leg). Text with no lone surrogate encodes under `pirUtf8Encode` to exactly
  // those standard UTF-8 bytes (a well-formed pair becomes one 4-byte
  // sequence), so both channels write the same bytes; a lone surrogate pins
  // the old base64 expression instead — its WTF-8 bytes cannot ride
  // `dataText` anyway (JSON.stringify escapes the unpaired unit and the
  // host's JSON payload parser rejects that escape). Non-string `text`
  // coerces exactly as `pirUtf8Encode` coerced it (`String(text ?? "")`).
  const data = String(text ?? "");
  const payload = { path: temp };
  if (pirHasLoneSurrogate(data)) payload.data = pirBase64EncodeBytes(pirUtf8Encode(data));
  else payload.dataText = data;
  pirHostNodeCall("fs", "writeFile", payload);
  try {
    pirHostNodeCall("fs", "rename", { from: temp, to: target });
  } catch (error) {
    try {
      pirHostNodeCall("fs", "unlink", { path: temp });
    } catch { /* best-effort temp cleanup */ }
    throw error;
  }
}

/** JSON-only deep clone (`structuredClone` is not part of this runtime). */
function pirDeepCloneJson(value) {
  if (value === null || typeof value !== "object") return value;
  if (Array.isArray(value)) return value.map(pirDeepCloneJson);
  const out = {};
  for (const key of Object.keys(value)) out[key] = pirDeepCloneJson(value[key]);
  return out;
}

function pirIsPlainObject(value) {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function pirIsAbsolutePath(p) {
  return /^([A-Za-z]:[\\/]|[\\/])/.test(p);
}

function pirToPosixPath(p) {
  return String(p).replace(/\\/g, "/");
}

function pirJoinPath() {
  const parts = [];
  for (let i = 0; i < arguments.length; i++) {
    const part = String(arguments[i] ?? "");
    if (part.length > 0) parts.push(part);
  }
  return parts.join("/").replace(/\/+/g, "/");
}

function pirDirname(p) {
  const normalized = pirToPosixPath(p).replace(/\/+$/, "");
  const index = normalized.lastIndexOf("/");
  if (index < 0) return "";
  if (index === 0) return "/";
  if (/^[A-Za-z]:$/.test(normalized.slice(0, index))) return normalized.slice(0, index) + "/";
  return normalized.slice(0, index);
}

function pirBasename(p) {
  const normalized = pirToPosixPath(p).replace(/\/+$/, "");
  const index = normalized.lastIndexOf("/");
  return index < 0 ? normalized : normalized.slice(index + 1);
}

function pirResolvePath(p, base) {
  const raw = String(p ?? "");
  const combined = pirIsAbsolutePath(raw) ? raw : pirJoinPath(base ?? ".", raw);
  return pirToPosixPath(combined);
}

function pirFindOnPath(executable) {
  const env = typeof process !== "undefined" && process ? process.env ?? {} : {};
  const pathValue = env.PATH ?? env.Path ?? env.path ?? "";
  const separator = pathValue.indexOf(";") !== -1 ? ";" : ":";
  for (const dir of String(pathValue).split(separator)) {
    if (!dir) continue;
    const candidate = pirJoinPath(dir, executable);
    if (pirFsExists(candidate)) return candidate;
  }
  return null;
}

/**
 * Project one selected session entry into LLM/runtime messages
 * (upstream session-manager.ts sessionEntryToContextMessages).
 */
export function sessionEntryToContextMessages(entry) {
  if (!entry || typeof entry !== "object") return [];
  if (entry.type === "message") {
    const message = entry.message;
    if (!message || typeof message !== "object") return [];
    if (message.role === "system" && message.content == null) return [Object.assign({}, message, { content: "" })];
    if (
      (message.role === "user" || message.role === "assistant" || message.role === "toolResult") &&
      message.content == null
    ) {
      return [Object.assign({}, message, { content: [] })];
    }
    return [message];
  }
  if (entry.type === "custom_message") {
    return [pirCreateCustomMessage(entry.customType, entry.content ?? [], entry.display, entry.details, entry.timestamp)];
  }
  if (entry.type === "branch_summary" && entry.summary) {
    return [pirCreateBranchSummaryMessage(entry.summary, entry.fromId, entry.timestamp)];
  }
  if (entry.type === "compaction") {
    const summary = pirCreateCompactionSummaryMessage(entry.summary, entry.tokensBefore, entry.timestamp);
    return entry.systemMessage ? [entry.systemMessage, summary] : [summary];
  }
  return [];
}

function pirBuildEntryIndex(entries, byId) {
  if (byId) return byId;
  const index = new Map();
  for (const entry of entries) index.set(entry.id, entry);
  return index;
}

function pirBuildSessionPath(entries, leafId, byId) {
  const index = pirBuildEntryIndex(entries, byId);
  let leaf;
  if (leafId === null) return [];
  if (typeof leafId === "string" && leafId.length > 0) leaf = index.get(leafId);
  if (!leaf) leaf = entries[entries.length - 1];
  if (!leaf) return [];
  const path = [];
  let current = leaf;
  while (current) {
    path.push(current);
    current = current.parentId ? index.get(current.parentId) : undefined;
  }
  path.reverse();
  return path;
}

function pirGetSessionContextSettings(path) {
  let thinkingLevel = "off";
  let model = null;
  for (const entry of path) {
    if (entry.type === "thinking_level_change") thinkingLevel = entry.thinkingLevel;
    else if (entry.type === "model_change") model = { provider: entry.provider, modelId: entry.modelId };
    else if (entry.type === "message" && entry.message && entry.message.role === "assistant") {
      model = { provider: entry.message.provider, modelId: entry.message.model };
    }
  }
  return { thinkingLevel, model };
}

/**
 * Build the active, compaction-aware session entry list
 * (upstream session-manager.ts buildContextEntries).
 */
export function buildContextEntries(entries, leafId, byId) {
  const list = Array.isArray(entries) ? entries : [];
  const path = pirBuildSessionPath(list, leafId, byId);
  let compaction = null;
  for (const entry of path) {
    if (entry.type === "compaction") compaction = entry;
  }
  if (!compaction) return path;
  const compactionIdx = path.findIndex((entry) => entry.id === compaction.id);
  if (compactionIdx < 0) return path;
  const contextEntries = [compaction];
  let foundFirstKept = false;
  for (let i = 0; i < compactionIdx; i++) {
    const entry = path[i];
    if (entry.id === compaction.firstKeptEntryId) foundFirstKept = true;
    if (foundFirstKept && !(entry.type === "message" && entry.message && entry.message.role === "system")) {
      contextEntries.push(entry);
    }
  }
  for (let i = compactionIdx + 1; i < path.length; i++) contextEntries.push(path[i]);
  return contextEntries;
}

function pirProjectContextEntry(entry, edit) {
  const messages = sessionEntryToContextMessages(entry);
  if (!edit) return messages;
  const replacement = edit.replacement;
  if (replacement === null) return [];
  return messages.map((message) => {
    if (
      message.role !== "user" && message.role !== "assistant" &&
      message.role !== "toolResult" && message.role !== "custom"
    ) {
      return message;
    }
    const content =
      (message.role === "assistant" || message.role === "toolResult") && typeof replacement.content === "string"
        ? [{ type: "text", text: replacement.content }]
        : replacement.content;
    return Object.assign({}, message, { content });
  });
}

/**
 * Build provenance-preserving, compaction-aware model context
 * (upstream session-manager.ts buildSessionProjection).
 */
export function buildSessionProjection(entries, leafId, byId) {
  const list = Array.isArray(entries) ? entries : [];
  const path = pirBuildSessionPath(list, leafId, byId);
  const settings = pirGetSessionContextSettings(path);
  const contextEntries = buildContextEntries(list, leafId, byId);
  const edits = new Map();
  for (const entry of contextEntries) {
    if (entry.type === "context_edit") edits.set(entry.targetId, entry);
  }
  const projectedEntries = contextEntries.map((sourceEntry, index) => ({
    sourceEntry,
    messages:
      sourceEntry.type === "compaction" && index > 0 ? [] : pirProjectContextEntry(sourceEntry, edits.get(sourceEntry.id)),
  }));
  const messages = [];
  for (const entry of projectedEntries) {
    for (const message of entry.messages) messages.push(message);
  }
  return { entries: projectedEntries, messages, thinkingLevel: settings.thinkingLevel, model: settings.model };
}

/**
 * Parse JSONL session content (upstream session-manager.ts parseSessionEntries).
 * Malformed lines are skipped.
 */
export function parseSessionEntries(content) {
  const entries = [];
  const lines = String(content ?? "").trim().split("\n");
  for (const line of lines) {
    if (!line.trim()) continue;
    try {
      entries.push(JSON.parse(line));
    } catch {
      // Skip malformed lines.
    }
  }
  return entries;
}

function pirGenerateShortId() {
  let id = "";
  for (let i = 0; i < 8; i++) id += "0123456789abcdef"[Math.floor(Math.random() * 16)];
  return id;
}

/**
 * Migrate v1 -> v2 (id/parentId tree) and v2 -> v3 (hookMessage role rename)
 * in place (upstream session-manager.ts migrateSessionEntries).
 */
export function migrateSessionEntries(entries) {
  const list = Array.isArray(entries) ? entries : [];
  let header;
  for (const entry of list) {
    if (entry && entry.type === "session") { header = entry; break; }
  }
  const version = (header && typeof header.version === "number" ? header.version : 1);
  if (version >= CURRENT_SESSION_VERSION) return;
  if (version < 2) {
    let prevId = null;
    for (const entry of list) {
      if (!entry || typeof entry !== "object") continue;
      if (entry.type === "session") { entry.version = 2; continue; }
      entry.id = pirGenerateShortId();
      entry.parentId = prevId;
      prevId = entry.id;
      if (entry.type === "compaction" && typeof entry.firstKeptEntryIndex === "number") {
        const targetEntry = list[entry.firstKeptEntryIndex];
        if (targetEntry && targetEntry.type !== "session") entry.firstKeptEntryId = targetEntry.id;
        delete entry.firstKeptEntryIndex;
      }
    }
  }
  if (version < 3) {
    for (const entry of list) {
      if (!entry || typeof entry !== "object") continue;
      if (entry.type === "session") { entry.version = 3; continue; }
      if (entry.type === "message" && entry.message && entry.message.role === "hookMessage") {
        entry.message.role = "custom";
      }
    }
  }
}

/** Latest compaction entry on the list, or null (upstream session-manager.ts). */
export function getLatestCompactionEntry(entries) {
  const list = Array.isArray(entries) ? entries : [];
  for (let i = list.length - 1; i >= 0; i--) {
    if (list[i] && list[i].type === "compaction") return list[i];
  }
  return null;
}

// --- token estimates for the transplanted compaction logic -----------------

function pirEstimateTextAndImageChars(content) {
  if (typeof content === "string") return content.length;
  if (!Array.isArray(content)) return 0;
  let chars = 0;
  for (const block of content) {
    if (!block || typeof block !== "object") continue;
    if (block.type === "text" && typeof block.text === "string") chars += block.text.length;
    else if (block.type === "image") chars += PIR_ESTIMATED_IMAGE_CHARS;
  }
  return chars;
}

/** Full upstream estimateTokens (compaction.ts) including summary message roles. */
function pirEstimateTokensFull(message) {
  if (!message || typeof message !== "object") return 0;
  let chars = 0;
  switch (message.role) {
    case "system": {
      chars = pirEstimateTextAndImageChars(message.content);
      if (message.sections) {
        for (const section of Object.values(message.sections)) {
          if (typeof section === "string" && section) chars += section.length;
        }
      }
      if (message.toolsAdded) chars += JSON.stringify(message.toolsAdded).length;
      return Math.ceil(chars / 4);
    }
    case "user":
    case "custom":
    case "toolResult":
      return Math.ceil(pirEstimateTextAndImageChars(message.content) / 4);
    case "assistant": {
      if (!Array.isArray(message.content)) return 0;
      for (const block of message.content) {
        if (!block || typeof block !== "object") continue;
        if (block.type === "text" && typeof block.text === "string") chars += block.text.length;
        else if (block.type === "thinking" && typeof block.thinking === "string") chars += block.thinking.length;
        else if (block.type === "toolCall") chars += String(block.name ?? "").length + JSON.stringify(block.arguments ?? {}).length;
      }
      return Math.ceil(chars / 4);
    }
    case "bashExecution":
      return Math.ceil((String(message.command ?? "").length + String(message.output ?? "").length) / 4);
    case "branchSummary":
    case "compactionSummary":
      return Math.ceil(String(message.summary ?? "").length / 4);
  }
  return 0;
}

function pirIsCutPointMessage(message) {
  switch (message.role) {
    case "user":
    case "assistant":
    case "bashExecution":
    case "custom":
    case "branchSummary":
    case "compactionSummary":
      return true;
  }
  return false;
}

function pirIsTurnStartMessage(message) {
  switch (message.role) {
    case "user":
    case "bashExecution":
    case "custom":
    case "branchSummary":
    case "compactionSummary":
      return true;
  }
  return false;
}

function pirIsTurnStartEntry(entry) {
  if (!entry || entry.type === "compaction") return false;
  return sessionEntryToContextMessages(entry).some(pirIsTurnStartMessage);
}

function pirFindValidCutPoints(entries, startIndex, endIndex) {
  const cutPoints = [];
  for (let i = startIndex; i < endIndex; i++) {
    const entry = entries[i];
    if (!entry || entry.type === "compaction") continue;
    if (sessionEntryToContextMessages(entry).some(pirIsCutPointMessage)) cutPoints.push(i);
  }
  return cutPoints;
}

/**
 * Find the user-role entry that starts the turn containing entryIndex
 * (upstream compaction.ts findTurnStartIndex). -1 when none found.
 */
export function findTurnStartIndex(entries, entryIndex, startIndex) {
  const list = Array.isArray(entries) ? entries : [];
  for (let i = entryIndex; i >= startIndex; i--) {
    if (list[i] && pirIsTurnStartEntry(list[i])) return i;
  }
  return -1;
}

/**
 * Find the cut point that keeps approximately keepRecentTokens
 * (upstream compaction.ts findCutPoint).
 */
export function findCutPoint(entries, startIndex, endIndex, keepRecentTokens) {
  const list = Array.isArray(entries) ? entries : [];
  const cutPoints = pirFindValidCutPoints(list, startIndex, endIndex);
  if (cutPoints.length === 0) {
    return { firstKeptEntryIndex: startIndex, turnStartIndex: -1, isSplitTurn: false };
  }
  let accumulatedTokens = 0;
  let cutIndex = cutPoints[0];
  for (let i = endIndex - 1; i >= startIndex; i--) {
    const entry = list[i];
    const messageTokens = sessionEntryToContextMessages(entry).reduce(
      (sum, message) => sum + pirEstimateTokensFull(message),
      0,
    );
    if (messageTokens === 0) continue;
    accumulatedTokens += messageTokens;
    if (accumulatedTokens >= keepRecentTokens) {
      cutIndex = cutPoints.find((candidate) => candidate >= i) ?? cutPoints[cutPoints.length - 1];
      break;
    }
  }
  while (cutIndex > startIndex) {
    const prevEntry = list[cutIndex - 1];
    if (!prevEntry || prevEntry.type === "compaction" || sessionEntryToContextMessages(prevEntry).length > 0) break;
    cutIndex--;
  }
  const cutEntry = list[cutIndex];
  const startsTurn = cutEntry ? pirIsTurnStartEntry(cutEntry) : false;
  const turnStartIndex = startsTurn ? -1 : findTurnStartIndex(list, cutIndex, startIndex);
  return {
    firstKeptEntryIndex: cutIndex,
    turnStartIndex,
    isSplitTurn: !startsTurn && turnStartIndex !== -1,
  };
}

/** Whether compaction should trigger (upstream compaction.ts shouldCompact). */
export function shouldCompact(contextTokens, contextWindow, settings) {
  if (!settings || !settings.enabled) return false;
  return contextTokens > contextWindow - settings.reserveTokens;
}

/**
 * Find the last valid assistant message usage from session entries
 * (upstream compaction.ts getLastAssistantUsage). Undefined when absent.
 */
export function getLastAssistantUsage(entries) {
  const list = Array.isArray(entries) ? entries : [];
  for (let i = list.length - 1; i >= 0; i--) {
    const entry = list[i];
    if (!entry || entry.type !== "message") continue;
    const message = entry.message;
    if (
      message && message.role === "assistant" && message.usage &&
      message.stopReason !== "aborted" && message.stopReason !== "error" &&
      calculateContextTokens(message.usage) > 0
    ) {
      return message.usage;
    }
  }
  return undefined;
}

// --- branch summarization (branch-summarization.ts ports) ------------------

function pirGetMessageFromEntry(entry) {
  if (!entry || typeof entry !== "object") return undefined;
  switch (entry.type) {
    case "message":
      if (!entry.message || entry.message.role === "toolResult") return undefined;
      return entry.message;
    case "custom_message":
      return pirCreateCustomMessage(entry.customType, entry.content, entry.display, entry.details, entry.timestamp);
    case "branch_summary":
      return pirCreateBranchSummaryMessage(entry.summary, entry.fromId, entry.timestamp);
    case "compaction":
      return pirCreateCompactionSummaryMessage(entry.summary, entry.tokensBefore, entry.timestamp);
    default:
      return undefined;
  }
}

function pirExtractFileOps(message, fileOps) {
  if (!message || typeof message !== "object") return;
  if (message.role === "toolResult") {
    const nested = message.nestedCalls && Array.isArray(message.nestedCalls.calls) ? message.nestedCalls.calls : [];
    for (const call of nested) pirAddFileOp(call && call.name, call && call.arguments, fileOps);
    return;
  }
  if (message.role !== "assistant" || !Array.isArray(message.content)) return;
  for (const block of message.content) {
    if (!block || typeof block !== "object" || block.type !== "toolCall") continue;
    pirAddFileOp(block.name, block.arguments, fileOps);
  }
}

function pirAddFileOp(toolName, args, fileOps) {
  const path = args && typeof args.path === "string" ? args.path : undefined;
  if (!path) return;
  if (toolName === "read") fileOps.read.add(path);
  else if (toolName === "write") fileOps.written.add(path);
  else if (toolName === "edit") fileOps.edited.add(path);
}

/**
 * Prepare entries for summarization with a token budget
 * (upstream branch-summarization.ts prepareBranchEntries).
 */
export function prepareBranchEntries(entries, tokenBudget) {
  const list = Array.isArray(entries) ? entries : [];
  const budget = typeof tokenBudget === "number" ? tokenBudget : 0;
  const messages = [];
  const fileOps = { read: new Set(), written: new Set(), edited: new Set() };
  let totalTokens = 0;
  for (const entry of list) {
    if (entry && entry.type === "branch_summary" && !entry.fromHook && entry.details) {
      const details = entry.details;
      if (Array.isArray(details.readFiles)) {
        for (const f of details.readFiles) fileOps.read.add(f);
      }
      if (Array.isArray(details.modifiedFiles)) {
        for (const f of details.modifiedFiles) fileOps.edited.add(f);
      }
    }
  }
  for (let i = list.length - 1; i >= 0; i--) {
    const entry = list[i];
    const message = pirGetMessageFromEntry(entry);
    if (!message) continue;
    pirExtractFileOps(message, fileOps);
    const tokens = pirEstimateTokensFull(message);
    if (budget > 0 && totalTokens + tokens > budget) {
      if (entry && (entry.type === "compaction" || entry.type === "branch_summary")) {
        if (totalTokens < budget * 0.9) {
          messages.unshift(message);
          totalTokens += tokens;
        }
      }
      break;
    }
    messages.unshift(message);
    totalTokens += tokens;
  }
  return { messages, fileOps, totalTokens };
}

/**
 * Collect the entries to summarize when navigating from oldLeafId to targetId
 * (upstream branch-summarization.ts collectEntriesForBranchSummary). Requires
 * a session-manager-like object exposing getBranch/getEntry.
 */
export function collectEntriesForBranchSummary(session, oldLeafId, targetId) {
  if (!session || typeof session.getBranch !== "function" || typeof session.getEntry !== "function") {
    throw new Error("collectEntriesForBranchSummary requires a session manager exposing getBranch and getEntry");
  }
  if (!oldLeafId) return { entries: [], commonAncestorId: null };
  const oldPath = new Set(session.getBranch(oldLeafId).map((entry) => entry.id));
  const targetPath = session.getBranch(targetId);
  let commonAncestorId = null;
  for (let i = targetPath.length - 1; i >= 0; i--) {
    if (oldPath.has(targetPath[i].id)) {
      commonAncestorId = targetPath[i].id;
      break;
    }
  }
  const entries = [];
  let current = oldLeafId;
  while (current && current !== commonAncestorId) {
    const entry = session.getEntry(current);
    if (!entry) break;
    entries.push(entry);
    current = entry.parentId;
  }
  entries.reverse();
  return { entries, commonAncestorId };
}

function pirContentText(content, fallback) {
  if (typeof content === "string") return content;
  if (Array.isArray(content)) {
    const parts = [];
    for (const block of content) {
      if (block && typeof block === "object" && typeof block.text === "string") parts.push(block.text);
    }
    return parts.length > 0 ? parts.join("\n") : fallback;
  }
  return fallback;
}

function pirTruncateForSummary(text, maxChars) {
  if (text.length <= maxChars) return text;
  const truncatedChars = text.length - maxChars;
  return text.slice(0, maxChars) + "\n\n[... " + truncatedChars + " more characters truncated]";
}

/**
 * Serialize LLM messages to text for summarization
 * (upstream compaction/utils.ts serializeConversation).
 */
export function serializeConversation(messages) {
  const list = Array.isArray(messages) ? messages : [];
  const parts = [];
  for (const msg of list) {
    if (!msg || typeof msg !== "object") continue;
    if (msg.role === "user") {
      const content = pirContentText(msg.content, "");
      if (content) parts.push("[User]: " + content);
    } else if (msg.role === "assistant") {
      const thinkingParts = [];
      const toolCalls = [];
      if (Array.isArray(msg.content)) {
        for (const block of msg.content) {
          if (!block || typeof block !== "object") continue;
          if (block.type === "thinking" && typeof block.thinking === "string") thinkingParts.push(block.thinking);
          else if (block.type === "toolCall") {
            const args = block.arguments && typeof block.arguments === "object" ? block.arguments : {};
            const argsStr = Object.entries(args).map(([k, v]) => k + "=" + JSON.stringify(v)).join(", ");
            toolCalls.push(String(block.name ?? "") + "(" + argsStr + ")");
          }
        }
      }
      if (thinkingParts.length > 0) parts.push("[Assistant thinking]: " + thinkingParts.join("\n"));
      if (Array.isArray(msg.content) && msg.content.some((block) => block && block.type === "text")) {
        parts.push("[Assistant]: " + pirContentText(msg.content, ""));
      }
      if (toolCalls.length > 0) parts.push("[Assistant tool calls]: " + toolCalls.join("; "));
    } else if (msg.role === "toolResult") {
      const content = pirContentText(msg.content, "");
      if (content) parts.push("[Tool result]: " + pirTruncateForSummary(content, 2000));
    }
  }
  return parts.join("\n\n");
}

/** Format bytes as human-readable size (upstream truncate.ts formatSize). */
export function formatSize(bytes) {
  const value = Number(bytes) || 0;
  if (value < 1024) return value + "B";
  if (value < 1024 * 1024) return (value / 1024).toFixed(1) + "KB";
  return (value / (1024 * 1024)).toFixed(1) + "MB";
}

function pirSplitLinesForCounting(content) {
  if (content.length === 0) return [];
  const lines = content.split("\n");
  if (content.endsWith("\n")) lines.pop();
  return lines;
}

function pirTruncateStringToBytesFromEnd(str, maxBytes) {
  const totalBytes = pirUtf8ByteLength(str);
  if (totalBytes <= maxBytes) return str;
  let bytes = 0;
  let index = str.length;
  while (index > 0) {
    const code = str.charCodeAt(index - 1);
    let width;
    if (code >= 0xdc00 && code <= 0xdfff && index >= 2) {
      width = 4;
      index -= 2;
    } else if (code < 0x80) width = 1;
    else if (code < 0x800) width = 2;
    else width = 3;
    if (bytes + width > maxBytes) {
      if (width === 4) index += 2; // keep the full surrogate pair out
      break;
    }
    bytes += width;
    if (width !== 4) index -= 1;
  }
  return str.slice(index);
}

/** Keep the first N lines/bytes (upstream truncate.ts truncateHead). */
export function truncateHead(content, options) {
  const text = String(content ?? "");
  const maxLines = options && options.maxLines !== undefined ? options.maxLines : DEFAULT_MAX_LINES;
  const maxBytes = options && options.maxBytes !== undefined ? options.maxBytes : DEFAULT_MAX_BYTES;
  const totalBytes = pirUtf8ByteLength(text);
  const lines = pirSplitLinesForCounting(text);
  const totalLines = lines.length;

  if (totalLines <= maxLines && totalBytes <= maxBytes) {
    return {
      content: text, truncated: false, truncatedBy: null, totalLines, totalBytes,
      outputLines: totalLines, outputBytes: totalBytes, lastLinePartial: false,
      firstLineExceedsLimit: false, maxLines, maxBytes,
    };
  }

  const firstLineBytes = lines.length > 0 ? pirUtf8ByteLength(lines[0]) : 0;
  if (lines.length > 0 && firstLineBytes > maxBytes) {
    return {
      content: "", truncated: true, truncatedBy: "bytes", totalLines, totalBytes,
      outputLines: 0, outputBytes: 0, lastLinePartial: false,
      firstLineExceedsLimit: true, maxLines, maxBytes,
    };
  }

  const outputLinesArr = [];
  let outputBytesCount = 0;
  let truncatedBy = "lines";
  for (let i = 0; i < lines.length && i < maxLines; i++) {
    const line = lines[i];
    const lineBytes = pirUtf8ByteLength(line) + (i > 0 ? 1 : 0);
    if (outputBytesCount + lineBytes > maxBytes) {
      truncatedBy = "bytes";
      break;
    }
    outputLinesArr.push(line);
    outputBytesCount += lineBytes;
  }
  if (outputLinesArr.length >= maxLines && outputBytesCount <= maxBytes) truncatedBy = "lines";

  const outputContent = outputLinesArr.join("\n");
  return {
    content: outputContent, truncated: true, truncatedBy, totalLines, totalBytes,
    outputLines: outputLinesArr.length, outputBytes: pirUtf8ByteLength(outputContent),
    lastLinePartial: false, firstLineExceedsLimit: false, maxLines, maxBytes,
  };
}

/** Keep the last N lines/bytes (upstream truncate.ts truncateTail). */
export function truncateTail(content, options) {
  const text = String(content ?? "");
  const maxLines = options && options.maxLines !== undefined ? options.maxLines : DEFAULT_MAX_LINES;
  const maxBytes = options && options.maxBytes !== undefined ? options.maxBytes : DEFAULT_MAX_BYTES;
  const totalBytes = pirUtf8ByteLength(text);
  const lines = pirSplitLinesForCounting(text);
  const totalLines = lines.length;

  if (totalLines <= maxLines && totalBytes <= maxBytes) {
    return {
      content: text, truncated: false, truncatedBy: null, totalLines, totalBytes,
      outputLines: totalLines, outputBytes: totalBytes, lastLinePartial: false,
      firstLineExceedsLimit: false, maxLines, maxBytes,
    };
  }

  const outputLinesArr = [];
  let outputBytesCount = 0;
  let truncatedBy = "lines";
  let lastLinePartial = false;
  for (let i = lines.length - 1; i >= 0 && outputLinesArr.length < maxLines; i--) {
    const line = lines[i];
    const lineBytes = pirUtf8ByteLength(line) + (outputLinesArr.length > 0 ? 1 : 0);
    if (outputBytesCount + lineBytes > maxBytes) {
      truncatedBy = "bytes";
      if (outputLinesArr.length === 0) {
        const truncatedLine = pirTruncateStringToBytesFromEnd(line, maxBytes);
        outputLinesArr.unshift(truncatedLine);
        outputBytesCount = pirUtf8ByteLength(truncatedLine);
        lastLinePartial = true;
      }
      break;
    }
    outputLinesArr.unshift(line);
    outputBytesCount += lineBytes;
  }
  if (outputLinesArr.length >= maxLines && outputBytesCount <= maxBytes) truncatedBy = "lines";

  const outputContent = outputLinesArr.join("\n");
  return {
    content: outputContent, truncated: true, truncatedBy, totalLines, totalBytes,
    outputLines: outputLinesArr.length, outputBytes: pirUtf8ByteLength(outputContent),
    lastLinePartial, firstLineExceedsLimit: false, maxLines, maxBytes,
  };
}

/** Truncate a single line with a `... [truncated]` suffix (upstream truncateLine). */
export function truncateLine(line, maxChars) {
  const text = String(line ?? "");
  const limit = maxChars === undefined ? PIR_GREP_MAX_LINE_LENGTH : maxChars;
  if (text.length <= limit) return { text, wasTruncated: false };
  return { text: text.slice(0, limit) + "... [truncated]", wasTruncated: true };
}

/** Minimal visual width wrapper (padding-aware; no ANSI width accounting). */
function pirWrapToWidth(line, width, paddingX) {
  const inner = Math.max(1, width - paddingX * 2);
  if (line.length <= inner) return [line];
  const out = [];
  for (let i = 0; i < line.length; i += inner) out.push(line.slice(i, i + inner));
  return out;
}

/**
 * Truncate text to a maximum number of visual (wrapped) lines
 * (upstream visual-truncate.ts; wraps by character width here).
 */
export function truncateToVisualLines(text, maxVisualLines, width, paddingX, keep) {
  const body = String(text ?? "");
  if (!body) return { visualLines: [], skippedCount: 0 };
  const pad = paddingX === undefined ? 0 : paddingX;
  const mode = keep === "start" ? "start" : "end";
  const allVisualLines = [];
  for (const logicalLine of body.split(/\r?\n/)) {
    for (const wrapped of pirWrapToWidth(logicalLine, Math.max(1, width), pad)) {
      allVisualLines.push(" ".repeat(pad) + wrapped);
    }
  }
  const limit = Math.max(0, Math.floor(maxVisualLines));
  if (allVisualLines.length <= limit) return { visualLines: allVisualLines, skippedCount: 0 };
  const visualLines = mode === "start" ? allVisualLines.slice(0, limit) : allVisualLines.slice(-limit);
  return { visualLines, skippedCount: allVisualLines.length - limit };
}

/** Dimension note for resized images (upstream image-resize.ts). */
export function formatDimensionNote(result) {
  if (!result || !result.wasResized) return undefined;
  const scale = result.originalWidth / result.width;
  return "[Image: original " + result.originalWidth + "x" + result.originalHeight +
    ", displayed at " + result.width + "x" + result.height +
    ". Multiply coordinates by " + scale.toFixed(2) + " to map to original image.]";
}

function pirValidateSkillName(name) {
  const errors = [];
  if (name.length > 64) errors.push("name exceeds 64 characters (" + name.length + ")");
  if (!/^[a-z0-9-]+$/.test(name)) errors.push("name contains invalid characters (must be lowercase a-z, 0-9, hyphens only)");
  if (name.startsWith("-") || name.endsWith("-")) errors.push("name must not start or end with a hyphen");
  if (name.indexOf("--") !== -1) errors.push("name must not contain consecutive hyphens");
  return errors;
}

function pirValidateSkillDescription(description) {
  const errors = [];
  if (typeof description !== "string" || description.trim() === "") {
    errors.push("description is required");
  } else if (description.length > 1024) {
    errors.push("description exceeds 1024 characters (" + description.length + ")");
  }
  return errors;
}

function pirLoadSkillFromFile(filePath, source) {
  const diagnostics = [];
  const isDeclaredSkill = pirBasename(filePath) === "SKILL.md";
  const raw = pirFsReadText(filePath);
  if (raw === undefined) {
    diagnostics.push({ type: "warning", message: "could not read skill file", path: filePath });
    return { skill: null, diagnostics };
  }
  let frontmatter;
  try {
    frontmatter = parseFrontmatter(raw).frontmatter;
  } catch (error) {
    if (isDeclaredSkill) {
      diagnostics.push({ type: "warning", message: String((error && error.message) || error), path: filePath });
    }
    return { skill: null, diagnostics };
  }
  const description = frontmatter.description;
  const hasDescription = typeof description === "string" && description.trim() !== "";
  if (!isDeclaredSkill && !hasDescription) return { skill: null, diagnostics };
  const skillDir = pirDirname(filePath);
  const parentDirName = pirBasename(skillDir);
  for (const error of pirValidateSkillDescription(description)) {
    diagnostics.push({ type: "warning", message: error, path: filePath });
  }
  const frontmatterName = typeof frontmatter.name === "string" ? frontmatter.name : undefined;
  const name = frontmatterName || parentDirName;
  for (const error of pirValidateSkillName(name)) {
    diagnostics.push({ type: "warning", message: error, path: filePath });
  }
  if (!hasDescription) return { skill: null, diagnostics };
  return {
    skill: {
      name,
      description: typeof description === "string" ? description : "",
      filePath,
      baseDir: skillDir,
      sourceInfo: pirSkillSourceInfo(filePath, skillDir, source),
      disableModelInvocation: frontmatter["disable-model-invocation"] === true,
    },
    diagnostics,
  };
}

function pirLoadSkillsFromDirInternal(dir, source, includeRootFiles) {
  const skills = [];
  const diagnostics = [];
  const entries = pirFsReaddir(dir);
  if (!entries) return { skills, diagnostics };

  // A directory containing SKILL.md is a skill root; it is not recursed into.
  for (const entry of entries) {
    if (!entry || entry.name !== "SKILL.md") continue;
    let isFile = entry.isFile === true;
    if (entry.isSymbolicLink === true) {
      const stat = pirFsStat(pirJoinPath(dir, entry.name));
      isFile = !!stat && stat.isFile === true;
    }
    if (!isFile) continue;
    const result = pirLoadSkillFromFile(pirJoinPath(dir, entry.name), source);
    if (result.skill) skills.push(result.skill);
    for (const diagnostic of result.diagnostics) diagnostics.push(diagnostic);
    return { skills, diagnostics };
  }

  for (const entry of entries) {
    if (!entry || typeof entry.name !== "string") continue;
    if (entry.name.startsWith(".") || entry.name === "node_modules") continue;
    const fullPath = pirJoinPath(dir, entry.name);
    let isDirectory = entry.isDirectory === true;
    let isFile = entry.isFile === true;
    if (entry.isSymbolicLink === true) {
      const stat = pirFsStat(fullPath);
      if (!stat) continue;
      isDirectory = stat.isDirectory === true;
      isFile = stat.isFile === true;
    }
    if (isDirectory) {
      const sub = pirLoadSkillsFromDirInternal(fullPath, source, false);
      for (const skill of sub.skills) skills.push(skill);
      for (const diagnostic of sub.diagnostics) diagnostics.push(diagnostic);
      continue;
    }
    if (!isFile || !includeRootFiles || !entry.name.endsWith(".md")) continue;
    const result = pirLoadSkillFromFile(fullPath, source);
    if (result.skill) skills.push(result.skill);
    for (const diagnostic of result.diagnostics) diagnostics.push(diagnostic);
  }
  return { skills, diagnostics };
}

/** Load skills from a directory (upstream skills.ts loadSkillsFromDir). */
export function loadSkillsFromDir(options) {
  const opts = options && typeof options === "object" ? options : {};
  const dir = pirToPosixPath(String(opts.dir ?? ""));
  const source = String(opts.source ?? "path");
  if (!dir) return { skills: [], diagnostics: [] };
  return pirLoadSkillsFromDirInternal(dir, source, true);
}

/**
 * Load skills from all configured locations (upstream skills.ts loadSkills).
 * Ignore-file handling (.gitignore/.ignore/.fdignore) is not applied here.
 */
export function loadSkills(options) {
  const opts = options && typeof options === "object" ? options : {};
  const resolvedCwd = pirResolvePath(opts.cwd === undefined ? "." : opts.cwd, ".");
  const resolvedAgentDir = pirResolvePath(opts.agentDir === undefined || !opts.agentDir ? getAgentDir() : opts.agentDir, ".");
  const skillPaths = Array.isArray(opts.skillPaths) ? opts.skillPaths : [];
  const includeDefaults = opts.includeDefaults !== false;

  const skillMap = new Map();
  const allDiagnostics = [];
  const collisionDiagnostics = [];

  function addSkills(result) {
    for (const diagnostic of result.diagnostics) allDiagnostics.push(diagnostic);
    for (const skill of result.skills) {
      const existing = skillMap.get(skill.name);
      if (existing) {
        collisionDiagnostics.push({
          type: "collision",
          message: 'name "' + skill.name + '" collision',
          path: skill.filePath,
          collision: {
            resourceType: "skill",
            name: skill.name,
            winnerPath: existing.filePath,
            loserPath: skill.filePath,
          },
        });
      } else {
        skillMap.set(skill.name, skill);
      }
    }
  }

  const userSkillsDir = pirJoinPath(resolvedAgentDir, "skills");
  const projectSkillsDir = pirResolvePath(pirJoinPath(CONFIG_DIR_NAME, "skills"), resolvedCwd);

  if (includeDefaults) {
    addSkills(pirLoadSkillsFromDirInternal(userSkillsDir, "user", true));
    addSkills(pirLoadSkillsFromDirInternal(projectSkillsDir, "project", true));
  }

  const isUnderPath = (target, root) => {
    const normalizedRoot = pirResolvePath(root, ".");
    if (target === normalizedRoot) return true;
    return target.startsWith(normalizedRoot.endsWith("/") ? normalizedRoot : normalizedRoot + "/");
  };

  for (const rawPath of skillPaths) {
    const resolvedPath = pirResolvePath(rawPath, resolvedCwd);
    const stat = pirFsStat(resolvedPath);
    if (!stat) {
      allDiagnostics.push({ type: "warning", message: "skill path does not exist", path: resolvedPath });
      continue;
    }
    const source = !includeDefaults && isUnderPath(resolvedPath, userSkillsDir)
      ? "user"
      : !includeDefaults && isUnderPath(resolvedPath, projectSkillsDir)
        ? "project"
        : "path";
    if (stat.isDirectory === true) {
      addSkills(pirLoadSkillsFromDirInternal(resolvedPath, source, true));
    } else if (stat.isFile === true && resolvedPath.endsWith(".md")) {
      const result = pirLoadSkillFromFile(resolvedPath, source);
      if (result.skill) addSkills({ skills: [result.skill], diagnostics: result.diagnostics });
      else for (const diagnostic of result.diagnostics) allDiagnostics.push(diagnostic);
    } else {
      allDiagnostics.push({ type: "warning", message: "skill path is not a markdown file", path: resolvedPath });
    }
  }

  const skills = [];
  for (const skill of skillMap.values()) skills.push(skill);
  return { skills, diagnostics: allDiagnostics.concat(collisionDiagnostics) };
}

function pirEscapeXml(str) {
  return String(str)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&apos;");
}

/**
 * Format skills for a system prompt (upstream skills.ts formatSkillsForPrompt).
 * Skills with disableModelInvocation are excluded.
 */
export function formatSkillsForPrompt(skills, fileReadTool) {
  const list = Array.isArray(skills) ? skills : [];
  const visibleSkills = list.filter((skill) => !(skill && skill.disableModelInvocation));
  if (visibleSkills.length === 0) return "";
  const loader = fileReadTool === undefined ? "read" : fileReadTool;
  const lines = [
    "\n\nThe following skills provide specialized instructions for specific tasks.",
    loader === "read"
      ? "Use the read tool to load a skill's file when the task matches its description."
      : loader === "bash"
        ? "Use bash to load a skill's file when the task matches its description."
        : "Load a skill's file when the task matches its description.",
    "When a skill file references a relative path, resolve it against the skill directory (parent of SKILL.md / dirname of the path) and use that absolute path in tool commands.",
    "",
    "<available_skills>",
  ];
  for (const skill of visibleSkills) {
    lines.push("  <skill>");
    lines.push("    <name>" + pirEscapeXml(skill.name) + "</name>");
    lines.push("    <description>" + pirEscapeXml(skill.description) + "</description>");
    lines.push("    <location>" + pirEscapeXml(skill.filePath) + "</location>");
    lines.push("  </skill>");
  }
  lines.push("</available_skills>");
  return lines.join("\n");
}

/**
 * Parse a `<skill name="..." location="...">...</skill>` block from a user
 * message (upstream agent-session.ts parseSkillBlock). Null when absent.
 */
export function parseSkillBlock(text) {
  const match = String(text ?? "").match(/^<skill name="([^"]+)" location="([^"]+)">\n([\s\S]*?)\n<\/skill>(?:\n\n([\s\S]+))?$/);
  if (!match) return null;
  return {
    name: match[1],
    location: match[2],
    content: match[3],
    userMessage: match[4] ? match[4].trim() || undefined : undefined,
  };
}

function pirLoadContextFileFromDir(dir) {
  for (const filename of PIR_CONTEXT_FILE_CANDIDATES) {
    const filePath = pirJoinPath(dir, filename);
    const stat = pirFsStat(filePath);
    if (!stat || stat.isFile !== true) continue;
    const content = pirFsReadText(filePath);
    if (content === undefined) continue;
    return { path: filePath, content: pirStripBom(content) };
  }
  return null;
}

/**
 * Load AGENTS/CLAUDE context files from the agent dir and the cwd chain
 * (upstream resource-loader.ts loadProjectContextFiles).
 */
export function loadProjectContextFiles(options) {
  const opts = options && typeof options === "object" ? options : {};
  const resolvedCwd = pirResolvePath(opts.cwd === undefined ? "." : opts.cwd, ".");
  const resolvedAgentDir = pirResolvePath(opts.agentDir === undefined || !opts.agentDir ? getAgentDir() : opts.agentDir, ".");
  const contextFiles = [];
  const seenPaths = new Set();
  const globalContext = pirLoadContextFileFromDir(resolvedAgentDir);
  if (globalContext) {
    contextFiles.push(globalContext);
    seenPaths.add(globalContext.path);
  }
  let currentDir = resolvedCwd;
  for (let depth = 0; depth < 64; depth++) {
    const entry = pirLoadContextFileFromDir(currentDir);
    if (entry && !seenPaths.has(entry.path)) {
      contextFiles.push(entry);
      seenPaths.add(entry.path);
    }
    const parentDir = pirDirname(currentDir);
    if (!parentDir || parentDir === currentDir) break;
    currentDir = parentDir;
  }
  return contextFiles;
}

/**
 * Whether the cwd has project-local resources gated by project trust
 * (upstream trust-manager.ts hasTrustRequiringProjectResources). Symlink
 * canonicalization is skipped in this runtime.
 */
export function hasTrustRequiringProjectResources(cwd) {
  const env = typeof process !== "undefined" && process ? process.env ?? {} : {};
  const homeDir = String(env.HOME || env.USERPROFILE || "");
  const userAgentsSkillsDir = homeDir ? pirJoinPath(pirToPosixPath(homeDir), ".agents/skills") : null;
  let currentDir = pirResolvePath(cwd === undefined ? "." : cwd, ".");
  const configDir = pirJoinPath(currentDir, CONFIG_DIR_NAME);
  for (const entry of PIR_TRUST_REQUIRING_PROJECT_CONFIG_RESOURCES) {
    if (pirFsExists(pirJoinPath(configDir, entry))) return true;
  }
  for (let depth = 0; depth < 64; depth++) {
    const agentsSkillsDir = pirJoinPath(currentDir, ".agents/skills");
    if (agentsSkillsDir !== userAgentsSkillsDir && pirFsExists(agentsSkillsDir)) return true;
    const parentDir = pirDirname(currentDir);
    if (!parentDir || parentDir === currentDir) return false;
    currentDir = parentDir;
  }
  return false;
}

/**
 * Read a stored credential for a provider from auth.json
 * (upstream auth-storage.ts readStoredCredential). Undefined when absent.
 */
export function readStoredCredential(providerId, authPath) {
  try {
    const path = authPath === undefined || authPath === null ? pirJoinPath(getAgentDir(), "auth.json") : String(authPath);
    const text = pirFsReadText(path);
    if (text === undefined) return undefined;
    const data = JSON.parse(pirStripBom(text));
    return data ? data[providerId] : undefined;
  } catch {
    return undefined;
  }
}

/**
 * Detect a supported image MIME type from a file path
 * (upstream mime.ts; sniffs the file header through the host fs bridge).
 * Returns null when the file is missing/unreadable or not a supported image.
 */
export function detectSupportedImageMimeTypeFromFile(filePath) {
  if (typeof filePath !== "string" || filePath.length === 0) return Promise.resolve(null);
  return Promise.resolve().then(() => {
    const bytes = pirFsReadBytes(filePath);
    if (!bytes) return null;
    return pirDetectImageMimeTypeFromBytes(bytes);
  });
}

function pirDetectImageMimeTypeFromBytes(buffer) {
  const startsWith = (bytes, offset) => {
    if (buffer.length < offset + bytes.length) return false;
    for (let i = 0; i < bytes.length; i++) if (buffer[offset + i] !== bytes[i]) return false;
    return true;
  };
  const startsWithAscii = (offset, text) => {
    if (buffer.length < offset + text.length) return false;
    for (let i = 0; i < text.length; i++) if (buffer[offset + i] !== text.charCodeAt(i)) return false;
    return true;
  };
  const readUint16LE = (offset) => (buffer[offset] ?? 0) + ((buffer[offset + 1] ?? 0) << 8);
  const readUint32BE = (offset) =>
    (buffer[offset] ?? 0) * 0x1000000 + ((buffer[offset + 1] ?? 0) << 16) +
    ((buffer[offset + 2] ?? 0) << 8) + (buffer[offset + 3] ?? 0);
  const readUint32LE = (offset) =>
    (buffer[offset] ?? 0) + ((buffer[offset + 1] ?? 0) << 8) +
    ((buffer[offset + 2] ?? 0) << 16) + (buffer[offset + 3] ?? 0) * 0x1000000;

  if (startsWith([0xff, 0xd8, 0xff], 0)) return buffer[3] === 0xf7 ? null : "image/jpeg";
  if (startsWith([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a], 0)) {
    const isPng = buffer.length >= 16 && readUint32BE(8) === 13 && startsWithAscii(12, "IHDR");
    if (!isPng) return null;
    let offset = 8;
    while (offset + 8 <= buffer.length) {
      const chunkLength = readUint32BE(offset);
      if (startsWithAscii(offset + 4, "acTL")) return null; // animated PNG
      if (startsWithAscii(offset + 4, "IDAT")) break;
      const nextOffset = offset + 8 + chunkLength + 4;
      if (nextOffset <= offset || nextOffset > buffer.length) break;
      offset = nextOffset;
    }
    return "image/png";
  }
  if (startsWithAscii(0, "GIF87a") || startsWithAscii(0, "GIF89a")) return "image/gif";
  if (startsWithAscii(0, "RIFF") && startsWithAscii(8, "WEBP")) return "image/webp";
  if (startsWithAscii(0, "BM") && buffer.length >= 26) {
    const declaredFileSize = readUint32LE(2);
    const pixelDataOffset = readUint32LE(10);
    const dibHeaderSize = readUint32LE(14);
    let ok = true;
    if (declaredFileSize !== 0 && declaredFileSize < 26) ok = false;
    if (pixelDataOffset < 14 + dibHeaderSize) ok = false;
    if (declaredFileSize !== 0 && pixelDataOffset >= declaredFileSize) ok = false;
    let colorPlanes = 0;
    let bitsPerPixel = 0;
    if (dibHeaderSize === 12) {
      colorPlanes = readUint16LE(22);
      bitsPerPixel = readUint16LE(24);
    } else if (dibHeaderSize >= 40 && dibHeaderSize <= 124) {
      colorPlanes = readUint16LE(26);
      bitsPerPixel = readUint16LE(28);
    } else {
      ok = false;
    }
    if (ok && colorPlanes === 1 && [1, 4, 8, 16, 24, 32].indexOf(bitsPerPixel) !== -1) return "image/bmp";
  }
  return null;
}

/** Resolve the bash shell configuration (upstream shell.ts getShellConfig). */
export function getShellConfig(customShellPath) {
  if (customShellPath) {
    if (pirFsExists(customShellPath)) return { shell: customShellPath, args: ["-c"] };
    throw new Error("Custom shell path not found: " + customShellPath);
  }
  const isWin = typeof process !== "undefined" && process && process.platform === "win32";
  if (isWin) {
    const env = process.env ?? {};
    const candidates = [];
    if (env.ProgramFiles) candidates.push(env.ProgramFiles + "\\Git\\bin\\bash.exe");
    if (env["ProgramFiles(x86)"]) candidates.push(env["ProgramFiles(x86)"] + "\\Git\\bin\\bash.exe");
    for (const candidate of candidates) {
      if (pirFsExists(candidate)) return { shell: candidate, args: ["-c"] };
    }
    const onPath = pirFindOnPath("bash.exe");
    if (onPath) return { shell: onPath, args: ["-c"] };
    throw new Error("No bash shell found. Install Git for Windows or add bash to PATH.");
  }
  if (pirFsExists("/bin/bash")) return { shell: "/bin/bash", args: ["-c"] };
  const onPath = pirFindOnPath("bash");
  if (onPath) return { shell: onPath, args: ["-c"] };
  return { shell: "sh", args: ["-c"] };
}

/** Resolve the PowerShell configuration (upstream shell.ts getPowerShellConfig). */
export function getPowerShellConfig() {
  const isWin = typeof process !== "undefined" && process && process.platform === "win32";
  if (!isWin) throw new Error("The powershell tool is only available on Windows.");
  const shell = pirFindOnPath("pwsh.exe") ?? pirFindOnPath("powershell.exe");
  if (!shell) throw new Error("No PowerShell executable found. Install PowerShell or add powershell.exe/pwsh.exe to PATH.");
  return { shell, args: PIR_POWERSHELL_ARGS.slice() };
}

function pirSplitContentLines(text) {
  const body = String(text ?? "");
  if (body.length === 0) return [];
  const lines = body.split("\n");
  if (body.endsWith("\n")) lines.pop();
  return lines;
}

/** LCS line diff; returns ops [{type: "equal"|"delete"|"insert", line}]. */
function pirDiffLineOps(oldLines, newLines) {
  const n = oldLines.length;
  const m = newLines.length;
  const ops = [];
  if (n * m > 4000000) {
    for (const line of oldLines) ops.push({ type: "delete", line });
    for (const line of newLines) ops.push({ type: "insert", line });
    return ops;
  }
  const width = m + 1;
  const dp = new Int32Array((n + 1) * width);
  for (let i = n - 1; i >= 0; i--) {
    for (let j = m - 1; j >= 0; j--) {
      dp[i * width + j] = oldLines[i] === newLines[j]
        ? dp[(i + 1) * width + (j + 1)] + 1
        : Math.max(dp[(i + 1) * width + j], dp[i * width + (j + 1)]);
    }
  }
  let i = 0;
  let j = 0;
  while (i < n && j < m) {
    if (oldLines[i] === newLines[j]) {
      ops.push({ type: "equal", line: oldLines[i] });
      i++;
      j++;
    } else if (dp[(i + 1) * width + j] >= dp[i * width + (j + 1)]) {
      ops.push({ type: "delete", line: oldLines[i] });
      i++;
    } else {
      ops.push({ type: "insert", line: newLines[j] });
      j++;
    }
  }
  while (i < n) { ops.push({ type: "delete", line: oldLines[i] }); i++; }
  while (j < m) { ops.push({ type: "insert", line: newLines[j] }); j++; }
  return ops;
}

/** Group ops into diff parts [{value, added?, removed?}] like the `diff` package. */
function pirDiffParts(oldText, newText) {
  const oldLines = pirSplitContentLines(oldText);
  const newLines = pirSplitContentLines(newText);
  const ops = pirDiffLineOps(oldLines, newLines);
  const newEndsWithNewline = String(newText ?? "").endsWith("\n") || String(newText ?? "").length === 0;
  const parts = [];
  let current = null;
  for (let index = 0; index < ops.length; index++) {
    const op = ops[index];
    const kind = op.type === "equal" ? "equal" : op.type === "delete" ? "removed" : "added";
    if (!current || current.kind !== kind) {
      current = { kind, lines: [] };
      parts.push(current);
    }
    current.lines.push(op.line);
  }
  const result = [];
  for (let index = 0; index < parts.length; index++) {
    const part = parts[index];
    const isLast = index === parts.length - 1;
    let value = part.lines.join("\n");
    const needsNewline = isLast
      ? (part.kind === "removed" ? String(oldText ?? "").endsWith("\n") : newEndsWithNewline)
      : true;
    if (needsNewline) value += "\n";
    result.push({
      value,
      added: part.kind === "added" ? true : undefined,
      removed: part.kind === "removed" ? true : undefined,
    });
  }
  return result;
}

/**
 * Display-oriented diff with line numbers and context
 * (upstream edit-diff.ts generateDiffString).
 */
export function generateDiffString(oldContent, newContent, contextLines) {
  const context = contextLines === undefined ? 4 : contextLines;
  const parts = pirDiffParts(oldContent, newContent);
  const output = [];
  const oldLines = String(oldContent ?? "").split("\n");
  const newLines = String(newContent ?? "").split("\n");
  const maxLineNum = Math.max(oldLines.length, newLines.length);
  const lineNumWidth = String(maxLineNum).length;

  let oldLineNum = 1;
  let newLineNum = 1;
  let lastWasChange = false;
  let firstChangedLine;

  for (let i = 0; i < parts.length; i++) {
    const part = parts[i];
    const raw = part.value.split("\n");
    if (raw[raw.length - 1] === "") raw.pop();

    if (part.added || part.removed) {
      if (firstChangedLine === undefined) firstChangedLine = newLineNum;
      for (const line of raw) {
        if (part.added) {
          output.push("+" + String(newLineNum).padStart(lineNumWidth, " ") + " " + line);
          newLineNum++;
        } else {
          output.push("-" + String(oldLineNum).padStart(lineNumWidth, " ") + " " + line);
          oldLineNum++;
        }
      }
      lastWasChange = true;
    } else {
      const nextPartIsChange = i < parts.length - 1 && (parts[i + 1].added || parts[i + 1].removed);
      const hasLeadingChange = lastWasChange;
      const hasTrailingChange = nextPartIsChange;

      if (hasLeadingChange && hasTrailingChange) {
        if (raw.length <= context * 2) {
          for (const line of raw) {
            output.push(" " + String(oldLineNum).padStart(lineNumWidth, " ") + " " + line);
            oldLineNum++;
            newLineNum++;
          }
        } else {
          const leadingLines = raw.slice(0, context);
          const trailingLines = raw.slice(raw.length - context);
          const skippedLines = raw.length - leadingLines.length - trailingLines.length;
          for (const line of leadingLines) {
            output.push(" " + String(oldLineNum).padStart(lineNumWidth, " ") + " " + line);
            oldLineNum++;
            newLineNum++;
          }
          output.push(" " + "".padStart(lineNumWidth, " ") + " ...");
          oldLineNum += skippedLines;
          newLineNum += skippedLines;
          for (const line of trailingLines) {
            output.push(" " + String(oldLineNum).padStart(lineNumWidth, " ") + " " + line);
            oldLineNum++;
            newLineNum++;
          }
        }
      } else if (hasLeadingChange) {
        const shownLines = raw.slice(0, context);
        const skippedLines = raw.length - shownLines.length;
        for (const line of shownLines) {
          output.push(" " + String(oldLineNum).padStart(lineNumWidth, " ") + " " + line);
          oldLineNum++;
          newLineNum++;
        }
        if (skippedLines > 0) {
          output.push(" " + "".padStart(lineNumWidth, " ") + " ...");
          oldLineNum += skippedLines;
          newLineNum += skippedLines;
        }
      } else if (hasTrailingChange) {
        const skippedLines = Math.max(0, raw.length - context);
        if (skippedLines > 0) {
          output.push(" " + "".padStart(lineNumWidth, " ") + " ...");
          oldLineNum += skippedLines;
          newLineNum += skippedLines;
        }
        for (const line of raw.slice(raw.length - context)) {
          output.push(" " + String(oldLineNum).padStart(lineNumWidth, " ") + " " + line);
          oldLineNum++;
          newLineNum++;
        }
      } else {
        for (const line of raw) {
          output.push(" " + String(oldLineNum).padStart(lineNumWidth, " ") + " " + line);
          oldLineNum++;
          newLineNum++;
        }
      }
      lastWasChange = false;
    }
  }

  return { diff: output.join("\n"), firstChangedLine };
}

/**
 * Standard unified patch with hunks (upstream edit-diff.ts
 * generateUnifiedPatch, header-only file names).
 */
export function generateUnifiedPatch(path, oldContent, newContent, contextLines) {
  const context = contextLines === undefined ? 4 : contextLines;
  const oldLines = pirSplitContentLines(oldContent);
  const newLines = pirSplitContentLines(newContent);
  const ops = pirDiffLineOps(oldLines, newLines);

  // Precompute old/new line positions before every op.
  const positions = [];
  let oldPos = 1;
  let newPos = 1;
  for (const op of ops) {
    positions.push({ oldPos, newPos });
    if (op.type !== "insert") oldPos++;
    if (op.type !== "delete") newPos++;
  }

  const changeIndices = [];
  for (let i = 0; i < ops.length; i++) {
    if (ops[i].type !== "equal") changeIndices.push(i);
  }

  const header = "--- " + path + "\n+++ " + path + "\n";
  if (changeIndices.length === 0) return header;

  // Merge change regions that are within 2*context of each other into hunks.
  const hunks = [];
  let start = Math.max(0, changeIndices[0] - context);
  let end = Math.min(ops.length - 1, changeIndices[0] + context);
  for (let i = 1; i < changeIndices.length; i++) {
    const index = changeIndices[i];
    if (index - context <= end + 1) {
      end = Math.min(ops.length - 1, index + context);
    } else {
      hunks.push([start, end]);
      start = Math.max(0, index - context);
      end = Math.min(ops.length - 1, index + context);
    }
  }
  hunks.push([start, end]);

  const output = [header];
  for (const [hunkStart, hunkEnd] of hunks) {
    let oldCount = 0;
    let newCount = 0;
    const body = [];
    for (let i = hunkStart; i <= hunkEnd; i++) {
      const op = ops[i];
      if (op.type === "equal") {
        body.push(" " + op.line);
        oldCount++;
        newCount++;
      } else if (op.type === "delete") {
        body.push("-" + op.line);
        oldCount++;
      } else {
        body.push("+" + op.line);
        newCount++;
      }
    }
    const oldStart = positions[hunkStart] ? positions[hunkStart].oldPos : 1;
    const newStart = positions[hunkStart] ? positions[hunkStart].newPos : 1;
    output.push("@@ -" + oldStart + "," + oldCount + " +" + newStart + "," + newCount + " @@\n");
    output.push(body.join("\n") + "\n");
  }
  return output.join("");
}

/**
 * Render a diff string with theme colors (upstream components/diff.ts;
 * intra-line inverse highlighting is dropped in the minimal renderer).
 */
export function renderDiff(diffText, _options) {
  const lines = String(diffText ?? "").split("\n");
  const result = [];
  for (const line of lines) {
    const first = line.length > 0 ? line[0] : " ";
    if (first === "-") result.push(pirAnsiFg("toolDiffRemoved", line));
    else if (first === "+") result.push(pirAnsiFg("toolDiffAdded", line));
    else result.push(pirAnsiFg("toolDiffContext", line));
  }
  return result.join("\n");
}

// ===========================================================================
// 10. Theme (theme.ts ports; headless color resolution degrades to
// passthrough for tokens without a resolved color)
// ===========================================================================

function pirColorToAnsi(value, isBackground, dim) {
  if (typeof value === "number") {
    return (isBackground ? "\u001b[48;5;" : "\u001b[38;5;") + value + "m";
  }
  if (typeof value === "string") {
    if (value === "") return isBackground ? "\u001b[49m" : "\u001b[39m";
    const match = /^#([0-9a-fA-F]{6})$/.exec(value);
    if (match) {
      const rgb = parseInt(match[1], 16);
      const prefix = isBackground ? "\u001b[48;2;" : "\u001b[38;2;";
      const ansi = prefix + ((rgb >> 16) & 255) + ";" + ((rgb >> 8) & 255) + ";" + (rgb & 255) + "m";
      return dim ? ansi + "\u001b[2m" : ansi;
    }
    return ""; // Unsupported color formats render as passthrough.
  }
  return "";
}

/**
 * Minimal Theme (upstream theme.ts): stores per-token ANSI escapes and
 * renders fg/bg/style text. Tokens without a resolved color pass text
 * through (upstream throws for unknown tokens; headless passthrough here is
 * the documented degradation — no color, correct text).
 */
export class Theme {
  constructor(fgColors, bgColors, mode, options) {
    const opts = options && typeof options === "object" ? options : {};
    this.name = opts.name;
    this.sourcePath = opts.sourcePath;
    this.sourceInfo = opts.sourceInfo;
    this._mode = mode === undefined || mode === null ? "truecolor" : mode;
    this._dim = new Set(Array.isArray(opts.dim) ? opts.dim : []);
    this._appearance = opts.appearance;
    this._fgAnsi = new Map();
    this._bgAnsi = new Map();
    this._colorValues = {};
    const foregrounds = fgColors && typeof fgColors === "object" ? fgColors : {};
    const backgrounds = bgColors && typeof bgColors === "object" ? bgColors : {};
    for (const token of Object.keys(foregrounds)) {
      const value = foregrounds[token];
      this._fgAnsi.set(token, pirColorToAnsi(value, false, this._dim.has(token)));
      this._colorValues[token] = typeof value === "number" ? value : String(value);
    }
    for (const token of Object.keys(backgrounds)) {
      const value = backgrounds[token];
      this._bgAnsi.set(token, pirColorToAnsi(value, true, false));
      this._colorValues[token] = typeof value === "number" ? value : String(value);
    }
  }

  get appearance() {
    return this._appearance ?? "dark";
  }

  get colors() {
    return Object.freeze(Object.assign({}, this._colorValues));
  }

  fg(color, text) {
    const ansi = this._fgAnsi.get(color);
    if (ansi === undefined || ansi === "") return String(text);
    if (this._dim.has(color)) return ansi + String(text) + "\u001b[22;39m";
    return ansi + String(text) + "\u001b[39m";
  }

  bg(color, text) {
    const ansi = this._bgAnsi.get(color);
    if (ansi === undefined || ansi === "") return String(text);
    return ansi + String(text) + "\u001b[49m";
  }

  style(text, options) {
    const opts = options && typeof options === "object" ? options : {};
    let prefix = "";
    let suffix = "";
    if (typeof opts.fg === "string") {
      const ansi = this._fgAnsi.get(opts.fg);
      if (ansi) { prefix += ansi; suffix = "\u001b[39m" + suffix; }
    } else if (typeof opts.fg === "number") {
      prefix += pirColorToAnsi(opts.fg, false, false);
      suffix = "\u001b[39m" + suffix;
    }
    if (typeof opts.bg === "string") {
      const ansi = this._bgAnsi.get(opts.bg);
      if (ansi) { prefix += ansi; suffix = "\u001b[49m" + suffix; }
    } else if (typeof opts.bg === "number") {
      prefix += pirColorToAnsi(opts.bg, true, false);
      suffix = "\u001b[49m" + suffix;
    }
    const dim = opts.dim === true || (typeof opts.fg === "string" && this._dim.has(opts.fg));
    if (dim) { prefix += "\u001b[2m"; suffix = "\u001b[22m" + suffix; }
    if (opts.bold) { prefix += "\u001b[1m"; suffix = "\u001b[22m" + suffix; }
    if (opts.italic) { prefix += "\u001b[3m"; suffix = "\u001b[23m" + suffix; }
    if (opts.underline) { prefix += "\u001b[4m"; suffix = "\u001b[24m" + suffix; }
    if (opts.inverse) { prefix += "\u001b[7m"; suffix = "\u001b[27m" + suffix; }
    if (opts.strikethrough) { prefix += "\u001b[9m"; suffix = "\u001b[29m" + suffix; }
    if (!prefix) return String(text);
    return prefix + String(text) + suffix;
  }

  bold(text) { return pirBold(text); }
  italic(text) { return pirItalic(text); }
  underline(text) { return pirUnderline(text); }
  inverse(text) { return "\u001b[7m" + text + "\u001b[27m"; }
  strikethrough(text) { return pirStrikethrough(text); }

  getFgAnsi(color) {
    const ansi = this._fgAnsi.get(color);
    if (ansi === undefined) return "";
    return this._dim.has(color) ? ansi + "\u001b[2m" : ansi;
  }

  getBgAnsi(color) {
    const ansi = this._bgAnsi.get(color);
    return ansi === undefined ? "" : ansi;
  }

  getColorMode() {
    return this._mode;
  }

  getThinkingBorderColor(level) {
    const token = level === "minimal" ? "thinkingMinimal"
      : level === "low" ? "thinkingLow"
      : level === "medium" ? "thinkingMedium"
      : level === "high" ? "thinkingHigh"
      : level === "xhigh" ? "thinkingXhigh"
      : level === "max" ? "thinkingMax"
      : "thinkingOff";
    const self = this;
    return function (str) { return self.fg(token, str); };
  }

  getBashModeBorderColor() {
    const self = this;
    return function (str) { return self.fg("bashMode", str); };
  }
}

// Global theme (shared across module instances via a globalThis symbol, like
// upstream). initTheme loads the named theme snapshot from the host when
// available and otherwise installs a passthrough theme.
const PIR_THEME_KEY =
  typeof Symbol !== "undefined" && Symbol.for
    ? Symbol.for("@earendil-works/pi-coding-agent:theme")
    : "__pirCodingAgentTheme";

function pirGetGlobalTheme() {
  try {
    return globalThis[PIR_THEME_KEY] ?? null;
  } catch {
    return null;
  }
}

function pirSetGlobalTheme(themeInstance) {
  try {
    globalThis[PIR_THEME_KEY] = themeInstance;
  } catch {
    // Non-configurable global; the getters fall back to host snapshots anyway.
  }
}

function pirThemeFromHostColors(colors, name) {
  const fgColors = {};
  for (const token of Object.keys(colors ?? {})) {
    if (typeof colors[token] === "string" || typeof colors[token] === "number") fgColors[token] = colors[token];
  }
  return new Theme(fgColors, {}, "truecolor", { name });
}

/**
 * Initialize the global theme (upstream theme.ts initTheme). When the host
 * reports a theme snapshot, its colors are used; otherwise the theme degrades
 * to passthrough. The optional watcher is a no-op in this runtime.
 */
export function initTheme(themeName, _enableWatcher) {
  try {
    let raw = null;
    if (typeof themeName === "string" && themeName.length > 0 &&
        typeof __pir_host !== "undefined" && __pir_host && typeof __pir_host.getThemeJsonByName === "function") {
      raw = __pir_host.getThemeJsonByName(themeName);
    }
    if (!raw && typeof __pir_host !== "undefined" && __pir_host && typeof __pir_host.getThemeJson === "function") {
      raw = __pir_host.getThemeJson();
    }
    let colors = null;
    if (raw) {
      const parsed = JSON.parse(raw);
      if (parsed && typeof parsed === "object" && parsed.colors && typeof parsed.colors === "object") {
        colors = parsed.colors;
      }
    }
    pirSetGlobalTheme(pirThemeFromHostColors(colors, themeName));
  } catch {
    pirSetGlobalTheme(new Theme({}, {}, "truecolor", { name: themeName }));
  }
}

function pirActiveThemeFg(role, text) {
  const themeInstance = pirGetGlobalTheme();
  if (themeInstance && typeof themeInstance.fg === "function") {
    try {
      return themeInstance.fg(role, text);
    } catch {
      // Fall through to the host snapshot.
    }
  }
  return pirAnsiFg(role, text);
}

function pirRenderTextComponent(label, text, width) {
  const w = Math.max(1, Math.floor(typeof width === "number" && width > 0 ? width : 80));
  const body = String(text === undefined || text === null ? "" : text);
  const firstLine = body.split(/\r?\n/)[0];
  const line = firstLine.length > 0 ? firstLine : label;
  return [line.length > w ? line.slice(0, w) : line];
}

function pirMessageText(message) {
  if (!message) return "";
  if (typeof message.summary === "string") return message.summary;
  const content = message.content;
  if (typeof content === "string") return content;
  if (Array.isArray(content)) {
    const parts = [];
    for (const block of content) {
      if (block && typeof block === "object" && typeof block.text === "string") parts.push(block.text);
    }
    return parts.join("\n");
  }
  return "";
}

/**
 * Create an event bus (upstream core/event-bus.ts createEventBus).
 * Handler errors are swallowed after reporting.
 */
export function createEventBus() {
  const channels = new Map();
  return {
    emit(channel, data) {
      const handlers = channels.get(channel);
      if (!handlers) return;
      for (const handler of handlers.slice()) {
        const safeHandler = async (value) => {
          try {
            await handler(value);
          } catch (err) {
            console.error("Event handler error (" + channel + "):", err);
          }
        };
        void safeHandler(data);
      }
    },
    on(channel, handler) {
      if (typeof handler !== "function") return () => {};
      if (!channels.has(channel)) channels.set(channel, []);
      const list = channels.get(channel);
      list.push(handler);
      return () => {
        const index = list.indexOf(handler);
        if (index !== -1) list.splice(index, 1);
      };
    },
    clear() {
      channels.clear();
    },
  };
}

/**
 * Create the extension runtime object with throwing action stubs
 * (upstream extensions/loader.ts createExtensionRuntime). Host code
 * rebinds action methods; pre-bind registrations are queued.
 */
export function createExtensionRuntime() {
  const notInitialized = () => {
    throw new Error("Extension runtime not initialized. Action methods cannot be called during extension loading.");
  };
  const state = {};
  const eventBusUnsubscribers = new Set();
  const assertActive = () => {
    if (state.staleMessage) throw new Error(state.staleMessage);
  };
  const runtime = {
    sendMessage: notInitialized,
    sendUserMessage: notInitialized,
    appendEntry: notInitialized,
    setSessionName: notInitialized,
    getSessionName: notInitialized,
    setLabel: notInitialized,
    getActiveTools: notInitialized,
    getAllTools: notInitialized,
    getSettings: notInitialized,
    setActiveTools: notInitialized,
    refreshTools: () => {},
    getCommands: notInitialized,
    setModel: () => Promise.reject(new Error("Extension runtime not initialized")),
    getThinkingLevel: notInitialized,
    setThinkingLevel: notInitialized,
    flagValues: new Map(),
    pendingProviderRegistrations: [],
    pendingNativeProviderRegistrations: [],
    pendingVirtualModelRegistrations: [],
    createContext: notInitialized,
    assertActive,
    invalidate(message) {
      if (state.staleMessage) return;
      state.staleMessage =
        message ??
        "This extension ctx is stale after session replacement or reload. Do not use a captured pi or command ctx after ctx.newSession(), ctx.fork(), ctx.switchSession(), or ctx.reload().";
      for (const unsubscribe of eventBusUnsubscribers) unsubscribe();
      eventBusUnsubscribers.clear();
    },
    trackEventBusSubscription(unsubscribe) {
      let active = true;
      const trackedUnsubscribe = () => {
        if (!active) return;
        active = false;
        eventBusUnsubscribers.delete(trackedUnsubscribe);
        unsubscribe();
      };
      eventBusUnsubscribers.add(trackedUnsubscribe);
      return trackedUnsubscribe;
    },
    registerProvider(name, config, extensionPath) {
      runtime.pendingProviderRegistrations.push({ name, config, extensionPath: extensionPath ?? "<unknown>" });
    },
    registerNativeProvider(provider, extensionPath) {
      runtime.pendingNativeProviderRegistrations.push({ provider, extensionPath: extensionPath ?? "<unknown>" });
    },
    unregisterProvider(name) {
      runtime.pendingProviderRegistrations = runtime.pendingProviderRegistrations.filter((r) => r.name !== name);
      runtime.pendingNativeProviderRegistrations = runtime.pendingNativeProviderRegistrations.filter(
        (r) => !r.provider || r.provider.id !== name,
      );
    },
    registerVirtualModel(definition, extensionPath) {
      runtime.pendingVirtualModelRegistrations.push({ definition, extensionPath: extensionPath ?? "<unknown>" });
    },
    unregisterVirtualModel(provider, id) {
      runtime.pendingVirtualModelRegistrations = runtime.pendingVirtualModelRegistrations.filter(
        ({ definition }) => !definition || definition.provider !== provider || definition.id !== id,
      );
    },
  };
  return runtime;
}

// ---------------------------------------------------------------------------
// createAgentSession (child-process sessions)
// ---------------------------------------------------------------------------
//
// Upstream creates an in-process AgentSession (agent-session.ts); the pir
// extension runtime cannot host a second full agent runtime in-process, so
// the pir counterpart spawns `pir print --mode json` child processes — the
// SAME launch shape upstream's compiled-binary host uses for detached
// runners (async-execution.js spawns `<binaryHost> --no-extensions ...
// --mode rpc`). One prompt = one child run; session continuity rides the
// `--session <file>` flag, so every prompt of a session shares the JSONL
// transcript the way upstream's shared SessionManager would.
//
// Consumption surface (pi-subagents child-session.js open(), the only
// pir-reachable caller): the returned `{ session }` supports
// subscribe / prompt / abort / dispose / hasQueuedMessages / bindExtensions /
// getActiveToolNames / getAllTools plus the messages / sessionFile /
// sessionId / model getters. `prompt(text)` resolves with the turn's reply
// text and REJECTS when the turn failed or was aborted (upstream
// print-mode.ts:139-156's terminal-message rule; the error text is the
// terminal message's `errorMessage`, else `Request <stopReason>` — see
// `pirAgentSessionRun`). Uncovered AgentSession methods throw the
// standard unavailable error. Degradations (documented): child runs carry no
// ambient extensions, skills, or prompt templates (--no-extensions, the
// binary-host precedent); steering/follow-up queues have no mid-run channel
// and throw; resourceLoader/modelRuntime/settingsManager options are ignored
// (the child resolves providers through the host CLI resolution order).

const PIR_CHILD_SESSION_BASE_ARGS = [
  "print", "--mode", "json", "--no-extensions", "--no-skills", "--no-prompt-templates",
];

/**
 * The extension paths the caller's resource loader resolved (see
 * `DefaultResourceLoader`: the child's extensions are loaded by the child
 * process, not in-process). Returns [] when no loader was passed, the loader
 * exposes no extension result, or the child must run extension-free.
 */
function pirChildSessionExtensionPaths(resourceLoader) {
  if (!resourceLoader || typeof resourceLoader !== "object") return [];
  let result;
  try {
    if (typeof resourceLoader.getExtensions !== "function") return [];
    result = resourceLoader.getExtensions();
  } catch {
    return [];
  }
  if (!result || !Array.isArray(result.extensions)) return [];
  const paths = [];
  for (const extension of result.extensions) {
    const path = extension && typeof extension.path === "string" ? extension.path : undefined;
    if (path && !path.startsWith("<") && !path.startsWith("builtin:")) paths.push(path);
  }
  return paths;
}

function pirAgentSessionExecutable() {
  try {
    const env = typeof process !== "undefined" && process ? process.env : undefined;
    const override = env && env.PIR_EXECUTABLE ? String(env.PIR_EXECUTABLE).trim() : "";
    if (override) return override;
  } catch { /* fall through to the host answer */ }
  try {
    if (typeof __pir_host !== "undefined" && typeof __pir_host.getExecutablePath === "function") {
      const exe = __pir_host.getExecutablePath();
      if (exe) return String(exe);
    }
  } catch { /* fall through to PATH resolution */ }
  return "pir";
}

function pirAgentSessionId() {
  const random = Math.floor(Math.random() * 0x100000000).toString(16).padStart(8, "0")
    + Math.floor(Math.random() * 0x100000000).toString(16).padStart(8, "0");
  return random;
}

function pirAgentSessionDefaultFile(cwd) {
  // Upstream getDefaultSessionDirPath's encoding (phone-sessions.ts repeats
  // it): strip one leading separator, map / \ : to -, wrap in --...--.
  const safe = "--" + String(cwd).replace(/^[/\\]/, "").replace(/[/\\:]/g, "-") + "--";
  const stamp = new Date().toISOString().replace(/[:.]/g, "-");
  return getAgentDir() + "/sessions/" + safe + "/" + stamp + "_" + pirAgentSessionId() + ".jsonl";
}

export async function createAgentSession(options) {
  const opts = options && typeof options === "object" ? options : {};
  const cwd = typeof opts.cwd === "string" && opts.cwd ? opts.cwd : process.cwd();

  // Session storage: the passed sessionManager supplies the file/dir. A pir
  // host-backed SessionManager plans its file immediately (fresh sessions
  // carry the planned path before the first turn creates the file); the ctx
  // (readonly) session manager answers getSessionFile from the live session.
  // An in-memory manager has no file (upstream `SessionManager.inMemory`), so
  // no path is invented for it.
  const inMemoryStorage = !!(opts.sessionManager
    && opts.sessionManager._pirOptions
    && opts.sessionManager._pirOptions.kind === "inMemory");
  let sessionFile;
  try {
    if (opts.sessionManager && typeof opts.sessionManager.getSessionFile === "function") {
      sessionFile = opts.sessionManager.getSessionFile();
    }
  } catch { /* fall through to a fresh default file */ }
  if (!inMemoryStorage && (!sessionFile || typeof sessionFile !== "string")) {
    sessionFile = pirAgentSessionDefaultFile(cwd);
  }

  const state = {
    cwd: cwd,
    sessionFile: sessionFile,
    sessionId: undefined,
    model: opts.model && typeof opts.model === "object" ? opts.model : undefined,
    thinkingLevel: typeof opts.thinkingLevel === "string" ? opts.thinkingLevel : undefined,
    tools: Array.isArray(opts.tools) ? opts.tools.map(String) : undefined,
    excludeTools: Array.isArray(opts.excludeTools) ? opts.excludeTools.map(String) : undefined,
    systemPrompt: typeof opts.systemPrompt === "string" ? opts.systemPrompt : undefined,
    apiKey: typeof opts.apiKey === "string" ? opts.apiKey : undefined,
    // Extensions the caller's resource loader resolved. The child process
    // loads them itself (`--extension <path>`); the runner never imports them
    // in-process (see `DefaultResourceLoader`).
    extensionPaths: pirChildSessionExtensionPaths(opts.resourceLoader),
    listeners: [],
    messages: [],
    running: undefined,
    disposed: false,
  };

  const emit = (event) => {
    for (const listener of state.listeners.slice()) {
      try {
        listener(event);
      } catch { /* a broken subscriber never breaks the stream */ }
    }
  };

  const session = {
    subscribe(listener) {
      if (typeof listener !== "function") return () => {};
      state.listeners.push(listener);
      return () => {
        state.listeners = state.listeners.filter((entry) => entry !== listener);
      };
    },
    prompt(text) {
      if (state.disposed) return Promise.reject(new Error("AgentSession.prompt: the session is disposed"));
      if (state.running) {
        // Upstream queues concurrent prompts; one pir print child serves one
        // prompt, so an overlapping prompt waits for the running turn first.
        // The queue advances on BOTH outcomes: a failed turn rejects ITS
        // prompt's promise (see pirAgentSessionRun) but is a turn outcome,
        // not a wedged session — chaining only the fulfilled path would make
        // every later prompt of this session reject with the stale error,
        // the same silent no-run class this bridge must not have.
        const runNext = () => (state.disposed
          ? Promise.reject(new Error("AgentSession.prompt: the session is disposed"))
          : pirAgentSessionRun(state, emit, text));
        state.running = state.running.then(runNext, runNext);
        return state.running;
      }
      state.running = pirAgentSessionRun(state, emit, text);
      return state.running;
    },
    steer() { throw pirUnavailable("AgentSession", "steer"); },
    followUp() { throw pirUnavailable("AgentSession", "followUp"); },
    abort() {
      const proc = state.currentProcess;
      if (proc && typeof proc.kill === "function") {
        state.aborted = true;
        proc.kill();
      }
      return true;
    },
    hasQueuedMessages() { return false; },
    async bindExtensions(_options) { /* child runs are extension-free */ },
    getActiveToolNames() { return []; },
    getAllTools() { return []; },
    dispose() {
      state.disposed = true;
      const proc = state.currentProcess;
      if (proc && typeof proc.kill === "function") proc.kill();
      state.listeners = [];
    },
  };
  Object.defineProperty(session, "messages", { get: () => state.messages.slice() });
  Object.defineProperty(session, "sessionFile", { get: () => state.sessionFile });
  Object.defineProperty(session, "sessionId", { get: () => state.sessionId });
  Object.defineProperty(session, "model", { get: () => state.model });
  Object.defineProperty(session, "agent", {
    get: () => ({ hasQueuedMessages: () => false }),
  });
  Object.defineProperty(session, "extensionRunner", {
    get: () => ({ hasHandlers: () => false }),
  });
  for (const name of [
    "refreshContext", "setCacheWarmingMode", "getCallableToolNames", "getToolDefinition",
    "setActiveToolsByName", "setScopedModels", "sendCustomMessage", "sendUserMessage",
    "clearQueue", "getSteeringMessages", "getFollowUpMessages", "waitForIdle", "setModel",
    "cycleModel", "setThinkingLevel", "cycleThinkingLevel", "getAvailableThinkingLevels",
    "supportsThinking", "setSteeringMode", "setFollowUpMode", "compact", "abortCompaction",
    "abortBranchSummary", "setAutoCompactionEnabled", "reload", "abortRetry",
    "setAutoRetryEnabled", "executeBash", "abortBash", "setSessionName", "navigateTree",
    "getUserMessagesForForking", "getSessionStats", "getContextUsage", "exportToHtml",
    "exportToJsonl", "getLastAssistantText",
  ]) {
    if (session[name] === undefined) {
      session[name] = () => {
        throw pirUnavailable("AgentSession", name);
      };
    }
  }
  return { session };
}

function pirAgentSessionText(message) {
  if (!message) return "";
  const content = message.content;
  if (typeof content === "string") return content;
  if (!Array.isArray(content)) return "";
  return content
    .map((block) => (block && block.type === "text" ? block.text || "" : ""))
    .join(" ");
}

/**
 * The terminal assistant message of one run read as a failure, or undefined
 * when the run completed. Upstream print-mode.ts:139-156 decides the run's
 * outcome from exactly this shape — `stopReason` `error` or `aborted` is a
 * failed run, reported as `errorMessage || "Request " + stopReason` (text
 * mode exits 1 on it). The `--mode json` child this bridge consumes exits 0
 * for those turns and reports the failure only in-band (the terminal
 * `message_end`/`turn_end` line; `src/coding_agent/json.rs`, pinned by
 * `tests/suites/cli_entry_contract.rs::json_mode_provider_error_exits_0_with_header_and_error_event`),
 * so the consumer MUST apply the same check to the terminal message itself:
 * resolving a failed turn would report a provider/transport failure as a
 * successful empty reply (`reply: ""`), the exact cross-turn silence this
 * bridge was hiding.
 *
 * `fromIndex` bounds the scan to the messages THIS run appended
 * (`state.messages` accumulates across prompts), so a later run never
 * inherits an earlier turn's failure.
 */
function pirAgentSessionTerminalError(messages, fromIndex) {
  for (let i = messages.length - 1; i >= fromIndex; i--) {
    const message = messages[i];
    if (!message || message.role !== "assistant") continue;
    const stopReason = message.stopReason;
    if (stopReason !== "error" && stopReason !== "aborted") return undefined;
    // Upstream's error text (`errorMessage || "Request " + stopReason`).
    if (typeof message.errorMessage === "string" && message.errorMessage.trim()) {
      return message.errorMessage;
    }
    return "Request " + stopReason;
  }
  return undefined;
}

/**
 * One prompt = one `pir print --mode json` child run (see the
 * `createAgentSession` note above).
 *
 * Resolves with the last assistant reply THIS run produced. Rejects on a
 * failed run, with upstream's error surface (print-mode.ts:139-156):
 *   - a terminal assistant message whose `stopReason` is `error` or
 *     `aborted` rejects with its `errorMessage` (else `Request <stopReason>`);
 *   - a locally aborted run (`session.abort()` kills the child before its
 *     terminal message can be delivered) rejects, with the delivered text
 *     when one raced through, else `Request aborted`;
 *   - a non-zero child exit keeps the child-failure rejection below.
 * A completed turn (including `toolUse`/`length`/`stop`) keeps resolving the
 * reply text exactly as before.
 */
function pirAgentSessionRun(state, emit, text) {
  return new Promise((resolve, reject) => {
    const args = PIR_CHILD_SESSION_BASE_ARGS.slice();
    if (state.extensionPaths && state.extensionPaths.length) {
      const noExtensionsIndex = args.indexOf("--no-extensions");
      if (noExtensionsIndex !== -1) args.splice(noExtensionsIndex, 1);
      for (const extensionPath of state.extensionPaths) {
        args.push("--extension", extensionPath);
      }
    }
    // In-memory storage has no file to continue from; the run is stateless.
    if (typeof state.sessionFile === "string" && state.sessionFile) {
      args.push("--session", state.sessionFile);
    }
    if (state.model && typeof state.model.provider === "string") {
      args.push("--provider", state.model.provider);
    }
    if (state.model && typeof state.model.id === "string") {
      args.push("--model", state.model.id);
    }
    if (state.thinkingLevel) args.push("--thinking", state.thinkingLevel);
    if (state.tools && state.tools.length) args.push("--tools", state.tools.join(","));
    if (state.excludeTools && state.excludeTools.length) {
      args.push("--exclude-tools", state.excludeTools.join(","));
    }
    if (state.systemPrompt) args.push("--system-prompt", state.systemPrompt);
    if (state.apiKey) args.push("--api-key", state.apiKey);
    // `--` guards prompt text that begins with "-" (clap end-of-flags).
    args.push("--", String(text === undefined ? "" : text));

    state.aborted = false;
    // Messages this run appended start here: the terminal check below must
    // never read a previous turn's outcome.
    const firstMessageIndex = state.messages.length;
    let child;
    try {
      child = pirSpawn(pirAgentSessionExecutable(), args, { cwd: state.cwd, windowsHide: true });
    } catch (error) {
      reject(error);
      return;
    }
    state.currentProcess = child;
    let stdoutText = "";
    let stderrText = "";
    const handleLine = (line) => {
      const trimmed = line.trim();
      if (!trimmed.startsWith("{")) return;
      let event;
      try {
        event = JSON.parse(trimmed);
      } catch {
        return;
      }
      if (!event || typeof event !== "object") return;
      if (event.type === "session" && typeof event.id === "string") {
        // The `--mode json` header line: adopt the child's session id.
        state.sessionId = event.id;
        return;
      }
      if (event.type === "message_end" && event.message) {
        state.messages.push(event.message);
      }
      emit(event);
    };
    child.stdout.on("data", (chunk) => {
      stdoutText += String(chunk);
      let newlineIndex;
      while ((newlineIndex = stdoutText.indexOf("\n")) !== -1) {
        handleLine(stdoutText.slice(0, newlineIndex));
        stdoutText = stdoutText.slice(newlineIndex + 1);
      }
    });
    child.stderr.on("data", (chunk) => {
      stderrText += String(chunk);
      if (stderrText.length > 8192) stderrText = stderrText.slice(-8192);
    });
    child.on("error", (error) => {
      if (state.currentProcess === child) state.currentProcess = undefined;
      reject(error);
    });
    child.on("close", (code) => {
      if (state.currentProcess === child) state.currentProcess = undefined;
      if (stdoutText.trim()) handleLine(stdoutText);
      if (state.aborted) {
        // The caller killed the child; fall back to upstream's aborted text
        // when the kill raced a terminal message that did make it out.
        const delivered = pirAgentSessionTerminalError(state.messages, firstMessageIndex);
        reject(new Error(delivered === undefined ? "Request aborted" : delivered));
        return;
      }
      if (code !== 0) {
        const detail = stderrText.trim().split("\n").pop() || "exit code " + code;
        reject(new Error("AgentSession.prompt child run failed: " + detail));
        return;
      }
      const terminalError = pirAgentSessionTerminalError(state.messages, firstMessageIndex);
      if (terminalError !== undefined) {
        reject(new Error(terminalError));
        return;
      }
      resolve(pirAgentSessionLastReply(state.messages, firstMessageIndex));
    });
  });
}

function pirAgentSessionLastReply(messages, fromIndex) {
  for (let i = messages.length - 1; i >= fromIndex; i--) {
    const message = messages[i];
    if (message && message.role === "assistant") return pirAgentSessionText(message);
  }
  return "";
}

// Real, host-backed `ModelRuntime` (upstream
// `packages/coding-agent/src/core/model-runtime.ts`, HEAD `7d75a0e3e`).
//
// Upstream owns the provider catalog + credential store in-process
// (`pi-ai` `createModels`, `builtinProviderCatalog.builtinProviders()`,
// `models.json`). pir's host already owns both: the agent session pushes its
// provider registry snapshot into the extension runtime before every prompt
// (`__pir_host.getModelsJson()` → `{models, providers}`, the same table
// `ctx.modelRegistry` reads) and exposes the provider registration ops.
// `ModelRuntime` is therefore a shape adapter over those host seams —
// no catalog, no credentials, and no HTTP client are re-created in JS.
//
// Host seams used:
//   * `getModelsJson()` / `getPhysicalModelJson(provider,id)` — the catalog.
//   * `hasAuthenticatedProvider(id)` — the credential snapshot.
//   * `registerProviderJson` / `unregisterProvider` /
//     `registerVirtualModelJson` / `unregisterVirtualModel` — provider
//     registration (`pi-subagents` child-session flushes the child loader's
//     pending registrations through these).
//   * `modelCompleteJson(model, context, options)` — the host completion
//     (`AgentSession::complete_fn_for`); the real network + auth path.
//
// Degradations (documented, not silent):
//   * `refresh({ allowNetwork: true })` re-reads the host snapshot; the host
//     owns the network catalog refresh, so `allowNetwork` cannot widen what
//     this runtime sees. Extensions that registered `refreshModels` hooks are
//     handled by the host's own provider pipeline, not here.
//   * `stream`/`streamSimple` deliver the host completion as a single
//     `text_delta` rather than incremental provider deltas (the host op is a
//     completion, not a provider stream); `completeSimple` = collect(stream)
//     exactly like upstream, so results are identical, timing is not.
//   * The catalog carries the host's field names (`context_window`,
//     `max_tokens`); `provider`, `id`, `api` and `input` are stable across
//     both spellings and are what consumers read.
//   * Credential mutation (`login`/`logout`/`setRuntimeApiKey`) stays
//     unavailable: the host owns the credential store and exposes no
//     extension write path for it.
const PIR_MODEL_RUNTIME_CATALOG_ERROR =
  "ModelRuntime: the pir host has no model catalog in this mode (no session model registry pushed)";

function pirModelRuntimeSnapshot() {
  let parsed;
  try {
    if (typeof __pir_host === "undefined" || typeof __pir_host.getModelsJson !== "function") {
      return { models: [], providers: [], error: undefined, bound: false };
    }
    parsed = JSON.parse(__pir_host.getModelsJson()) ?? {};
  } catch (error) {
    return { models: [], providers: [], error: String((error && error.message) || error), bound: true };
  }
  const models = [];
  if (Array.isArray(parsed.models)) {
    for (const model of parsed.models) {
      if (model && typeof model === "object" && typeof model.provider === "string" && typeof model.id === "string") {
        models.push(model);
      }
    }
  }
  const providers = Array.isArray(parsed.providers)
    ? parsed.providers.filter((provider) => provider && typeof provider === "object")
    : [];
  return { models, providers, error: undefined, bound: true };
}

function pirModelTypeOf(model) {
  const type = model && model.type;
  return typeof type === "string" && type ? type : "chat";
}

function pirModelRuntimeHostAvailable() {
  return typeof __pir_host !== "undefined" && __pir_host !== undefined && __pir_host !== null;
}

/** A one-shot assistant message stream over a host completion (see header). */
function pirHostCompletionStream(model, context, options) {
  const events = [];
  const waiters = [];
  let finished = false;
  let resultMessage;
  let settle;
  const done = new Promise((resolve) => { settle = resolve; });

  const push = (event) => {
    events.push(event);
    for (const waiter of waiters.splice(0)) waiter();
  };
  const end = (message) => {
    resultMessage = message;
    finished = true;
    for (const waiter of waiters.splice(0)) waiter();
    settle(message);
  };
  const fail = (message) => {
    push({ type: "error", reason: "error", error: message });
    end(message);
  };

  const modelJson = JSON.stringify(model ?? {});
  const contextJson = JSON.stringify(context ?? {});
  const optionsJson = JSON.stringify(options ?? {});

  if (!pirModelRuntimeHostAvailable() || typeof __pir_host.modelCompleteJson !== "function") {
    fail({ role: "assistant", content: [], errorMessage: pirUnavailable("ModelRuntime", "streamSimple").message });
    return pirCompletionStreamHandle(() => events, () => resultMessage, () => finished, waiters, done);
  }

  pirMicrotask(async () => {
    let payload;
    try {
      const raw = await __pir_host.modelCompleteJson(modelJson, contextJson, optionsJson);
      payload = typeof raw === "string" ? JSON.parse(raw) : raw;
    } catch (error) {
      fail({ role: "assistant", content: [], errorMessage: String((error && error.message) || error) });
      return;
    }
    if (!payload || payload.ok !== true || !payload.message) {
      fail({
        role: "assistant",
        content: [],
        errorMessage: String((payload && payload.error) || "ModelRuntime.streamSimple: the host returned no message"),
      });
      return;
    }
    const message = payload.message;
    push({ type: "start", partial: message });
    const content = Array.isArray(message.content) ? message.content : [];
    for (let index = 0; index < content.length; index++) {
      const block = content[index];
      if (block && block.type === "text") {
        push({ type: "text_start", contentIndex: index, partial: message });
        push({ type: "text_delta", contentIndex: index, delta: String(block.text ?? ""), partial: message });
        push({ type: "text_end", contentIndex: index, content: String(block.text ?? ""), partial: message });
      } else if (block && block.type === "thinking") {
        push({ type: "thinking_start", contentIndex: index, partial: message });
        push({ type: "thinking_delta", contentIndex: index, delta: String(block.thinking ?? ""), partial: message });
        push({ type: "thinking_end", contentIndex: index, content: String(block.thinking ?? ""), partial: message });
      }
    }
    push({ type: "done", reason: "stop", message });
    end(message);
  });

  return pirCompletionStreamHandle(() => events, () => resultMessage, () => finished, waiters, done);
}

/** Async-iterable + `result()` handle matching `AssistantMessageEventStream`. */
function pirCompletionStreamHandle(eventsOf, resultOf, finishedOf, waiters, done) {
  return {
    async *[Symbol.asyncIterator]() {
      let index = 0;
      for (;;) {
        const pending = eventsOf();
        while (index < pending.length) {
          yield pending[index++];
        }
        if (finishedOf()) return;
        await new Promise((resolve) => {
          waiters.push(resolve);
          // Close the finish/delivery race: a flush between the check above
          // and the wait must not strand this iterator forever.
          if (finishedOf() || eventsOf().length > index) resolve();
        });
      }
    },
    result() { return done; },
  };
}

function pirMicrotask(fn) {
  if (typeof queueMicrotask === "function") queueMicrotask(fn);
  else Promise.resolve().then(fn);
}

export class ModelRuntime {
  constructor() {
    /** False for a bare `new ModelRuntime()` (no host binding). */
    this._pirBound = false;
    this._pirModels = [];
    this._pirProviders = [];
    this._pirError = undefined;
    this._pirRegisteredProviders = new Map();
    this._pirRegisteredNative = new Map();
    this._pirVirtualModels = [];
  }

  static async create(options) {
    const runtime = new ModelRuntime();
    runtime._pirBound = true;
    runtime._pirOptions = options && typeof options === "object" ? options : {};
    runtime.updateModelSnapshot();
    // Upstream refreshes on create; the host owns the catalog, so this is the
    // snapshot pull (no network path exists from the extension runtime).
    await runtime.refresh({ allowNetwork: false });
    return runtime;
  }

  // -- catalog snapshot ----------------------------------------------------
  updateModelSnapshot() {
    const snapshot = pirModelRuntimeSnapshot();
    this._pirModels = snapshot.models;
    this._pirProviders = snapshot.providers;
    this._pirError = snapshot.error;
  }

  providerIds() {
    const ids = new Set();
    for (const provider of this._pirProviders) if (typeof provider.id === "string") ids.add(provider.id);
    for (const model of this._pirModels) ids.add(model.provider);
    for (const id of this._pirRegisteredProviders.keys()) ids.add(id);
    for (const id of this._pirRegisteredNative.keys()) ids.add(id);
    return ids;
  }

  getProviders() { return this._pirProviders.slice(); }

  getProvider(providerId) {
    return this._pirProviders.find((provider) => provider && provider.id === providerId);
  }

  getAllModels() { return this._pirModels.slice(); }
  getModels() { return this._pirModels.slice(); }

  getModelsOfType(type, providerId) {
    return this._pirModels.filter(
      (model) => pirModelTypeOf(model) === type && (providerId === undefined || model.provider === providerId),
    );
  }

  getModel(providerId, id) {
    return this._pirModels.find((model) => model.provider === providerId && model.id === id);
  }

  getModelOfType(type, providerId, id) {
    return this.getModelsOfType(type, providerId).find((model) => model.id === id);
  }

  getAvailableOfType(type, providerId) {
    return this.getModelsOfType(type, providerId).filter((model) => this.hasConfiguredAuth(model.provider));
  }

  getError() { return this._pirError; }

  async getAvailable(providerId) {
    const models = providerId === undefined
      ? this._pirModels
      : this._pirModels.filter((model) => model.provider === providerId);
    return models.filter((model) => this.hasConfiguredAuth(model.provider));
  }

  async getAllAvailable(providerId) { return this.getAvailable(providerId); }

  async refresh(options) {
    // `allowNetwork` is accepted for upstream shape compatibility; the host
    // owns the provider catalog, so the extension runtime always answers from
    // the host snapshot (see the section header).
    void options;
    this.updateModelSnapshot();
    if (this._pirError !== undefined) throw new Error(this._pirError);
  }

  buildProviderSnapshot() { this.updateModelSnapshot(); }
  rebuildProviders() { this.updateModelSnapshot(); }
  runAvailabilityRefresh() { return this.refresh({ allowNetwork: false }); }
  queueAvailabilityRefresh() { return this.refresh({ allowNetwork: false }); }
  refreshProviderAvailability() { return this.refresh({ allowNetwork: false }); }
  recomposeProvider(providerId) { return this.getProvider(providerId); }
  composeProvider(providerId) { return this.getProvider(providerId); }
  enqueueCredentialOperation(_providerId, operation) { return Promise.resolve().then(operation); }
  synchronizeCredentialState() { return Promise.resolve(); }

  // -- credentials (host snapshot; no extension write path) ----------------
  hasConfiguredAuth(modelOrProviderId) {
    const providerId = typeof modelOrProviderId === "string"
      ? modelOrProviderId
      : modelOrProviderId && modelOrProviderId.provider;
    if (typeof providerId !== "string" || !providerId) return false;
    try {
      if (typeof __pir_host === "undefined" || typeof __pir_host.hasAuthenticatedProvider !== "function") return false;
      return __pir_host.hasAuthenticatedProvider(providerId) === true;
    } catch {
      return false;
    }
  }

  checkAuth(providerId) {
    if (this.hasConfiguredAuth(providerId)) return { providerId, type: "api_key" };
    return undefined;
  }

  isUsingOAuth(providerId) {
    const auth = this.checkAuth(providerId);
    return auth !== undefined && auth.type === "oauth";
  }

  isUsingSubscription(providerId) {
    const auth = this.checkAuth(providerId);
    return auth !== undefined && auth.type === "subscription";
  }

  async getAuth(providerId) { return this.checkAuth(providerId); }
  async listCredentials() { return []; }
  getProviderAuthStatus(providerId) { return this.checkAuth(providerId); }

  prepareRequest(model) { return { model, headers: undefined, apiKey: undefined }; }
  getApiKeyAndHeaders(model) {
    const providerId = model && model.provider;
    return { apiKey: undefined, headers: undefined, configured: this.hasConfiguredAuth(providerId) };
  }

  // -- provider registration (host ops) -----------------------------------
  registerProvider(name, config) {
    this._pirRegisteredProviders.set(name, config);
    if (pirModelRuntimeHostAvailable() && typeof __pir_host.registerProviderJson === "function") {
      __pir_host.registerProviderJson(JSON.stringify({ id: name, config: config ?? {} }));
    }
  }

  unregisterProvider(name) {
    this._pirRegisteredProviders.delete(name);
    if (pirModelRuntimeHostAvailable() && typeof __pir_host.unregisterProvider === "function") {
      __pir_host.unregisterProvider(name);
    }
  }

  registerNativeProvider(provider) {
    const id = provider && provider.id;
    if (typeof id !== "string" || !id) throw new TypeError("registerNativeProvider requires a provider with a string id");
    this._pirRegisteredNative.set(id, provider);
    if (pirModelRuntimeHostAvailable() && typeof __pir_host.registerProviderJson === "function") {
      __pir_host.registerProviderJson(JSON.stringify({ id, native: true, provider }));
    }
  }

  unregisterNativeProvider(providerId) {
    this._pirRegisteredNative.delete(providerId);
    if (pirModelRuntimeHostAvailable() && typeof __pir_host.unregisterProvider === "function") {
      __pir_host.unregisterProvider(providerId);
    }
  }

  registerVirtualModel(definition) {
    if (!definition || typeof definition !== "object") {
      throw new TypeError("registerVirtualModel requires a definition object");
    }
    this._pirVirtualModels.push(definition);
    if (pirModelRuntimeHostAvailable() && typeof __pir_host.registerVirtualModelJson === "function") {
      __pir_host.registerVirtualModelJson(JSON.stringify(definition));
    }
  }

  unregisterVirtualModel(provider, id) {
    this._pirVirtualModels = this._pirVirtualModels.filter(
      (definition) => !(definition.provider === provider && definition.id === id),
    );
    if (pirModelRuntimeHostAvailable() && typeof __pir_host.unregisterVirtualModel === "function") {
      __pir_host.unregisterVirtualModel(provider, id);
    }
  }

  getRegisteredProviderConfig(providerId) { return this._pirRegisteredProviders.get(providerId); }
  getRegisteredProviderIds() {
    return [...this._pirRegisteredProviders.keys(), ...this._pirRegisteredNative.keys()];
  }
  getRegisteredNativeProvider(providerId) { return this._pirRegisteredNative.get(providerId); }

  // -- completion / streaming (host completion op) -------------------------
  stream(model, context, options) {
    if (!this._pirBound) throw pirUnavailable("ModelRuntime", "stream");
    return pirHostCompletionStream(model, context, options);
  }

  streamSimple(model, context, options) {
    if (!this._pirBound) throw pirUnavailable("ModelRuntime", "streamSimple");
    return pirHostCompletionStream(model, context, options);
  }

  async complete(model, context, options) {
    if (!this._pirBound) throw pirUnavailable("ModelRuntime", "complete");
    const stream = pirHostCompletionStream(model, context, options);
    return await stream.result();
  }

  async completeSimple(model, context, options) {
    if (!this._pirBound) throw pirUnavailable("ModelRuntime", "completeSimple");
    return await this.complete(model, context, options);
  }
}

// Real, host-backed `SettingsManager` (upstream
// `packages/coding-agent/src/core/settings-manager.ts`, HEAD `7d75a0e3e`).
//
// Upstream reads and writes `<agentDir>/settings.json` (global scope) and
// `<cwd>/.pi/settings.json` (project scope, trust-gated), deep-merging the
// project scope over the global one. This port does the same through the host
// fs bridge (`__pir_host.node`), so the two files, the migration rules, and
// the merge order are the ones the host's own `SettingsManager`
// (`src/config/settings.rs`) reads — the shim never invents a second format.
//
// Host reuse: when the extension runtime has no fs bridge (a host that does
// not grant extension file access), the manager seeds the global scope from
// the host-pushed effective snapshot (`__pir_host.getSettingsJson()`, the
// value the host's own manager produced) instead of failing closed. Writes in
// that mode stay in memory and are reported through `drainErrors()`, exactly
// like an upstream write that could not reach its file.
//
// Degradations (documented, not silent):
//   * `proper-lockfile` inter-process locking is not reproduced; writes are
//     atomic (temp sibling + rename) but two processes writing the same
//     settings.json at once can still lose one update.
//   * The typed-field warning/error diagnostics of the host's parse path
//     degrade to a JSON parse error per scope (recorded in `errors`).
const PIR_SETTINGS_DEFAULTS = {
  transport: "auto",
  steeringMode: "one-at-a-time",
  followUpMode: "one-at-a-time",
  compaction: { enabled: true, reserveTokens: 16384, keepRecentTokens: 20000 },
  branchSummary: { reserveTokens: 16384, skipPrompt: false },
  retry: {
    enabled: true,
    maxRetries: 3,
    baseDelayMs: 2000,
    maxAgentDelayMs: 60000,
    provider: { maxRetryDelayMs: 60000 },
  },
  hideThinkingBlock: false,
  quietStartup: false,
  defaultProjectTrust: "ask",
  cacheWarming: "streaming",
  terminal: { showImages: true },
  warnings: { anthropicExtraUsage: true },
};
const PIR_DEFAULT_COMPACTION_TOKEN_SETTINGS = {
  reserveTokens: PIR_SETTINGS_DEFAULTS.compaction.reserveTokens,
  keepRecentTokens: PIR_SETTINGS_DEFAULTS.compaction.keepRecentTokens,
};
const PIR_CACHE_WARMING_MODES = ["off", "streaming", "idle"];
const PIR_DEFAULT_TOOL_NAMES = ["read", "bash", "edit", "write"];

function pirSettingsDeepMergeObjects(base, overrides) {
  const result = { ...base };
  for (const key of Object.keys(overrides)) {
    const overrideValue = overrides[key];
    if (overrideValue === undefined) continue;
    const baseValue = base[key];
    result[key] = pirIsPlainObject(baseValue) && pirIsPlainObject(overrideValue)
      ? pirSettingsDeepMergeObjects(baseValue, overrideValue)
      : overrideValue;
  }
  return result;
}

function pirIsToolModifier(entry) {
  return typeof entry === "string" && (entry.startsWith("+") || entry.startsWith("-"));
}

/** Upstream `mergeDefaultTools`: a modifier-only list appends to the base. */
function pirSettingsMergeDefaultTools(base, overrides) {
  if (overrides === undefined) return base;
  if (!Array.isArray(base) || !Array.isArray(overrides) || !overrides.every(pirIsToolModifier)) return overrides;
  return [...base, ...overrides];
}

/** Upstream `resolveDefaultTools`. */
function pirSettingsResolveDefaultTools(entries) {
  const list = Array.isArray(entries) ? entries : [];
  const plain = list.filter((entry) => !pirIsToolModifier(entry));
  let tools = plain.length > 0 || list.length === 0 ? plain : PIR_DEFAULT_TOOL_NAMES.slice();
  for (const entry of list) {
    if (!pirIsToolModifier(entry)) continue;
    const name = entry.slice(1);
    const index = tools.indexOf(name);
    if (entry.startsWith("+") && index === -1 && name) tools.push(name);
    else if (entry.startsWith("-") && index !== -1) tools.splice(index, 1);
  }
  return tools;
}

/** Upstream `deepMergeSettings`: per-field deep merge plus the defaultTools rule. */
function pirSettingsDeepMerge(base, overrides) {
  const merged = pirSettingsDeepMergeObjects(base, overrides);
  const defaultTools = pirSettingsMergeDefaultTools(base.defaultTools, overrides.defaultTools);
  if (defaultTools !== undefined) merged.defaultTools = defaultTools;
  return merged;
}

/** Upstream `migrateSettings`: legacy keys rewritten on read. */
function pirSettingsMigrate(settings) {
  const value = pirIsPlainObject(settings) ? settings : {};
  if ("queueMode" in value && !("steeringMode" in value)) {
    value.steeringMode = value.queueMode;
    delete value.queueMode;
  }
  if (!("transport" in value) && typeof value.websockets === "boolean") {
    value.transport = value.websockets ? "websocket" : "sse";
    delete value.websockets;
  }
  if (pirIsPlainObject(value.skills)) {
    const skills = value.skills;
    if (skills.enableSkillCommands !== undefined && value.enableSkillCommands === undefined) {
      value.enableSkillCommands = skills.enableSkillCommands;
    }
    if (Array.isArray(skills.customDirectories) && skills.customDirectories.length > 0) {
      value.skills = skills.customDirectories;
    } else {
      delete value.skills;
    }
  }
  if (pirIsPlainObject(value.retry)) {
    const retry = value.retry;
    const provider = pirIsPlainObject(retry.provider) ? retry.provider : undefined;
    if (typeof retry.maxDelayMs === "number" && (provider === undefined || provider.maxRetryDelayMs === undefined || provider.maxRetryDelayMs === null)) {
      retry.provider = { ...(provider ?? {}), maxRetryDelayMs: retry.maxDelayMs };
    }
    delete retry.maxDelayMs;
  }
  return value;
}

/** Strip a UTF-8 BOM (Node's `stripBom`; the host bridge decodes raw bytes). */
function pirStripUtf8Bom(text) {
  return String(text ?? "").replace(/^\ufeff/, "");
}

function pirSettingsPathsFor(cwd, agentDir) {
  const resolvedCwd = pirResolvePath(cwd ?? ".", pirToPosixPath(process.cwd ? process.cwd() : "."));
  const resolvedAgentDir = pirResolvePath(agentDir ?? getAgentDir(), pirToPosixPath(process.cwd ? process.cwd() : "."));
  return {
    cwd: resolvedCwd,
    agentDir: resolvedAgentDir,
    global: pirJoinPath(resolvedAgentDir, "settings.json"),
    project: pirJoinPath(resolvedCwd, CONFIG_DIR_NAME, "settings.json"),
  };
}

/** The host's own effective settings snapshot (`__pir_host.getSettingsJson`). */
function pirHostSettingsSnapshot() {
  try {
    if (typeof __pir_host === "undefined" || typeof __pir_host.getSettingsJson !== "function") return undefined;
    const raw = __pir_host.getSettingsJson();
    if (typeof raw !== "string") return undefined;
    const parsed = JSON.parse(raw);
    return pirIsPlainObject(parsed) ? parsed : undefined;
  } catch {
    return undefined;
  }
}

function pirHostSessionCwd() {
  try {
    if (typeof __pir_host === "undefined" || typeof __pir_host.getContextJson !== "function") return undefined;
    const parsed = JSON.parse(__pir_host.getContextJson());
    return parsed && typeof parsed.cwd === "string" ? pirToPosixPath(parsed.cwd) : undefined;
  } catch {
    return undefined;
  }
}

/**
 * File-backed storage over the host fs bridge (upstream `FileSettingsStorage`).
 * `withLock` reads the current file text, hands it to `fn`, and writes back
 * the returned string (the upstream contract; `undefined` means "no write").
 */
class FileSettingsStorage {
  constructor(cwd, agentDir) {
    const paths = pirSettingsPathsFor(cwd, agentDir);
    this.globalSettingsPath = paths.global;
    this.projectSettingsPath = paths.project;
    // Host-scoped reuse (see the section header): only the manager built for
    // the host's own cwd/agentDir may seed from the host snapshot.
    this.hostSnapshotPath =
      paths.agentDir === pirToPosixPath(getAgentDir()) && paths.cwd === (pirHostSessionCwd() ?? paths.cwd);
  }

  read(scope) {
    const path = scope === "global" ? this.globalSettingsPath : this.projectSettingsPath;
    if (!pirFsExists(path)) return undefined;
    const text = pirFsReadText(path);
    if (text === undefined) {
      // No fs bridge (host without extension file access). The global scope
      // falls back to the host's own effective snapshot when it applies.
      if (scope === "global" && this.hostSnapshotPath) {
        const snapshot = pirHostSettingsSnapshot();
        if (snapshot !== undefined) return JSON.stringify(snapshot, null, 2);
      }
      throw new Error("settings storage is not readable through the host fs bridge");
    }
    return text;
  }

  write(scope, text) {
    const path = scope === "global" ? this.globalSettingsPath : this.projectSettingsPath;
    pirFsWriteText(path, text);
  }
}

/** Upstream `InMemorySettingsStorage` (the `inMemory` / `fromStorage` seam). */
class InMemorySettingsStorage {
  constructor() {
    this.global = undefined;
    this.project = undefined;
  }

  withLockScoped(scope, fn) {
    const current = scope === "global" ? this.global : this.project;
    const next = fn(current);
    if (next !== undefined) {
      if (scope === "global") this.global = next;
      else this.project = next;
    }
  }
}

export class SettingsManager {
  constructor() {
    this._pirStorage = undefined;
    this._pirGlobal = {};
    this._pirProject = {};
    this._pirSettings = {};
    this._pirProjectTrusted = true;
    this._pirErrors = [];
    this._pirPaths = {};
    this._pirModified = { global: new Set(), project: new Set() };
    this._pirModifiedNested = { global: new Map(), project: new Map() };
    this._pirLoadErrors = { global: null, project: null };
    this._pirWriteQueue = Promise.resolve();
  }

  static create(cwd, agentDir, options) {
    const dir = agentDir === undefined ? getAgentDir() : agentDir;
    const paths = pirSettingsPathsFor(cwd, dir);
    const storage = new FileSettingsStorage(cwd, dir);
    return SettingsManager.fromStorageWithPaths(storage, options, { global: paths.global, project: paths.project });
  }

  static fromStorage(storage, options) {
    return SettingsManager.fromStorageWithPaths(storage, options, {});
  }

  static fromStorageWithPaths(storage, options, settingsPaths) {
    const opts = options && typeof options === "object" ? options : {};
    const paths = settingsPaths && typeof settingsPaths === "object" ? settingsPaths : {};
    const manager = new SettingsManager();
    manager._pirStorage = storage;
    manager._pirPaths = paths;
    manager._pirProjectTrusted = opts.projectTrusted ?? true;
    const globalLoad = SettingsManager.tryLoadFromStorage(storage, "global");
    const projectLoad = SettingsManager.tryLoadFromStorage(storage, "project", manager._pirProjectTrusted);
    manager._pirGlobal = globalLoad.settings;
    manager._pirProject = projectLoad.settings;
    manager._pirLoadErrors = { global: globalLoad.error, project: projectLoad.error };
    if (globalLoad.error) manager._pirRecordError("global", globalLoad.error);
    if (projectLoad.error) manager._pirRecordError("project", projectLoad.error);
    manager._pirRekey();
    return manager;
  }

  static inMemory(settings, options) {
    const storage = new InMemorySettingsStorage();
    const initial = pirSettingsMigrate(pirDeepCloneJson(settings && typeof settings === "object" ? settings : {}));
    storage.withLockScoped("global", () => JSON.stringify(initial, null, 2));
    return SettingsManager.fromStorage(storage, options);
  }

  static loadFromStorage(storage, scope, projectTrusted) {
    const trusted = projectTrusted === undefined ? true : projectTrusted;
    if (scope === "project" && !trusted) return {};
    const text = SettingsManager._pirStorageRead(storage, scope);
    if (!text) return {};
    return pirSettingsMigrate(JSON.parse(pirStripUtf8Bom(text)));
  }

  static tryLoadFromStorage(storage, scope, projectTrusted) {
    try {
      return { settings: SettingsManager.loadFromStorage(storage, scope, projectTrusted), error: null };
    } catch (error) {
      return { settings: {}, error: error instanceof Error ? error : new Error(String(error)) };
    }
  }

  /** Read one scope from either a host-backed storage or a JS storage object. */
  static _pirStorageRead(storage, scope) {
    if (storage instanceof FileSettingsStorage) return storage.read(scope);
    let current;
    if (storage && typeof storage.withLock === "function") {
      storage.withLock(scope, (text) => {
        current = text;
        return undefined;
      });
    } else if (storage && typeof storage.withLockScoped === "function") {
      storage.withLockScoped(scope, (text) => {
        current = text;
        return undefined;
      });
    } else {
      throw new Error("SettingsManager storage must implement withLock(scope, fn)");
    }
    return current;
  }

  static _pirStorageWrite(storage, scope, text) {
    if (storage instanceof FileSettingsStorage) {
      storage.write(scope, text);
      return;
    }
    if (storage && typeof storage.withLock === "function") {
      storage.withLock(scope, () => text);
      return;
    }
    if (storage && typeof storage.withLockScoped === "function") {
      storage.withLockScoped(scope, () => text);
      return;
    }
    throw new Error("SettingsManager storage must implement withLock(scope, fn)");
  }

  _pirRekey() {
    this._pirSettings = pirSettingsDeepMerge(this._pirGlobal, this._pirProject);
  }

  _pirRecordError(scope, error, path) {
    this._pirErrors.push({
      scope,
      ...(path ? { path } : {}),
      error: error instanceof Error ? error : new Error(String(error)),
    });
  }

  _pirMarkModified(scope, field, nestedKey) {
    this._pirModified[scope].add(field);
    if (nestedKey === undefined) return;
    const nested = this._pirModifiedNested[scope];
    if (!nested.has(field)) nested.set(field, new Set());
    nested.get(field).add(nestedKey);
  }

  _pirClearModified(scope) {
    this._pirModified[scope].clear();
    this._pirModifiedNested[scope].clear();
  }

  _pirAssertProjectTrustedForWrite() {
    if (!this._pirProjectTrusted) {
      throw new Error("Refusing to write project settings for an untrusted project.");
    }
  }

  /** Persist the modified global fields; unknown keys in the file survive. */
  _pirPersist(scope, snapshot, modifiedFields, modifiedNested) {
    if (!this._pirStorage) return;
    const storage = this._pirStorage;
    const current = SettingsManager._pirStorageRead(storage, scope);
    const currentFile = current ? pirSettingsMigrate(JSON.parse(pirStripUtf8Bom(current))) : {};
    const merged = { ...currentFile };
    for (const field of modifiedFields) {
      const value = snapshot[field];
      if (modifiedNested.has(field) && pirIsPlainObject(value)) {
        const baseNested = pirIsPlainObject(currentFile[field]) ? currentFile[field] : {};
        const mergedNested = { ...baseNested };
        for (const nestedKey of modifiedNested.get(field)) mergedNested[nestedKey] = value[nestedKey];
        merged[field] = mergedNested;
      } else {
        merged[field] = value;
      }
    }
    SettingsManager._pirStorageWrite(storage, scope, JSON.stringify(merged, null, 2));
  }

  _pirSave() {
    this._pirRekey();
    if (this._pirLoadErrors.global) return;
    const snapshot = pirDeepCloneJson(this._pirGlobal);
    const modified = new Set(this._pirModified.global);
    const nested = new Map();
    for (const [field, keys] of this._pirModifiedNested.global.entries()) nested.set(field, new Set(keys));
    this._pirEnqueueWrite("global", () => this._pirPersist("global", snapshot, modified, nested));
  }

  _pirSaveProjectSettings(settings) {
    this._pirAssertProjectTrustedForWrite();
    this._pirProject = pirDeepCloneJson(settings);
    this._pirRekey();
    if (this._pirLoadErrors.project) return;
    const snapshot = pirDeepCloneJson(this._pirProject);
    const modified = new Set(this._pirModified.project);
    const nested = new Map();
    for (const [field, keys] of this._pirModifiedNested.project.entries()) nested.set(field, new Set(keys));
    this._pirEnqueueWrite("project", () => this._pirPersist("project", snapshot, modified, nested));
  }

  _pirEnqueueWrite(scope, task) {
    this._pirWriteQueue = this._pirWriteQueue
      .then(() => {
        if (scope === "project") this._pirAssertProjectTrustedForWrite();
        task();
        this._pirClearModified(scope);
      })
      .catch((error) => {
        this._pirRecordError(scope, error, this._pirPaths[scope]);
      });
  }

  /** The active fields of a scope, or of the effective merged settings. */
  _pirRead(field) {
    return this._pirSettings[field];
  }

  getSettings() { return pirDeepCloneJson(this._pirSettings); }
  getGlobalSettings() { return pirDeepCloneJson(this._pirGlobal); }
  getProjectSettings() { return pirDeepCloneJson(this._pirProject); }
  isProjectTrusted() { return this._pirProjectTrusted; }

  setProjectTrusted(trusted) {
    if (this._pirProjectTrusted === !!trusted) return;
    this._pirProjectTrusted = !!trusted;
    this._pirClearModified("project");
    if (!this._pirProjectTrusted) {
      this._pirProject = {};
      this._pirLoadErrors.project = null;
      this._pirRekey();
      return;
    }
    const load = SettingsManager.tryLoadFromStorage(this._pirStorage, "project", true);
    this._pirProject = load.settings;
    this._pirLoadErrors.project = load.error;
    if (load.error) this._pirRecordError("project", load.error);
    this._pirRekey();
  }

  async reload() {
    await this._pirWriteQueue;
    const globalLoad = SettingsManager.tryLoadFromStorage(this._pirStorage, "global");
    if (!globalLoad.error) {
      this._pirGlobal = globalLoad.settings;
      this._pirLoadErrors.global = null;
    } else {
      this._pirLoadErrors.global = globalLoad.error;
      this._pirRecordError("global", globalLoad.error);
    }
    this._pirClearModified("global");
    this._pirClearModified("project");
    const projectLoad = SettingsManager.tryLoadFromStorage(this._pirStorage, "project", this._pirProjectTrusted);
    if (!projectLoad.error) {
      this._pirProject = projectLoad.settings;
      this._pirLoadErrors.project = null;
    } else {
      this._pirLoadErrors.project = projectLoad.error;
      this._pirRecordError("project", projectLoad.error);
    }
    this._pirRekey();
  }

  applyOverrides(overrides) {
    if (!pirIsPlainObject(overrides)) return;
    this._pirSettings = pirSettingsDeepMerge(this._pirSettings, overrides);
  }

  save() { this._pirSave(); }
  saveProjectSettings(settings) { this._pirSaveProjectSettings(settings); }

  updateProjectSettings(field, update) {
    this._pirAssertProjectTrustedForWrite();
    const project = pirDeepCloneJson(this._pirProject);

    this._pirMarkModified("project", field);
    this._pirSaveProjectSettings(project);
  }

  setCompactionEnabled(enabled) {
    if (!pirIsPlainObject(this._pirGlobal.compaction)) this._pirGlobal.compaction = {};

  _pirCompactionTokenSetting(field, model) {
    const compaction = this._pirRead("compaction") ?? {};
    const ordinary = compaction[field];
    if (ordinary !== undefined && (typeof ordinary !== "number" || !Number.isSafeInteger(ordinary) || ordinary < 0)) {
      throw new Error("Invalid compaction." + field + " setting: " + String(ordinary) + ". Expected a non-negative safe integer.");
    }
    const modelKey = model && model.provider !== undefined ? model.provider + "/" + model.id : undefined;
    const entry = modelKey !== undefined ? compaction.modelOverrides?.[modelKey] : undefined;
    if (entry !== undefined && !pirIsPlainObject(entry)) {
      throw new Error('Invalid compaction.modelOverrides["' + modelKey + '"] setting: ' + String(entry) + ". Expected an object.");
    }
    const override = entry ? entry[field] : undefined;
    if (override !== undefined && (typeof override !== "number" || !Number.isSafeInteger(override) || override < 0)) {
      throw new Error('Invalid compaction.modelOverrides["' + modelKey + '"].' + field + " setting: " + String(override) + ". Expected a non-negative safe integer.");
    }
    return override ?? ordinary ?? PIR_DEFAULT_COMPACTION_TOKEN_SETTINGS[field];
  }
  getCompactionReserveTokens(model) { return this._pirCompactionTokenSetting("reserveTokens", model); }
  getCompactionKeepRecentTokens(model) { return this._pirCompactionTokenSetting("keepRecentTokens", model); }

// Real, host-backed `DefaultResourceLoader` (upstream
// `core/resource-loader.ts`, HEAD `7d75a0e3e`).
//
// Upstream loads a session's extensions, skills, prompts, themes, context
// files and system prompt from disk (plus packages and inline factories) into
// this process. pir's child sessions are separate processes (see the
// `createAgentSession` note above), so the loader splits the work honestly:
//
//   * EXTENSIONS: the loader resolves the requested paths (the caller's
//     `additionalExtensionPaths` plus, unless `noExtensions`, the settings
//     `extensions` list) to on-disk entry files and reports them through
//     `getExtensions()`. `createAgentSession` forwards them to the child
//     process as `--extension <path>`, so the child genuinely loads them —
//     this is what makes `requiredExtensions` verifiable (a path that cannot
//     be resolved yields an error entry, which the caller turns into a hard
//     failure instead of a silent skip).
//   * SKILLS / CONTEXT FILES / SYSTEM PROMPT: real disk reads through the
//     same ports the rest of this shim uses (`loadSkills`,
//     `loadProjectContextFiles`) and gated by `noSkills`/`noContextFiles`.
//   * INLINE EXTENSION FACTORIES (`extensionFactories`, e.g. pi-subagents'
//     child hooks and codemode): reported as load errors. A JS closure cannot
//     cross the child-process boundary, so the child cannot run them; the
//     child's tools/providers come from its own host resolution instead. This
//     is the documented boundary of the child-session design, surfaced as an
//     error entry rather than pretended success.
//   * PROMPTS / THEMES: empty results — the child process resolves its own
//     (the shim launches it with `--no-prompt-templates`; a theme is
//     presentation state the runner has no terminal for).
export class DefaultResourceLoader {
  constructor(options) {
    const opts = options && typeof options === "object" ? options : {};
    this.options = opts;
    this._pirCwd = pirResolvePath(opts.cwd === undefined ? "." : opts.cwd, pirToPosixPath(process.cwd()));
    this._pirAgentDir = pirResolvePath(
      opts.agentDir === undefined || !opts.agentDir ? getAgentDir() : opts.agentDir,
      pirToPosixPath(process.cwd()),
    );
    this._pirSettings =
      opts.settingsManager ?? SettingsManager.create(this._pirCwd, this._pirAgentDir, { projectTrusted: true });
    this._pirAdditionalExtensionPaths = (Array.isArray(opts.additionalExtensionPaths) ? opts.additionalExtensionPaths : [])
      .map(String);
    this._pirAdditionalSkillPaths = (Array.isArray(opts.additionalSkillPaths) ? opts.additionalSkillPaths : []).map(String);
    this._pirExtensionFactories = Array.isArray(opts.extensionFactories) ? opts.extensionFactories.slice() : [];
    this._pirNoExtensions = opts.noExtensions === true;
    this._pirNoSkills = opts.noSkills === true;
    this._pirNoContextFiles = opts.noContextFiles === true;
    this._pirExtensionsResult = { extensions: [], errors: [], runtime: createExtensionRuntime() };
    this._pirSkills = [];
    this._pirSkillDiagnostics = [];
    this._pirAgentsFiles = [];
    this._pirSystemPrompt = typeof opts.systemPrompt === "string" ? opts.systemPrompt : undefined;
    this._pirSystemPromptSource = undefined;
    this._pirAppendSystemPrompt = Array.isArray(opts.appendSystemPrompt) ? opts.appendSystemPrompt.map(String) : [];
    // Read by pi-subagents' `resetExtensionCacheOnReload` (child-session.js):
    // a loader carrying this field claims the reload-cache reset capability.
    this.loaded = false;
  }

  /** Resolve one extension request to an on-disk entry file. */
  _pirResolveExtensionPath(raw) {
    const text = String(raw ?? "").trim();
    if (!text) return undefined;
    if (text.startsWith("builtin:")) return undefined;
    let candidate = pirResolvePath(text, this._pirCwd);
    const stat = pirFsStat(candidate);
    if (stat && stat.isDirectory === true) {
      for (const entry of ["index.ts", "index.js", "index.mjs", "index.cjs"]) {
        const nested = pirJoinPath(candidate, entry);
        if (pirFsExists(nested)) return nested;
      }
      return undefined;
    }
    return stat && stat.isFile === true ? candidate : undefined;
  }

  async reload(options) {
    const opts = options && typeof options === "object" ? options : {};
    void opts;
    const errors = [];
    const extensions = [];
    const seen = new Set();
    const requested = this._pirAdditionalExtensionPaths.slice();
    if (!this._pirNoExtensions && this._pirSettings && typeof this._pirSettings.getExtensionPaths === "function") {
      for (const path of this._pirSettings.getExtensionPaths()) requested.push(path);
    }
    for (const raw of requested) {
      if (raw.startsWith("builtin:")) {
        // Built-in extension names (`builtin:mcp` and friends) are host-owned
        // factories, not files: they cannot be handed to the child process.
        errors.push({
          path: raw,
          error: new Error(
            "built-in extensions (" + raw + ") are host-integrated and cannot be loaded into a child session",
          ),
        });
        continue;
      }
      const resolved = this._pirResolveExtensionPath(raw);
      if (resolved === undefined) {
        errors.push({ path: raw, error: new Error("extension path could not be resolved: " + raw) });
        continue;
      }
      if (seen.has(resolved)) continue;
      seen.add(resolved);
      extensions.push({ path: resolved, source: "path" });
    }
    for (const factory of this._pirExtensionFactories) {
      const name = factory && typeof factory === "object" && typeof factory.name === "string" ? factory.name : "<inline>";
      errors.push({
        path: "<inline:" + name + ">",
        error: new Error(
          "inline extension factories cannot cross the child-session process boundary; the child resolves its own tools and providers",
        ),
      });
    }

    this._pirSkills = [];
    this._pirSkillDiagnostics = [];
    if (!this._pirNoSkills) {
      const result = loadSkills({
        cwd: this._pirCwd,
        agentDir: this._pirAgentDir,
        skillPaths: this._pirAdditionalSkillPaths,
      });
      this._pirSkills = Array.isArray(result.skills) ? result.skills : [];
      this._pirSkillDiagnostics = Array.isArray(result.diagnostics) ? result.diagnostics : [];
    }

    this._pirAgentsFiles = [];
    if (!this._pirNoContextFiles) {
      const files = loadProjectContextFiles({ cwd: this._pirCwd, agentDir: this._pirAgentDir });
      this._pirAgentsFiles = Array.isArray(files) ? files : [];
    }

    let result = { extensions, errors, runtime: createExtensionRuntime() };
    if (typeof this.options.extensionsOverride === "function") {
      const overridden = this.options.extensionsOverride(result);
      if (overridden && typeof overridden === "object") result = overridden;
    }
    this._pirExtensionsResult = result;
    this.loaded = true;
  }

  getExtensions() { return this._pirExtensionsResult; }
  getSkills() { return { skills: this._pirSkills, diagnostics: this._pirSkillDiagnostics }; }
  getPrompts() { return { prompts: [], diagnostics: [] }; }
  getThemes() { return { themes: [], diagnostics: [] }; }
  getAgentsFiles() { return { agentsFiles: this._pirAgentsFiles }; }
  getSystemPrompt() { return this._pirSystemPrompt; }
  getSystemPromptSource() { return this._pirSystemPromptSource; }
  getAppendSystemPrompt() { return this._pirAppendSystemPrompt.slice(); }

  async extendResources() { return this.getExtensions(); }
  async loadProjectTrustExtensions() {
    await this.reload();
    return this.getExtensions();
  }

  /**
   * Upstream tracks the APPEND_SYSTEM.md/`--append-system-prompt` file paths
   * that produced `appendSystemPrompt`; those files are read by the child
   * process here, so the loader has no source list to report.
   */
  getAppendSystemPromptSources() { return []; }
}

/** Model resolution helpers (delegate to the passed runtime and surface its
 * error in the result channel when the runtime is a stub). */
export function resolveCliModel(options) {
  const opts = options && typeof options === "object" ? options : {};
  const cliModel = opts.cliModel;
  const cliProvider = opts.cliProvider;
  if (!cliModel) return { model: undefined, warning: undefined, error: undefined };
  const modelRuntime = opts.modelRuntime;
  let availableModels;
  try {
    if (!modelRuntime || typeof modelRuntime.getModels !== "function") {
      return {
        model: undefined,
        warning: undefined,
        error: "ModelRuntime.getModels is not available in the pir extension runtime",
      };
    }
    availableModels = Array.from(modelRuntime.getModels());
  } catch (error) {
    return {
      model: undefined,
      warning: undefined,
      error: String((error && error.message) || error),
    };
  }
  if (availableModels.length === 0) {
    return {
      model: undefined,
      warning: undefined,
      error: "No models available. Check your installation or add models to models.json.",
    };
  }
  const providerMap = new Map();
  for (const model of availableModels) {
    if (model && typeof model.provider === "string") providerMap.set(model.provider.toLowerCase(), model.provider);
  }
  let provider = cliProvider ? providerMap.get(String(cliProvider).toLowerCase()) : undefined;
  if (cliProvider && !provider) {
    return {
      model: undefined,
      warning: undefined,
      error: 'Unknown provider "' + cliProvider + '". Use --list-models to see available providers/models.',
    };
  }
  let pattern = String(cliModel);
  if (!provider) {
    const slashIndex = pattern.indexOf("/");
    if (slashIndex !== -1) {
      const maybeProvider = providerMap.get(pattern.slice(0, slashIndex).toLowerCase());
      if (maybeProvider) {
        provider = maybeProvider;
        pattern = pattern.slice(slashIndex + 1);
      }
    }
  }
  const matches = availableModels.filter((model) => {
    if (!model) return false;
    if (provider && model.provider !== provider) return false;
    return model.id === pattern || (typeof model.id === "string" && model.id.includes(pattern));
  });
  if (matches.length === 0) {
    return {
      model: undefined,
      warning: undefined,
      error: 'No models match pattern "' + cliModel + '".',
    };
  }
  return { model: matches[0], thinkingLevel: opts.cliThinking, warning: undefined, error: undefined };
}

/** Resolve model scope patterns against the passed runtime (upstream
 * model-resolver.ts resolveModelScopeWithDiagnostics). */
/** Upstream `minimatch`-style glob over one candidate string (case-insensitive). */
function pirGlobMatch(candidate, pattern) {
  const source = String(pattern).replace(/[.+^${}()|\\]/g, "\\$&").replace(/\*/g, ".*").replace(/\?/g, ".");
  try {
    return new RegExp("^" + source + "$", "i").test(String(candidate));
  } catch {
    return false;
  }
}

/** Upstream `findExactModelReferenceMatch`: exact `provider/model` or bare id. */
function pirFindExactModelReferenceMatch(reference, models) {
  const trimmed = String(reference ?? "").trim();
  if (!trimmed) return undefined;
  const normalized = trimmed.toLowerCase();
  const canonical = models.filter((model) => (model.provider + "/" + model.id).toLowerCase() === normalized);
  if (canonical.length === 1) return canonical[0];
  if (canonical.length > 1) return undefined;
  const slashIndex = trimmed.indexOf("/");
  if (slashIndex !== -1) {
    const provider = trimmed.slice(0, slashIndex).trim();
    const modelId = trimmed.slice(slashIndex + 1).trim();
    if (provider && modelId) {
      const providerMatches = models.filter(
        (model) => model.provider.toLowerCase() === provider.toLowerCase() && model.id === modelId,
      );
      if (providerMatches.length === 1) return providerMatches[0];
      if (providerMatches.length > 1) return undefined;
    }
  }
  const idMatches = models.filter((model) => model.id === trimmed);
  return idMatches.length === 1 ? idMatches[0] : undefined;
}

export async function resolveModelScopeWithDiagnostics(patterns, modelRuntime, _options) {
  const list = Array.isArray(patterns) ? patterns : [];
  const scopedModels = [];
  const diagnostics = [];
  let availableModels;
  try {
    availableModels = await modelRuntime.getAvailable(undefined, _options);
  } catch (error) {
    diagnostics.push({ type: "warning", code: "unavailable", message: String((error && error.message) || error) });
    return { scopedModels, diagnostics };
  }
  const models = Array.isArray(availableModels) ? availableModels.slice() : [];
  for (const pattern of list) {
    const text = String(pattern ?? "");
    let thinkingLevel;
    let base = text;
    const colonIndex = text.lastIndexOf(":");
    if (colonIndex !== -1) {
      const maybeLevel = text.slice(colonIndex + 1);
      if (PIR_VALID_THINKING_LEVELS.indexOf(maybeLevel) !== -1) {
        thinkingLevel = maybeLevel;
        base = text.slice(0, colonIndex);
      }
    }
    if (base === "*") {
      for (const model of models) {
        if (!scopedModels.some((entry) => entry.model === model)) scopedModels.push({ model, thinkingLevel });
      }
      continue;
    }
    // Glob patterns match `provider/modelId` or the bare model id, exactly like
    // upstream (`resolveModelScopeFromModels` + minimatch).
    const hasGlob = base.includes("*") || base.includes("?") || base.includes("[");
    if (hasGlob) {
      const matches = models.filter(
        (model) => model && (pirGlobMatch(model.provider + "/" + model.id, base) || pirGlobMatch(model.id, base)),
      );
      if (matches.length === 0) {
        diagnostics.push({ type: "warning", code: "no-match", message: 'No models match pattern "' + pattern + '"', pattern });
        continue;
      }
      for (const model of matches) {
        if (!scopedModels.some((entry) => entry.model === model)) scopedModels.push({ model, thinkingLevel });
      }
      continue;
    }
    const exact = pirFindExactModelReferenceMatch(base, models);
    const matches = exact
      ? [exact]
      : models.filter((model) =>
        model && (model.id === base || model.id === text ||
          (typeof model.id === "string" && model.id.includes(base)) ||
          (typeof model.name === "string" && model.name.toLowerCase().includes(base.toLowerCase()))));
    if (matches.length === 0) {
      diagnostics.push({ type: "warning", code: "no-match", message: 'No models match pattern "' + pattern + '"', pattern });
      continue;
    }
    for (const model of matches) {
      if (!scopedModels.some((entry) => entry.model === model)) scopedModels.push({ model, thinkingLevel });
    }
  }
  return { scopedModels, diagnostics };
}

const PIR_VALID_THINKING_LEVELS = ["off", "minimal", "low", "medium", "high", "xhigh", "max"];

/**
 * Parse CLI arguments (upstream cli/args.ts parseArgs). Tool-name syntax
 * validation (getToolListError) is not applied here; the parsed list is
 * returned as-is.
 */
export function parseArgs(args) {
  const list = Array.isArray(args) ? args : [];
  const result = { messages: [], fileArgs: [], unknownFlags: new Map(), diagnostics: [] };
  const pushList = (key, value) => {
    result[key] = result[key] ?? [];
    result[key].push(value);
  };
  const splitCommas = (value) => String(value).split(",").map((s) => s.trim()).filter((s) => s.length > 0);

  for (let i = 0; i < list.length; i++) {
    const arg = list[i];
    if (arg === "--") {
      for (const positionalArg of list.slice(i + 1)) {
        if (positionalArg.startsWith("@")) result.fileArgs.push(positionalArg.slice(1));
        else result.messages.push(positionalArg);
      }
      break;
    } else if (arg === "--help" || arg === "-h") {
      result.help = true;
    } else if (arg === "--version" || arg === "-v") {
      result.version = true;
    } else if (arg === "--mode") {
      const mode = list[i + 1];
      if (mode === undefined || mode.startsWith("-")) {
        result.diagnostics.push({ type: "error", message: "--mode requires text, json, or rpc" });
        continue;
      }
      i++;
      if (mode !== "text" && mode !== "json" && mode !== "rpc") {
        result.diagnostics.push({ type: "error", message: 'Invalid mode "' + mode + '". Valid values: text, json, rpc' });
        continue;
      }
      result.mode = mode;
    } else if (arg === "--continue" || arg === "-c") {
      result.continue = true;
    } else if (arg === "--resume" || arg === "-r") {
      result.resume = true;
    } else if (arg === "--provider" && i + 1 < list.length) {
      result.provider = list[++i];
    } else if (arg === "--model" && i + 1 < list.length) {
      result.model = list[++i];
    } else if (arg === "--api-key" && i + 1 < list.length) {
      result.apiKey = list[++i];
    } else if (arg === "--system-prompt" && i + 1 < list.length) {
      result.systemPrompt = list[++i];
    } else if (arg === "--append-system-prompt" && i + 1 < list.length) {
      pushList("appendSystemPrompt", list[++i]);
    } else if (arg === "--name" || arg === "-n") {
      if (i + 1 < list.length) result.name = list[++i];
      else result.diagnostics.push({ type: "error", message: "--name requires a value" });
    } else if (arg === "--no-session") {
      result.noSession = true;
    } else if (arg === "--session" && i + 1 < list.length) {
      result.session = list[++i];
    } else if (arg === "--session-id" && i + 1 < list.length) {
      result.sessionId = list[++i];
    } else if (arg === "--fork" && i + 1 < list.length) {
      result.fork = list[++i];
    } else if (arg === "--session-dir" && i + 1 < list.length) {
      result.sessionDir = list[++i];
    } else if (arg === "--models" && i + 1 < list.length) {
      result.models = splitCommas(list[++i]);
    } else if (arg === "--no-tools" || arg === "-nt") {
      result.noTools = true;
    } else if (arg === "--no-builtin-tools" || arg === "-nbt") {
      result.noBuiltinTools = true;
    } else if ((arg === "--tools" || arg === "-t") && i + 1 < list.length) {
      result.tools = splitCommas(list[++i]);
    } else if ((arg === "--exclude-tools" || arg === "-xt") && i + 1 < list.length) {
      result.excludeTools = splitCommas(list[++i]);
    } else if (arg === "--thinking" && i + 1 < list.length) {
      const level = list[++i];
      if (PIR_VALID_THINKING_LEVELS.indexOf(level) !== -1) result.thinking = level;
      else {
        result.diagnostics.push({
          type: "warning",
          message: 'Invalid thinking level "' + level + '". Valid values: ' + PIR_VALID_THINKING_LEVELS.join(", "),
        });
      }
    } else if (arg === "--print" || arg === "-p") {
      result.print = true;
      const next = list[i + 1];
      if (next !== undefined && !next.startsWith("@") && (!next.startsWith("-") || next.startsWith("---"))) {
        result.messages.push(next);
        i++;
      }
    } else if (arg === "--export" && i + 1 < list.length) {
      result.export = list[++i];
    } else if ((arg === "--extension" || arg === "-e") && i + 1 < list.length) {
      pushList("extensions", list[++i]);
    } else if (arg === "--no-extensions" || arg === "-ne") {
      result.noExtensions = true;
    } else if (arg === "--no-mcp") {
      result.noMcp = true;
    } else if (arg === "--skill" && i + 1 < list.length) {
      pushList("skills", list[++i]);
    } else if (arg === "--prompt-template" && i + 1 < list.length) {
      pushList("promptTemplates", list[++i]);
    } else if (arg === "--theme" && i + 1 < list.length) {
      pushList("themes", list[++i]);
    } else if (arg === "--use-theme") {
      const themeName = list[i + 1];
      if (themeName === undefined || themeName.startsWith("-")) {
        result.diagnostics.push({ type: "error", message: "--use-theme requires a theme name" });
      } else {
        result.useTheme = themeName;
        i++;
      }
    } else if (arg === "--no-skills" || arg === "-ns") {
      result.noSkills = true;
    } else if (arg === "--no-prompt-templates" || arg === "-np") {
      result.noPromptTemplates = true;
    } else if (arg === "--no-themes") {
      result.noThemes = true;
    } else if (arg === "--no-context-files" || arg === "-nc") {
      result.noContextFiles = true;
    } else if (arg === "--list-models") {
      if (i + 1 < list.length && !list[i + 1].startsWith("-") && !list[i + 1].startsWith("@")) {
        result.listModels = list[++i];
      } else {
        result.listModels = true;
      }
    } else if (arg === "--tui-mode") {
      const mode = list[i + 1];
      if (mode === "regular" || mode === "fullscreen") {
        result.tuiMode = mode;
        i++;
      } else if (mode === undefined || mode.startsWith("-")) {
        result.diagnostics.push({ type: "error", message: "--tui-mode requires regular or fullscreen" });
      } else {
        i++;
        result.diagnostics.push({ type: "error", message: 'Invalid TUI mode "' + mode + '". Valid values: regular, fullscreen' });
      }
    } else if (arg === "--verbose") {
      result.verbose = true;
    } else if (arg === "--approve" || arg === "-a") {
      result.projectTrustOverride = true;
    } else if (arg === "--no-approve" || arg === "-na") {
      result.projectTrustOverride = false;
    } else if (arg === "--offline") {
      result.offline = true;
    } else if (arg.startsWith("@")) {
      result.fileArgs.push(arg.slice(1));
    } else if (arg.startsWith("--")) {
      const eqIndex = arg.indexOf("=");
      if (eqIndex !== -1) {
        result.unknownFlags.set(arg.slice(2, eqIndex), arg.slice(eqIndex + 1));
      } else {
        const flagName = arg.slice(2);
        const next = list[i + 1];
        if (next !== undefined && !next.startsWith("-") && !next.startsWith("@")) {
          result.unknownFlags.set(flagName, next);
          i++;
        } else {
          result.unknownFlags.set(flagName, true);
        }
      }
    } else if (arg.startsWith("-") && !arg.startsWith("--")) {
      result.diagnostics.push({ type: "error", message: "Unknown option: " + arg });
    } else if (!arg.startsWith("-")) {
      result.messages.push(arg);
    }
  }
  return result;
}

/**
 * The pi package directory (upstream config.ts getPackageDir). The pir
 * runtime has no package resolution: PI_PACKAGE_DIR and the standard
 * agent-dir npm layout are honored, falling back to the agent dir.
 */
export function getPackageDir() {
  const env = typeof process !== "undefined" && process ? process.env ?? {} : {};
  if (env.PI_PACKAGE_DIR) {
    const override = pirToPosixPath(env.PI_PACKAGE_DIR);
    if (pirFsExists(override)) return override;
  }
  const installed = pirJoinPath(getAgentDir(), "npm/node_modules/@earendil-works/pi-coding-agent");
  if (pirFsExists(installed)) return installed;
  return getAgentDir();
}

/** Path to the pi README.md (upstream config.ts getReadmePath). */
export function getReadmePath() {
  return pirJoinPath(getPackageDir(), "README.md");
}

/** Path to the pi docs directory (upstream config.ts getDocsPath). */
export function getDocsPath() {
  return pirJoinPath(getPackageDir(), "docs");
}

/** Path to the pi examples directory (upstream config.ts getExamplesPath). */
export function getExamplesPath() {
  return pirJoinPath(getPackageDir(), "examples");
}

/**
 * Credentials changed successfully, but the local model/auth snapshot could
 * not be synchronized (upstream model-runtime.ts).
 */
export class CredentialSynchronizationError extends Error {
  constructor(providerId, operation, credential, options) {
    super(
      "Credential " + operation + " committed for " + providerId + ", but local synchronization failed",
      options,
    );
    this.name = "CredentialSynchronizationError";
    this.providerId = providerId;
    this.operation = operation;
    this.credential = credential;
  }
}
